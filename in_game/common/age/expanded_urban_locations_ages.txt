# Expanded Urban Locations - Inverted Age Efficiency
# Vanilla has efficiency DECREASE over time (1.0 -> 0.1), which makes no sense
# Buildings should become MORE efficient as technology advances, not less
# This inverts the scaling so later ages benefit more from building levels

	efficiency = 0.1
}

	efficiency = 0.2
}

	efficiency = 0.4
}
INJECT:age_4_reformation = {
	efficiency = 0.6
}
INJECT:age_5_absolutism = {
	efficiency = 0.8
}
INJECT:age_6_revolutions = {
	efficiency = 1.0
}
