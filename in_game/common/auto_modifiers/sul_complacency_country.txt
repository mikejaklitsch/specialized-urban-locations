# Complacency — "bread and circuses."
#
# scales_with = complacency × 0.01 (magnitudes = effect at complacency 100).
#
# Enticing: pop growth, prosperity, prestige, demand. People thrive.
# Poison: production drops, military decays, institutions stagnate,
# building costs rise, estates bloat.

REPLACE:complacency_impact = {
	scales_with = {
		value = complacency
		multiply = 0.01
	}

	# --- PP drag (coupling layer) ---
	power_projection = -100

	# --- Enticing: comfort and demand (what your people want) ---

	global_population_growth = 0.005
	global_monthly_prosperity = 0.01
	monthly_prestige = 0.005
	hire_artist_cost_modifier = -0.5
	global_pop_food_consumption = 0.25		# +25% demand at full complacency

	# --- Poison: stagnation and institutional decay ---

	global_production_efficiency = -0.10

	research_speed_modifier = -0.5
	country_cabinet_efficiency = -0.5
	global_institution_growth_modifier = -0.5
	embrace_institution_cost_modifier = 5.0

	# --- Poison: military softness ---

	army_tradition_decay = 0.01
	navy_tradition_decay = 0.01
	global_manpower_modifier = -0.3
	global_sailors_modifier = -0.3
	land_morale_modifier = -0.25
	naval_morale_modifier = -0.25
	discipline = -0.10

	# --- Poison: economic bloat and fragility ---

	global_build_buildings_cost = 0.002
	court_spending_cost = 0.10
	diplomatic_capacity = -2.0
	add_accepted_culture_cost_modifier = 1.0
	declaring_war_cost_modifier = 1.0
	monthly_rebel_growth = 0.01
	global_migration_speed_modifier = -0.003
	pop_join_rebel_threshold = 0.2
}
