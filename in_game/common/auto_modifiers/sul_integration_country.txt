# Diplomatic reputation feeds into annexation speed.
# +1 flat annexation speed per 10 displayed diplomatic reputation.
# modifier:diplomatic_reputation returns displayed_value / 10,
# so multiply by 1.0 (= 10/10) to get +1 per 10 displayed.

sul_integration_diplo_rep_annexation = {
	scales_with = { value = modifier:diplomatic_reputation }

	annexation_speed_base = 1.0
}
