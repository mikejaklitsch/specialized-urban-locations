# Diplomatic reputation feeds into annexation speed and power projection.
# modifier:diplomatic_reputation returns displayed_value / 10,
# so multiply by 1.0 (= 10/10) to get +1 annexation per 10 displayed,
# and 2.0 gives +2 PP per 1 displayed dip rep.

sul_integration_diplo_rep_annexation = {
	scales_with = { value = modifier:diplomatic_reputation }

	annexation_speed_base = 1.0
	power_projection = 2
}
