# The Gold Standard - Auto Modifiers (Player Only)
# AI equivalents are applied in the monthly pulse via direct effects.

# Minting efficiency from market prices

sul_minting_market_efficiency = {
	potential_trigger = { is_ai = no }

	scales_with = sul_minting_cached_minting_efficiency

	minting_income_factor = 1.0
}

# Debasement inflation

sul_minting_debasement_inflation = {
	potential_trigger = { is_ai = no has_variable = sul_minting_debasement_level }

	scales_with = sul_minting_debasement_inflation_effect

	monthly_inflation = 1.0
}

# Minting inflation scaling: adjusts vanilla inflation to match
# price-adjusted income rather than just slider position

sul_minting_inflation_scaling = {
	potential_trigger = { is_ai = no }

	scales_with = sul_minting_inflation_adjustment

	monthly_inflation = 1.0
}

# Rebasement cost: monthly gold expense for coin recall

sul_minting_rebasement_gold = {
	potential_trigger = { is_ai = no has_variable = sul_minting_rebasement_level }

	scales_with = sul_minting_rebasement_cost

	monthly_gold_income = -1.0
}

# Rebasement deflation: monthly inflation reduction from coin recall

sul_minting_rebasement_deflation = {
	potential_trigger = { is_ai = no has_variable = sul_minting_rebasement_level }

	scales_with = sul_minting_rebasement_deflation_effect

	monthly_inflation = 1.0
}
