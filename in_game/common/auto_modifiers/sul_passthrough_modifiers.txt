# Vanilla applies a +produced_in_market_bonus to all countries by default,
# giving producers a price advantage when selling in their own market. This
# discourages export — producers earn more selling locally than exporting.
#
# This INJECT cancels that bonus by subtracting an equal value, so producers
# are indifferent between selling locally and exporting. Combined with the
# stockpile-driven price crush in the HIGH branch, producer markets actively
# push goods out instead of hoarding them locally.
#
# Adopted from yosiu_market_stockpiles workshop mod.
# =============================================================================

INJECT:country_base_values = {
	produced_in_market_bonus = -0.2
}
