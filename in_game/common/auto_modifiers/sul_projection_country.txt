# Power Projection — scale layer (INJECT into vanilla) + structural drag layer.
#
# SCALE LAYER: INJECT all PP-scaled effects into vanilla's power_projection
#   auto_modifier. A separate auto_modifier with scales_with = power_projection
#   does not work — engine ignores it.
#
# STRUCTURAL DRAG LAYER: separate auto_modifiers that *write* power_projection
#   from country state.
#
# Sign rule: positive PP always improves the player's situation. Coefficients
# below are placeholders — tune by feel during testing.

# ============================================================================
# Scale layer — separate auto_modifier following vanilla's _impact pattern
# (war_exhaustion_impact scales_with war_exhaustion). INJECT into the vanilla
# power_projection block fails (self-referential name collision).
# ============================================================================

power_projection_impact = {
	scales_with = power_projection

	global_pop_assimilation_speed_modifier = 0.005
	global_pop_conversion_speed_modifier = 0.005
	global_war_score_cost = -0.003
	global_distance_from_capital_speed_propagation = 0.005
	subject_loyalty = 0.5
	diplomatic_annexation_cost = -0.003
	antagonism_taking_land_giving_modifier = -0.005
	levy_recovery_modifier = 0.02
	global_levy_size_modifier = 0.0025
}

# ============================================================================
# Structural drag layer
# ============================================================================

sul_projection_drag_control = {
	scales_with = sul_projection_control_drag

	power_projection = 25
}

sul_projection_drag_culture = {
	scales_with = sul_projection_culture_drag

	power_projection = 25
}

sul_projection_drag_religion = {
	scales_with = sul_projection_religion_drag

	power_projection = 25
}

sul_projection_drag_stability = {
	scales_with = sul_projection_stability_drag

	# Stability ranges -100..+100 in EU5, so 0.1 caps the contribution at ±10 PP.
	power_projection = 0.1
}

sul_projection_drag_subjects = {
	scales_with = sul_projection_subject_drag

	# Variable is -(subject_load / overlord_strength). Each multiple of the
	# overlord's country_strength in subjects gives -25 PP. Linear, no cap.
	power_projection = 50
}

# Administrative overhead — raw territory size.
# -0.05 PP per location: 100 locs = -5 PP, 500 = -25, 1000 = -50.

sul_projection_drag_size = {
	scales_with = sul_projection_size_drag

	power_projection = -0.05
}

# Government rank. +10 PP per rank tier (county=10, duchy=20, kingdom=30).

sul_projection_rank = {
	scales_with = sul_projection_rank_value

	power_projection = 1
}

# Military strength — regular army and navy relative to expected size.
# +10 PP per multiple of expected. At 1x = +10 PP, at 2x = +20 PP, at 0.5x = +5 PP.
# Army uses a custom ratio that excludes levies (regular_army_size / expected_army_size).

sul_projection_army = {
	scales_with = sul_projection_regular_army_ratio

	power_projection = 10
}

sul_projection_navy = {
	scales_with = navy_size_percentage

	power_projection = 10
}
