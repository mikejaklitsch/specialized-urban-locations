# Power Projection — unified scale layer + structural drag layer.
#
# Monthly grow/decay system. Auto_modifiers write to monthly_sul_power_projection
# (monthly contributions). A monthly scripted effect applies contributions and
# decay to var:sul_power_projection. The scale layer auto_modifier reads the
# variable via script_value to produce PP-scaling effects.
#
# This sidesteps engine limitations where auto_modifiers cannot read modifier
# values set by other auto_modifiers.

# ============================================================================
# Structural drag layer — writes monthly contributions
# ============================================================================

sul_projection_drag_culture = {
	scales_with = sul_projection_culture_drag

	monthly_sul_power_projection = 0.25
}

sul_projection_drag_religion = {
	scales_with = sul_projection_religion_drag

	monthly_sul_power_projection = 0.2
}

# Diplomatic overextension — ratio of used to total diplomatic capacity.

sul_projection_drag_diplomacy = {
	scales_with = sul_projection_diplo_usage_ratio

	monthly_sul_power_projection = -0.5
}

# ============================================================================
# Scale layer — reads var:sul_power_projection via script_value.
# ============================================================================

sul_power_projection_impact = {
	potential_trigger = { has_variable = sul_power_projection }

	scales_with = sul_power_projection_value

	# --- Projection benefits (positive PP = advantage) ---
	subject_loyalty = 0.5                                    # +50
	global_pop_assimilation_speed_modifier = 0.01            # +100%
	global_pop_conversion_speed_modifier = 0.01              # +100%
	global_distance_from_capital_speed_propagation = 0.005   # +50%
	global_war_score_cost = -0.005                           # -50%
	diplomatic_annexation_cost = -0.005                      # -50%
	antagonism_taking_land_giving_modifier = -0.01           # -100%
	sul_trade_maintenance_efficiency = 0.0025                # +25%
	global_integration_speed_modifier = 0.01                 # +100%
	settle_country_cost_modifier = -0.005                    # -50%
	casus_belli_creation_speed_modifier = 0.005              # +50%

	# --- Military ---
	levy_recovery_modifier = 0.02                            # +200%
	global_levy_size_modifier = 0.02                         # +200%

	# --- Complacency coupling (PP builds complacency) ---
	monthly_complacency = 0.005                              # +0.5/mo

	# --- PP cost layer (stability, estates, civic drag) ---
	stability_decay = -0.0001                                # -1%
	global_estate_target_satisfaction = -0.002               # -20%
	global_estate_power = -0.005                             # -50%

	# --- AI behavior ---
	aggressiveness_modifier = 0.05                           # +5
	control_importance_modifier = -0.05                      # -5
	gold_importance_modifier = -0.05                         # -5
}

# ============================================================================
# Vanilla power_projection auto_modifier — emptied.
# ============================================================================

# Vanilla bridge — feeds var:sul_power_projection back into the engine's
# power_projection value so hardcoded vanilla systems (AI weighting, war
# score, subject interactions) that read power_projection keep working.
# Separate from sul_power_projection_impact so the vanilla value never
# leaks into the player-facing effects tooltip.

sul_projection_vanilla_bridge = {
	potential_trigger = { has_variable = sul_power_projection }

	scales_with = sul_power_projection_value

	power_projection = 1
}

REPLACE:power_projection = {
	scales_with = power_projection
}
