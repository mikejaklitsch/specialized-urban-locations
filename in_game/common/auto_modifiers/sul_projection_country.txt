# Power Projection — scale layer + structural drag layer.
#
# SCALE LAYER: one sul_power_projection_impact auto_modifier holding every
#   effect that scales with PP (benefits, complacency coupling, PP costs).
#
# STRUCTURAL DRAG LAYER: separate auto_modifiers that *write* power_projection
#   from country state.
#
# Sign rule: positive PP always improves the player's situation. Coefficients
# below are placeholders — tune by feel during testing.

# ============================================================================
# Scale layer — single auto_modifier scaling with power_projection. Follows
# vanilla's _impact pattern (war_exhaustion_impact scales_with war_exhaustion).
# Prefixed sul_ because we add it; INJECT into the vanilla power_projection
# block fails (self-referential name collision).
#
# Holds every effect that scales with PP:
#   - projection benefits (assimilation, conversion, war score, subjects…)
#   - complacency coupling (PP builds complacency)
#   - PP cost layer (stability drain, estate resentment, civic drag)
# ============================================================================

# Positive PP effects — applies when PP > 0, capped at +200.

sul_projection_positive = {
	potential_trigger = { power_projection > 0 }
	scales_with = {
		value = power_projection
		max = 200
	}

	subject_loyalty = 0.25                                   # +50
	global_pop_assimilation_speed_modifier = 0.005           # +100%
	global_pop_conversion_speed_modifier = 0.005             # +100%
	global_distance_from_capital_speed_propagation = 0.0025  # +50%
	global_war_score_cost = -0.0025                          # -50%
	diplomatic_annexation_cost = -0.0025                     # -50%
	antagonism_taking_land_giving_modifier = -0.005          # -100%
	sul_trade_maintenance_efficiency = 0.00125               # +25%
	monthly_complacency = 0.0015                             # +0.3/mo
	stability_decay = -0.000025                               # -0.5%
	global_estate_target_satisfaction = -0.001               # -20%
	global_estate_power = -0.0025                            # -50%
	levy_recovery_modifier = 0.01                            # +200%
	global_levy_size_modifier = 0.01                         # +200%
	global_integration_speed_modifier = 0.005                # +100%
	settle_country_cost_modifier = -0.0025                   # -50%
	casus_belli_creation_speed_modifier = 0.0025             # +50%
	aggressiveness_modifier = 0.025                          # +5
	control_importance_modifier = -0.025                     # -5
	gold_importance_modifier = -0.025                        # -5
}

# Negative PP effects — applies when PP < 0, capped at -200.

sul_projection_negative = {

	global_war_score_cost = -0.0025                          # +50%
	diplomatic_annexation_cost = -0.0025                     # +50%
	antagonism_taking_land_giving_modifier = -0.005          # +100%
	sul_trade_maintenance_efficiency = 0.00125               # -25%
	monthly_complacency = 0.0015                             # -0.3/mo
	stability_decay = 0.0001                                  # -2%
	global_estate_target_satisfaction = -0.001               # +20%
	global_estate_power = -0.0025                            # +50%
	global_integration_speed_modifier = 0.005                # -100%
	settle_country_cost_modifier = -0.0025                   # +50%
	casus_belli_creation_speed_modifier = 0.0025             # -50%
	aggressiveness_modifier = 0.025                          # -5
	control_importance_modifier = -0.025                     # +5
	gold_importance_modifier = -0.025                        # +5
}

# ============================================================================
# Structural drag layer
# ============================================================================

sul_projection_drag_control = {
	scales_with = sul_projection_control_drag

	power_projection = 50
}

sul_projection_drag_culture = {
	scales_with = sul_projection_culture_drag

	power_projection = 25
}

sul_projection_drag_religion = {
	scales_with = sul_projection_religion_drag

	power_projection = 25
}

# Diplomatic overextension — ratio of used to total diplomatic capacity.
# 0% used = 0 PP, 100% used = -50 PP. Overcommitted diplomacy erodes projection.

sul_projection_drag_diplomacy = {
	scales_with = sul_projection_diplo_usage_ratio

	power_projection = -50
}

# Government rank. +10 PP per rank tier (county=10, duchy=20, kingdom=30).

sul_projection_rank = {
	scales_with = sul_projection_rank_value

	power_projection = 1
}
