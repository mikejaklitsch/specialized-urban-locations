REPLACE:entrepot = {
	is_foreign = no
	pop_type = burghers
	max_levels = 1

	category = trade_category

	employment_size = trade_employment

	town = yes
	city = yes

	expensive = yes
	build_time = market_build_time

	location_potential = {
		AND = {
			is_market_center = yes
			var:specialization_designation = flag:commercial
		}
	}

	remove_if = {
		location = {
			OR = {
				is_market_center = no
				var:specialization_designation != flag:commercial
			}
		}
	}
	modifier = {
		merchant_capacity_from_building = 5
	}

	market_center_modifier = {
		local_trade_center_power = 0.1
	}

	possible_production_methods = {
		market_center_building_maintenance
	}
	construction_demand = market_center_building_construction
}
