REPLACE:fruit_orchard = {
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	increase_per_level_cost = 0.15
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = yes

	remove_if = {
		location = {
			has_farming_specialization = no
		}
	}

	location_potential = {
		AND = {
			OR = {
				vegetation = grasslands
				vegetation = farmland
				vegetation = woods
				vegetation = jungle
			}
			has_farming_specialization = yes
		}
	}

	build_time = rural_build_time
	construction_demand = farm_constructions
	unique_production_methods = { fruit_orchard_maintenance }
}

REPLACE:sugar_plantation = {
	is_foreign = no
	pop_type = slaves
	max_levels = plantation_cap
	increase_per_level_cost = 0.15
	employment_size = plantation_employment
	category = colonial_category

	location_potential = {
		AND = {
			raw_material = goods:sugar
			OR = {
				is_overseas_for_owner = yes
				owner ?= { is_colonial_subject = yes }
			}
			has_farming_specialization = yes
		}
	}

	remove_if = {
		location = {
			has_farming_specialization = no
		}
	}

	rural_settlement = yes
	town = yes
	city = yes
	price = expand_rgo_farming
	always_add_demands = yes
	build_time = plantation_build_time

	modifier = {
		local_sugar_output_modifier = 0.1
	}

	construction_demand = plantation_construction
	unique_production_methods = { sugar_plantation_maintenance }
}
