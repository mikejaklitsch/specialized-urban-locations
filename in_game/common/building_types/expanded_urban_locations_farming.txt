REPLACE:fruit_orchard = {
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = yes

	remove_if = {
		location = {
			NOR = {
				location_rank = location_rank:rural_settlement
				var:specialization_designation = flag:farming
			}
		}
	}

	location_potential = {
		AND = {
			OR = {
				vegetation = grasslands
				vegetation = farmland
				vegetation = woods
				vegetation = jungle
			}
			OR = {
				location_rank = location_rank:rural_settlement
				var:specialization_designation = flag:farming
			}
		}
	}

	unique_production_methods = {
		fruit_orchard_maintenance = {
			produced = fruit
			output = 0.3
			lumber = 0.17
			category = building_maintenance
		}
	}

	build_time = rural_build_time
	construction_demand = farm_constructions
}
