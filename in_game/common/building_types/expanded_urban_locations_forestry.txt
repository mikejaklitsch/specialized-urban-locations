REPLACE:lumber_mill = {
	is_foreign = no
	max_levels = rural_building_cap
	pop_type = laborers
	employment_size = rural_peasant_produce_employment
	rural_settlement = yes
	town = yes
	city = yes
	location_potential = {
		AND = {
			OR = {
				vegetation = woods
				vegetation = forest
				vegetation = farmland
				vegetation = jungle
			}
			NOT = {
				raw_material = goods:lumber
			}
			OR = {
				location_rank = location_rank:rural_settlement
				var:specialization_designation = flag:forestry
			}
		}
	}

	remove_if = {
		location = {
			NOR = {
				location_rank = location_rank:rural_settlement
				var:specialization_designation = flag:forestry
			}
		}
	}

	category = rgo_building_category
	build_time = rural_build_time

	unique_production_methods = {
		lumber_mill_maintenance = {
			produced = lumber
			output = 1
			tools = 0.4
			category = building_maintenance
		}
	}

	construction_demand = basic_construction_needs
}

	is_foreign = no
	max_levels = {
