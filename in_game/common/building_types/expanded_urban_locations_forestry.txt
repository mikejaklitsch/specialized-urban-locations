REPLACE:lumber_mill = {
	is_foreign = no
	max_levels = rural_building_cap
	pop_type = laborers
	employment_size = rural_peasant_produce_employment
	rural_settlement = yes
	town = yes
	city = yes
	location_potential = {
		AND = {
			OR = {
				vegetation = woods
				vegetation = forest
				vegetation = farmland
				vegetation = jungle
			}
			NOT = {
				raw_material = goods:lumber
			}
			OR = {
				location_rank = location_rank:rural_settlement
				has_forestry_specialization = yes
			}
		}
	}

	remove_if = {
		location = {
			NOR = {
				location_rank = location_rank:rural_settlement
				has_forestry_specialization = yes
			}
		}
	}

	category = rgo_building_category
	build_time = rural_build_time
	construction_demand = basic_construction_needs
	unique_production_methods = { lumber_mill_maintenance }
}

REPLACE:elephant_hunting_grounds = {
	is_foreign = no
	max_levels = {
		value = rural_building_cap
		if = {
			limit = {
				owner ?= {
					has_variable = ori_double_elephant_hunting_ground_level_flag
				}
			}
			add = {
				desc = "INCREASED_MAX_LEVEL_ELEPHANT_HUNTING_GROUND"
				value = 2
			}
		}
	}
	pop_type = peasants
	category = rgo_building_category
	employment_size = elephant_hunting_grounds_employment
	rural_settlement = yes
	town = yes
	city = yes
	country_potential = {
		produced_in_country:elephants > 0
	}

	location_potential = {
		AND = {
			raw_material = goods:elephants
			OR = {
				location_rank = location_rank:rural_settlement
				has_forestry_specialization = yes
			}
		}
	}

	remove_if = {
		location = {
			NOR = {
				location_rank = location_rank:rural_settlement

		}
	}

	build_time = rural_build_time

	modifier = {
		local_monthly_food_modifier = -0.05

	}

	construction_demand = village_construction
	unique_production_methods = { elephant_hunting_grounds_maintenance }
}
