REPLACE:clay_pit = {
	is_foreign = no
	max_levels = rural_building_cap
	pop_type = laborers
	employment_size = rural_peasant_produce_employment
	rural_settlement = yes
	town = yes
	city = yes
	location_potential = {
		AND = {
			OR = {
				is_coastal = yes
				has_river = yes
				is_adjacent_to_lake = yes
			}
			NOT = {
				raw_material = goods:clay
			}
			OR = {
				location_rank = location_rank:rural_settlement
				var:specialization_designation = flag:gathering
			}
		}
	}

	remove_if = {
		location = {
			NOR = {
				location_rank = location_rank:rural_settlement
				var:specialization_designation = flag:gathering
			}
		}
	}
	category = rgo_building_category
	unique_production_methods = {
		clay_pit_maintenance = {
			produced = clay
			output = 1
			lumber = 0.28
			category = building_maintenance
		}
	}
	build_time = rural_build_time
	construction_demand = basic_construction_needs
}
