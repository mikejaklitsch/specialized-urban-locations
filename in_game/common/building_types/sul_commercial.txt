# Custom cottage industry buildings

sul_rural_ropemaker = {
	is_foreign = no
	pop_type = laborers
	max_levels = guild_max_level
	category = consumer_goods_category
	employment_size = 0.4
	rural_settlement = yes
	town = yes
	city = yes

	location_potential = {
		OR = {
			is_produced_in_location_market = goods:lumber
			is_produced_in_location_market = goods:tar
			is_produced_in_location_market = goods:fiber_crops
		}

		sul_has_commercial_specialization = yes
	}

	build_time = rural_build_time

	possible_production_methods = {
		sul_rural_rope_and_pitch
		sul_rural_fiber_cordage
	}

	construction_demand = village_construction
}
