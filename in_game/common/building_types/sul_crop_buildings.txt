# Terrain-gated crop buildings — food goods production based on geography
# Output at 75% of corresponding RGO, 20% of output diverted to provisions
# Non-farmland vegetation requires river/lake access (except drought-resistant crops)

REPLACE:fruit_orchard = {
	icon = sul_rgo_fruit # PLACEHOLDER ICON TODO
	is_foreign = no
	max_levels = sul_extraction_max_level
	pop_type = peasants
	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = yes
	important_for_AI = yes
	ai_forbid_shutdown = yes

	location_potential = {
		OR = {
			vegetation = farmland
			AND = {
				OR = {
					vegetation = grasslands
					vegetation = woods
					vegetation = jungle
				}
				OR = {
					has_river = yes
					is_adjacent_to_lake = yes
				}
			}
		}
		OR = {
			climate = mediterranean
			climate = subtropical
			climate = tropical
		}

		is_produced_in_location_market = goods:fruit

		sul_has_farming_specialization = yes
	}

	build_time = rural_build_time

	unique_production_methods = {
		sul_fruit_orchard_maintenance = {
			tools = 0.05
			produced = fruit
			output = 0.75
			debug_max_profit = -1
			category = guild_input
		}

		sul_fruit_orchard_provisions = {
			fruit = 0.15
			produced = provisions
			output = 0.15
			debug_max_profit = -1
			category = building_maintenance
		}
	}

	construction_demand = rgo_construction_farming

	modifier = {
		sul_is_extraction_building = yes
		local_monthly_development = -0.01

		local_population_growth = 0.00015
	}
}
