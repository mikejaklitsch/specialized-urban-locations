	increase_per_level_cost = 0.0
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap

	employment_size = rural_peasant_produce_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = yes

	location_potential = {
		AND = {
			OR = {
				vegetation = grasslands
				vegetation = farmland
				vegetation = woods
				vegetation = jungle
			}
			sul_has_farming_specialization = yes
		}
	}

	modifier = {
		local_monthly_food = 1.0
		# Farming: +0.05 food, +0.01 raw, +0.005 rgo
		local_monthly_food_modifier = 0.05
		local_raw_material_output = 0.02
		local_max_rgo_size_modifier = 0.02
	}

	build_time = rural_build_time
	construction_demand = farm_constructions
	unique_production_methods = {
		sul_fruit_orchard_maintenance = {
			produced = fruit
			output = 0.3
			lumber = 0.17
			category = building_maintenance
		}
	}
}
