REPLACE:mercury_patio = {
	increase_per_level_cost = 0.0
	is_foreign = no
	max_levels = mercury_patio_max_level

	pop_type = laborers
	category = rgo_building_category
	employment_size = advanced_peasant_building_employment
	rural_settlement = no
	town = yes
	city = yes
	location_potential = {
		AND = {
			OR = {
				raw_material = goods:silver
				raw_material = goods:goods_gold
			}
			sul_has_mining_specialization = yes
		}
	}

	build_time = guild_build_time
	construction_demand = mercury_patio_construction

	modifier = {
		local_goods_gold_output_modifier = 1.0
		local_silver_output_modifier = 1.0
		# Precious metals: no raw material bonus
	}
	unique_production_methods = {
		sul_mercury_patio_maintenance = {
			mercury = 1.0
			category = building_maintenance
		}
	}
}

REPLACE:local_smelters = {
	increase_per_level_cost = 0.0
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap

	employment_size = rural_peasant_enhance_employment
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = yes
	location_potential = {
		AND = {
			OR = {
				raw_material = goods:iron
				raw_material = goods:copper
				raw_material = goods:tin
				raw_material = goods:lead
			}
			sul_has_mining_specialization = yes
		}
	}

	modifier = {
		local_iron_output_modifier = 0.1
		local_copper_output_modifier = 0.1
		local_tin_output_modifier = 0.1
		local_lead_output_modifier = 0.1
		# Raw Extraction: +0.02 raw, +0.005 rgo
		local_raw_material_output = 0.02
		local_max_rgo_size_modifier = 0.02
		# Mining goods output
		local_tools_output_modifier = 0.0075
		local_weaponry_output_modifier = 0.0075
		local_cannons_output_modifier = 0.0075
		local_firearms_output_modifier = 0.0075
		local_steel_output_modifier = 0.0075
		local_jewelry_output_modifier = 0.0075
	}

	build_time = rural_build_time
	construction_demand = farm_constructions
	unique_production_methods = {
		sul_local_smelters_maintenance = {
			tools = 0.025
			lumber = 0.05
			coal = 0.05
			category = building_maintenance
		}
	}
}
