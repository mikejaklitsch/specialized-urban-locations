# Mining specialization — custom buildings

sul_iron_mine = {
	is_foreign = no
	max_levels = sul_extraction_max_level
	pop_type = laborers
	employment_size = rural_peasant_produce_employment
	rural_settlement = yes
	town = yes
	city = yes

	location_potential = {
		OR = {
			topography = mountains
			topography = hills
		}
		NOT = {
			raw_material = goods:iron
		}

		sul_has_mining_specialization = yes
	}

	category = rgo_building_category

	possible_production_methods = {
		sul_iron_mine_maintenance
		sul_crude_iron_mine_maintenance
	}

	build_time = rural_build_time
	construction_demand = basic_construction_needs

	modifier = {
		sul_is_extraction_building = yes
		sul_extraction_size = 0.25
		sul_production_size = -0.25
		local_population_growth = 0.00015
	}
}
