# Mining specialization — custom buildings

sul_iron_mine = {
	is_foreign = no
	max_levels = rural_building_cap
	pop_type = laborers
	employment_size = rural_peasant_produce_employment
	rural_settlement = yes
	town = yes
	city = yes

	location_potential = {
		OR = {
			topography = mountains
			topography = hills
		}
		NOT = {
			raw_material = goods:iron
		}

		sul_has_mining_specialization = yes
	}

	category = rgo_building_category

	possible_production_methods = {
		sul_iron_mine_maintenance
		sul_crude_iron_mine_maintenance
	}

	build_time = rural_build_time
	construction_demand = basic_construction_needs
}

REPLACE:local_smelters = {
	is_foreign = no
	pop_type = laborers
	max_levels = rural_building_cap
	employment_size = rural_peasant_enhance_employment
	category = rgo_building_category
	rural_settlement = yes
	ai_forbid_shutdown = yes

	location_potential = {
		OR = {
			raw_material = goods:iron
			raw_material = goods:copper
			raw_material = goods:tin
			raw_material = goods:lead
		}

		sul_has_mining_specialization = yes
	}

	unique_production_methods = {
		sul_local_smelters_maintenance = {
			tools = 0.03
			lumber = 0.05
			coal = 0.05
			category = building_maintenance
		}
	}

	modifier = {
		local_iron_output_modifier = 0.1
		local_copper_output_modifier = 0.1
		local_tin_output_modifier = 0.1
		local_lead_output_modifier = 0.1
	}

	build_time = rural_build_time
	construction_demand = farm_constructions
}

	is_foreign = no
	pop_type = laborers

	rural_settlement = yes
	town = no
	city = no
