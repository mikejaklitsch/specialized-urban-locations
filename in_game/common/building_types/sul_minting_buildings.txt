# TGS Monetization — event building modifier replacements
# Must be kept in sync with vanilla.

INJECT:wisselbank = {
	capital_country_modifier = { minting_inflation_threshold = 0.02 sul_minting_monetization = 0.10 }

	city = no
	town = no
	rural_settlement = no

	commercial_city = yes
	commercial_town = yes
	farming_city = yes
	farming_town = yes
	gathering_city = yes
	gathering_town = yes
	mining_city = yes
	mining_town = yes
	woodland_city = yes
	woodland_town = yes
	on_built = {
		location = { add_to_variable_list = { name = sul_epbm_buildings target = prev } }
		sul_epbm_on_building_built = yes
	}
	on_destroyed = {
		location = { remove_list_variable = { name = sul_epbm_buildings target = prev } }
		sul_epbm_on_building_destroyed = yes
	}
}
