sul_inland_wharf = { # River Wharf
	increase_per_level_cost = 0.0
	is_foreign = no
	max_levels = 1

	pop_type = laborers

	category = trade_category

	employment_size = dock_employment

	rural_settlement = yes
	town = yes
	city = yes

	location_potential = {
		has_river = yes
	}

	build_time = medium_port_building_time

	construction_demand = town_building_construction

	modifier = {
		local_merchant_capacity = 0.4
		local_merchant_power = 0.4
		local_market_access = 0.02
		local_manpower = 0.001
		local_distance_from_capital_cost_modifier = -0.025
		# Trade/Commercial: baseline penalty only
	}

	unique_production_methods = {

			tar = 0.1
			naval_supplies = 0.05
			tools = 0.1
			lumber = 0.1
			category = building_maintenance
		}
	}
}
