inland_wharf = { # River Wharf
	is_foreign = no
	max_levels = 1
	
	pop_type = laborers

	category = trade_category

	employment_size = dock_employment

	rural_settlement = yes
	town = yes
	city = yes

	location_potential = {
		has_river = yes
	}

	build_time = medium_port_building_time

	construction_demand = town_building_construction

	modifier = {
		merchant_capacity_from_building = 0.25
		merchant_power_from_building = 0.25
		local_market_access = 0.02
		local_sailors = 0.001
		local_distance_from_capital_cost_modifier = -0.025
	}

	unique_production_methods = { inland_wharf_maintenance }
}
