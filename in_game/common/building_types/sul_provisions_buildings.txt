# ============================================
# PROVISIONS BUILDINGS
# ============================================
# Provisions system flow:
#   1. Cookhouse converts raw food goods → provisions (trade good)
#   2. Villages/windmill supplement with small-scale provisions production
#   3. Provisions trades on the market like any good
#   4. Granary consumes provisions, provides local_monthly_food (town/city)
#   5. Storehouse does the same at smaller scale (rural)
#
# Cookhouse is deliberately unprofitable — food processing is a cost of
# civilization, not a business. Margins target ~-5% to -10%.

# ============================================
# COOKHOUSE — Food processing building
# ============================================
# Converts raw food goods into provisions via regional recipes.
# Primary provisions producer. Available at all settlement types.

sul_cookhouse = {
	is_foreign = no
	max_levels = guild_max_level
	pop_type = peasants
	employment_size = 1
	category = consumer_goods_category
	rural_settlement = yes
	town = yes
	city = yes
	ai_forbid_shutdown = no
	important_for_AI = yes
	can_close = yes

	build_time = workshop_build_time

	# Base recipe — pick one
	unique_production_methods = {
		# Wheat + livestock: European staple
		# Cost: 2.0 + 1.2 + 0.1 = 3.3, Revenue: 3.0, Margin: -9%
		sul_wheat_pottage = {
			wheat = 2.0
			livestock = 0.8
			salt = 0.1
			produced = provisions
			output = 1.0
			category = building_maintenance
		}

		# Rice + legumes: South/Southeast Asian staple
		# Cost: 1.75 + 1.5 + 0.1 = 3.35, Revenue: 3.0, Margin: -10%
		sul_rice_and_legumes = {
			rice = 1.75
			legumes = 1.5
			salt = 0.1
			produced = provisions
			output = 1.0
			category = building_maintenance
		}

		# Fish + millet: Coastal/African
		# Cost: 2.0 + 1.0 + 0.2 = 3.2, Revenue: 3.0, Margin: -6%
		sul_salted_fish_stew = {
			fish = 2.0
			millet = 1.0
			salt = 0.2
			produced = provisions
			output = 1.0
			category = building_maintenance
		}

		# Maize + livestock + potato: American staple
		# Cost: 1.2 + 1.2 + 0.5 = 2.9, Revenue: 3.0, Margin: +3%
		sul_maize_stew = {
			maize = 1.2
			livestock = 0.8
			potato = 0.5
			produced = provisions
			output = 1.0
			category = building_maintenance
		}

		# Pemmican: Frontier/pastoral
		# Cost: 2.25 + 0.5 + 0.2 = 2.95, Revenue: 3.0, Margin: +2%
		sul_pemmican = {
			livestock = 1.5
			wild_game = 0.5
			fruit = 0.2
			produced = provisions
			output = 1.0
			category = building_maintenance
		}

		# Olive bread: Mediterranean
		# Cost: 1.5 + 1.0 + 0.5 = 3.0, Revenue: 3.0, Margin: 0%
		sul_olive_bread = {
			wheat = 1.5
			olives = 1.0
			legumes = 0.5
			produced = provisions
			output = 1.0
			category = building_maintenance
		}
	}

	construction_demand = workshop_construction

	modifier = {
		local_provisions_output_modifier = 0.00
	}
}

# ============================================
# STOREHOUSE — Rural food distribution
# ============================================
# Weaker rural counterpart to the granary.
# Lower capacity, lower throughput, but keeps rural settlements fed.

sul_storehouse = {
	is_foreign = no
	max_levels = 2
	pop_type = peasants
	employment_size = 0.5
	category = infrastructure_category
	rural_settlement = yes

	build_time = rural_build_time

	unique_production_methods = {
		sul_storehouse_maintenance = {
			lumber = 0.1
			provisions = 0.5
			category = building_maintenance
		}
	}

	construction_demand = village_construction

	modifier = {
		local_food_capacity = 200

	}

	ai_forbid_shutdown = yes
}
