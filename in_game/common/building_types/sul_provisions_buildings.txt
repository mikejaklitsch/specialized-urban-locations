# ============================================
# PROVISIONS BUILDINGS
# ============================================
# Provisions system flow:
#   1. Cookhouse converts raw food goods → provisions (trade good)
#   2. Villages/windmill supplement with small-scale provisions production
#   3. Provisions trades on the market like any good
#   4. Granary consumes provisions, provides local_monthly_food (town/city)
#   5. Storehouse does the same at smaller scale (rural)
#
# Cookhouse is the laborer-scale provisions producer.
# Villages (peasant) produce 1.0 provisions; cookhouse (laborer) produces 4.0.
# Mirrors the rural_granary (peasant, 15 food) → granary (laborer, 60 food) relationship.

# ============================================

	category = infrastructure_category
	rural_settlement = yes

	# Maintenance: masonry for structure + provisions for food distribution
	unique_production_methods = {
		sul_granary_maintenance = { provisions = 1.0 category = building_maintenance }
	}

	construction_demand = granary_construction

	modifier = {
		local_food_capacity = 600
		local_monthly_food = 60
		local_max_rgo_size_modifier = 0.005
		local_production_efficiency = 0.005
	}
}
