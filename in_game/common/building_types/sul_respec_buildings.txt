@sul_mining_type = 1
@sul_farming_type = 2
@sul_gathering_type = 3
@sul_woodland_type = 4
@sul_commercial_type = 5

# Specialization carrier buildings. Hidden from the build menu, queued by
# generic actions. Construction consumes goods and gold over time; canceling
# refunds everything. on_built applies the specialization then self-destructs.

# ========================================================================
# MINING
# ========================================================================

sul_respec_mining_rural = {
	is_special = yes
	max_levels = 1
	category = military_category

	rural_settlement = yes
	town = no
	city = no

	build_time = sul_respec_rural_build_time
	price = sul_respec_mining_rural_price
	construction_demand = sul_respec_mining_rural_construction

	country_potential = { always = no }
	allow = { always = no }
	can_destroy = {}

	on_built = {
		sul_respec_on_built_effect = {
			SPEC_TYPE = @sul_mining_type
			MODIFIER = sul_mining_rural_specialization
			BUILDING_TYPE = sul_respec_mining_rural
			COUNT_VAR = sul_mining_count
			WEIGHT = 1
		}
	}
}
