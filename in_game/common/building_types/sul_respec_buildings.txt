@sul_mining_type = 1
@sul_farming_type = 2
@sul_gathering_type = 3
@sul_woodland_type = 4
@sul_commercial_type = 5

# Specialization carrier buildings. Hidden from the build menu, queued by
# generic actions. Gold-only cost via vanilla rank upgrade prices.
# on_built applies the specialization then remove_if self-destructs.

# ========================================================================
# MINING
# ========================================================================

sul_respec_mining_rural = {
	is_special = yes
	max_levels = 1
	pop_type = nobles
	employment_size = 0
	category = government_category

	mining_rural = yes
	farming_rural = yes
	gathering_rural = yes
	woodland_rural = yes
	commercial_rural = yes

	build_time = sul_respec_rural_build_time
	price = rural_settlement_upgrade
	construction_demand = basic_construction_needs

	unique_production_methods = {
		sul_respec_mining_rural_pm = {
			masonry = 0.1
			category = building_construction
		}
	}

	country_potential = {
		always = no
	}

	allow = {
		always = no
	}

	can_destroy = {  }

	remove_if = { location = { modifier:sul_mining_specialization = yes } }

	on_built = {
		sul_respec_on_built_effect = {
			SPEC = mining
			COUNT_VAR = sul_mining_count
			WEIGHT = 1
		}
	}
}
