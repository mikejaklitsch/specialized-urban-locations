# ============================================
# VILLAGE BUILDING MODIFICATIONS
# ============================================
# Makes village buildings effectively free by granting +1 max building level per level
# Also applies positive externality modifiers to support their industries
# market_village has no vanilla location_potential, so INJECT works

INJECT:market_village = {
	town = yes
	city = yes

	location_potential = {
		OR = {
			sul_has_commercial_specialization = yes
			sul_has_mining_specialization = yes
		}
	}

	modifier = {
		free_building_levels = 1
		local_monthly_food = 1.0
		# Commercial goods output
		local_cloth_output_modifier = 0.005
		local_fine_cloth_output_modifier = 0.005
		local_books_output_modifier = 0.005
		local_naval_supplies_output_modifier = 0.005
		# Mining goods output
		local_tools_output_modifier = 0.005
		local_weaponry_output_modifier = 0.005
		local_cannons_output_modifier = 0.005
		local_firearms_output_modifier = 0.005
		local_steel_output_modifier = 0.005
		local_jewelry_output_modifier = 0.005
	}
}

REPLACE:fishing_village = {
	increase_per_level_cost = 0.0
	is_foreign = no
	max_levels = rural_building_cap
	pop_type = peasants
	category = village_category
	employment_size = village_employment_size
	rural_settlement = yes
	town = yes
	city = yes
	AI_optimization_flag_coastal = yes

	location_potential = {
		sul_has_gathering_specialization = yes
	}

	build_time = village_build_time
	construction_demand = village_construction

	modifier = {
		free_building_levels = 1
		local_monthly_food = 1.5
		# Force Multiplier: +0.10 food, +0.01 raw, +0.01 rgo
		local_monthly_food_modifier = 0.10
		local_raw_material_output = 0.01
		local_max_rgo_size_modifier = 0.01
		# Gathering goods output
		local_glass_output_modifier = 0.01
		local_pottery_output_modifier = 0.01
		local_porcelain_output_modifier = 0.01
		local_lacquerware_output_modifier = 0.01
	}

	unique_production_methods = {
		sul_fishing_village_maintenance = {
			produced = fish
			output = 0.5
			salt = 0.08
			tools = 0.02
			category = building_maintenance
		}
	}
}
