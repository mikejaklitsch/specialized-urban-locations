# =============================================================================
# PASSTHROUGH WAREHOUSE BUILDINGS
# =============================================================================
# Adapted from yosiu_market_stockpiles workshop mod.
#
# Each warehouse has a self-cancelling production method: output = 1 unit
# of the good, input = 0.97 units of the same good. Net flow into the
# supply pool is +0.03/tick (scaled by production efficiency).
#
# The paired sul_<good>_oversupply temp demand (applied in
# sul_passthrough_per_good's HIGH branch with scale =
# sul_passthrough_warehouse_correction) exactly cancels this +0.03 at the
# flow layer, so the stockpile never actually grows from the warehouse.
#
# The warehouse's PURPOSE is to inflate the goods_supply_in_market reading
# that vanilla's trade routing engine consumes, without leaking real goods
# into inventory. This is the only way to create "extra exportable supply"
# in EU5 since there is no add_temporary_supply effect.
#
# free_building_levels = 1 in the modifier block means these warehouses
# always exist with 1 free level on every location, without needing to be
# built manually.
# =============================================================================

	pop_type = peasants
	max_levels = 1

	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = yes
	can_close = no

	allow = {
		always = no
	}

	country_potential = {
		always = no
	}

	unique_production_methods = {
		sul_horses_warehouse_pm = {
			produced = horses
			output = 1
			horses = 0.97
			category = government_activities
			debug_max_profit = 0.04
		}
	}

	modifier = { free_building_levels = 1 }

	build_time = rural_build_time
	construction_demand = basic_construction_needs
}
