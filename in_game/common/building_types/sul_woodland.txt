	is_foreign = no

	max_levels = rural_building_cap

	forbidden_for_estates = yes
	category = rgo_building_category

		OR = {
			vegetation = woods
			vegetation = forest

		}

		sul_has_woodland_specialization = yes

		sul_sawmill_maintenance = { tools = 0.05 category = building_maintenance }
	}

	build_time = rural_build_time
	construction_demand = farm_constructions
}
