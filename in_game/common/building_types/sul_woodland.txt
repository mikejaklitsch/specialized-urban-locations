REPLACE:lumber_mill = {
	increase_per_level_cost = 0.0
	is_foreign = no
	max_levels = rural_building_cap
	
	pop_type = laborers
	employment_size = rural_peasant_produce_employment
	rural_settlement = yes
	town = yes
	city = yes
	location_potential = {
		AND = {
			OR = {
				vegetation = woods
				vegetation = forest
				vegetation = farmland
				vegetation = jungle
			}
			NOT = { raw_material = goods:lumber }
			sul_has_woodland_specialization = yes
		}
	}


	modifier = {
		# Externality: Raw Extraction
		local_monthly_food_modifier = -0.05
		local_raw_material_output = 0.01
		local_max_rgo_size_modifier = 0.01
		local_production_efficiency = -0.005
	}

	category = rgo_building_category
	build_time = rural_build_time
	construction_demand = basic_construction_needs
	unique_production_methods = {
		sul_lumber_mill_maintenance = {
			produced = lumber
			output = 1
			tools = 0.4
			category = building_maintenance
		}
	}
}
