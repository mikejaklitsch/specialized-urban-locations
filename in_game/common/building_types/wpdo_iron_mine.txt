# Iron Mine — produces iron from mountainous terrain without iron deposits.
# Unlocked by iron_working advance.
# Topography-based PMs: mountains (best), hills, plateaus (worst).

wpdo_iron_mine = {
	is_foreign = no
	max_levels = rural_building_cap
	pop_type = laborers
	employment_size = rural_peasant_produce_employment
	rural_settlement = yes
	town = yes
	city = yes

	location_potential = {
		OR = {
			topography = mountains
			topography = plateau
			topography = hills
		}
		NOT = {
			raw_material = goods:iron
		}
	}

	category = rgo_building_category

	unique_production_methods = {
		# Mountains: richest veins, 0.5 output, ~50% margin
		# Cost: 0.15*3 + 0.2*1.5 + 0.125*2 = 1.0, Revenue: 1.5
		wpdo_iron_mine_mountain_maintenance = {
			tools = 0.15
			lumber = 0.2
			coal = 0.125
			produced = iron
			output = 0.5

			allow = {
				scope:target ?= { topography = mountains }
			}

			debug_max_profit = 1
			category = building_maintenance
		}

		# Hills: moderate deposits, 0.4 output, ~37% margin
		# Cost: 0.15*3 + 0.15*1.5 + 0.1*2 = 0.875, Revenue: 1.2
		wpdo_iron_mine_hills_maintenance = {
			tools = 0.15
			lumber = 0.15
			coal = 0.1
			produced = iron
			output = 0.4

			allow = {
				scope:target ?= { topography = hills }
			}

			debug_max_profit = 1
			category = building_maintenance
		}

		# Plateaus: shallow deposits, 0.3 output, ~22% margin
		# Cost: 0.12*3 + 0.12*1.5 + 0.1*2 = 0.74, Revenue: 0.9
		wpdo_iron_mine_plateau_maintenance = {
			tools = 0.12
			lumber = 0.12
			coal = 0.1
			produced = iron
			output = 0.3

			allow = {
				scope:target ?= { topography = plateau }
			}

			category = building_maintenance
		}
	}

	build_time = rural_build_time
	price = wpdo_iron_mine_price
	construction_demand = wpdo_iron_mine_constructions
}
