
wpdo_rgo_chili = {
	is_foreign = no
	pop_type = laborers
	max_levels = wpdo_rgo_max_level_base
	employment_size = employment_size_RGO
	category = rgo_building_category
	rural_settlement = yes
	town = yes
	city = yes
	price = expand_rgo_farming
	is_indestructible = yes

	location_potential = {
		can_build_RGO = { GOOD = chili }
	}

	unique_production_methods = {
		wpdo_chili_maintenance = {
			tools = 0.05
			produced = chili
			output = 1
			debug_max_profit = 33
			category = building_maintenance
		}
	}

	modifier = { free_building_levels = 1 }

	build_time = build_time_RGO
	construction_demand = rgo_construction_light

	on_construction_started = {
		location ?= { change_variable = { name = wpdo_rgo_constructing add = 1 } }
	}

	on_construction_ended = {
		location ?= { change_variable = { name = wpdo_rgo_constructing add = -1 } }
	}
}
