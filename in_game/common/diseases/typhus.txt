
typhus = {
	monthly_resistance_reduction = 0.001

	monthly_spawn_chance = {
		if = { limit = { disease_is_active = scope:disease } value = 0 }
		else = { value = 0.5 }
	}

	spawn = {
		random_ownable_location = {
			limit = {
				population > 1
				any_unit_in_location = { is_army = yes owner = { at_war = yes } }
			}

			spawn_disease = {
				disease = scope:disease
				value = 0.5
				save_scope_as = outbreak_scope
			}
		}
	}

	r0 = {
		if = {
			limit = { modifier:sul_is_rural_settlement_rank_tier = yes }
			value = { 1.2 1.8 }
		}
		else_if = {
			limit = { modifier:sul_is_town_rank_tier = yes }
			value = { 1.8 2.5 }
		}
		else = { value = { 2.5 3.5 } }
		if = { limit = { prosperity < 0 } multiply = 3 }
		if = {
			limit = {
				OR = {
					has_building_with_at_least_one_level = market_village
					has_building_with_at_least_one_level = marketplace
					has_building_with_at_least_one_level = slave_market
					has_building_with_at_least_one_level = entrepot
					has_building_with_at_least_one_level = trading_hub
					has_building_with_at_least_one_level = stock_exchange
				}
			}

			multiply = 1.2
		}
		multiply = { value = "disease_presence(scope:disease)" multiply = 2 add = 1 }
	}

	calc_interval_days = { 24 36 }

	location_stagnation_chance = {
		# Very aggressive distance falloff — typhus stays local
		if = {
			limit = { "disease_presence(scope:disease)" > 0.05 }
			value = "distance_to_squared(scope:disease.origin)"
			multiply = 0.005
			multiply = 0.025
		}
		else = {
			value = "distance_to_squared(scope:disease.origin)"
			multiply = 0.0005
		}
		if = {
			limit = { "disease_resistance(scope:disease)" > 0.7 }
			add = { value = "disease_resistance(scope:disease)" subtract = 0.5 }
		}
		if = { limit = { "disease_resistance(scope:disease)" > 0.9 } add = 0.2 }
	}

	sub_unit_stagnation_chance = {
		value = 0.5
		subtract = { value = "disease_presence(scope:disease)" multiply = 0.5 }
	}

	percentage_to_meet_their_fate_on_calc = { value = 0.35 }

	location_modifier = {
		potential_trigger = {
			disease_presence = { disease = scope:disease value >= 0.2 }
			NOT = {
				disease_has_stagnated = scope:disease
			}
		}

		local_forced_attrition = 5
		local_prosperity_decay = 0.1
	}

	mortality_rate = { 0.15 0.50 }

	character_mortality_chance = {
		value = 0.06
		multiply = "disease_presence(scope:disease)"
	}

	on_spread_to_country = { trigger_event_non_silently = diseases.1 }

	map_color = {
		lerp = {
			min_color = rgb { 211 157 128 }
			max_color = rgb { 129 47 3 }
			factor = "disease_presence(scope:disease)"
		}
	}

	secondary_map_color = {
		if = {
			limit = { disease_affects_pops_here = scope:disease }
			value = rgb { 100 120 188 }
		}
	}

	specific_pop_type_effect = {
		pop_type = nobles
		multiplier = 0.3
	}
	specific_pop_type_effect = {
		pop_type = clergy
		multiplier = 0.1
	}
	specific_pop_type_effect = {
		pop_type = burghers
		multiplier = 0.5
	}
}
