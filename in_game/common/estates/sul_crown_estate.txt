# =============================================================================
# CROWN ESTATE MODIFICATIONS
# =============================================================================
#
# Adds building upkeep reduction scaling with crown power.
# Linear scaling: -75% at 0% crown power → -25% at 100% crown power
#
# INJECT adds to vanilla values, so we offset:
# - Vanilla low_power: +2.0, we want -3.0 final → inject -5.0
# - Vanilla high_power: -1.0, we want +3.0 final → inject +4.0
# - Vanilla power: none, we want -2.5 → inject -2.5
#
# Final values after combining with vanilla:
# At 0%:   (-3 × 0.25) + (-2.5 × 0)   = -75%
# At 25%:  (-3 × 0)    + (-2.5 × 0.25) = -62.5%
# At 50%:  (+3 × 0.25) + (-2.5 × 0.5)  = -50%
# At 100%: (+3 × 0.75) + (-2.5 × 1)    = -25%
# =============================================================================

INJECT:crown_estate = {
	low_power = {
		building_upkeep_costs = -5.0
	}
	high_power = {
		building_upkeep_costs = 4.0
	}
	power = {
		building_upkeep_costs = -2.5
	}
}
