# Estate power blocks: food_purchase_cost + building_upkeep_costs
#
# Each non-crown estate reduces food purchase cost and building upkeep
# proportional to its power. The engine's power = {} block scales linearly
# by estate_power (0–1), so combined non-crown estates at 70% total power
# produce food_purchase_cost = -0.70 and building_upkeep_costs = -0.70.

INJECT:crown_estate = {
	low_power = {
		building_upkeep_costs = -2.0 # + vanilla 2.0 = net 0
	}

	high_power = {
		building_upkeep_costs = 1.0 # + vanilla -1.0 = net 0
		power_projection = -5 # cancel vanilla's +5
	}
}
