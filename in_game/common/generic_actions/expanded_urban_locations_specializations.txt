# Expanded Urban Locations - Specialization Designation Actions
# These actions allow players to designate a specialization for towns/cities
# Costs gold + prestige, destroys non-matching buildings, applies modifier
# scope:actor = country, scope:target = location

designate_mining_specialization = {
	type = owncountry

	potential = {
	}

	allow = {
	}

	ai_tick = never

	price = price:designate_specialization_price

	select_trigger = {
		looking_for_a = location
		target_flag = target
		source = actor
		name = "designate_mining_select"

		visible = {
			owner = scope:actor
			is_town_or_city = yes
		}

		enabled = {
			has_mining_resources = yes
			NOT = { var:specialization_designation = flag:mining }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				# Remove any existing specialization modifiers
				clear_all_specialization_modifiers = yes

				# Set the specialization variable
				set_variable = { name = specialization_designation value = flag:mining }

				# Destroy incompatible buildings
				destroy_non_mining_buildings = yes

				# Apply appropriate modifier based on current rank
				if = {
					limit = { location_rank = location_rank:town }
					add_location_modifier = { modifier = mining_town_specialization }
				}
				else_if = {
					limit = { location_rank = location_rank:city }
					add_location_modifier = { modifier = mining_city_specialization }
				}
			}
		}
	}
}
