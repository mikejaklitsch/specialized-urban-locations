# Expanded Urban Locations - Specialization Designation Actions
# These actions allow players to designate a specialization for towns/cities
# Costs gold + prestige, destroys non-matching buildings, applies modifier
# scope:actor = country, scope:target = location

designate_mining_specialization = {
	type = owncountry

	potential = {
	}

	allow = {
	}

	ai_tick = monthly
	ai_tick_frequency = 12

	ai_prerequisite = {
		at_war = no
		treasury >= 100
	}

	ai_will_do = {
		# Base inertia - AI resists changing specializations
		add = {
			desc = "inertia"
			value = -100
		}

		# Strong desire if location has mining resources but wrong specialization
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					has_mining_resources = yes
					has_any_specialization = yes
					has_mining_specialization = no
				}
			}
			add = {
				desc = "resource_mismatch_correction"
				value = 150
			}
		}

		# Bonus for cities (higher value targets)
		if = {
			limit = {
				exists = scope:target
				scope:target = { location_rank = location_rank:city }
			}
			add = {
				desc = "city_priority"
				value = 25
			}
		}
	}

	price = price:designate_specialization_price

	select_trigger = {
		looking_for_a = location
		target_flag = target
		source = actor
		name = "designate_mining_select"

		visible = {
			owner = scope:actor
			is_town_or_city = yes
		}

		enabled = {
			has_mining_resources = yes
			has_mining_specialization = no
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				clear_all_specialization_modifiers = yes
				if = {
					limit = { location_rank = location_rank:town }
					set_variable = { name = specialization_designation value = eul_mining_town_value }
				}
				else = {
					set_variable = { name = specialization_designation value = eul_mining_city_value }
				}
				destroy_non_mining_buildings = yes
				apply_specialization_modifier = yes
			}
		}
	}
}

				}
			}
			add = {
				desc = "market_center_should_be_commercial"
				value = 200
			}
		}

		# Moderate desire for high harbor locations
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					modifier:harbor_suitability > 0.7
					has_commercial_specialization = no
				}
			}
			add = {
				desc = "harbor_commercial_potential"
