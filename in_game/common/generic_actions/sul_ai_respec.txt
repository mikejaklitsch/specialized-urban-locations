# Local variable winner values: 1=mining, 2=farming, 3=woodland, 4=gathering, 5=commercial
@sul_winner_mining = 1
@sul_winner_farming = 2
@sul_winner_woodland = 3
@sul_winner_gathering = 4
@sul_winner_commercial = 5

sul_ai_respecialize = {
	type = owncountry
	show_in_gui_list = no
	show_message = no

	potential = {
	}

	allow = {
	}

	price = price:sul_respec_price

	price_modifier = {
		add = scope:actor.modifier:global_build_buildings_cost
		if = {
			limit = { exists = scope:target }
			add = scope:target.modifier:local_build_buildings_cost
			add = scope:target.modifier:local_build_new_buildings_cost

			if = {
				limit = { scope:target = { modifier:sul_is_town_rank_tier = yes } }
				add = 1
			}

			if = {
				limit = { scope:target = { modifier:sul_is_city_rank_tier = yes } }
				add = 2
			}
		}
	}

	should_execute_price = no

	ai_tick = monthly

	ai_tick_frequency = {
		value = 120

		divide = {
			value = num_locations
			divide = 50
			min = 1
		}

		min = 12
		max = 120
	}

	ai_prerequisite = {
		is_real_country = yes
		current_year >= 1338
		num_loans = 0
		gold >= 200
		monthly_balance >= 0
	}

	ai_will_do = {
		if = { limit = { exists = scope:target } add = { desc = "respec_score" value = scope:target.sul_respec_score } }
	}

	select_trigger = {
		looking_for_a = location
		target_flag = target
		source = actor
		name = "ai_respec_select"

		visible = {
			exists = owner
			owner = scope:actor
			OR = {
				modifier:sul_is_rural_settlement_rank_tier = yes
				modifier:sul_is_town_rank_tier = yes
				modifier:sul_is_city_rank_tier = yes
			}
		}

		enabled = {
			NOT = { has_location_modifier = sul_recently_respecialized }

			controller = owner
			market_access >= 0.3
			local_control >= 0.3
		}

		pre_evaluation_sort_value = {
			value = local_control
			multiply = 100
			subtract = location_tax_base
		}

		pre_evaluation_number_to_evaluate_fully = 15
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				# Tournament: find best eligible non-current spec
				set_local_variable = { name = best value = -1000 }
				set_local_variable = { name = winner value = 0 }

				# Mining
				if = {
					limit = { NOT = { sul_has_mining_specialization = yes } sul_can_specialize_mining = yes }

					set_local_variable = { name = score value = sul_attractiveness_mining }

					if = {
						limit = { local_var:score > local_var:best }
						set_local_variable = { name = best value = local_var:score }
						set_local_variable = { name = winner value = @sul_winner_mining }
					}
				}

				# Farming
				if = {
					limit = { NOT = { sul_has_farming_specialization = yes } sul_can_specialize_farming = yes }

					set_local_variable = { name = score value = sul_attractiveness_farming }

					if = {
						limit = { local_var:score > local_var:best }
						set_local_variable = { name = best value = local_var:score }
						set_local_variable = { name = winner value = @sul_winner_farming }
					}
				}

				# Woodland
				if = {
					limit = { NOT = { sul_has_woodland_specialization = yes } sul_can_specialize_woodland = yes }

					set_local_variable = { name = score value = sul_attractiveness_woodland }

					if = {
						limit = { local_var:score > local_var:best }
						set_local_variable = { name = best value = local_var:score }
						set_local_variable = { name = winner value = @sul_winner_woodland }
					}
				}

				# Gathering
				if = {
					limit = { NOT = { sul_has_gathering_specialization = yes } sul_can_specialize_gathering = yes }

					set_local_variable = { name = score value = sul_attractiveness_gathering }

					if = {
						limit = { local_var:score > local_var:best }
						set_local_variable = { name = best value = local_var:score }
						set_local_variable = { name = winner value = @sul_winner_gathering }
					}
				}

				# Commercial
				if = {
					limit = { NOT = { sul_has_commercial_specialization = yes } }

					set_local_variable = { name = score value = sul_attractiveness_commercial }

					if = {
						limit = { local_var:score > local_var:best }
						set_local_variable = { name = best value = local_var:score }
						set_local_variable = { name = winner value = @sul_winner_commercial }
					}
				}

				if = { limit = { local_var:winner = @sul_winner_mining } sul_respec_to_mining = yes }
				else_if = { limit = { local_var:winner = @sul_winner_farming } sul_respec_to_farming = yes }
				else_if = { limit = { local_var:winner = @sul_winner_woodland } sul_respec_to_woodland = yes }
				else_if = { limit = { local_var:winner = @sul_winner_gathering } sul_respec_to_gathering = yes }
				else_if = { limit = { local_var:winner = @sul_winner_commercial } sul_respec_to_commercial = yes }
			}
		}
	}
}
