sul_add_all_locations_to_hre = {
	type = owncountry
	show_in_gui_list = yes

	potential = {
		is_member_of_international_organization = international_organization:hre
	}

	allow = {
		any_owned_location = {
			sul_io_flood_fill_can_add_location = { IO = international_organization:hre }
			any_neighbor_location = {
				is_owned_by_international_organization = international_organization:hre
			}
		}
	}

	effect = {
		scope:actor = {
			sul_add_all_eligible_to_io = { IO = international_organization:hre }
		}
	}

	ai_will_do = {
		value = 0
	}

	ai_tick = never
}

sul_expand_hre_from_location = {
	type = owncountry
	show_in_gui_list = yes

	potential = {
		is_member_of_international_organization = international_organization:hre
	}

	allow = {
		any_owned_location = {
			sul_io_flood_fill_can_add_location = { IO = international_organization:hre }
			any_neighbor_location = {
				is_owned_by_international_organization = international_organization:hre
			}
		}
	}

	select_trigger = {
		looking_for_a = location
		target_flag = target
		source = actor
		source_flags = only_actual_locations
		name = "sul_expand_hre_from_location_select"
		none_available_msg_key = "sul_expand_hre_from_location_none"
		column = {
			data = name
		}
		column = {
			data = population
		}
		visible = {
			sul_io_flood_fill_can_add_location = { IO = international_organization:hre }
		}
		enabled = {
			any_neighbor_location = {
				is_owned_by_international_organization = international_organization:hre
			}
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				sul_expand_io_from_location = { IO = international_organization:hre }
			}
		}
	}

	ai_will_do = {
		value = 0
	}

	ai_tick = never
}
