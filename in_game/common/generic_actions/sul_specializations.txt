# Player-facing specialization designation actions. Each action takes a town
# or city, charges gold + prestige, destroys non-matching buildings, and
# applies the rank modifier. scope:actor = country, scope:target = location.

@sul_mining_type = 1
@sul_farming_type = 2
@sul_gathering_type = 3
@sul_woodland_type = 4
@sul_commercial_type = 5

sul_designate_mining_specialization = {
	type = owncountry

	potential = {
	}

	allow = {
	}

	price = {
		if = {
			limit = { exists = scope:target scope:target = { location_rank = location_rank:rural_settlement } }
			value = price:sul_respec_mining_rural_price
		}
		else_if = {
			limit = { exists = scope:target scope:target = { location_rank = location_rank:town } }
			value = price:sul_respec_mining_town_price
		}
		else = { value = price:sul_respec_mining_city_price }
	}
	price_modifier = {
		add = scope:actor.modifier:global_build_buildings_cost
		if = {
			limit = { exists = scope:target }
			add = scope:target.modifier:local_build_buildings_cost
			add = scope:target.modifier:local_build_new_buildings_cost
		}
	}
	should_execute_price = no

	ai_tick = never

	select_trigger = {
		looking_for_a = location
		target_flag = target
		source = actor
		name = "designate_mining_select"

		visible = {
			owner = scope:actor
			OR = {
				location_rank = location_rank:rural_settlement
				location_rank = location_rank:town
				location_rank = location_rank:city
			}
		}

		enabled = {
			# Mining eligibility: resources OR terrain
			OR = {
				# Mining resources
				raw_material = goods:coal
				raw_material = goods:iron
				raw_material = goods:copper
				raw_material = goods:goods_gold
				raw_material = goods:silver
				raw_material = goods:stone
				raw_material = goods:tin
				raw_material = goods:lead
				raw_material = goods:alum
				raw_material = goods:gems
				raw_material = goods:marble
				raw_material = goods:mercury
				# Mining terrain
				topography = mountains
				topography = wetlands
				is_adjacent_to_lake = yes
			}
			NOT = {
				sul_has_mining_specialization = yes
			}
			NOT = {
				has_location_modifier = sul_recently_respecialized
			}
			NOT = { sul_has_respec_construction = yes }

			controller = owner
		}
	}
	effect = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				if = {
					limit = { location_rank = location_rank:rural_settlement }
					construct_building = { building_type = building_type:sul_respec_mining_rural }
				}
				else_if = {
					limit = { location_rank = location_rank:town }
					construct_building = { building_type = building_type:sul_respec_mining_town }
				}
				else = {
					construct_building = { building_type = building_type:sul_respec_mining_city }
				}
			}
		}
	}
}
