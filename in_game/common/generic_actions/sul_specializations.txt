# Player-facing specialization designation actions. Each action selects a
# location, displays price (not charged — carrier building handles payment),
# and queues a carrier building whose on_built fires the respec.
# scope:actor = country, scope:target = location.

sul_designate_mining_specialization = {
	type = owncountry

	potential = {
	}

	allow = {
	}

	price = price:sul_respec_price

	price_modifier = {
		value = 1
		if = {
			limit = {
				exists = scope:target
				scope:target = { modifier:sul_is_city_rank_tier = yes }
			}

			multiply = 20
		}
		else_if = {
			limit = {
				exists = scope:target
				scope:target = { modifier:sul_is_town_rank_tier = yes }
			}

			multiply = 5
		}

		add = scope:actor.modifier:global_build_buildings_cost
		if = {
			limit = { exists = scope:target }
			add = scope:target.modifier:local_build_buildings_cost
			add = scope:target.modifier:local_build_new_buildings_cost
		}
	}

	should_execute_price = no

	ai_tick = never

	select_trigger = {
		looking_for_a = location
		target_flag = target
		source = actor
		name = "designate_mining_select"

		visible = {
			owner = scope:actor
			OR = {
				modifier:sul_is_rural_settlement_rank_tier = yes
				modifier:sul_is_town_rank_tier = yes
				modifier:sul_is_city_rank_tier = yes
			}
		}

		enabled = {
			OR = {
				raw_material = goods:coal
				raw_material = goods:iron
				raw_material = goods:copper
				raw_material = goods:goods_gold
				raw_material = goods:silver
				raw_material = goods:stone
				raw_material = goods:tin
				raw_material = goods:lead
				raw_material = goods:alum
				raw_material = goods:gems
				raw_material = goods:marble
				raw_material = goods:mercury
				topography = mountains
				topography = wetlands
				is_adjacent_to_lake = yes
			}
			NOT = {
				sul_has_mining_specialization = yes
			}
			NOT = {
				has_location_modifier = sul_recently_respecialized
			}

			controller = owner
		}
	}
	effect = {
		if = {
			limit = { exists = scope:target }

			scope:target = {
				custom_tooltip = SUL_RESPEC_BUILD_TIME
				sul_respec_show_destroy_non_mining = yes
				if = {
					limit = { is_market_center = yes }
					custom_tooltip = SUL_RESPEC_MARKET_DESTROYED
				}

				custom_tooltip = SUL_RESPEC_PENALTY_APPLIED

				hidden_effect = {
					sul_respec_to_mining = yes
				}
			}
		}
	}
}
