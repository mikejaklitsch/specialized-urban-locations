# Specialized Urban Locations - Specialization Designation Actions
# These actions allow players to designate a specialization for towns/cities
# Costs gold + prestige, destroys non-matching buildings, applies modifier
# scope:actor = country, scope:target = location
@sul_mining_type = 1
@sul_farming_type = 2
@sul_gathering_type = 3
@sul_woodland_type = 4
@sul_commercial_type = 5

sul_designate_mining_specialization = {
	type = owncountry

	potential = {
	}

	allow = {
	}

	ai_tick = monthly
	ai_tick_frequency = 12

	ai_prerequisite = {
		is_real_country = yes
		current_year >= 1338
		num_loans = 0
		gold >= 50
		monthly_balance >= 0
	}

	ai_will_do = {
		# === AFFORDABILITY + INERTIA (branched by rank) ===
		add = { desc = "inertia" value = 0 }
		if = {
			limit = {
				exists = scope:target
				scope:target = { location_rank = location_rank:town }
			}

			add = { desc = "town_inertia" value = -5 }
			if = {
				limit = {
					scope:actor = {
						OR = {
							gold < 300
							prestige < 5
						}
					}
				}

				add = { desc = "cannot_afford" value = -1000 }
			}
		}
		else_if = {
			limit = {
				exists = scope:target
				scope:target = { location_rank = location_rank:city }
			}

			add = { desc = "city_inertia" value = -10 }
			if = {
				limit = {
					scope:actor = {
						OR = {
							gold < 1000
							prestige < 10
						}
					}
				}

				add = { desc = "cannot_afford" value = -1000 }
			}
		}

		# === TERRAIN BONUSES (else_if chain - only one terrain can match) ===
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					sul_can_specialize_mining = yes
					NOT = {
						sul_has_mining_specialization = yes
					}
				}
			}

			# Wetlands (bog iron): best for mining - iron is more valuable than stone
			if = {
				limit = { scope:target = { topography = wetlands } }
				add = { desc = "wetlands_bog_iron" value = 3 }
			}

			# Mountains: good for mining
			else_if = {
				limit = { scope:target = { topography = mountains } }
				add = { desc = "mountains_terrain" value = 2 }
			}

			# Other terrain: no terrain bonus (resources still get value bonus)
		}

		# Gold: massive bonus - should always be mining
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					raw_material = goods:goods_gold
					NOT = {
						sul_has_mining_specialization = yes
					}
				}
			}

			add = { desc = "gold_resource" value = 100 }
		}

		# Silver: massive bonus - should always be mining
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					raw_material = goods:silver
					NOT = {
						sul_has_mining_specialization = yes
					}
				}
			}

			add = { desc = "silver_resource" value = 100 }
		}

		# === RAW MATERIAL DEMAND PRESSURE ===
		# Location's actual raw material demand/supply ratio, ±5 scale
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					sul_can_specialize_mining = yes
					NOT = {
						sul_has_mining_specialization = yes
					}
				}
			}

			scope:target.market ?= {
				add = {
					desc = "raw_material_demand"
					value = "goods_demand_in_market(scope:target.raw_material)"

					divide = {
						value = "goods_supply_in_market(scope:target.raw_material)"
						min = 0.1
					}

					subtract = 1
					max = 1
					min = -1
					multiply = 5
				}
			}
		}

		# Wetland iron bonus (bog iron smelter potential)
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					sul_can_specialize_mining = yes
					NOT = {
						sul_has_mining_specialization = yes
					}

					topography = wetlands

					market ?= {
						is_key_in_global_variable_map = { name = sul_iron_pressure target = this }
					}
				}
			}

			scope:target.market ?= {
				add = {
					desc = "wetland_iron_demand"
					value = "global_variable_map(sul_iron_pressure|this)"
				}
			}
		}

		# === FINISHED GOODS COMPOSITE DEMAND ===
		# Average demand pressure across mining finished goods (tools, weaponry, etc.)
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					sul_can_specialize_mining = yes
					NOT = {
						sul_has_mining_specialization = yes
					}
					market ?= {
						is_key_in_global_variable_map = {
							name = sul_mining_finished_composite
							target = this
						}
					}
				}
			}

			scope:target.market ?= {
				add = {
					desc = "finished_goods_demand"
					value = "global_variable_map(sul_mining_finished_composite|this)"
				}
			}
		}

		# === STAY PUT: Resist changing away from optimal spec/terrain matches ===
		# Branched by current specialization to reduce evaluations (was 16 flat checks, now max 4)
		if = {
			limit = {
				exists = scope:target
				scope:target = { sul_has_mining_specialization = yes }
			}

			# Raw material value inertia
			scope:target.raw_material ?= {
				add = {
					desc = "raw_material_value"
					value = "price_in_market(scope:target.market)"
					multiply = -1
				}
			}

			# Precious metals: never change
			if = {
				limit = { scope:target = { raw_material = goods:goods_gold } }
				add = { desc = "stay_put_gold" value = -100 }
			}
			else_if = {
				limit = { scope:target = { raw_material = goods:silver } }
				add = { desc = "stay_put_silver" value = -100 }
			}

			# Terrain bonuses
			if = {
				limit = { scope:target = { topography = mountains } }
				add = { desc = "stay_put_mining" value = -8 }
			}
			else_if = {
				limit = { scope:target = { topography = wetlands } }
				add = { desc = "stay_put_mining" value = -4 }
			}
		}
		else_if = {
			limit = {
				exists = scope:target
				scope:target = { sul_has_farming_specialization = yes }
			}

			scope:target.raw_material ?= {
				add = {
					desc = "raw_material_value"
					value = "price_in_market(scope:target.market)"
					multiply = -1
				}
			}
			if = {
				limit = { scope:target = { vegetation = farmland } }
				add = { desc = "stay_put_farming" value = -8 }
			}
			else_if = {
				limit = { scope:target = { topography = flatland vegetation = grasslands } }
				add = { desc = "stay_put_farming" value = -5 }
			}
			else_if = {
				limit = { scope:target = { vegetation = grasslands } }
				add = { desc = "stay_put_farming" value = -2 }
			}
		}
		else_if = {
			limit = {
				exists = scope:target
				scope:target = { sul_has_woodland_specialization = yes }
			}

			scope:target.raw_material ?= {
				add = {
					desc = "raw_material_value"
					value = "price_in_market(scope:target.market)"
					multiply = -1
				}
			}
			if = {
				limit = {
					scope:target = {
						OR = {
							vegetation = forest
							vegetation = woods
						}
					}
				}

				add = { desc = "stay_put_woodland" value = -8 }
			}
			else_if = {
				limit = {
					scope:target = {
						OR = {
							topography = hills
							topography = plateau
						}
					}
				}

				add = { desc = "stay_put_woodland" value = -2 }
			}
		}
		else_if = {
			limit = {
				exists = scope:target
				scope:target = { sul_has_gathering_specialization = yes }
			}

			scope:target.raw_material ?= {
				add = {
					desc = "raw_material_value"
					value = "price_in_market(scope:target.market)"
					multiply = -1
				}
			}
			if = {
				limit = { scope:target = { is_coastal = yes } }
				add = { desc = "stay_put_gathering" value = -8 }
			}
			else_if = {
				limit = { scope:target = { has_river = yes } }
				add = { desc = "stay_put_gathering" value = -7 }
			}
			else_if = {
				limit = {
					scope:target = {
						OR = {
							vegetation = forest
							vegetation = woods
						}
					}
				}

				add = { desc = "stay_put_gathering" value = -2 }
			}
		}
		else_if = {
			limit = {
				exists = scope:target
				scope:target = { sul_has_commercial_specialization = yes }
			}
			if = {
				limit = { scope:target = { is_market_center = yes } }
				add = { desc = "stay_put_commercial" value = -10 }
			}
			else_if = {
				limit = { scope:target = { modifier:harbor_suitability > 0.7 } }
				add = { desc = "stay_put_commercial" value = -5 }
			}
			else_if = {
				limit = {
					scope:target = {
						OR = {
							is_coastal = yes
							has_river = yes
						}
					}
				}

				add = { desc = "stay_put_commercial" value = -2 }
			}
		}

		# === BUILDING LEVEL PENALTY (num_buildings * -0.15) ===
		if = {
			limit = { exists = scope:target }

			add = {
				desc = "building_levels"
				value = scope:target.total_building_levels
				multiply = -0.15
			}
		}

		# === DIVERSITY BONUS (percentage-based) ===
		# Uses script value that calculates fractional deviation from expected
		# Positive if under-represented, negative if over-represented
		add = { desc = "diversity_pressure" value = scope:actor.sul_mining_diversity }
	}

	price = {
		if = {
			limit = {
				exists = scope:target
				scope:target = { location_rank = location_rank:rural_settlement }
			}

			value = price:sul_designate_specialization_price_rural
		}
		else_if = {
			limit = {
				exists = scope:target
				scope:target = { location_rank = location_rank:town }
			}

			value = price:sul_designate_specialization_price_town
		}
		else = { value = price:sul_designate_specialization_price_city }
	}
	select_trigger = {
		looking_for_a = location
		target_flag = target
		source = actor
		name = "designate_mining_select"

		visible = {
			owner = scope:actor
			OR = {
				location_rank = location_rank:rural_settlement
				location_rank = location_rank:town
				location_rank = location_rank:city
			}
		}

		enabled = {
			# Mining eligibility: resources OR terrain
			OR = {
				# Mining resources
				raw_material = goods:coal
				raw_material = goods:iron
				raw_material = goods:copper
				raw_material = goods:goods_gold
				raw_material = goods:silver
				raw_material = goods:stone
				raw_material = goods:tin
				raw_material = goods:lead
				raw_material = goods:alum
				raw_material = goods:gems
				raw_material = goods:marble
				raw_material = goods:mercury
				# Mining terrain
				topography = mountains
				topography = wetlands
				is_adjacent_to_lake = yes
			}
			NOT = {
				sul_has_mining_specialization = yes
			}
			NOT = {
				has_location_modifier = sul_recently_respecialized
			}

			controller = owner
		}
	}
	effect = {
		if = {
			limit = { exists = scope:target }
			# === UPDATE SPECIALIZATION COUNTS ===
			scope:actor = {
				# Initialize counts if they don't exist (existing saves)
				if = {
					limit = {
						NOT = {
							has_variable = sul_mining_count
						}
					}

					sul_initialize_specialization_counts = yes
				}

				# Decrement old type (if changing from another specialization)
				scope:target = { save_temporary_scope_as = sul_loc }
				sul_decrement_spec_count = yes
				# Increment new type
				change_variable = { name = sul_mining_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}

			scope:target = {
				sul_clear_all_specialization_modifiers = yes
				set_variable = { name = sul_spec_type value = @sul_mining_type }
				sul_update_designation = yes
				# Inlined instead of sul_apply_specialization_modifier so tooltip preview
				# shows the correct new modifier (preview evaluates limits before effects run)
				if = {
					limit = { location_rank = location_rank:rural_settlement }
					add_location_modifier = { modifier = sul_mining_rural_specialization }
				}
				else_if = {
					limit = { location_rank = location_rank:town }
					add_location_modifier = { modifier = sul_mining_town_specialization }
				}
				else = { add_location_modifier = { modifier = sul_mining_city_specialization } }
				sul_destroy_non_mining_buildings = yes
				add_location_modifier = { modifier = sul_recently_respecialized years = 5 }
			}
		}
	}
}

	potential = {
	}

		monthly_balance >= 0
	}

	ai_will_do = {
		# === AFFORDABILITY + INERTIA (branched by rank) ===
		add = { desc = "inertia" value = 0 }
		if = {
			limit = {
				exists = scope:target
				scope:target = { location_rank = location_rank:town }
			}

			add = { desc = "town_inertia" value = -5 }
			if = {

				limit = {

						OR = {
							gold < 1000
							prestige < 10
						}
					}
				}

				add = { desc = "cannot_afford" value = -1000 }
			}
		}

		# === TERRAIN BONUSES ===
		# Farmland vegetation: big bonus (+8)
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					vegetation = farmland
					sul_can_specialize_farming = yes
					NOT = {
						sul_has_farming_specialization = yes
					}
				}
			}

			add = { desc = "farmland_terrain" value = 4 }
		}

		# Flatland + Grasslands: moderate bonus (+5)
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					topography = flatland
					vegetation = grasslands
					sul_can_specialize_farming = yes
					NOT = {
						sul_has_farming_specialization = yes
					}
				}
			}

			add = { desc = "plains_terrain" value = 2 }
		}

		# Grasslands only (not flatland): small bonus (+2)
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					vegetation = grasslands
					NOT = {
						topography = flatland
					}
					NOT = {
						vegetation = farmland
					}

					sul_can_specialize_farming = yes
					NOT = {
						sul_has_farming_specialization = yes
					}
				}
			}

			add = { desc = "grasslands_terrain" value = 1 }
		}

		# River access: small bonus (+2)
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					has_river = yes
					sul_can_specialize_farming = yes
					NOT = {
						sul_has_farming_specialization = yes
					}
				}
			}

			add = { desc = "river_irrigation" value = 1 }
		}

		# === GOVERNMENT PREFERENCES ===
		# Tribes prefer farming: small bonus (+2)
		if = {
			limit = {
				scope:actor = {
					OR = {
						government_type = government_type:tribe
						government_type = government_type:steppe_horde
					}
				}

				exists = scope:target
				scope:target = {
					sul_can_specialize_farming = yes
					NOT = {
						sul_has_farming_specialization = yes
					}
				}
			}

			add = { desc = "tribal_farming_preference" value = 2 }
		}

		# === RAW MATERIAL DEMAND PRESSURE ===
		# Location's actual raw material demand/supply ratio, ±5 scale
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					sul_can_specialize_farming = yes
					NOT = {
						sul_has_farming_specialization = yes
					}
				}
			}

			scope:target.market ?= {
				add = {
					desc = "raw_material_demand"
					value = "goods_demand_in_market(scope:target.raw_material)"

					divide = {
						value = "goods_supply_in_market(scope:target.raw_material)"
						min = 0.1
					}

					subtract = 1
					max = 1
					min = -1
					multiply = 5
				}
			}
		}

		# === FINISHED GOODS COMPOSITE DEMAND ===
		# Average demand pressure across farming finished goods (beer, liquor, leather, wine)
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					sul_can_specialize_farming = yes
					NOT = {
						sul_has_farming_specialization = yes
					}
					market ?= {
						is_key_in_global_variable_map = {
							name = sul_farming_finished_composite
							target = this
						}
					}
				}
			}

			scope:target.market ?= {
				add = {
					desc = "finished_goods_demand"
					value = "global_variable_map(sul_farming_finished_composite|this)"
				}
			}
		}

		# === FOOD PRICE BONUS (strong for farming - main food producer) ===
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					sul_can_specialize_farming = yes
					NOT = {
						sul_has_farming_specialization = yes
					}
				}
			}

			scope:target = {
				market ?= {
					add = { desc = "food_price_bonus" value = sul_farming_food_price_bonus }
				}
			}
		}

		# === STAY PUT: Resist changing away from optimal spec/terrain matches ===

		if = {
			limit = {
				exists = scope:target
				scope:target = { sul_has_mining_specialization = yes }
			}

			scope:target.raw_material ?= {
				add = {
					desc = "raw_material_value"
					value = "price_in_market(scope:target.market)"
					multiply = -1
				}
			}
			if = {
				limit = { scope:target = { raw_material = goods:goods_gold } }
				add = { desc = "stay_put_gold" value = -100 }
			}
			else_if = {
				limit = { scope:target = { raw_material = goods:silver } }
				add = { desc = "stay_put_silver" value = -100 }
			}
			if = {

			else_if = {
				limit = { scope:target = { topography = flatland vegetation = grasslands } }
				add = { desc = "stay_put_farming" value = -5 }
			}

				limit = { scope:target = { vegetation = grasslands } }

		else_if = {

				scope:target = { sul_has_woodland_specialization = yes }
			}
