# Specialized Urban Locations - Specialization Designation Actions
# These actions allow players to designate a specialization for towns/cities
# Costs gold + prestige, destroys non-matching buildings, applies modifier
# scope:actor = country, scope:target = location

designate_mining_specialization = {
	type = owncountry

	potential = {
	}

	allow = {
	}

	ai_tick = monthly
	ai_tick_frequency = 12

	ai_prerequisite = {
		current_year >= 1338
		num_loans = 0
		gold >= 50
		monthly_balance >= 0
	}

	ai_will_do = {
		# === AFFORDABILITY + INERTIA (branched by rank) ===
		add = { desc = "inertia" value = -6 }
		if = {
			limit = { exists = scope:target scope:target = { location_rank = location_rank:town } }
			add = { desc = "town_inertia" value = -5 }
			if = {
				limit = { scope:actor = { OR = { gold < 300 prestige < 5 } } }
				add = { desc = "cannot_afford" value = -1000 }
			}
		}
		else_if = {
			limit = { exists = scope:target scope:target = { location_rank = location_rank:city } }
			add = { desc = "city_inertia" value = -10 }
			if = {
				limit = { scope:actor = { OR = { gold < 1000 prestige < 10 } } }
				add = { desc = "cannot_afford" value = -1000 }
			}
		}

		# === TERRAIN BONUSES (else_if chain - only one terrain can match) ===
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					can_specialize_mining = yes
					NOT = { has_mining_specialization = yes }
				}
			}
			# Wetlands (bog iron): best for mining - iron is more valuable than stone
			if = {
				limit = { scope:target = { topography = wetlands } }
				add = { desc = "wetlands_bog_iron" value = 3 }
			}
			# Mountains: good for mining
			else_if = {
				limit = { scope:target = { topography = mountains } }
				add = { desc = "mountains_terrain" value = 2 }
			}
			# Hills or plateau: marginal mining terrain
			else_if = {
				limit = { scope:target = { OR = { topography = hills topography = plateau } } }
				add = { desc = "hills_plateau_terrain" value = 1 }
			}
			# Other terrain: no terrain bonus (resources still get value bonus)
		}

		# === RESOURCE VALUE BONUS ===
		# Add bonus equal to the location's raw material market price
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					can_specialize_mining = yes
					NOT = { has_mining_specialization = yes }
				}
			}
			scope:target.raw_material = {
				add = "price_in_market(scope:actor.capital.market)"
			}
		}

		# Gold: massive bonus - should always be mining
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					raw_material = goods:goods_gold
					NOT = { has_mining_specialization = yes }
				}
			}
			add = {
				desc = "gold_resource"
				value = 100
			}
		}

		# Silver: massive bonus - should always be mining
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					raw_material = goods:silver
					NOT = { has_mining_specialization = yes }
				}
			}
			add = {
				desc = "silver_resource"
				value = 100
			}
		}

		# === MARKET PRICE BONUSES (uses cached scaled values from market scope) ===
		# Formula: (price / base_price - 1) × 8, capped at -8 to +16
		# Positive = shortage (high prices encourage production)
		# Negative = surplus (low prices discourage production)

		# Raw mining goods (iron) price bonus - all location types
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					can_specialize_mining = yes
					NOT = { has_mining_specialization = yes }
				}
			}
			scope:target = {
				market ?= {
					add = {
						desc = "iron_price_bonus"
						value = location.var:sul_mining_raw_price_bonus
					}
				}
			}
		}

		# Finished mining goods (tools) price bonus - towns/cities only
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					can_specialize_mining = yes
					NOT = { has_mining_specialization = yes }
					NOT = { location_rank = location_rank:rural_settlement }
				}
			}
			scope:target = {
				market ?= {
					add = {
						desc = "tools_price_bonus"
						value = location.var:sul_mining_finished_price_bonus
					}
				}
			}
		}

		# === MARKET SUPPLY/DEMAND RATIO ===
		# Iron shortage/surplus: ±5 based on demand vs supply ratio
		# Uses cached pressure value, falls back to calculation if cache missing
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					can_specialize_mining = yes
					NOT = { has_mining_specialization = yes }
					market ?= { location = { has_variable = sul_iron_pressure } }
				}
			}
			scope:target = {
				market ?= {
					add = {
						desc = "iron_market_pressure"
						value = location.var:sul_iron_pressure
					}
				}
			}
		}
		# Fallback for uncached markets
		else_if = {
			limit = {
				exists = scope:target
				scope:target = {
					can_specialize_mining = yes
					NOT = { has_mining_specialization = yes }
				}
			}
			scope:target = {
				market ?= {
					add = {
						desc = "iron_market_pressure"
						value = sul_iron_market_pressure
					}
				}
			}
		}

		# === STAY PUT: Resist changing away from optimal spec/terrain matches ===
		# Branched by current specialization to reduce evaluations (was 16 flat checks, now max 4)
		if = {
			limit = { exists = scope:target scope:target = { has_mining_specialization = yes } }
			# Resource value penalty
			scope:target.raw_material = { subtract = "price_in_market(scope:actor.capital.market)" }
			# Precious metals: never change
			if = {
				limit = { scope:target = { raw_material = goods:goods_gold } }
				add = { desc = "stay_put_gold" value = -100 }
			}
			else_if = {
				limit = { scope:target = { raw_material = goods:silver } }
				add = { desc = "stay_put_silver" value = -100 }
			}
			# Terrain bonuses
			if = {
				limit = { scope:target = { topography = mountains } }
				add = { desc = "stay_put_mining" value = -8 }
			}
			else_if = {
				limit = { scope:target = { topography = wetlands } }
				add = { desc = "stay_put_mining" value = -4 }
			}
			else_if = {
				limit = { scope:target = { OR = { topography = hills topography = plateau } } }
				add = { desc = "stay_put_mining" value = -2 }
			}
		}
		else_if = {
			limit = { exists = scope:target scope:target = { has_farming_specialization = yes } }
			scope:target.raw_material = { subtract = "price_in_market(scope:actor.capital.market)" }
			if = {
				limit = { scope:target = { vegetation = farmland } }
				add = { desc = "stay_put_farming" value = -8 }
			}
			else_if = {
				limit = { scope:target = { topography = flatland vegetation = grasslands } }
				add = { desc = "stay_put_farming" value = -5 }
			}
			else_if = {
				limit = { scope:target = { vegetation = grasslands } }
				add = { desc = "stay_put_farming" value = -2 }
			}
		}
		else_if = {
			limit = { exists = scope:target scope:target = { has_woodland_specialization = yes } }
			scope:target.raw_material = { subtract = "price_in_market(scope:actor.capital.market)" }
			if = {
				limit = { scope:target = { OR = { vegetation = forest vegetation = woods } } }
				add = { desc = "stay_put_woodland" value = -8 }
			}
			else_if = {
				limit = { scope:target = { OR = { topography = hills topography = plateau } } }
				add = { desc = "stay_put_woodland" value = -2 }
			}
		}
		else_if = {
			limit = { exists = scope:target scope:target = { has_gathering_specialization = yes } }
			scope:target.raw_material = { subtract = "price_in_market(scope:actor.capital.market)" }
			if = {
				limit = { scope:target = { is_coastal = yes } }
				add = { desc = "stay_put_gathering" value = -8 }
			}
			else_if = {
				limit = { scope:target = { has_river = yes } }
				add = { desc = "stay_put_gathering" value = -7 }
			}
			else_if = {
				limit = { scope:target = { OR = { vegetation = forest vegetation = woods } } }
				add = { desc = "stay_put_gathering" value = -2 }
			}
		}
		else_if = {
			limit = { exists = scope:target scope:target = { has_commercial_specialization = yes } }
			scope:target.raw_material = { subtract = "price_in_market(scope:actor.capital.market)" }
			if = {
				limit = { scope:target = { is_market_center = yes } }
				add = { desc = "stay_put_commercial" value = -10 }
			}
			else_if = {
				limit = { scope:target = { modifier:harbor_suitability > 0.7 } }
				add = { desc = "stay_put_commercial" value = -5 }
			}
			else_if = {
				limit = { scope:target = { OR = { is_coastal = yes has_river = yes } } }
				add = { desc = "stay_put_commercial" value = -2 }
			}
		}

		# === BUILDING LEVEL PENALTY (num_buildings * -0.15) ===
		if = {
			limit = { exists = scope:target }
			add = {
				desc = "building_levels"
				value = scope:target.total_building_levels
				multiply = -0.15
			}
		}

		# === DIVERSITY BONUS (percentage-based) ===
		# Uses script value that calculates fractional deviation from expected
		# Positive if under-represented, negative if over-represented
		add = {
			desc = "diversity_pressure"
			value = scope:actor.sul_mining_diversity
		}
	}

	price = {
		if = {
			limit = { exists = scope:target scope:target = { location_rank = location_rank:rural_settlement } }
			value = price:designate_specialization_price_rural
		}
		else_if = {
			limit = { exists = scope:target scope:target = { location_rank = location_rank:town } }
			value = price:designate_specialization_price_town
		}
		else = {
			value = price:designate_specialization_price_city
		}
	}

	select_trigger = {
		looking_for_a = location
		target_flag = target
		source = actor
		name = "designate_mining_select"

		visible = {
			owner = scope:actor
			is_rural_or_higher = yes
		}

		enabled = {
			can_specialize_mining = yes
			NOT = { has_mining_specialization = yes }
			NOT = { has_location_modifier = sul_specialization_cooldown }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }

			# === AI TRACKING (France/Bohemia only) ===
			if = {
				limit = {
					scope:actor = {
						is_ai = yes
						OR = {
							tag = FRA
							tag = BOH
						}
					}
				}

				# Count mining specializations
				scope:actor = {
					if = {
						limit = { NOT = { has_variable = ai_mining_changes } }
						set_variable = { name = ai_mining_changes value = 1 }
					}
					else = {
						change_variable = { name = ai_mining_changes add = 1 }
					}
				}

				# Count by location rank
				if = {
					limit = { scope:target = { location_rank = location_rank:rural_settlement } }
					scope:actor = {
						if = {
							limit = { NOT = { has_variable = ai_changes_rural } }
							set_variable = { name = ai_changes_rural value = 1 }
						}
						else = {
							change_variable = { name = ai_changes_rural add = 1 }
						}
					}
				}
				else_if = {
					limit = { scope:target = { location_rank = location_rank:town } }
					scope:actor = {
						if = {
							limit = { NOT = { has_variable = ai_changes_town } }
							set_variable = { name = ai_changes_town value = 1 }
						}
						else = {
							change_variable = { name = ai_changes_town add = 1 }
						}
					}
				}
				else = {
					scope:actor = {
						if = {
							limit = { NOT = { has_variable = ai_changes_city } }
							set_variable = { name = ai_changes_city value = 1 }
						}
						else = {
							change_variable = { name = ai_changes_city add = 1 }
						}
					}
				}

				# Track repeat changes on location
				scope:target = {
					if = {
						limit = { NOT = { has_variable = times_specialized } }
						set_variable = { name = times_specialized value = 1 }
					}
					else = {
						change_variable = { name = times_specialized add = 1 }
					}
				}

				# Count repeat changes (location changed 2+ times)
				if = {
					limit = { scope:target = { has_variable = times_specialized var:times_specialized >= 2 } }
					scope:actor = {
						if = {
							limit = { NOT = { has_variable = ai_repeat_changes } }
							set_variable = { name = ai_repeat_changes value = 1 }
						}
						else = {
							change_variable = { name = ai_repeat_changes add = 1 }
						}
					}
				}
			}

			# === UPDATE SPECIALIZATION COUNTS ===
			scope:actor = {
				# Initialize counts if they don't exist (existing saves)
				if = {
					limit = { NOT = { has_variable = sul_mining_count } }
					sul_initialize_specialization_counts = yes
				}
				# Decrement old type (if changing from another specialization)
				if = {
					limit = { scope:target = { has_mining_specialization = yes } }
					change_variable = { name = sul_mining_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
				else_if = {
					limit = { scope:target = { has_farming_specialization = yes } }
					change_variable = { name = sul_farming_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 2 }
				}
				else_if = {
					limit = { scope:target = { has_woodland_specialization = yes } }
					change_variable = { name = sul_woodland_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
				else_if = {
					limit = { scope:target = { has_gathering_specialization = yes } }
					change_variable = { name = sul_gathering_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
				else_if = {
					limit = { scope:target = { has_commercial_specialization = yes } }
					change_variable = { name = sul_commercial_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
				# Increment new type
				change_variable = { name = sul_mining_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}

			scope:target = {
				clear_all_specialization_modifiers = yes
				if = {
					limit = { location_rank = location_rank:rural_settlement }
					set_variable = { name = specialization_designation value = sul_mining_rural_value }
					add_location_modifier = { modifier = mining_rural_specialization }
				}
				else_if = {
					limit = { location_rank = location_rank:town }
					set_variable = { name = specialization_designation value = sul_mining_town_value }
					add_location_modifier = { modifier = mining_town_specialization }
				}
				else = {
					set_variable = { name = specialization_designation value = sul_mining_city_value }
					add_location_modifier = { modifier = mining_city_specialization }
				}
				destroy_non_mining_buildings = yes
				add_location_modifier = {
					modifier = sul_specialization_cooldown
					years = 5
				}
			}
		}
	}
}

					NOT = { has_farming_specialization = yes }
				}
			}
			add = {
				desc = "farmland_terrain"

			}
		}

		# Flatland + Grasslands: moderate bonus (+5)
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					topography = flatland
					vegetation = grasslands
					can_specialize_farming = yes
					NOT = { has_farming_specialization = yes }
				}
			}
			add = {
				desc = "plains_terrain"
				value = 2
			}
		}

		# Grasslands only (not flatland): small bonus (+2)
		if = {
			limit = {
				exists = scope:target
				scope:target = {
					vegetation = grasslands
					NOT = { topography = flatland }
					NOT = { vegetation = farmland }
					can_specialize_farming = yes
					NOT = { has_farming_specialization = yes }
				}

					NOT = { has_farming_specialization = yes }
				}
			}
			add = {
				desc = "river_irrigation"
				value = 1
			}
		}

		# === GOVERNMENT PREFERENCES ===
		# Tribes prefer farming: small bonus (+2)
		if = {
			limit = {
				scope:actor = {
					OR = {
						government_type = government_type:tribe
						government_type = government_type:steppe_horde
					}
				}
				exists = scope:target
				scope:target = {
					can_specialize_farming = yes
					NOT = { has_farming_specialization = yes }
				}
			}

					add = {
						desc = "wool_market_pressure"

		# Wheat (food) shortage/surplus: ±5 based on demand vs supply ratio

		# === STAY PUT: Resist changing away from optimal spec/terrain matches ===

			scope:target.raw_material = { subtract = "price_in_market(scope:actor.capital.market)" }
			if = {
				limit = { scope:target = { vegetation = farmland } }
				add = { desc = "stay_put_farming" value = -8 }
			}
			else_if = {
				limit = { scope:target = { topography = flatland vegetation = grasslands } }
				add = { desc = "stay_put_farming" value = -5 }
			}
			else_if = {
				limit = { scope:target = { vegetation = grasslands } }
				add = { desc = "stay_put_farming" value = -2 }
			}
		}
		else_if = {
			limit = { exists = scope:target scope:target = { has_woodland_specialization = yes } }
			scope:target.raw_material = { subtract = "price_in_market(scope:actor.capital.market)" }
			if = {
				limit = { scope:target = { OR = { vegetation = forest vegetation = woods } } }
				add = { desc = "stay_put_woodland" value = -8 }
			}
			else_if = {
				limit = { scope:target = { OR = { topography = hills topography = plateau } } }
				add = { desc = "stay_put_woodland" value = -2 }
			}
		}
		else_if = {
			limit = { exists = scope:target scope:target = { has_gathering_specialization = yes } }
			scope:target.raw_material = { subtract = "price_in_market(scope:actor.capital.market)" }
			if = {
				limit = { scope:target = { is_coastal = yes } }
				add = { desc = "stay_put_gathering" value = -8 }
			}
			else_if = {
				limit = { scope:target = { has_river = yes } }
				add = { desc = "stay_put_gathering" value = -7 }
			}
			else_if = {
				limit = { scope:target = { OR = { vegetation = forest vegetation = woods } } }
				add = { desc = "stay_put_gathering" value = -2 }
			}
		}
		else_if = {
			limit = { exists = scope:target scope:target = { has_commercial_specialization = yes } }
			if = {
				limit = { scope:target = { is_market_center = yes } }
				add = { desc = "stay_put_commercial" value = -10 }
			}
			else_if = {
				limit = { scope:target = { modifier:harbor_suitability > 0.7 } }
				add = { desc = "stay_put_commercial" value = -5 }
			}
			else_if = {
				limit = { scope:target = { OR = { is_coastal = yes has_river = yes } } }
				add = { desc = "stay_put_commercial" value = -2 }
			}
		}

		# === BUILDING LEVEL PENALTY (num_buildings * -0.15) ===
		if = {
			limit = { exists = scope:target }
			add = {
				desc = "building_levels"
				value = scope:target.total_building_levels
				multiply = -0.15
			}
		}

		# === DIVERSITY BONUS (percentage-based) ===
		# Uses script value that calculates fractional deviation from expected
		# Positive if under-represented, negative if over-represented
		add = {
			desc = "diversity_pressure"
			value = scope:actor.sul_commercial_diversity
		}
	}

	price = {
		if = {
			limit = { exists = scope:target scope:target = { location_rank = location_rank:rural_settlement } }
			value = price:designate_specialization_price_rural
		}
		else_if = {
			limit = { exists = scope:target scope:target = { location_rank = location_rank:town } }
			value = price:designate_specialization_price_town
		}
		else = {
			value = price:designate_specialization_price_city
		}
	}

	select_trigger = {
		looking_for_a = location
		target_flag = target
		source = actor
		name = "designate_commercial_select"

		visible = {
			owner = scope:actor
			is_rural_or_higher = yes
		}

		# Commercial has no resource requirement - but cannot already be commercial
		enabled = {
			NOT = { has_commercial_specialization = yes }
			NOT = { has_location_modifier = sul_specialization_cooldown }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }

			# === AI TRACKING (France/Bohemia only) ===
			if = {
				limit = {
					scope:actor = {
						is_ai = yes
						OR = {
							tag = FRA
							tag = BOH
						}
					}
				}

				# Count commercial specializations
				scope:actor = {
					if = {
						limit = { NOT = { has_variable = ai_commercial_changes } }
						set_variable = { name = ai_commercial_changes value = 1 }
					}
					else = {
						change_variable = { name = ai_commercial_changes add = 1 }
					}
				}

				# Count by location rank
				if = {
					limit = { scope:target = { location_rank = location_rank:rural_settlement } }
					scope:actor = {
						if = {
							limit = { NOT = { has_variable = ai_changes_rural } }
							set_variable = { name = ai_changes_rural value = 1 }
						}
						else = {
							change_variable = { name = ai_changes_rural add = 1 }
						}
					}
				}
				else_if = {
					limit = { scope:target = { location_rank = location_rank:town } }
					scope:actor = {
						if = {
							limit = { NOT = { has_variable = ai_changes_town } }
							set_variable = { name = ai_changes_town value = 1 }
						}
						else = {
							change_variable = { name = ai_changes_town add = 1 }
						}
					}
				}
				else = {
					scope:actor = {
						if = {
							limit = { NOT = { has_variable = ai_changes_city } }
							set_variable = { name = ai_changes_city value = 1 }
						}
						else = {
							change_variable = { name = ai_changes_city add = 1 }
						}
					}
				}

				# Track repeat changes on location
				scope:target = {
					if = {
						limit = { NOT = { has_variable = times_specialized } }
						set_variable = { name = times_specialized value = 1 }
					}
					else = {
						change_variable = { name = times_specialized add = 1 }
					}
				}

				# Count repeat changes (location changed 2+ times)
				if = {
					limit = { scope:target = { has_variable = times_specialized var:times_specialized >= 2 } }
					scope:actor = {
						if = {
							limit = { NOT = { has_variable = ai_repeat_changes } }
							set_variable = { name = ai_repeat_changes value = 1 }
						}
						else = {
							change_variable = { name = ai_repeat_changes add = 1 }
						}
					}
				}
			}

			# === UPDATE SPECIALIZATION COUNTS ===
			scope:actor = {
				# Initialize counts if they don't exist (existing saves)
				if = {
					limit = { NOT = { has_variable = sul_mining_count } }
					sul_initialize_specialization_counts = yes
				}
				# Decrement old type (if changing from another specialization)
				if = {
					limit = { scope:target = { has_mining_specialization = yes } }
					change_variable = { name = sul_mining_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
				else_if = {
					limit = { scope:target = { has_farming_specialization = yes } }
					change_variable = { name = sul_farming_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 2 }
				}
				else_if = {
					limit = { scope:target = { has_woodland_specialization = yes } }
					change_variable = { name = sul_woodland_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
				else_if = {
					limit = { scope:target = { has_gathering_specialization = yes } }
					change_variable = { name = sul_gathering_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
				else_if = {
					limit = { scope:target = { has_commercial_specialization = yes } }
					change_variable = { name = sul_commercial_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
				# Increment new type
				change_variable = { name = sul_commercial_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}

			scope:target = {
				clear_all_specialization_modifiers = yes
				if = {
					limit = { location_rank = location_rank:rural_settlement }
					set_variable = { name = specialization_designation value = sul_commercial_rural_value }
					add_location_modifier = { modifier = commercial_rural_specialization }
				}
				else_if = {
					limit = { location_rank = location_rank:town }
					set_variable = { name = specialization_designation value = sul_commercial_town_value }
					add_location_modifier = { modifier = commercial_town_specialization }
				}
				else = {
					set_variable = { name = specialization_designation value = sul_commercial_city_value }
					add_location_modifier = { modifier = commercial_city_specialization }
				}
				destroy_non_commercial_buildings = yes
				add_location_modifier = {
					modifier = sul_specialization_cooldown
					years = 5
				}
			}
		}
	}
}
