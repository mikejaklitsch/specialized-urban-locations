# =============================================================================
# SUL GOODS DEMAND OVERRIDES
# =============================================================================
# Increases base demand (demand_add) for key goods.
# Keeps vanilla demand_multiply untouched for patch safety.
#
# Design: peasants/laborers should demand nearly as much beer/liquor as nobles.
# Wine stays an upper-class drink. Pottery and medicaments get a general boost.
# =============================================================================

# -----------------------------------------------------------------------------
# Alcohol — peasants/laborers drink beer and liquor, wine is for the elite
# -----------------------------------------------------------------------------

# Beer: noble = 0.016 (vanilla). Peasant target ~0.005 (~3:1 ratio)

INJECT:beer = {
	demand_add = {
		peasants = 0.005
		laborers = 0.004
		soldiers = 0.005
	}
}

# Liquor: noble = 0.008 (vanilla). Peasant target ~0.0025 (~3:1 ratio)

INJECT:liquor = {
	demand_add = {
		peasants = 0.0025
		laborers = 0.002
		soldiers = 0.0025
	}
}

# Wine: 2x base demand across the board

INJECT:wine = {
	demand_add = {
		all = 0.0008
	}
}

# -----------------------------------------------------------------------------
# Produced goods — general demand increase
# -----------------------------------------------------------------------------

# Pottery: 2x base (everyone needs pots)

INJECT:pottery = {
	demand_add = {
		all = 0.004
	}
}

# Medicaments: 2x base

INJECT:medicaments = {
	demand_add = {
		all = 0.0008
	}
}
