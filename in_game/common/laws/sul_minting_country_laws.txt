# TGS Monetization — country-specific law replacements
# Must be kept in sync with vanilla.

REPLACE:kor_currency = {
	law_category = socioeconomic

	potential = {
		has_unlocked_law_trigger = { type = kor_currency }
	}

	kor_barter_economy = {
		country_modifier = {
			scale = {
				add = {
					desc = capital_economy_vs_traditional_economy
					value = societal_value:capital_economy_vs_traditional_economy
					multiply = 0.01
					min = 0
				}
			}
			trade_land_movement_cost_modifier = -0.2
		}
		country_modifier = { monthly_towards_traditional_economy = societal_value_minor_monthly_move }

		years = 2
	}

	kor_metal_coinage = {
		country_modifier = { minting_inflation_threshold = 0.01 sul_minting_monetization = 0.05 }
		years = 2
	}

	kor_jeohwa = {
		allow = {
			capital.market = { "stockpile_in_market(goods:paper)" > 20 }
		}

		country_modifier = { monthly_towards_capital_economy = societal_value_monthly_move }
		country_modifier = {
			scale = {
				add = {
					desc = capital_economy_vs_traditional_economy
					value = societal_value:capital_economy_vs_traditional_economy
					multiply = -1
					min = 0
				}
			}
			bank_interest = -0.0001
		}

		on_activate = {
			capital.market = { add_goods_supply = { amount = -20 goods = goods:paper } }
		}

		years = 2
	}

	kor_chinese_currency = {
		potential = {
			country_exists = c:CHI
		}

		country_modifier = {
			scale = {
				add = {
					desc = "kor_scale_chinese_currency"
					value = "c:CHI.opinion(root)"
					multiply = 0.01
				}
			}
			trade_efficiency = small_trade_efficiency_bonus
		}

		years = 2
	}
}
