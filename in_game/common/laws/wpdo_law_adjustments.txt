# HRE Perpetual Diet: vanilla +10% dev modifier on diet location

TRY_INJECT:perpetual_diet_law = {
	perpetual_diet_policy = {
		allow = {
			OR = {
				international_organization_has_policy = policy:hre_bi_camerial_imperial_diet_policy
				international_organization_has_policy = policy:hre_tri_camerial_imperial_diet_policy
			}
		}

		on_activate = {
			hre_on_reform_passed_effect = {
				amount = imperial_authority_standard_law
			}

			international_organization_unlock_parliament_issue_effect = {
				type = hre_establish_perpetual_diet
			}

			hre_lock_law = { law = perpetual_diet_law }
		}

		location_modifier = {
			# applies to diet location
			potential_trigger = {
				custom_tooltip = {
					text = perpetual_diet
					hre_is_perpetual_diet_location = yes

					owner = {
						is_member_of_international_organization = international_organization:hre
					}

					controller = {
						OR = {
							is_member_of_international_organization = international_organization:hre
							NOT = {
								is_at_war_with = international_organization:hre.leader_country
							}
						}
					}
				}
			}

			# vanilla local_monthly_development_modifier = 0.1 removed
			local_monthly_development = 0.01             # flat: imperial diet
			local_monthly_prosperity = 0.01
		}

		on_deactivate = {
			international_organization_lock_parliament_issue_effect = {
				type = hre_establish_perpetual_diet
			}

			every_international_organization_owned_location = {
				remove_variable = diet_location
			}
		}

		# How much the Emperor wants to propose a policy
		wants_propose_policy = {
			add = { desc = "POLICY_BIAS_LEADER_WANTS" value = 33 }
			add = hre_default_emperor_policy_desire
		}

		# How much the princes want to vote for / against the policy
		wants_this_policy_bias = {
			add = { desc = "POLICY_BIAS_PRINCE_WANTS" value = 25 }
			add = hre_default_policy_opinion
		}
	}
}

# Middle Kingdom: rice-based tax gives celestial governors +0.2 and tributaries +0.1 food

TRY_INJECT:tax_base_system = {
	rice_based_tax = {
		years = 2

		on_pay_price = {
			change_celestial_authority = {
				value = celestial_authority_very_strong_penalty
			}

			international_organization:middle_kingdom = {
				leader_country ?= {
					add_government_power = government_power_ultimate_penalty

					add_gold = {
						value = this.monthly_income_trade_and_tax

						multiply = {
							value = -6

							add = {
								value = -1

								multiply = {
									value = international_organization:middle_kingdom.total_members
									divide = 10
								}
							}
						}

						min = -100000
					}
				}
			}
		}

		country_modifier = {
			# Emperor benefits
			potential_trigger = { is_leader_of_international_organization = international_organization:middle_kingdom }
			trade_efficiency = small_trade_efficiency_bonus
		}
		country_modifier = {
			# Celestial Governor benefits
			potential_trigger = {
				international_organization:middle_kingdom = {
					country_has_special_status = {
						type = special_status:celestial_governor
						country = root
					}
				}
			}

			# vanilla global_monthly_food_modifier = 0.2 removed
			global_rice_output_modifier = 0.10
			global_wheat_output_modifier = 0.05
			global_fish_output_modifier = 0.05
			tax_income_efficiency = small_tax_income_efficiency_bonus
		}
		country_modifier = {
			# Tributary benefits
			potential_trigger = {
				international_organization:middle_kingdom = {
					NOT = {
						country_has_special_status = {
							type = special_status:celestial_governor
							country = root
						}
					}
				}
				NOT = {
					is_leader_of_international_organization = international_organization:middle_kingdom
				}
			}

			# vanilla global_monthly_food_modifier = 0.1 removed
			global_rice_output_modifier = 0.05
			global_wheat_output_modifier = 0.025
			global_fish_output_modifier = 0.025
			tax_income_efficiency = tiny_tax_income_efficiency_bonus
		}

		wants_this_policy_bias = {
			if = {
				limit = { scope:recipient = { international_organization_has_leader = yes } }

				add = {
					desc = "POLICY_BIAS_OPINION_OF_EMPEROR"
					value = "opinion(scope:recipient.leader_country)"
				}
				add = {
					desc = "IO_LEADER_DIP_REP"
					value = scope:recipient.leader_country.modifier:diplomatic_reputation
				}
			}
			else = { add = { desc = "VOTE_BASE_VALUE" value = 20 } }
		}
	}
}

# Jurchen Confederation: shared crops policy gives +0.1 food

TRY_INJECT:economic_framework = {
	shared_crops_policy = {
		wants_this_policy_bias = {
			add = { desc = "POLICY_BIAS_GOOD_IDEA" value = 20 }
		}

		on_activate = {
			leader_country = {
				set_variable = {
					name = economic_framework_cooldown_variable
					value = yes
					years = 5
				}

				add_tribal_cohesion = tribal_cohesion_weak_penalty
				if = {
					limit = { ruler_or_regent ?= { has_variable = difficult_sinicization_variable } }
					add_tribal_cohesion = tribal_cohesion_mild_penalty
					add_stability = stability_mild_penalty
				}
			}
		}

		years = 5

		country_modifier = {
			monthly_towards_traditional_economy = societal_value_monthly_move
			# vanilla global_monthly_food_modifier = 0.1 removed
			global_rice_output_modifier = 0.05
			global_wheat_output_modifier = 0.025
			global_livestock_output_modifier = 0.025
			monthly_towards_unsinicized = societal_value_monthly_move
			monthly_tribal_cohesion = 0.03
		}
	}
}
