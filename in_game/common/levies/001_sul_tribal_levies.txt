# Tribal levies — override vanilla to use levy_tribal_size instead of
# levy_generic_infantry_size. Tribal societies mobilize a much larger
# share of population.

REPLACE:levy_tribal_cavalry = {
	size = levy_tribal_size

	allowed_pop_type = tribesmen

	allow = {
		location = {
			market = { OR = { is_produced_in_market = goods:horses is_traded_in_market = goods:horses } }
		}
	}

	country_allow = {
		has_estate_privilege = estate_privilege:tribes_tribal_levies
		NOT = { government_type = government_type:steppe_horde }
	}

	unit = a_tribal_cavalry
}

REPLACE:levy_moa_hunters = {
	size = levy_tribal_size

	allowed_pop_type = peasants
	allowed_pop_type = tribesmen

	allow = {
		unit_moa_hunter_pop_trigger = yes
	}

	unit = a_moa_hunters
}

REPLACE:levy_tribesmen = {
	size = levy_tribal_size

	allowed_pop_type = peasants
	allowed_pop_type = tribesmen

	unit = a_tribesmen
}
