# =============================================================================
# LOCATION RANK — Food Modifier Removal
# =============================================================================
# Vanilla rural_settlement gives +10% local_monthly_food_modifier.
# This is an abstract food boost that bypasses the provisions system.

INJECT:rural_settlement = {
	rank_modifier = {
		local_monthly_food_modifier = -0.1       # nullify vanilla +0.10
		local_max_rgo_size_modifier = -1.0       # nullify vanilla +1.0 (replaced by flat)
		sul_rgo_size = 4                         # flat: rural settlement bonus
	}
}

# Development equilibrium: towns +10 dev (0.01 flat), cities +25 dev (0.025 flat)
INJECT:town = {
	rank_modifier = {
		local_monthly_development = 0.01
	}
}

INJECT:city = {
	rank_modifier = {
		local_monthly_development = 0.025
	}
}
