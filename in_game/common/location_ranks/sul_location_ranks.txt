# Specialized Urban Locations - Location Rank Overrides
# Population thresholds: Town 10k, City 50k (vanilla: 5k, 30k)
# Zero base population capacity - comes from buildings/food instead
# Negative base population growth - requires food surplus to grow
# Increased food consumption in urban areas
# Rural settlements have reduced raw material output

REPLACE:town = {
	rank_modifier = {
		local_burghers_desired_pop = 0.2
		local_laborers_desired_pop = 0.2
		local_soldiers_desired_pop = 0.1
		local_nobles_desired_pop = 0.025
		local_clergy_desired_pop = 0.05

		local_max_control = 0.05
		# Pop capacity comes from buildings/food, not rank
		local_population_capacity_modifier = 0.10
		# Food capacity comes from buildings, not rank
		local_integration_speed_modifier = -0.25

		# Negative base growth - requires food surplus to grow
		local_population_growth = -0.0025

		free_building_levels = 25

	}

	country_modifier = {
		monthly_doom = 0.01
	}

	allow = {
		population >= 10
		modifier:cannot_upgrade_location = no
	}

	color = color_rank_duchy
	is_established_city = yes

	build_time = 365
	construction_demand = build_town_demand

	frame_tier = 2
	show_in_label = yes
}
