# Decluttered War UI — CMF Registration & Settings Init
# on_game_start: initial setup (fires once)
# monthly_weather_pulse: re-register CMF + init new countries (observer-safe, no country scope)

on_game_start = {
	on_actions = {
		dwui_on_game_start
	}
}

dwui_on_game_start = {
	effect = {
		cmf_add_action_bar_element = { element = dwui_settings_config }

		every_country = {
			if = {
				limit = {
					NOT = {
						has_variable = dwui_initialized
					}
				}

				dwui_init_settings = yes
			}
		}
	}
}

# monthly_weather_pulse has no country scope — safe for observers

monthly_weather_pulse = {
	on_actions = {
		dwui_monthly_cmf_check
	}
}

dwui_monthly_cmf_check = {
	effect = {
		# Re-register CMF element (idempotent — engine deduplicates)
		cmf_add_action_bar_element = { element = dwui_settings_config }

		# Initialize settings for any country that hasn't been set up yet
		every_country = {
			if = {
				limit = {
					NOT = {
						has_variable = dwui_initialized
					}
				}

				dwui_init_settings = yes
			}
		}
	}
}
