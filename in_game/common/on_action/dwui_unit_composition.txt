# Decluttered War UI — Unit Composition Recalculation Hooks

# Monthly recalculation for all armies

monthly_country_pulse = {
	on_actions = {
		dwui_monthly_recalc
	}
}

dwui_monthly_recalc = {
	effect = {
		every_army = { dwui_calculate_unit_composition = yes }
	}
}

# Recalculate after battles

on_battle_won = {
	on_actions = {
		dwui_battle_recalc
	}
}

on_battle_lost = {
	on_actions = {
		dwui_battle_recalc
	}
}

dwui_battle_recalc = {
	effect = {
		scope:actor = { dwui_calculate_unit_composition = yes }
		scope:target = { dwui_calculate_unit_composition = yes }
	}
}
