# SUL — DWUI Unit Composition Recalc (battle triggers)

on_battle_won = {
	on_actions = { sul_dwui_battle_recalc }
}

on_battle_lost = {
	on_actions = { sul_dwui_battle_recalc }
}

sul_dwui_battle_recalc = {
	effect = {
		scope:actor = { sul_dwui_calculate_unit_composition = yes }
		scope:target = { sul_dwui_calculate_unit_composition = yes }
	}
}
