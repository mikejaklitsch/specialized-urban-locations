# Unified hooks into vanilla hardcoded on_actions.
# Sequential on_actions for order-dependent work; events = { } for parallelized.
# Monthly/yearly country pulses use hidden events (in_game/events/sul_*_events.txt)
# for engine-level parallelization by country.

# Order: WPDO RGO map first (locations need rgo_building_type before spec init),
# then SUL specialization rebuild, then WPDO demand system + economy.

on_game_start = {
	on_actions = {
		sul_rgo_init
		sul_rgo_removal_start
		sul_rank_setup
		sul_initialize_all
		sul_war_on_game_start
		sul_integration_on_game_start
		sul_minting_initialize_all
		sul_epbm_initialize_all
		sul_migration_on_game_start
		sul_set_unowned_development
		sul_proximity_on_game_start
	}
}

# Economy init deferred to after-lobby: on_game_start fires during
# partial country init, so modifier reads (e.g. enrichment rates from
# sul_compute_budget_pressure) return none. CMF's after-lobby hook
# fires once countries are fully initialized.
# Order: economy pass (produces national power + estate gold) → wealth
# seed (needs those) → WPP recompute (fills in capital returns).

on_game_start_after_lobby = {
	on_actions = {
		sul_initialize_economy
		sul_wealth_seed_all
		sul_init_recompute_wpp
	}
}

# Order: SUL spec logic first (may destroy/rebuild buildings),
# then WPDO trim (adjusts RGO to new rank cap).

on_location_changed_rank = {
	on_actions = { sul_on_rank_changed }
}

# then SUL validation (may change spec based on new material).
