# Unified hooks into vanilla hardcoded on_actions.
# Only on_actions sub-blocks here — actual logic lives in respective files.
# Specialization System: sul_on_actions.txt
# Provisions System: sul_on_actions.txt, sul_swap_buildings.txt

# =============================================================================
# GAME START
# =============================================================================
# Order: WPDO RGO map first (locations need rgo_building_type before spec init),
# then SUL specialization rebuild, then WPDO demand system + economy.

on_game_start = {
	on_actions = {
		sul_rgo_init
		sul_rgo_removal_start
		sul_initialize_all
		sul_initialize_economy
	}
}

# =============================================================================
# LOCATION RANK CHANGED (root = location)
# =============================================================================
# Order: SUL spec logic first (may destroy/rebuild buildings),
# then WPDO trim (adjusts RGO to new rank cap).

on_location_changed_rank = {
	on_actions = { sul_on_rank_changed }
}

# =============================================================================
# RAW MATERIAL CHANGED (root = location)
# =============================================================================
# Order: WPDO cache update first (just a variable set),
# then SUL validation (may change spec based on new material).

on_raw_material_changed = {
	on_actions = { sul_on_raw_material_changed }
}

# =============================================================================
# MONTHLY COUNTRY PULSE (root = country)
# =============================================================================

monthly_country_pulse = {
	on_actions = {
		# Specialization system
		sul_version_check
		sul_clear_market_cache
		sul_refresh_market_cache
		# Provisions system
		sul_update_spending_rates
		sul_swap_buildings
		sul_player_wage_update
		sul_ai_monthly_wage_payment
	}
}

# =============================================================================
# YEARLY COUNTRY PULSE (root = country)
# =============================================================================

yearly_country_pulse = {
	on_actions = {
		sul_ai_yearly_wage_accumulate
		sul_cleanup_dead_units
		sul_yearly_rgo_trim
	}
}

# =============================================================================
# WEATHER MONTHLY PULSE (scopeless, ~every 4-5 game days)
# =============================================================================

weather_monthly_pulse = {
	on_actions = { sul_batch_location_update }
}

# =============================================================================
# FOUR YEARLY COUNTRY PULSE (root = country)
# =============================================================================

four_yearly_country_pulse = {
	random_events = { 10 = sul.1 }
}
