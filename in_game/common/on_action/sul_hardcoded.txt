# Handle modifier tier transitions when location rank changes
# This ensures the correct modifier tier (rural/town/city) is applied when a
# specialized location upgrades or downgrades via normal gameplay
# root = location

on_location_changed_rank = {
	on_actions = {
		sul_on_location_rank_changed
	}
}

on_game_start = {
	on_actions = {
		sul_game_start_setup
		sul_refresh_market_cache
	}
}

monthly_country_pulse = {
	on_actions = {
		sul_refresh_market_cache
		sul_monthly_estate_income
	}
}

yearly_country_pulse = {
	on_actions = {
		sul_yearly_estate_income
	}
}

# Wrapper on_actions for events (can't add events directly to existing on_actions)
sul_monthly_estate_income = {
	events = {
		sul_estate_income.1
	}
}

sul_yearly_estate_income = {
	events = {
		sul_estate_income.2
	}
}

# Handle raw material changes - recalculate eligibility and validate specialization
# root = location, scope:old_goods = previous trade goods, scope:new_goods = new trade goods
on_raw_material_changed = {
	on_actions = {
		sul_on_raw_material_changed
	}
}

sul_on_raw_material_changed = {
	effect = {
		# Only process owned locations
		if = {
			limit = { exists = owner }
			# Recalculate eligibility based on new resources
			sul_calculate_specialization_eligibility = yes
			# Validate current specialization - switch to commercial if invalid
			sul_validate_specialization = yes
			# Allow re-specialization - raw material change clears cooldown
			remove_location_modifier = sul_recently_respecialized
		}
	}
}

sul_game_start_setup = {
	effect = {
		# Single pass: calculate eligibility for all valid land, initialize owned locations
		every_location_in_the_world = {
			limit = { is_ownable = yes }

			# Always calculate eligibility (for potential map modes)
			sul_calculate_specialization_eligibility = yes

			# Only initialize owned locations
			if = {
				limit = { exists = owner }

				if = {
					limit = { location_rank = location_rank:rural_settlement }
					sul_assign_specialization_by_resources = yes
					sul_destroy_non_matching_buildings = yes
					sul_apply_specialization_modifier = yes
				}
				else_if = {
					limit = { OR = { location_rank = location_rank:town location_rank = location_rank:city } }
					# Save building count before specialization
					sul_save_building_levels = yes
					# Assign and destroy
					sul_assign_specialization_by_resources = yes
					sul_destroy_non_matching_buildings = yes
					# Restore building levels to prior count (includes RGO + guild buildings)
					sul_restore_building_levels = yes
					sul_apply_specialization_modifier = yes
				}

				# Initialize capacity pressure modifier
				sul_update_capacity_pressure = yes
			}
		}

		# Initialize local GDP for all countries (after location loop)
		every_country = {
			sul_update_all_location_gdp = yes
		}
	}
}

sul_on_location_rank_changed = {
	effect = {
		# If already has specialization, update variable for new rank and reapply modifier
		if = {
			limit = { sul_has_any_specialization = yes }
			# Clear all existing specialization modifiers
			sul_clear_all_specialization_modifiers = yes

			# Update variable to match new rank (using constants from expanded_urban_locations_specialization_constants.txt)
			if = {
				limit = { location_rank = location_rank:rural_settlement }
				# Convert to rural values
				if = {
					limit = { sul_has_mining_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_mining_rural_value }
				}
				else_if = {
					limit = { sul_has_farming_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_farming_rural_value }
				}
				else_if = {
					limit = { sul_has_gathering_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_gathering_rural_value }
				}
				else_if = {
					limit = { sul_has_woodland_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_woodland_rural_value }
				}
				else_if = {
					limit = { sul_has_commercial_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_commercial_rural_value }
				}
			}
			else_if = {
				limit = { location_rank = location_rank:town }
				# Convert to town values
				if = {
					limit = { sul_has_mining_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_mining_town_value }
				}
				else_if = {
					limit = { sul_has_farming_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_farming_town_value }
				}
				else_if = {
					limit = { sul_has_gathering_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_gathering_town_value }
				}
				else_if = {
					limit = { sul_has_woodland_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_woodland_town_value }
				}
				else_if = {
					limit = { sul_has_commercial_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_commercial_town_value }
				}
				# Destroy buildings that don't match specialization
				sul_destroy_non_matching_buildings = yes
			}
			else_if = {
				limit = { location_rank = location_rank:city }
				# Convert to city values
				if = {
					limit = { sul_has_mining_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_mining_city_value }
				}
				else_if = {
					limit = { sul_has_farming_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_farming_city_value }
				}
				else_if = {
					limit = { sul_has_gathering_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_gathering_city_value }
				}
				else_if = {
					limit = { sul_has_woodland_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_woodland_city_value }
				}
				else_if = {
					limit = { sul_has_commercial_specialization = yes }
					set_variable = { name = sul_specialization_designation value = sul_commercial_city_value }
				}
			}

			# Apply the correct modifier tier based on new rank
			sul_apply_specialization_modifier = yes
		}
		# New rural/town/city without specialization - assign based on resources
		# Note: Buildings are NOT granted on rank-up, only on game start
		else_if = {
			limit = {
				OR = {
					location_rank = location_rank:rural_settlement
					location_rank = location_rank:town
					location_rank = location_rank:city
				}
			}
			sul_assign_specialization_by_resources = yes
			# Destroy buildings that don't match the assigned specialization
			if = {
				limit = { OR = { location_rank = location_rank:town location_rank = location_rank:city } }
				sul_destroy_non_matching_buildings = yes
			}
			sul_apply_specialization_modifier = yes
		}
	}
}

# ============================================
# MARKET DEMAND CACHE SYSTEM
# Caches supply/demand ratios and high-demand flags on market center locations
# to avoid recalculating for every AI location evaluation
# NOTE: Variables stored on market center location because markets don't support variables
# Uses market.location to access the market center
# ============================================

# Refresh market cache if expired or not initialized
# Called on game start and monthly - only updates when cache doesn't exist
sul_refresh_market_cache = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = sul_market_demand_cached } }

			# Single pass through all markets
			every_market_in_world = {
				save_temporary_scope_as = current_market

				# Cache pressure values and price bonuses
				sul_update_market_cache = yes

				# Calculate trade spread for commercial specialization
				# Formula: sum of |price/base - 1| × base for all goods
				location = {
					set_variable = { name = sul_trade_spread value = 0 }
				}

				# Accumulate price deviation for each good in THIS market
				every_goods = {
					save_temporary_scope_as = current_good
					scope:current_market.location = {
						change_variable = {
							name = sul_trade_spread
							add = "scope:current_good.sul_goods_price_deviation"
						}
					}
				}

				# Normalize to 0-10 range (typical active market has ~50-100 total)
				location = {
					change_variable = { name = sul_trade_spread divide = 10 }
					clamp_variable = { name = sul_trade_spread min = 0 max = 10 }
				}
			}

			set_global_variable = {
				name = sul_market_demand_cached
				value = 1
				days = 2
			}
		}
	}
}
