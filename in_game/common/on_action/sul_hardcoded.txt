# Handle modifier tier transitions when location rank changes
# This ensures the correct modifier tier (rural/town/city) is applied when a
# specialized location upgrades or downgrades via normal gameplay
# root = location

on_location_changed_rank = {
	on_actions = {
		on_location_updates_specialization_by_rankup
	}
}

on_game_start = {
	on_actions = {
		specialized_locations_setup
		sul_refresh_market_cache
	}
}

monthly_country_pulse = {
	on_actions = {
		sul_refresh_market_cache
	}
}

# Handle raw material changes - recalculate eligibility and validate specialization
# root = location, scope:old_goods = previous trade goods, scope:new_goods = new trade goods
on_raw_material_changed = {
	on_actions = {
		sul_on_raw_material_changed
	}
}

sul_on_raw_material_changed = {
	effect = {
		# Only process owned locations
		if = {
			limit = { exists = owner }
			# Recalculate eligibility based on new resources
			sul_calculate_specialization_eligibility = yes
			# Validate current specialization - switch to commercial if invalid
			sul_validate_specialization = yes
			# Clear specialization cooldown - raw material change allows immediate re-specialization
			if = {
				limit = { has_location_modifier = sul_specialization_cooldown }
				remove_location_modifier = sul_specialization_cooldown
			}
		}
	}
}

specialized_locations_setup = {
	effect = {
		# Single pass: calculate eligibility for all valid land, initialize owned locations
		every_location_in_the_world = {
			limit = { sul_is_valid_land_location = yes }

			# Always calculate eligibility (for potential map modes)
			sul_calculate_specialization_eligibility = yes

			# Only initialize owned locations
			if = {
				limit = { exists = owner }

				if = {
					limit = { is_rural_settlement = yes }
					# Assign specialization FIRST so village building requirements are met
					assign_specialization_by_resources = yes
					destroy_non_matching_buildings = yes
					grant_starter_village = yes
					apply_specialization_modifier = yes
				}
				else_if = {
					limit = { is_town_or_city = yes }
					assign_specialization_by_resources = yes
					destroy_non_matching_buildings = yes
					grant_specialization_starter_buildings = yes
					grant_specialization_rgo_building = yes
					apply_specialization_modifier = yes
				}
			}
		}
	}
}

on_location_updates_specialization_by_rankup = {
	effect = {
		# If already has specialization, update variable for new rank and reapply modifier
		if = {
			limit = { has_any_specialization = yes }
			# Clear all existing specialization modifiers
			clear_all_specialization_modifiers = yes

			# Update variable to match new rank (using constants from expanded_urban_locations_specialization_constants.txt)
			if = {
				limit = { location_rank = location_rank:rural_settlement }
				# Convert to rural values
				if = {
					limit = { has_mining_specialization = yes }
					set_variable = { name = specialization_designation value = sul_mining_rural_value }
				}
				else_if = {
					limit = { has_farming_specialization = yes }
					set_variable = { name = specialization_designation value = sul_farming_rural_value }
				}
				else_if = {
					limit = { has_gathering_specialization = yes }
					set_variable = { name = specialization_designation value = sul_gathering_rural_value }
				}
				else_if = {
					limit = { has_woodland_specialization = yes }
					set_variable = { name = specialization_designation value = sul_woodland_rural_value }
				}
				else_if = {
					limit = { has_commercial_specialization = yes }
					set_variable = { name = specialization_designation value = sul_commercial_rural_value }
				}
			}
			else_if = {
				limit = { location_rank = location_rank:town }
				# Convert to town values
				if = {
					limit = { has_mining_specialization = yes }
					set_variable = { name = specialization_designation value = sul_mining_town_value }
				}
				else_if = {
					limit = { has_farming_specialization = yes }
					set_variable = { name = specialization_designation value = sul_farming_town_value }
				}
				else_if = {
					limit = { has_gathering_specialization = yes }
					set_variable = { name = specialization_designation value = sul_gathering_town_value }
				}
				else_if = {
					limit = { has_woodland_specialization = yes }
					set_variable = { name = specialization_designation value = sul_woodland_town_value }
				}
				else_if = {
					limit = { has_commercial_specialization = yes }
					set_variable = { name = specialization_designation value = sul_commercial_town_value }
				}
				# When upgrading from rural to town, destroy buildings and grant starter buildings
				destroy_non_matching_buildings = yes
				grant_specialization_starter_buildings = yes
				grant_specialization_rgo_building = yes
			}
			else_if = {
				limit = { location_rank = location_rank:city }
				# Convert to city values
				if = {
					limit = { has_mining_specialization = yes }
					set_variable = { name = specialization_designation value = sul_mining_city_value }
				}
				else_if = {
					limit = { has_farming_specialization = yes }
					set_variable = { name = specialization_designation value = sul_farming_city_value }
				}
				else_if = {
					limit = { has_gathering_specialization = yes }
					set_variable = { name = specialization_designation value = sul_gathering_city_value }
				}
				else_if = {
					limit = { has_woodland_specialization = yes }
					set_variable = { name = specialization_designation value = sul_woodland_city_value }
				}
				else_if = {
					limit = { has_commercial_specialization = yes }
					set_variable = { name = specialization_designation value = sul_commercial_city_value }
				}
			}

			# Apply the correct modifier tier based on new rank
			apply_specialization_modifier = yes
		}
		# New rural/town/city without specialization - assign based on resources
		else_if = {
			limit = { is_rural_or_higher = yes }
			# Grant starter village for new rural settlements
			if = {
				limit = { is_rural_settlement = yes }
				grant_starter_village = yes
			}
			assign_specialization_by_resources = yes
			# Only destroy buildings for towns/cities, not rural
			if = {
				limit = { is_town_or_city = yes }
				destroy_non_matching_buildings = yes
				grant_specialization_starter_buildings = yes
				grant_specialization_rgo_building = yes
			}
			apply_specialization_modifier = yes
		}
	}
}

# ============================================
# MARKET DEMAND CACHE SYSTEM
# Caches supply/demand ratios and high-demand flags on market center locations
# to avoid recalculating for every AI location evaluation
# NOTE: Variables stored on market center location because markets don't support variables
# Uses market.location to access the market center
# ============================================

# Refresh market cache if expired or not initialized
# Called on game start and monthly - only updates when cache doesn't exist
sul_refresh_market_cache = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = sul_market_demand_cached } }

			# Update standard market caches (pressure values, price bonuses)
			every_market_in_world = {
				sul_update_market_cache = yes
			}

			# Calculate trade spread for commercial specialization
			# Step 1: Reset trade_spread on all market centers
			every_market_in_world = {
				sul_calculate_trade_spread = yes
			}
			# Step 2: Iterate all goods, accumulating price deviation to each market center
			every_goods = {
				sul_accumulate_trade_spread_for_markets = yes
			}
			# Step 3: Normalize trade_spread to a reasonable AI weight range
			# Weighted formula: sum of |price/base - 1| × base for all goods
			# With ~30 goods and avg base ~10, typical active market might have ~50-100 total
			every_market_in_world = {
				location = {
					# Divide by 10 to scale to 0-10 range for typical markets
					change_variable = { name = sul_trade_spread divide = 10 }
					clamp_variable = { name = sul_trade_spread min = 0 max = 10 }
				}
			}

			set_global_variable = {
				name = sul_market_demand_cached
				value = 1
				days = 2
			}
		}
	}
}
