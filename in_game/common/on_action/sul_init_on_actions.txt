sul_set_unowned_development = {
	effect = {
		every_location_in_the_world = {
			limit = {
				is_ownable = yes
				has_owner = no
			}
			change_development = {
				value = sul_development_equilibrium
				subtract = development
				round = yes
			}
		}
	}
}

sul_initialize_all = {
	effect = {
		sul_specialization_full_rebuild = yes
		set_global_variable = { name = sul_version value = sul_specialization_version_value }
	}
}

# Fires on weather_monthly_pulse (once globally) if the version marker is
# missing or stale. No reentrance guard needed — pulse is global, not per-country.

sul_version_check = {
	trigger = {
		OR = {
			NOT = { has_global_variable = sul_version }
			global_var:sul_version < sul_specialization_version_value
		}
	}

	effect = {
		sul_specialization_full_rebuild = yes
		set_global_variable = { name = sul_version value = sul_specialization_version_value }
	}
}
