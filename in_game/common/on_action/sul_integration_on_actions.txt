
sul_integration_on_game_start = {
	effect = {
		debug_log = "SUL_TRACE: integration_on_game_start ENTER"
		sul_integration_initialize = yes
	}
}

# Fires on weather_monthly_pulse (once globally) if the version marker is
# missing or stale. No reentrance guard needed — pulse is global, not per-country.

sul_integration_init_check = {
	trigger = {
		OR = {
			NOT = { has_global_variable = sul_integration_version }
			global_var:sul_integration_version < sul_integration_version_value
		}
	}

	effect = {
		debug_log = "SUL_TRACE: integration_init_check ENTER"
		sul_integration_initialize = yes
	}
}

sul_integration_on_location_conquered = {
	trigger = { integration_level = conquered }

	effect = {
		debug_log = "SUL_TRACE: integration_on_location_conquered ENTER"
		sul_integration_track_location = yes
		sul_integration_apply_capacity_bonus = yes
	}
}

# Iterate only tracked conquered locations, not all owned locations.
# Locations that advanced past conquered get untracked and cleaned up.
# Locations still conquered get their modifier size refreshed
# (population and capacity may have shifted since last month).

sul_integration_monthly_update = {
	effect = {
		debug_log = "SUL_TRACE: integration_monthly_update ENTER"
		# Pass 1: update or tag for removal.
		# Avoids mutating the conquered list during iteration.
		every_in_global_list = {
			variable = sul_integration_conquered_locations

			if = {
				limit = { NOT = { integration_level = conquered } }

				remove_location_modifier = sul_integration_capacity_bonus

				add_to_global_variable_list = { name = sul_integration_remove_queue target = this }
			}
			else = { sul_integration_apply_capacity_bonus = yes }
		}

		# Pass 2: drain the removal queue.
		every_in_global_list = {
			variable = sul_integration_remove_queue
			remove_list_global_variable = { name = sul_integration_conquered_locations target = this }
		}

		clear_global_variable_list = sul_integration_remove_queue
	}
}
