# =============================================================================
# INITIALIZATION & VERSION CHECK
# =============================================================================
# Rebuild version: major * 100 + minor (patch changes don't trigger rebuild)
# Bump this when save-breaking changes require re-initialization
@sul_rebuild_version = 10 # 0.1.x
# Bouvet Island (location:bouvet_island): impassable location used as mod state anchor.
# sul_active modifier applied here — stripped by engine when mod is removed.
# sul_version variable stored here for version-based rebuild detection.

# Game start: full rebuild + stamp version

sul_initialize_all = {
	effect = {
		sul_full_rebuild = yes

		# Initialization: accelerate pop promotion to populate starting buildings
		every_country = {
			add_country_modifier = { modifier = sul_init_pop_rush days = 45 }
		}

		# Stamp version on Bouvet Island
		location:bouvet_island = {
			remove_location_modifier = sul_active
			add_location_modifier = { modifier = sul_active days = -1 }
			set_variable = { name = sul_version value = @sul_rebuild_version }
		}
	}
}

# Monthly version check: triggers rebuild if marker is missing or outdated
# Handles first-time init on existing save, version upgrades, and mod remove/re-add

sul_version_check = {
	trigger = {
		OR = {
			# Modifier missing = mod was removed and re-added (or first init on existing save)
			location:bouvet_island = {
				NOT = {
					has_location_modifier = sul_active
				}
			}

			# Modifier present but version outdated = mod was updated
			location:bouvet_island = {
				has_location_modifier = sul_active
				has_variable = sul_version
				var:sul_version < @sul_rebuild_version
			}
		}

		# Guard: don't let multiple countries trigger rebuild in the same tick
		NOT = {
			has_global_variable = sul_rebuilding
		}
	}

	effect = {
		set_global_variable = { name = sul_rebuilding value = yes }
		sul_full_rebuild = yes
		# Stamp version on Bouvet Island
		location:bouvet_island = {
			remove_location_modifier = sul_active
			add_location_modifier = { modifier = sul_active days = -1 }
			set_variable = { name = sul_version value = @sul_rebuild_version }
		}

		remove_global_variable = sul_rebuilding
	}
}

sul_clear_market_cache = {
	effect = {
		if = {
			limit = {
				has_global_variable = sul_market_demand_cached
				has_global_variable = sul_cache_month
				NOT = {
					global_var:sul_cache_month = current_month
				}
			}

			remove_global_variable = sul_market_demand_cached
		}

		# Track current month
		set_global_variable = { name = sul_cache_month value = current_month }
	}
}

# Location rank changed: SUL spec logic first, then WPDO RGO trim
# root = location

sul_on_rank_changed = {
	effect = {
		# Specialization: refresh modifier or assign new spec, destroy mismatched buildings
		if = {
			limit = { sul_has_any_specialization = yes }
			sul_refresh_specialization_modifier = yes
			if = {
				limit = {
					OR = {
						location_rank = location_rank:town
						location_rank = location_rank:city
					}
				}

				sul_destroy_non_matching_buildings = yes
			}
		}
		else_if = {
			limit = {
				OR = {
					location_rank = location_rank:rural_settlement
					location_rank = location_rank:town
					location_rank = location_rank:city
				}
			}

			sul_assign_specialization_by_resources = yes
			sul_apply_specialization_modifier = yes
			if = {
				limit = {
					OR = {
						location_rank = location_rank:town
						location_rank = location_rank:city
					}
				}

				sul_destroy_non_matching_buildings = yes
			}
		}

		# Provisions: trim RGO to new rank cap
		sul_trim_rgo_levels = yes
	}
}

# Raw material changed: WPDO cache update first, then SUL validation
# root = location, scope:old_goods, scope:new_goods

sul_on_raw_material_changed = {
	effect = {
		# Provisions: update RGO building type cache from new raw material
		set_variable = {
			name = sul_rgo_building_type
			value = "global_variable_map(sul_rgo_map|raw_material)"
		}

		# Specialization: recalculate eligibility, validate current spec
		if = {
			limit = { exists = owner }
			sul_calculate_specialization_eligibility = yes
			sul_validate_specialization = yes
			remove_location_modifier = sul_recently_respecialized # new resource = free respec
		}
	}
}

sul_initialize_economy = {
	effect = {
		sul_init_demand_maps = yes

		every_country = {
			limit = { is_real_country = yes sul_is_valid_location_owner = yes }

			# Seed budget pressure defaults (1 = neutral) before economy update
			set_variable = { name = sul_budget_pressure_nobles value = 1 }
			set_variable = { name = sul_budget_pressure_clergy value = 1 }
			set_variable = { name = sul_budget_pressure_burghers value = 1 }
			set_variable = { name = sul_budget_pressure_peasants value = 1 }

			# Seed spending rates before economy update
			set_variable = {
				name = sul_spending_nobles

				value = {
					value = 1
					subtract = modifier:estate_enrichment
					subtract = @sul_enrichment_offset_nobles
					min = 0
					max = 1
				}
			}
			set_variable = {
				name = sul_spending_clergy

				value = {
					value = 1
					subtract = modifier:estate_enrichment
					subtract = @sul_enrichment_offset_clergy
					min = 0
					max = 1
				}
			}
			set_variable = {
				name = sul_spending_burghers

				value = {
					value = 1
					subtract = modifier:estate_enrichment
					subtract = @sul_enrichment_offset_burghers
					min = 0
					max = 1
				}
			}
			set_variable = {
				name = sul_spending_commoners

				value = {
					value = 1
					subtract = modifier:estate_enrichment
					subtract = @sul_enrichment_offset_commoners
					min = 0
					max = 1
				}
			}

			every_owned_location = { sul_init_zero_wpp = yes }

			sul_update_country_economy = yes
		}
	}
}

# Yearly RGO level trim — ensures no location exceeds max cap

sul_yearly_rgo_trim = {
	trigger = { is_real_country = yes }

	effect = {
		every_owned_location = {
			limit = {
				has_variable = sul_rgo_building_type
				sul_location_rgo_building_level > sul_location_max_rgo_building_level_rounded
			}

			sul_trim_rgo_levels = yes
		}
	}
}

# Dead Unit Cleanup — destroys AI armies with 0 strength.

sul_cleanup_dead_units = {
	effect = {
		if = {
			limit = { is_ai = yes }
			every_army = { limit = { strength_percentage <= 0 } destroy_unit = yes }
		}
	}
}
