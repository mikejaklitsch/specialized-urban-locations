# Specialized Urban Locations - On Actions
# =============================================================================
# MONTHLY COUNTRY PULSE
# =============================================================================

monthly_country_pulse = {
	on_actions = {
		sul_clear_market_cache
		sul_refresh_market_cache
		sul_player_economy_update
	}
}

sul_player_economy_update = {
	trigger = {
		is_ai = no
	}

	effect = {
		sul_update_all_location_gdp = yes
	}
}

# sul_player_capacity_update removed - effect was not implemented
# =============================================================================
# LOCATION OWNERSHIP CHANGE
# =============================================================================
# Hook into hardcoded on_action

on_location_changed_owner = {
	on_actions = {
		sul_on_location_ownership_changed
	}
}

# root = location, scope:loser = previous owner, scope:winner = new owner

sul_on_location_ownership_changed = {
	effect = {
		# Assign specialization if location doesn't have one (colonization, etc.)
		if = {
			limit = {
				NOR = {
					sul_has_mining_specialization = yes
					sul_has_farming_specialization = yes
					sul_has_woodland_specialization = yes
					sul_has_gathering_specialization = yes
					sul_has_commercial_specialization = yes
				}
			}

			sul_assign_specialization_by_resources = yes
			sul_apply_specialization_modifier = yes
			# Clear any cooldown from previous owner
			remove_location_modifier = sul_recently_respecialized
		}

		# Update economy for new owner (estate power -> GDP -> gdp_pc)
		sul_update_location_economy = yes
		# Now location definitely has a specialization - update counts
		# Decrement loser's counts (if loser exists and has tracking variables)
		if = {
			limit = {
				exists = scope:loser
				scope:loser = { has_variable = sul_mining_count }
			}

			save_temporary_scope_as = sul_loc
			scope:loser = { sul_decrement_spec_count = yes }
		}

		# Increment winner's counts
		if = {
			limit = { exists = scope:winner }
			if = {
				limit = {
					scope:winner = {
						NOT = {
							has_variable = sul_mining_count
						}
					}
				}

				scope:winner = { sul_initialize_specialization_counts = yes }
			}
			else = {
				save_temporary_scope_as = sul_loc
				scope:winner = { sul_increment_spec_count = yes }
			}
		}
	}
}
