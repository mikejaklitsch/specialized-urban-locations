# Specialized Urban Locations - On Actions

# =============================================================================
# MONTHLY COUNTRY PULSE
# =============================================================================

monthly_country_pulse = {
	on_actions = {
		sul_init_unowned_locations
		sul_init_country_variables
		sul_clear_market_cache
		sul_refresh_market_cache
		sul_player_economy_update
		sul_monthly_estate_income
		sul_player_capacity_update
	}
}

# One-time initialization of unowned locations (save migration)
# Runs once globally, independent of per-country init
sul_init_unowned_locations = {
	trigger = {
		NOT = { has_global_variable = sul_unowned_locations_initialized }
	}
	effect = {
		every_location_in_the_world = {
			limit = {
				is_ownable = yes
				NOT = { exists = owner }
			}
			set_variable = { name = sul_noble_gdp_pc value = 1 }
			set_variable = { name = sul_clergy_gdp_pc value = 1 }
			set_variable = { name = sul_burgher_gdp_pc value = 1 }
			set_variable = { name = sul_commoner_gdp_pc value = 1 }
			set_variable = { name = sul_tribesman_gdp_pc value = 1 }
			set_variable = { name = sul_slave_gdp_pc value = 1 }
		}
		set_global_variable = { name = sul_unowned_locations_initialized value = yes }
	}
}

# Initialize country-scope variables if missing (runs first each month)
# Also initializes location variables for save migration
sul_init_country_variables = {
	trigger = {
		NOT = { has_variable = sul_nobles_taxes_lost }
	}
	effect = {
		# Country-scope taxes lost
		set_variable = { name = sul_nobles_taxes_lost value = 0 }
		set_variable = { name = sul_clergy_taxes_lost value = 0 }
		set_variable = { name = sul_burghers_taxes_lost value = 0 }
		set_variable = { name = sul_peasants_taxes_lost value = 0 }
		set_variable = { name = sul_tribes_taxes_lost value = 0 }
		# Location-scope initialization (save migration)
		# sul_update_all_location_gdp calls sul_update_location_economy for each location
		# which handles: estate power -> GDP -> gdp_pc cache
		sul_update_all_location_gdp = yes
	}
}

sul_player_economy_update = {
	trigger = {
		is_ai = no
	}
	effect = {
		sul_update_all_location_gdp = yes
	}
}

sul_player_capacity_update = {
	trigger = {
		is_ai = no
	}
	effect = {
		sul_update_all_location_capacity = yes
	}
}

# =============================================================================
# LOCATION OWNERSHIP CHANGE
# =============================================================================

# Hook into hardcoded on_action
on_location_changed_owner = {
	on_actions = {
		sul_on_location_ownership_changed
	}
}

# root = location, scope:loser = previous owner, scope:winner = new owner
sul_on_location_ownership_changed = {
	effect = {
		# Assign specialization if location doesn't have one (colonization, etc.)
		if = {
			limit = {
				NOR = {
					sul_has_mining_specialization = yes
					sul_has_farming_specialization = yes
					sul_has_woodland_specialization = yes
					sul_has_gathering_specialization = yes
					sul_has_commercial_specialization = yes
				}
			}
			sul_assign_specialization_by_resources = yes
			sul_apply_specialization_modifier = yes
			# Clear any cooldown from previous owner
			remove_location_modifier = sul_recently_respecialized
		}

		# Update economy for new owner (estate power -> GDP -> gdp_pc)
		sul_update_location_economy = yes

		# Now location definitely has a specialization - update counts

		# Decrement loser's counts (if loser exists and has tracking variables)
		if = {
			limit = {
				exists = scope:loser
				scope:loser = { has_variable = sul_mining_count }
			}
			if = {
				limit = { sul_has_mining_specialization = yes }
				scope:loser = {
					change_variable = { name = sul_mining_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
			}
			else_if = {
				limit = { sul_has_farming_specialization = yes }
				scope:loser = {
					change_variable = { name = sul_farming_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 2 }
				}
			}
			else_if = {
				limit = { sul_has_woodland_specialization = yes }
				scope:loser = {
					change_variable = { name = sul_woodland_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
			}
			else_if = {
				limit = { sul_has_gathering_specialization = yes }
				scope:loser = {
					change_variable = { name = sul_gathering_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
			}
			else_if = {
				limit = { sul_has_commercial_specialization = yes }
				scope:loser = {
					change_variable = { name = sul_commercial_count subtract = 1 }
					change_variable = { name = sul_total_weighted subtract = 1 }
				}
			}
		}

		# Increment winner's counts
		if = {
			limit = { exists = scope:winner }
			# Initialize if winner doesn't have tracking yet
			if = {
				limit = { scope:winner = { NOT = { has_variable = sul_mining_count } } }
				scope:winner = { sul_initialize_specialization_counts = yes }
			}
			# Only increment if variables already existed (initialization already counted this location)
			else = {
				if = {
					limit = { sul_has_mining_specialization = yes }
					scope:winner = {
						change_variable = { name = sul_mining_count add = 1 }
						change_variable = { name = sul_total_weighted add = 1 }
					}
				}
				else_if = {
					limit = { sul_has_farming_specialization = yes }
					scope:winner = {
						change_variable = { name = sul_farming_count add = 1 }
						change_variable = { name = sul_total_weighted add = 2 }
					}
				}
				else_if = {
					limit = { sul_has_woodland_specialization = yes }
					scope:winner = {
						change_variable = { name = sul_woodland_count add = 1 }
						change_variable = { name = sul_total_weighted add = 1 }
					}
				}
				else_if = {
					limit = { sul_has_gathering_specialization = yes }
					scope:winner = {
						change_variable = { name = sul_gathering_count add = 1 }
						change_variable = { name = sul_total_weighted add = 1 }
					}
				}
				else_if = {
					limit = { sul_has_commercial_specialization = yes }
					scope:winner = {
						change_variable = { name = sul_commercial_count add = 1 }
						change_variable = { name = sul_total_weighted add = 1 }
					}
				}
			}
		}
	}
}
