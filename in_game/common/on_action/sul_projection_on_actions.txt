# Power Projection — on_action handlers.
# Wired into vanilla on_actions through sul_hardcoded.txt.

# Game start: initialize subject drag variables for every country so the
# auto_modifier has values on the first tick instead of waiting a month.

sul_projection_on_game_start = {
	effect = {
		every_country = { sul_projection_recompute_subject_drag = yes }
	}
}

# Monthly per-country pass: refresh iteration-based drag variables.
# Cheap (formula-only) drags handle themselves through their auto_modifier
# script_values and don't need updating here.

sul_projection_monthly_update = {
	effect = {
		sul_projection_recompute_subject_drag = yes
	}
}
