# Batched respec score cache. Walks at most 166 stale locations per tick,
# ordered by oldest sul_last_respec_cache first. ~10 year full world cycle.
@sul_respec_batch_size = 166

sul_respec_cache_batch = {
	effect = {
		ordered_sul_respec_cache_location = {
			max = @sul_respec_batch_size

			order_by = {
				value = 0
				if = {
					limit = { has_variable = sul_last_respec_cache }
					subtract = var:sul_last_respec_cache
				}
			}

			sul_cache_respec_location = yes
		}
	}
}

# Full world init pass for game start / version migration.
sul_respec_cache_init = {
	effect = {
		every_location_in_the_world = {
			limit = { has_owner = yes }
			sul_cache_respec_location = yes
		}
	}
}
