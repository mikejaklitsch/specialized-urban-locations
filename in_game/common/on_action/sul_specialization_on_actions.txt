@sul_mining_type = 1
@sul_farming_type = 2
@sul_gathering_type = 3
@sul_woodland_type = 4
@sul_commercial_type = 5

# Combined handlers for shared vanilla hooks - specialization logic plus
# sibling-system dispatch. These are the entry points registered in
# sul_hardcoded.txt for on_location_changed_rank, on_raw_material_changed,
# and on_location_changed_owner.

# Location rank changed: SUL spec logic first, then RGO trim to new rank cap
# root = location

sul_on_rank_changed = {
	effect = {
		# Redirect vanilla rural_settlement back to the correct specialization rural
		if = {
			limit = {
				location_rank = location_rank:rural_settlement
				has_variable = sul_spec_type
			}
			if = {
				limit = { var:sul_spec_type = 1 }
				change_location_rank = location_rank:mining_rural
			}
			else_if = {
				limit = { var:sul_spec_type = 2 }
				change_location_rank = location_rank:farming_rural
			}
			else_if = {
				limit = { var:sul_spec_type = 3 }
				change_location_rank = location_rank:gathering_rural
			}
			else_if = {
				limit = { var:sul_spec_type = 4 }
				change_location_rank = location_rank:woodland_rural
			}
			else_if = {
				limit = { var:sul_spec_type = 5 }
				change_location_rank = location_rank:commercial_rural
			}
		}

		if = {
			limit = { OR = { modifier:sul_is_rural_settlement_rank_tier = yes modifier:sul_is_town_rank_tier = yes modifier:sul_is_city_rank_tier = yes } }

			# Detect spec change: var:sul_spec_type has OLD spec,
			# the new rank's boolean modifier identifies the NEW spec.
			# Nested if to prevent var:sul_spec_type evaluation on uninitialized locations.
			if = {
				limit = { has_variable = sul_spec_type }
				if = {
				limit = {
					OR = {
						AND = { var:sul_spec_type = @sul_mining_type NOT = { modifier:sul_mining_specialization = yes } }
						AND = { var:sul_spec_type = @sul_farming_type NOT = { modifier:sul_farming_specialization = yes } }
						AND = { var:sul_spec_type = @sul_gathering_type NOT = { modifier:sul_gathering_specialization = yes } }
						AND = { var:sul_spec_type = @sul_woodland_type NOT = { modifier:sul_woodland_specialization = yes } }
						AND = { var:sul_spec_type = @sul_commercial_type NOT = { modifier:sul_commercial_specialization = yes } }
					}
				}

				# === SPECIALIZATION CHANGED ===
				# Primary respec path is on_built (carrier buildings).
				# This fallback catches script-driven rank changes.
				sul_specialization_destroy_and_penalize = yes

				if = {
					limit = { exists = owner }
					owner = {
						if = {
							limit = { NOT = { has_variable = sul_mining_count } }
							sul_specialization_count_initialize = yes
						}
						prev = { save_temporary_scope_as = sul_loc }
						sul_specialization_count_decrement = yes
						sul_specialization_count_increment = yes
					}
				}

				if = {
					limit = {
						is_market_center = yes
						NOT = { sul_has_commercial_specialization = yes }
					}
					sul_migration_on_market_destroyed = yes
					market = { destroy_market = {} }
				}

				if = {
					limit = {
						sul_has_commercial_specialization = yes
						NOT = { is_market_center = yes }
					}
					create_market = {
						builder = owner
						location = prev
						price = price:create_market
						price_modifier = 0
						instant = yes
					}
					sul_migration_on_market_created = yes
				}
			}
			else = {
				# === TIER CHANGED (upgrade/downgrade within same spec) ===
			}
			}

			# Update GUI bridge variables (sets sul_spec_type to new value)
			sul_specialization_update_designation = yes
		}

		sul_trim_rgo_levels = yes
	}
}

# Raw material changed: rebuild RGO under the new good, then revalidate spec
# root = location, scope:old_goods, scope:new_goods

sul_on_raw_material_changed = {
	effect = {
		# Save old RGO level so we can rebuild the new one at the same count.
		# sul_rebuild_rgo_if_empty's prev check uses raw_material directly,
		# so timing of the var update no longer affects the destroy/rebuild flow.
		save_temporary_scope_value_as = {
			name = sul_old_rgo_levels
			value = sul_location_rgo_building_level
		}

		set_variable = {
			name = sul_rgo_building_type
			value = "global_variable_map(sul_rgo_map|raw_material)"
		}

		# Destroy old RGO via the OLD raw material in scope:old_goods
		if = {
			limit = { scope:sul_old_rgo_levels > 0 }

			change_building_level_in_location = {
				building = "global_variable_map(sul_rgo_map|scope:old_goods)"
				value = { value = scope:sul_old_rgo_levels multiply = -1 }
			}
		}

		# Build new RGO with same level count (or 1 if was empty)
		if = {
			limit = { exists = raw_material }

			change_building_level_in_location = {
				building = "global_variable_map(sul_rgo_map|raw_material)"
				value = { value = scope:sul_old_rgo_levels min = 1 }
			}
		}

		if = {
			limit = { exists = owner }
			sul_specialization_calculate_eligibility = yes
			sul_specialization_validate = yes
			remove_location_modifier = sul_recently_respecialized # new resource = free respec
		}
	}
}

# Owner changed: seed RGO for new colonies that have none
# root = location, scope:loser, scope:winner

sul_on_location_changed_owner = {
	effect = {
		if = {
			limit = { exists = scope:winner NOT = { exists = scope:loser } }
			sul_on_location_colonized = yes
		}
		else_if = {
			limit = { exists = scope:loser NOT = { exists = scope:winner } }
			sul_on_location_decolonized = yes
		}

		if = {
			limit = { exists = raw_material }

			set_variable = {
				name = sul_rgo_building_type
				value = "global_variable_map(sul_rgo_map|raw_material)"
			}
		}
	}
}
