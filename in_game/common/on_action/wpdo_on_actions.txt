# Provisions on_actions — called from sul_hardcoded.txt unified dispatcher.
# Rank change and raw material change are in combined handlers (sul_on_actions.txt).

wpdo_rgo_init = {
	effect = {
		wpdo_init_rgo_map = yes

		every_location_in_the_world = {
			limit = { is_ownable = yes }
			set_variable = { name = wpdo_rgo_constructing value = 0 }

			set_variable = {
				name = wpdo_rgo_building_type
				value = "global_variable_map(wpdo_rgo_map|raw_material)"
			}
		}
	}
}

wpdo_rgo_removal_start = {
	effect = {
		every_country = { trigger_event_silently = { id = rgo_removal.01 } }
	}
}

# =============================================================================
# DEMAND SYSTEM INITIALIZATION
# =============================================================================

wpdo_initialize_all = {
	effect = {
		wpdo_init_demand_maps = yes

		every_country = {
			limit = { is_real_country = yes wpdo_is_valid_location_owner = yes }

			# Seed budget pressure defaults (1 = neutral) before economy update
			set_variable = { name = wpdo_budget_pressure_nobles value = 1 }
			set_variable = { name = wpdo_budget_pressure_clergy value = 1 }
			set_variable = { name = wpdo_budget_pressure_burghers value = 1 }
			set_variable = { name = wpdo_budget_pressure_peasants value = 1 }

			every_owned_location = { wpdo_init_zero_wpp = yes }

			wpdo_update_country_economy = yes
		}
	}
}

# =============================================================================
# WPP BATCH UPDATE
# =============================================================================

@wpdo_ai_batch_size = 333

wpdo_batch_location_update = {
	effect = {
		ordered_wpdo_stale_ai_location = {
			max = @wpdo_ai_batch_size

			order_by = {
				value = 0
				if = {
					limit = { has_variable = wpdo_last_update }
					subtract = var:wpdo_last_update
				}
			}

			wpdo_update_location_wpp = yes
			set_variable = { name = wpdo_last_update value = current_year }
		}
	}
}

# =============================================================================
# WAGE SYSTEM
# =============================================================================

wpdo_player_wage_update = {
	trigger = { is_ai = no is_real_country = yes }

	effect = {
		wpdo_accumulate_wages = yes
		wpdo_pay_estate_wages = yes
	}
}

# Yearly RGO level trim — ensures no location exceeds max cap

wpdo_yearly_rgo_trim = {
	trigger = { is_real_country = yes }

	effect = {
		every_owned_location = {
			limit = {
				has_variable = wpdo_rgo_building_type
				wpdo_location_rgo_building_level > wpdo_location_max_rgo_building_level_rounded
			}

			wpdo_trim_rgo_levels = yes
		}
	}
}

wpdo_ai_yearly_wage_accumulate = {
	trigger = { is_ai = yes is_real_country = yes }

	effect = {
		wpdo_accumulate_wages = yes
	}
}

wpdo_ai_monthly_wage_payment = {
	trigger = {
		is_ai = yes
		is_real_country = yes
		has_variable = wpdo_monthly_wages_nobles
	}

	effect = {
		wpdo_pay_estate_wages = yes
	}
}
