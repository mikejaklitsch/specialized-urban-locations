# Building Swap — replaces vanilla army/navy buildings with wpdo_ clones.
# Runs once per country via monthly_country_pulse.
# Also cleans up dead AI armies yearly.

wpdo_swap_buildings = {
	effect = {
		if = {
			limit = {
				NOT = {
					has_variable = wpdo_swap_done
				}
			}

			set_variable = wpdo_swap_done

			every_owned_location = {
				# --- Army buildings ---
				if = {
					limit = {
						location_building_level = { building_type = building_type:armory value > 0 }
					}

					change_building_level_in_location = {
						building = building_type:wpdo_armory
						value = "location_building_level(building_type:armory)"
					}
					change_building_level_in_location = {
						building = building_type:armory

						value = {
							add = "location_building_level(building_type:armory)"
							multiply = -1
						}
					}

							building_type = building_type:training_fields

					}

							add = "location_building_level(building_type:training_fields)"
							multiply = -1
						}

					limit = {

							add = "location_building_level(building_type:naval_supplies_mill)"
							multiply = -1
						}
					}
				}
			}
		}
	}
}

# Dead Unit Cleanup — destroys AI armies with 0 strength.

wpdo_cleanup_dead_units = {
	effect = {
		if = {
			limit = { is_ai = yes }
			every_army = { limit = { strength_percentage <= 0 } destroy_unit = yes }
		}
	}
}
