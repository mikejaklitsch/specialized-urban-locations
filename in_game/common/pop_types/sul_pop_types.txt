# Allow all pop types to grow naturally (not just peasants/tribesmen)

INJECT:nobles = {
	grow = yes
	pop_food_consumption = 25.0
}

INJECT:clergy = {
	grow = yes
	pop_food_consumption = 6.5
}

INJECT:burghers = {
	grow = yes
	pop_food_consumption = 6.5

	# which breaks our all-flat development system. Replace with flat value.
	literacy_impact = {
		local_monthly_development_modifier = -0.25   # cancel vanilla +0.25
		local_monthly_development = 0.025            # flat: literate merchants develop (0.5×)
	}
}

	grow = yes
	pop_food_consumption = 1.5
}

INJECT:soldiers = {
	grow = yes
}

INJECT:peasants = {
	pop_food_consumption = 0.95
}

INJECT:slaves = {
