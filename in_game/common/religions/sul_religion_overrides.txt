# RELIGION FOOD MODIFIER NULLIFICATION
# ============================================
# Religions with abstract food bonuses; replace with specific crops.

# Iroquoian Totemism (Orendism): vanilla +10% food (Three Sisters agriculture)

	definition_modifier = {
		global_monthly_food_modifier = -0.1          # nullify vanilla +0.10
		global_maize_output_modifier = 0.05
		global_legumes_output_modifier = 0.03
		global_wild_game_output_modifier = 0.02
	}
}
