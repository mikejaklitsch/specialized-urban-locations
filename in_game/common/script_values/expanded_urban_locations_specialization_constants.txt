# ============================================
# DIVERSITY PRESSURE (percentage-based, cached)
# ============================================
# Uses cached country variables updated only when specializations change
# Variables: var:eul_mining_count, var:eul_farming_count, etc.
# var:eul_total_weighted = farming×2 + others×1
# Result is positive if under-represented, negative if over-represented
# Returns 0 if variables not yet initialized (first action by this country)

# Mining diversity pressure: (expected - actual) / expected × 5
eul_mining_diversity = {
	if = {
		limit = { NOT = { has_variable = eul_total_weighted } }
		value = 0
	}
	else = {
		value = var:eul_total_weighted
		divide = 6
		subtract = var:eul_mining_count
		multiply = 6
		divide = { value = var:eul_total_weighted min = 1 }
		multiply = 5
	}
}

# ============================================
# MARKET SUPPLY/DEMAND PRESSURE (for AI weights)
# ============================================
# These calculate (demand/supply - 1) × scale
# Positive = shortage (demand > supply), encourages production
# Negative = surplus (supply > demand), discourages production
# Uses min = 0.1 on supply to avoid divide-by-zero
# Clamped to ±1 before scaling to prevent extreme swings
# Scale factor of 5 gives max ±5 AI weight impact

# Mining: Iron market pressure (called from market scope)
eul_iron_market_pressure = {
	value = "goods_demand_in_market(goods:iron)"
	divide = {
		value = "goods_supply_in_market(goods:iron)"
		min = 0.1
	}
	subtract = 1
	max = 1
	min = -1
	multiply = 5
}
