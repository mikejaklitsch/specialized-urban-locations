# Production caps: driven by development + population.
# Higher tiers scale harder with development.

REPLACE:guild_max_level = {
	add = { desc = "BUILDING_LEVEL_BASE" value = 1 }
	add = { desc = "BUILDING_LEVEL_DEVELOPMENT" value = development multiply = 0.1 }
	add = { desc = "BUILDING_LEVEL_POPULATION" value = population multiply = 0.02 }
}

# Extraction cap: driven by inverse development + population.

REPLACE:rural_building_cap = {
	add = { desc = "BUILDING_LEVEL_BASE" value = 1 }
	add = {
		desc = "SUL_INVERSE_DEVELOPMENT"
		value = 100
		subtract = development
		multiply = 0.1
	}
	add = { desc = "BUILDING_LEVEL_POPULATION" value = population multiply = 0.02 }
}

REPLACE:irrigant_cap = {
	add = { desc = "BUILDING_LEVEL_BASE" value = 2 }
	add = {
		desc = "SUL_INVERSE_DEVELOPMENT"
		value = 100
		subtract = development
		multiply = 0.1
	}
	add = { desc = "BUILDING_LEVEL_POPULATION" value = population multiply = 0.02 }
	add = { desc = "BUILDING_LEVEL_FROM_OWNER_MODIFIERS" add = owner.modifier:irrigant_cap_level }
}

# Granary: driven by population + location rank tier.

sul_granary_max_level = {
	add = { desc = "BUILDING_LEVEL_BASE" value = 1 }
	add = { desc = "BUILDING_LEVEL_POPULATION" value = population multiply = 0.015 }
	if = {
		limit = { modifier:sul_is_city_rank_tier = yes }
		add = { desc = "SUL_RANK_TIER_CITY" value = 4 }
	}
	else_if = {
		limit = { modifier:sul_is_town_rank_tier = yes }
		add = { desc = "SUL_RANK_TIER_TOWN" value = 2 }
	}
}

sul_rural_granary_max_level = {
	add = { desc = "BUILDING_LEVEL_BASE" value = 1 }
	add = { desc = "BUILDING_LEVEL_POPULATION" value = population multiply = 0.008 }
}

# Wrappers referenced by building_types files.

sul_extraction_max_level = {
	value = rural_building_cap
}

sul_production_max_level = {
	value = guild_max_level
}
