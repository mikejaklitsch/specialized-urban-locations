# Building caps: production scales with development, extraction inverse.
# Dev term scaled by min(pop/250, 1) — full effect at 250 pops.
# Production = dev × multiplier × min(pop/250, 1) + rank + pop.
# Extraction = (100 - dev) × 0.25 × min(pop/250, 1) + rank + pop.
# Rank adds flat bonus: +2 town, +5 city (shared by both).

REPLACE:guild_max_level = {
	add = {
		desc = "SUL_CAP_DEVELOPMENT"
		value = development
		multiply = 0.25

		multiply = {
			value = population
			divide = 250
			max = 1
		}
	}

	if = { limit = { modifier:sul_is_city_rank_tier = yes } add = { desc = "SUL_CAP_RANK" value = 5 } }
	else_if = { limit = { modifier:sul_is_town_rank_tier = yes } add = { desc = "SUL_CAP_RANK" value = 2 } }

	add = {
		desc = "SUL_CAP_POPULATION"
		value = population
		multiply = 0.02
	}

	min = 0
}
