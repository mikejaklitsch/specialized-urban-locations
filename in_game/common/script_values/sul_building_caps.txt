sul_efficiency_building_caps = {
	if = {
		limit = {
			location_rank = location_rank:rural_settlement
		}
		value = 1
	}
	else = {
		value = max_rgo_workers
	}
}

# Flat bonus for guild buildings from specialization
REPLACE:guild_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	add = {
		desc = "BUILDING_LEVEL_DEVELOPMENT"
		value = development
		multiply = 0.1
	}
	add = {
		desc = "BUILDING_LEVEL_POPULATION"
		value = population
		multiply = 0.05
	}
	if = {
		limit = {
			location_rank ?= location_rank:city
		}
		add = {
			desc = "BUILDING_LEVEL_IS_CITY"
			value = 5
		}
	}
	if = {
		limit = {
			location_rank ?= location_rank:rural_settlement
			has_variable = extra_levels_clothmaker_variable
		}
		add = {
			desc = "[ShowMissionTaskName('mission_cloth_makers')]"
			value = 2
		}
	}
	# Flat bonus from specialization: +5 towns, +10 cities
	if = {
		limit = { sul_is_mining_town = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_mining_town_specialization" value = 5 }
	}
	else_if = {
		limit = { sul_is_farming_town = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_farming_town_specialization" value = 5 }
	}
	else_if = {
		limit = { sul_is_gathering_town = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_gathering_town_specialization" value = 5 }
	}
	else_if = {
		limit = { sul_is_woodland_town = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_woodland_town_specialization" value = 5 }
	}
	else_if = {
		limit = { sul_is_commercial_town = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_commercial_town_specialization" value = 5 }
	}
	else_if = {
		limit = { sul_is_mining_city = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_mining_city_specialization" value = 10 }
	}
	else_if = {
		limit = { sul_is_farming_city = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_farming_city_specialization" value = 10 }
	}
	else_if = {
		limit = { sul_is_gathering_city = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_gathering_city_specialization" value = 10 }
	}
	else_if = {
		limit = { sul_is_woodland_city = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_woodland_city_specialization" value = 10 }
	}
	else_if = {
		limit = { sul_is_commercial_city = yes }
		add = { desc = "STATIC_MODIFIER_NAME_sul_commercial_city_specialization" value = 10 }
	}
}
