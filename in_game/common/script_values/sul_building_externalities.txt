# =============================================================================
# BASELINE COMPONENTS (for tooltip breakdown)
# =============================================================================
# These show just the baseline penalty from building_levels static modifier

sul_total_building_levels = {
	value = total_building_levels
}

# Effective building levels = sum of building levels weighted by employment ratio
# Formula: (employed_workers / max_workers) * building_level for each building
sul_effective_building_levels = {
	every_buildings_in_location = {
		add = {
			value = building_employed_amount
			multiply = building_level
			divide = building_employment_size_amount
		}
	}
}

# FOOD MODIFIER FROM BUILDINGS (location scope)
# -----------------------------------------------------------------------------
# Sum of local_monthly_food_modifier from all buildings at this location
# Includes baseline from building_levels static modifier (-0.02 per level)
# Uses effective building levels (employment-weighted) to match game calculations
sul_building_food_modifier = {
	# Baseline: -0.02 per effective building level (from building_levels static modifier)
	add = { value = sul_effective_building_levels multiply = -0.02 }

	# Building contributions use effective levels via iterator
	every_buildings_in_location = {
		# Force Multipliers (+0.10)
		if = {
			limit = {
				OR = {
					building_type = building_type:irrigation_systems
					building_type = building_type:fishing_village
					building_type = building_type:farming_village
				}
			}
			add = { value = building_employed_amount multiply = building_level divide = building_employment_size_amount multiply = 0.10 }
		}
		# Farming buildings (+0.05)
		else_if = {
			limit = {
				OR = {
					building_type = building_type:fruit_orchard
					building_type = building_type:sheep_farms
				}
			}
			add = { value = building_employed_amount multiply = building_level divide = building_employment_size_amount multiply = 0.05 }
		}
		# Food-adjacent buildings (+0.02 = net neutral with baseline)
		else_if = {
			limit = {
				OR = {
					building_type = building_type:windmill
					building_type = building_type:granary
					building_type = building_type:elephant_hunting_grounds
					building_type = building_type:tannery
					building_type = building_type:tanning_workshop
					building_type = building_type:tannery_manufactory
					building_type = building_type:tannery_mill
					building_type = building_type:brewery
					building_type = building_type:beer_workshop
					building_type = building_type:brewery_manufactory
					building_type = building_type:brewery_mill
					building_type = building_type:distillers_guild
					building_type = building_type:distillers_workshop
					building_type = building_type:distiller_manufactory
					building_type = building_type:distiller_mill
					building_type = building_type:winery
					building_type = building_type:winery_manufactory
				}
			}
			add = { value = building_employed_amount multiply = building_level divide = building_employment_size_amount multiply = 0.02 }
		}
		# All other buildings get no food bonus - just baseline penalty
	}
}
