# =============================================================================
# DEVELOPMENT TOOLTIP - Categorized Source Values (location scope)
# =============================================================================
# Script values that mirror static modifier contributions so the tooltip
# can group sources into Growth vs Decay sections.
#
# VALUES MUST STAY IN SYNC with the flat local_monthly_development injected
# in sul_location_modifier.txt, sul_vegetation.txt, sul_topography.txt,
# sul_climates.txt, and sul_building_overrides.txt.
#
# "Other" is computed as a catch-all: engine total minus known categories.
# This captures trade imports/exports, war modifiers, devastation,
# cheap/expensive food, building-specific bonuses, burgher literacy,
# starvation/overpopulation, and event modifiers.
# =============================================================================

# --- GRAND TOTAL (engine static modifiers + scripted delta) ---

sul_dev_total = {
	add = modifier:local_monthly_development
	add = sul_development_delta
}

# --- GROWTH SOURCES (always >= 0) ---

# Infrastructure: building levels drive development
# Mirrors: building_levels { local_monthly_development = 0.0012 }

sul_dev_infrastructure = {
	value = total_building_levels
	multiply = 0.0012
}

# Geography: permanent positive features
# Mirrors: coastal +0.01, river_flowing_through +0.01, farmland +0.01

sul_dev_geography = {
	value = 0
	if = {
		limit = { is_coastal = yes }
		add = 0.01
	}
	if = {
		limit = { has_river = yes }
		add = 0.01
	}
	if = {
		limit = { vegetation = farmland }
		add = 0.01
	}
}

# Roads: connectivity bonus
# Mirrors: has_road +0.005, has_road_connected_to_capital +0.025

sul_dev_roads = {
	value = 0
	if = {
		limit = { num_roads > 0 }
		add = 0.005
	}
	if = {
		limit = { has_road_to_capital = yes }
		add = 0.025
	}
}

# sul_dev_wealth_growth — defined in sul_development.txt (0 to +0.08)
# sul_dev_upper_class_growth — defined in sul_development.txt (0 to +0.05)
# sul_dev_institution_growth — defined in sul_development.txt (0 to +0.048)

# --- DECAY SOURCES (always <= 0) ---

# Population pressure: more people strain infrastructure
# Mirrors: total_population { local_monthly_development = -0.0002 }

sul_dev_pop_pressure = {
	value = population
	multiply = -0.0002
}

# Terrain: vegetation, topography, and climate penalties
# Mirrors flat values from sul_vegetation.txt, sul_topography.txt, sul_climates.txt

sul_dev_terrain = {
	value = 0
	# Vegetation
	if = {
		limit = { vegetation = desert }
		subtract = 0.025
	}
	if = {
		limit = { vegetation = woods }
		subtract = 0.005
	}
	if = {
		limit = { vegetation = forest }
		subtract = 0.01
	}
	if = {
		limit = { vegetation = jungle }
		subtract = 0.025
	}

	# Topography
	if = {
		limit = { topography = mountains }
		subtract = 0.025
	}
	if = {
		limit = { topography = hills }
		subtract = 0.01
	}
	if = {
		limit = { topography = plateau }
		subtract = 0.005
	}
	if = {
		limit = { topography = wetlands }
		subtract = 0.01
	}

	# Climate
	if = {
		limit = { climate = tropical }
		subtract = 0.005
	}
	if = {
		limit = { climate = arctic }
		subtract = 0.01
	}
}

# Development equilibrium: higher development = more decay pressure
# Mirrors: development { local_monthly_development = -0.001 }
# At 50 dev: -0.05/mo (1×), at 100 dev: -0.10/mo (2×)

sul_dev_equilibrium = {
	value = development
	multiply = -0.001
}

# --- MIXED SOURCES (can be positive or negative) ---

# Control: above 50% = growth, below 50% = decay
# Mirrors: control_50 { local_monthly_development = 0.1 }
# Engine scales by (control - 0.5), so value = 0.1 * (control - 0.5)

sul_dev_control = {
	value = local_control
	subtract = 0.5
	multiply = 0.1
}

# Other: catch-all for everything not explicitly categorized
# Includes: trade imports/exports, war modifiers, devastation,
# cheap/expensive food, building bonuses, burgher literacy, event modifiers

sul_dev_other = {
	value = modifier:local_monthly_development
	subtract = sul_dev_infrastructure
	subtract = sul_dev_geography
	subtract = sul_dev_roads
	subtract = sul_dev_pop_pressure
	subtract = sul_dev_terrain
	subtract = sul_dev_equilibrium
	subtract = sul_dev_control
}
