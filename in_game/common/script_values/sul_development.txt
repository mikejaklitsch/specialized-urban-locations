# =============================================================================
# DYNAMIC DEVELOPMENT SYSTEM - Scripted Components
# =============================================================================
# Wealth, upper class, and institutions need scripting (no engine scaler).
# All other development drivers use engine-scaled static modifiers:
#   building_levels  → local_monthly_development (growth from infrastructure)
#   total_population → local_monthly_development (decay from pop pressure)
#   inverse_control  → local_monthly_development (decay from lawlessness)
#   prosperity       → local_monthly_development (food surplus, vanilla +0.004)
#   overpopulation   → local_monthly_development (binary brake)
#   province_starving→ local_monthly_development (binary brake)
#   geography        → flat injections in vegetation/topography/climate
#   war              → flat injections in siege/looted/blockade modifiers
# =============================================================================

# --- MACROS ---
@sul_dev_wealth_ceiling = 0.05       # max monthly dev from wealth (1× baseline)
@sul_dev_wealth_k = 75              # GDP/capita at half-max growth
@sul_dev_upper_class_scale = 0.15    # multiplied by upper class pop ratio
@sul_dev_upper_class_cap = 0.05      # max monthly dev from upper class
@sul_dev_institution_bonus = 0.004   # per institution present at location

# =============================================================================
# WEALTH GROWTH: S-curve based on GDP per capita
# sul_gdp_per_capita cached in sul_update_location_gdp
# Formula: ceiling × gdp_pc / (gdp_pc + k)
# At GDP/cap=75: +0.025. At GDP/cap=200: +0.036.
# Returns 0 to +0.05
# =============================================================================

sul_dev_wealth_growth = {
	value = 0
	if = {
		limit = { has_variable = sul_gdp_per_capita var:sul_gdp_per_capita > 0 }

		add = {
			value = var:sul_gdp_per_capita
			divide = { value = var:sul_gdp_per_capita add = @sul_dev_wealth_k }
			multiply = @sul_dev_wealth_ceiling
		}
	}

	min = 0
	max = @sul_dev_wealth_ceiling
}

# =============================================================================
# UPPER CLASS GROWTH: linear from actual upper class population %
# sul_upper_class_ratio cached in sul_update_location_gdp
# (nobles + clergy + burghers) / total_population
# Typical range 5-30%. At 33% → +0.05 (cap).
# Returns 0 to +0.05
# =============================================================================

sul_dev_upper_class_growth = {
	value = 0
	if = {
		limit = { has_variable = sul_upper_class_ratio var:sul_upper_class_ratio > 0 }

		add = {
			value = var:sul_upper_class_ratio
			multiply = @sul_dev_upper_class_scale
		}
	}

	min = 0
	max = @sul_dev_upper_class_cap
}

# =============================================================================
# INSTITUTION GROWTH: +0.004 per institution present at this location
# Rewards institutional spread — more advanced locations develop faster.
# Early game (2-3 institutions): +0.008-0.012
# Mid game (6-7): +0.024-0.028
# Late game (10-12): +0.040-0.048
# =============================================================================

sul_dev_institution_growth = {
	value = 0
	if = {
		limit = { has_institution = institution:feudalism }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:legalism }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:meritocracy }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:banking }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:professional_armies }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:printing_press }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:pike_and_shot }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:global_trade }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:artillery_institution }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:scientific_revolution }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:military_revolution }
		add = @sul_dev_institution_bonus
	}
	if = {
		limit = { has_institution = institution:industrialization }
		add = @sul_dev_institution_bonus
	}
}

# =============================================================================
# COMBINED DELTA (location scope)
# =============================================================================
# Scripted components: wealth + upper class + institutions.
# Everything else is engine-driven.
# =============================================================================

sul_development_delta = {
	add = sul_dev_wealth_growth
	add = sul_dev_upper_class_growth
	add = sul_dev_institution_growth
}
