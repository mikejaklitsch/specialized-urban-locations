# =============================================================================
# WPP READER + TIER GATES (location scope, called from pop scope via location.)
# =============================================================================
# sul_read_wpp: raw WPP lookup from variable map
# sul_read_tier_*: max(0, WPP - threshold) per tier, used by pop_demands.txt
#
# Thresholds must match TIER_THRESHOLD in demand_calculator.py
# =============================================================================

@sul_tier_threshold_necessity = 0
@sul_tier_threshold_basic = 0.05
@sul_tier_threshold_common = 0.12
@sul_tier_threshold_upper = 0.25
@sul_tier_threshold_luxury = 0.5
@sul_tier_threshold_exotic = 1.5

sul_read_wpp = {
	value = 0
	if = {
		limit = {
			has_variable_map = sul_wealth_per_pop
			is_key_in_variable_map = { name = sul_wealth_per_pop target = prev.pop_type }
		}

		add = "variable_map(sul_wealth_per_pop|prev.pop_type)"
	}
}

sul_read_tier_necessity = {

	subtract = @sul_tier_threshold_necessity

# Budget pressure: per-estate spending intensity from accumulated gold.
# Location scope (called via location.sul_demand_*) — uses prev to reach pop scope.

sul_budget_pressure = {
	value = 1
	if = {
		limit = {
			prev.pop_type = pop_type:nobles
			owner = { has_variable = sul_budget_pressure_nobles }
		}

		value = owner.var:sul_budget_pressure_nobles
	}
	if = {
		limit = {
			prev.pop_type = pop_type:clergy
			owner = { has_variable = sul_budget_pressure_clergy }
		}

		value = owner.var:sul_budget_pressure_clergy
	}
	if = {
		limit = {
			prev.pop_type = pop_type:burghers
			owner = { has_variable = sul_budget_pressure_burghers }
		}

		value = owner.var:sul_budget_pressure_burghers
	}
	if = {
		limit = {
			OR = {
				prev.pop_type = pop_type:peasants
				prev.pop_type = pop_type:laborers
				prev.pop_type = pop_type:soldiers
			}

			owner = { has_variable = sul_budget_pressure_peasants }
		}

		value = owner.var:sul_budget_pressure_peasants
	}
}

# Granary cap — scales with development and population
# ~3-4 in a small town, ~8-10 in a large city

sul_granary_max_level = {
	add = { desc = "BUILDING_LEVEL_BASE" value = 1 }
	add = {
		desc = "BUILDING_LEVEL_DEVELOPMENT"
		value = development
		multiply = 0.05
	}
	add = {
		desc = "BUILDING_LEVEL_POPULATION"
		value = population
		multiply = 0.01
	}
	if = {
		limit = { location_rank ?= location_rank:city }
		add = { desc = "BUILDING_LEVEL_IS_CITY" value = 2 }
	}
}

# ===================
# RGO MAX LEVEL FORMULA
# ===================
# Explicit script value with desc lines for tooltip display.
# Mirrors the sul_rgo_size static modifier contributions.

sul_rgo_max_level_base = {
	# Local flat: base(2) + dev(0.1) + pop(0.04) + farmland(0.5) + rural(4) + granary(0.05/lvl)
	add = { desc = "SUL_RGO_LEVEL_FLAT" value = modifier:sul_rgo_size }

	# Country flat: advances, laws, privileges, religion, societal values
	if = {
		limit = { has_owner = yes }

		add = {
			desc = "SUL_RGO_LEVEL_COUNTRY"
			value = owner.modifier:global_sul_rgo_size
		}

		if = {
			limit = { location_rank = location_rank:rural_settlement }

			add = {
				desc = "SUL_RGO_LEVEL_RURAL"
				value = owner.modifier:global_sul_rgo_size_in_rural
			}
		}

		if = {
			limit = { is_town_or_city = yes }

			add = {
				desc = "SUL_RGO_LEVEL_URBAN"
				value = owner.modifier:global_sul_rgo_size_in_non_rural
			}
		}
	}

	# Specialization tier bonus
	if = {
		limit = { sul_has_any_specialization = yes }
		if = {
			limit = { location_rank = location_rank:city }
			add = { desc = "SUL_RGO_LEVEL_SPEC_CITY" value = 10 }
		}
		else_if = {
			limit = { is_town_or_city = yes }
			add = { desc = "SUL_RGO_LEVEL_SPEC_TOWN" value = 5 }
		}
		else = { add = { desc = "SUL_RGO_LEVEL_SPEC_RURAL" value = 3 } }
	}
}
