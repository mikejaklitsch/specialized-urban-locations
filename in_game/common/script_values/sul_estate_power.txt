# =============================================================================
# LOCAL TAX BASE SHARE CALCULATIONS
# =============================================================================
# Calculates each estate's tax base share at a location (excluding crown)
# Power = pop_count × tax_weight
# Share = estate_power / total_power
# =============================================================================

# -----------------------------------------------------------------------------
# TAX BASE WEIGHTS (from wiki - Tax Base Share calculation)
# -----------------------------------------------------------------------------
@sul_nobles_weight = 150
@sul_clergy_weight = 25
@sul_burghers_weight = 20
@sul_peasants_weight = 1
@sul_tribes_weight = 0.01

# -----------------------------------------------------------------------------
# INDIVIDUAL ESTATE TAX WEIGHT (location scope)
# -----------------------------------------------------------------------------

sul_local_nobles_power = {
	value = num_pop_type:nobles
	multiply = @sul_nobles_weight
}

sul_local_clergy_power = {
	value = num_pop_type:clergy
	multiply = @sul_clergy_weight
}

sul_local_burghers_power = {
	value = num_pop_type:burghers
	multiply = @sul_burghers_weight
}

sul_local_peasants_power = {
	value = num_pop_type:peasants
	multiply = @sul_peasants_weight
}

sul_local_laborers_power = {
	# Laborers use peasants estate
	value = num_pop_type:laborers
	multiply = @sul_peasants_weight
}

sul_local_soldiers_power = {
	# Soldiers use peasants estate
	value = num_pop_type:soldiers
	multiply = @sul_peasants_weight
}

sul_local_tribes_power = {
	value = num_pop_type:tribesmen
	multiply = @sul_tribes_weight
}

sul_local_slaves_power = {
	# Slaves use peasants estate
	value = num_pop_type:slaves
	multiply = @sul_peasants_weight
}

# -----------------------------------------------------------------------------
# AGGREGATED ESTATE POWER (location scope)
# -----------------------------------------------------------------------------
# Nobles estate includes only nobles pop type
sul_local_nobles_estate_total = {
	add = sul_local_nobles_power
}

# Peasants estate includes peasants + laborers + soldiers + slaves
sul_local_peasants_estate_total = {
	add = sul_local_peasants_power
	add = sul_local_laborers_power
	add = sul_local_soldiers_power
	add = sul_local_slaves_power
}

# -----------------------------------------------------------------------------
# TOTAL NON-CROWN POWER (location scope)
# -----------------------------------------------------------------------------
sul_local_total_estate_power = {
	add = sul_local_nobles_estate_total
	add = sul_local_clergy_power
	add = sul_local_burghers_power
	add = sul_local_peasants_estate_total
	add = sul_local_tribes_power
	min = 0.001  # prevent division by zero
}

# -----------------------------------------------------------------------------
# POWER SHARES (location scope) - returns 0.0 to 1.0
# -----------------------------------------------------------------------------
sul_local_nobles_share = {
	value = sul_local_nobles_estate_total
	divide = sul_local_total_estate_power
}

sul_local_clergy_share = {
	value = sul_local_clergy_power
	divide = sul_local_total_estate_power
}

sul_local_burghers_share = {
	value = sul_local_burghers_power
	divide = sul_local_total_estate_power
}

sul_local_peasants_share = {
	value = sul_local_peasants_estate_total
	divide = sul_local_total_estate_power
}

sul_local_tribes_share = {
	value = sul_local_tribes_power
	divide = sul_local_total_estate_power
}
