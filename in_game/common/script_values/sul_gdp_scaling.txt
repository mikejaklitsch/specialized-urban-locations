# =============================================================================
# GDP PER CAPITA SCALING FOR POP DEMAND
# =============================================================================
#
# Reads pre-cached GDP per capita values from location variables.
# Caching happens monthly in sul_cache_gdp_per_capita (called from GDP update).
#
# Raw scale (sul_gdp_per_capita_scale) ranges 1.0 to ~5.0 by pop type.
# Category wrappers apply elasticity dampening:
#   result = 1 + (raw - 1) * elasticity
#
# Elasticity controls how much wealth affects demand for each category:
#   necessity = 0.2  (inelastic — everyone needs beer regardless of wealth)
#   common    = 0.5  (moderate — wealth helps but isn't required)
#   upper     = 0.8  (elastic — mostly wealth-driven consumption)
#   luxury    = 1.0  (fully elastic — GDP multiplier passes through unchanged)
#
# Example: noble with raw multiplier 3.5
#   necessity: 1 + (3.5 - 1) * 0.2 = 1.50  (modest boost)
#   luxury:    1 + (3.5 - 1) * 1.0 = 3.50  (full boost)
# =============================================================================

sul_gdp_per_capita_scale = {
	value = 1
	if = {
		limit = {
			pop_type = pop_type:nobles
			exists = location
			location = { has_variable = sul_gdp_pc_nobles }
		}

		multiply = location.var:sul_gdp_pc_nobles
	}
	else_if = {
		limit = {
			pop_type = pop_type:clergy
			exists = location
			location = { has_variable = sul_gdp_pc_clergy }
		}

		multiply = location.var:sul_gdp_pc_clergy
	}
	else_if = {
		limit = {
			pop_type = pop_type:burghers
			exists = location
			location = { has_variable = sul_gdp_pc_burghers }
		}

		multiply = location.var:sul_gdp_pc_burghers
	}
	else_if = {
		limit = {
			pop_type = pop_type:soldiers
			exists = location
			location = { has_variable = sul_gdp_pc_soldiers }
		}

		multiply = location.var:sul_gdp_pc_soldiers
	}
	else_if = {
		limit = {
			pop_type = pop_type:laborers
			exists = location
			location = { has_variable = sul_gdp_pc_laborers }
		}

		multiply = location.var:sul_gdp_pc_laborers
	}
	else_if = {
		limit = {
			pop_type = pop_type:peasants
			exists = location
			location = { has_variable = sul_gdp_pc_peasants }
		}

		multiply = location.var:sul_gdp_pc_peasants
	}
	else_if = {
		limit = {
			pop_type = pop_type:tribesmen
			exists = location
			location = { has_variable = sul_gdp_pc_tribesmen }
		}

		multiply = location.var:sul_gdp_pc_tribesmen
	}
	else_if = {
		limit = {
			pop_type = pop_type:slaves
			exists = location
			location = { has_variable = sul_gdp_pc_slaves }
		}

		multiply = location.var:sul_gdp_pc_slaves
	}
}

# =============================================================================
# ELASTICITY-DAMPENED CATEGORY WRAPPERS
# =============================================================================
# Formula: 1 + (raw - 1) * elasticity
# Each reads sul_gdp_per_capita_scale then dampens by category.
# =============================================================================

# Necessity: elasticity = 0.2 (inelastic — beer, cloth, food, fuel)

sul_gdp_scale_necessity = {
	value = sul_gdp_per_capita_scale
	subtract = 1
	multiply = 0.2
	add = 1
}

# Common: elasticity = 0.5 (moderate — furniture, leather, tools, weapons)

sul_gdp_scale_common = {
	value = sul_gdp_per_capita_scale
	subtract = 1
	multiply = 0.5
	add = 1
}

# Upper: elasticity = 0.8 (elastic — wine, books, fine cloth, colonial goods)

sul_gdp_scale_upper = {
	value = sul_gdp_per_capita_scale
	subtract = 1
	multiply = 0.8
	add = 1
}

# Luxury: elasticity = 1.0 (fully elastic — jewelry, spices, precious goods)

sul_gdp_scale_luxury = {
	value = sul_gdp_per_capita_scale
	subtract = 1
	multiply = 1.0
	add = 1
}
