# =============================================================================
# GDP PER CAPITA SCALING FOR POP DEMAND
# =============================================================================
#
# Reads pre-cached GDP per capita values from location variables.
# Caching happens monthly in sul_cache_gdp_per_capita (called from GDP update).
# =============================================================================

sul_gdp_per_capita_scale = {
	value = 1
	if = {
		limit = { exists = location location = { has_variable_map = sul_gdp_pc } }
		multiply = { value = "location.variable_map(sul_gdp_pc|this.pop_type)" }
	}
}
