# The Gold Standard - Minting Efficiency
# Market-price-based minting profitability system.
# Scope: country (all values)

@sul_minting_vanilla_rate = 25
@sul_minting_efficiency_floor = -1.0001
@sul_minting_inflation_factor = 0.005

# ---- Cached Variables ----
# Set by sul_minting_rebuild_minting_goods on init and policy change.

sul_minting_goods_count = {
	value = 0

	if = { limit = { has_variable = sul_minting_goods_count } add = var:sul_minting_goods_count }

	min = 1
}

sul_minting_target_value = {
	value = 0

	if = { limit = { has_variable = sul_minting_target_value } add = var:sul_minting_target_value }
}

# ---- Base Monetization ----
# Derived from the economy: wages capture domestic monetary circulation,
# trade share captures commercial activity.

sul_minting_monetization_base = {
	value = 0

	if = { limit = { has_variable = sul_monthly_wages_nobles } add = var:sul_monthly_wages_nobles }

	if = { limit = { has_variable = sul_monthly_wages_clergy } add = var:sul_monthly_wages_clergy }

	if = { limit = { has_variable = sul_monthly_wages_burghers } add = var:sul_monthly_wages_burghers }

	if = { limit = { has_variable = sul_monthly_wages_peasants } add = var:sul_monthly_wages_peasants }

	divide = { value = total_population min = 1 }
}

sul_minting_trade_monetization_value = {
	value = monthly_trade_income

	divide = { value = monthly_income_trade_and_tax min = 1 }

	min = 0
}

# ---- Market Cost ----
# All consumers now read var:sul_minting_market_cost (cached monthly/yearly).
# The 70-block live computation has been removed for performance.

# ---- Coin Value Chain ----

# Monetization premium: face value of coins above raw metal content.
# Base 100%, modified by the same sources as minting_inflation_threshold.

sul_minting_mint_monetization_value = {
	value = modifier:sul_minting_monetization
}

# Face value before debasement: target_value × (1 + monetization)

sul_minting_mint_output_value = {
	value = sul_minting_target_value

	multiply = { value = 1 add = sul_minting_mint_monetization_value }
}

# Face value after debasement: output × (1 + d)

sul_minting_debased_output_value = {
	value = sul_minting_mint_output_value

	multiply = {
		value = 1
		if = { limit = { has_variable = sul_minting_debasement_level } add = var:sul_minting_debasement_level }
	}
}

# ---- Profit ----

# Profit per complete set of minting goods

sul_minting_mint_profit_per_set = {
	value = sul_minting_debased_output_value
	subtract = var:sul_minting_market_cost
}

# Average profit per unit of goods demanded

sul_minting_mint_profit_per_good = {
	value = sul_minting_mint_profit_per_set
	divide = sul_minting_goods_count
}

# ---- Demand ----

# Demand for each individual good type: slider × tax_base / 100

sul_minting_demand_per_type = {
	value = slider_minting_value
	multiply = country_tax_base
	divide = 100
}

# Total demand across all good types

sul_minting_goods_demand = {
	value = slider_minting_value
	multiply = country_tax_base
	divide = 100
	multiply = sul_minting_goods_count
}

# ---- Total Income ----

# Final monthly minting income: demand × profit_per_unit

sul_minting_total_income = {
	value = sul_minting_goods_demand
	multiply = sul_minting_mint_profit_per_good
}

# ---- Minting Efficiency (E) ----
# Cancels vanilla's fixed income rate, replaces with profit-based income.
# Vanilla pays 25 per demand unit. We set E so 25 × (1+E) = profit.
# Formula: (debased_output - market_cost) / 25 - 1

sul_minting_efficiency_from_cached_cost = {
	value = sul_minting_debased_output_value
	subtract = var:sul_minting_market_cost
	divide = @sul_minting_vanilla_rate
	subtract = 1
	min = @sul_minting_efficiency_floor
}

# Player: live formula from cached market cost.
# AI: monthly-cached variable.

sul_minting_cached_minting_efficiency = {
	value = 0

	if = {
		limit = { is_ai = yes }
		if = { limit = { has_variable = sul_minting_efficiency } add = var:sul_minting_efficiency }
	}
	else = { add = sul_minting_efficiency_from_cached_cost }
}

# (1 + E): fraction of vanilla income that remains

sul_minting_modifier_scalar = {
	value = 1
	add = sul_minting_cached_minting_efficiency
}

# ---- Inflation Adjustment ----
# Unified inflation formula: income uses (1+d), inflation uses (1 + penalty×d)
# where penalty = 2×(1+inflation) — Gresham's Law: debasement is 2× as
# inflationary as legitimate monetization, worsening with accumulated inflation.

sul_minting_vanilla_minting_inflation = {
	value = 0

	if = {
		limit = { slider_minting_value > modifier:minting_inflation_threshold }

		add = { value = slider_minting_value subtract = modifier:minting_inflation_threshold }

		multiply = @sul_minting_inflation_factor
	}
}

# ---- Debasement Inflation Penalty (Gresham's Law) ----
# Each point of debasement is 2x as inflationary as legitimate monetization,
# worsening with accumulated inflation. Confidence erosion accelerates.

sul_minting_debasement_inflation_penalty = {
	value = 2

	multiply = { value = 1 add = inflation }
}

# Inflation-weighted face value: uses heavier debasement multiplier.
# Income uses (1+d), inflation uses (1 + penalty×d).

sul_minting_inflation_output_value = {
	value = sul_minting_mint_output_value

	multiply = {
		value = 1
		if = {
			limit = { has_variable = sul_minting_debasement_level }
			add = { value = var:sul_minting_debasement_level multiply = sul_minting_debasement_inflation_penalty }
		}
	}
}

sul_minting_inflation_scalar = {
	value = sul_minting_inflation_output_value
	subtract = var:sul_minting_market_cost
	divide = @sul_minting_vanilla_rate
	min = 0
}

sul_minting_unified_inflation_adjustment = {
	value = sul_minting_vanilla_minting_inflation

	multiply = { value = sul_minting_inflation_scalar subtract = 1 }
}

# Display value: total minting inflation = vanilla × inflation_scalar

sul_minting_total_minting_inflation = {
	value = sul_minting_vanilla_minting_inflation
	multiply = sul_minting_inflation_scalar
}

# ---- AI Modifier Sizes ----
# Used by add_country_modifier { size = ... }, which takes the literal
# multiplier on the modifier's base value (no implicit +1).

sul_minting_ai_minting_size = {
	value = var:sul_minting_efficiency
}

sul_minting_ai_inflation_size = {
	value = sul_minting_unified_inflation_adjustment

	if = {
		limit = {
			has_variable = sul_minting_rebasement_level
			var:sul_minting_rebasement_level > 0
			inflation > 0
		}
		add = sul_minting_rebasement_deflation_effect
	}
}

sul_minting_ai_rebasement_size = {
	value = sul_minting_rebasement_cost
}
