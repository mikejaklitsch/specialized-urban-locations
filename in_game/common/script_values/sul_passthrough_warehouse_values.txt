# Passthrough warehouse stockpile math.
#
# Called from market scope (inside sul_passthrough_per_good's
# scope:current_market block). Uses the workshop mod's script value
# chaining pattern: helpers that read modifiers/stockpile from the right
# scope, composed into the main formula.
#
# Scope chain paths like scope:X.location.modifier:Y DO NOT WORK in EU5 -
# only single-hop prefixes (location.X, owner.X, prev.X) parse correctly.
# That's why an intermediate location-scoped helper is needed for the
# modifier reads.
#
# None of these read goods_demand_in_market - that read is stale after
# remove_temporary_demand inside the same effect execution.

# Location-scoped helper: returns the location's max stockpile capacity.
# Referenced via location.sul_passthrough_local_capacity from market scope.

sul_passthrough_local_capacity = {
	value = "modifier:maximum_stockpile_capacity"
}

sul_passthrough_fill_ratio = {
	value = "stockpile_in_market(scope:current_good)"
	divide = location.sul_passthrough_local_capacity
}

sul_passthrough_upper_value = {
	value = local_var:sul_pt_cap
	multiply = 0.05
}

sul_passthrough_lower_value = {
	value = local_var:sul_pt_cap
	multiply = 0.049
}

sul_passthrough_upper_overflow = {
	value = local_var:sul_pt_fill
	subtract = 0.05
	multiply = 100
	min = 0
}

# HIGH branch demand scale. Piecewise linear, evaluated via if/else_if so
# only ONE tier's formula fires per call (checked highest-to-lowest).
# Each tier is a single linear function:
#   start_value + (fill - tier_start) × slope
# where start_value matches the previous tier's endpoint to keep the curve
# continuous without summing all prior contributions.
#
# Tier boundaries (start → end value as fill goes through the range):
#   Base (≤ 50%):    workshop's gentle formula, 0 → 12
#   Tier 1 (50-75%): 12 → 112  (slope 400)
#   Tier 2 (75-95%): 112 → 412 (slope 1500)
#   Tier 3 (> 95%):  412 → 812 (slope 8000)
#
# Max demand reduction at 100% fill = -812 (with _stockpile template -1).
# Tune slopes (400 / 1500 / 8000) and starting offsets (12 / 112 / 412)
# together to keep the curve continuous.

# Reads local_var:sul_pt_fill (pre-computed per-good in sul_passthrough_per_good)

sul_passthrough_supply_effect = {
	value = 0

	if = {
		limit = { local_var:sul_pt_fill > 0.95 }

		add = {
			value = local_var:sul_pt_fill
			subtract = 0.95
			multiply = 8000
			add = 412
		}
	}

	else_if = {
		limit = { local_var:sul_pt_fill > 0.75 }

		add = {
			value = local_var:sul_pt_fill
			subtract = 0.75
			multiply = 1500
			add = 112
		}
	}

	else_if = {
		limit = { local_var:sul_pt_fill > 0.5 }

		add = {
			value = local_var:sul_pt_fill
			subtract = 0.5
			multiply = 400
			add = 12
		}
	}

	else = {
		add = {
			value = "stockpile_in_market(scope:current_good)"
			subtract = sul_passthrough_upper_value
			divide = 18
			min = 1
			divide = 95
			multiply = sul_passthrough_upper_overflow
		}
	}
}

# Warehouse efficiency correction - matches the warehouse building's net
# output per tick (+0.03 base + local/global production_efficiency).
# The paired _oversupply demand uses this scale to cancel the warehouse's
# contribution at the flow layer, so the warehouse inflates the supply
# reading without actually growing the stockpile.

sul_passthrough_warehouse_correction = {
	value = 0.03
	add = location.modifier:local_production_efficiency
	add = location.owner.modifier:global_production_efficiency
}

# LOW branch: price-up demand scale. Bounded small negative so it nudges
# the price up without spiking.

sul_passthrough_low_supply_effect = {
	value = -0.1

	multiply = {
		value = 0.05
		subtract = local_var:sul_pt_fill
		min = 0
		max = 0.05
	}

	multiply = 20
	max = -0.01
	min = -0.5
}

sul_passthrough_low_stockpile_effect = {
	value = 0.05
	subtract = local_var:sul_pt_fill
	min = 0
	multiply = local_var:sul_pt_cap
	multiply = 0.1
}
