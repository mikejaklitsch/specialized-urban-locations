# Per pop-type per tier breakdown values for the GUI budget tooltip.
# Returns max(0, wpp - threshold) × num_pops. The GUI then multiplies by
# demand_add (from the sul_da_* map) to get per-good quantity.
#
# Evaluated on tooltip hover only - this 36-value table never runs in the
# economy hot path. The hot path uses sul_demand_* (in sul_pop_demand_wpp.txt)
# called from pop_demands.txt.
#
# Tier thresholds redeclared here because @constants are file-local.

@sul_tier_threshold_necessity = 0
@sul_tier_threshold_basic = 0.05
@sul_tier_threshold_common = 0.12
@sul_tier_threshold_upper = 0.25
@sul_tier_threshold_luxury = 0.5
@sul_tier_threshold_exotic = 1.5

# --- NOBLES ---

sul_nobles_necessity = {
	value = {
		value = "variable_map(sul_wealth_per_pop|pop_type:nobles)"
		subtract = @sul_tier_threshold_necessity
		min = 0
	}

	multiply = num_pop_type:nobles
}
