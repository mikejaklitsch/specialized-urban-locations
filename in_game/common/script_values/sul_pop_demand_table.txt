# Per pop-type per tier breakdown values for the GUI budget tooltip.
#
# Each value = demand_base × weight (constant tiers) or
# demand_base × WPP × weight (growth tiers).
#
# These are per-pop values matching the demand column in the WPP tooltip.
# Evaluated on tooltip hover only — never in the economy hot path.

# Tier weights (must match sul_pop_demand_wpp.txt)
@sul_w_necessity = 1.65
@sul_w_basic = 0.90
@sul_w_common = 0.45
@sul_w_upper = 0.21
@sul_w_luxury = 0.16
@sul_w_exotic = 0.10

sul_nobles_necessity = {
