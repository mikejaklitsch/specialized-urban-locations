# Per pop-type per tier breakdown values for the GUI budget tooltip.
#
# Each value = demand_base × weight × num_pops (constant tiers) or
# demand_base × WPP × weight × num_pops (growth tiers).
#
# These match the continuous percentage model in sul_pop_demand_wpp.txt
# exactly — constant-weight tiers saturate, growth-weight tiers scale
# with WPP. The result is the location's total demand for that
# pop type at that tier, before per-good splitting.
#
# Evaluated on tooltip hover only — never in the economy hot path.

# Tier weights (must match sul_pop_demand_wpp.txt)
@sul_w_necessity = 0.55
@sul_w_basic = 0.30
@sul_w_common = 0.15
@sul_w_upper = 0.45
@sul_w_luxury = 0.35
@sul_w_exotic = 0.20

# --- NOBLES ---

sul_nobles_necessity = {
	value = "variable_map(sul_demand_base|pop_type:nobles)"
	multiply = @sul_w_necessity
	multiply = num_pop_type:nobles
}

sul_nobles_basic = {
	value = "variable_map(sul_demand_base|pop_type:nobles)"
	multiply = @sul_w_basic
	multiply = num_pop_type:nobles
}

sul_nobles_common = {
	value = "variable_map(sul_demand_base|pop_type:nobles)"
	multiply = @sul_w_common
	multiply = num_pop_type:nobles
}

sul_nobles_upper = {
	value = "variable_map(sul_demand_base|pop_type:nobles)"
	multiply = "variable_map(sul_wealth_per_pop|pop_type:nobles)"
	multiply = @sul_w_upper
	multiply = num_pop_type:nobles
}

sul_nobles_luxury = {
	value = "variable_map(sul_demand_base|pop_type:nobles)"
	multiply = "variable_map(sul_wealth_per_pop|pop_type:nobles)"
	multiply = @sul_w_luxury
	multiply = num_pop_type:nobles
}

	value = "variable_map(sul_demand_base|pop_type:nobles)"
