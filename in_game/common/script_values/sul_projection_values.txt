# Power Projection — drag computations
# All values are SIGNED so the auto_modifier coefficient stays positive.
# Negative result = penalty (PP drops), zero = neutral, positive = bonus.
# Scope: country.

# Cultural non-acceptance share, signed negative.
# (accepted_pop / total_pop) - 1
#   all pops in accepted cultures → 0
#   half pops accepted → -0.5
#   no pops accepted → -1.0

sul_projection_culture_drag = {
	value = total_accepted_culture_population
	divide = total_population
	subtract = 1
}

# Religious friction — fraction of pops NOT in the country's own religion.
# Captures both heretics (same group) and heathens (different group), unlike
# heretic_population_fraction which only counts the same-group case.
# Mirrors the cultural drag shape:
#   100% in country religion → 0 (neutral)
#   50% in country religion → -0.5
#   0% in country religion → -1 (max drag)

sul_projection_religion_drag = {
	value = "religion_percentage_in_country(root.religion)"
	subtract = 1
}

# Diplomatic capacity usage ratio — used / total.
# 0 = no slots used, 1.0 = fully committed. Auto_modifier multiplies by -0.25.

sul_debug_diplo_used = {
	value = used_diplomatic_capacity
}

sul_debug_diplo_base = {
	value = modifier:diplomatic_capacity
}

sul_debug_diplo_modifier = {
	value = modifier:diplomatic_capacity_modifier
}

sul_debug_diplo_total = {
	value = modifier:diplomatic_capacity

	multiply = { value = 1 add = modifier:diplomatic_capacity_modifier }
}

# Reads the PP variable. Used by auto_modifier scales_with and GUI display.

sul_power_projection_value = {
	value = var:sul_power_projection
}

# GUI display alias.

sul_power_projection_display = {
	value = var:sul_power_projection
}

# Net monthly change for top bar delta display (contributions + decay).
# Decay is positive when PP is negative (pulls toward 0).

sul_power_projection_monthly_delta = {
	value = modifier:monthly_sul_power_projection

	add = {
		value = var:sul_power_projection
		multiply = -0.05
		multiply = { value = 1 add = { value = complacency multiply = 0.01 } }
	}
}

sul_projection_diplo_usage_ratio = {
	value = used_diplomatic_capacity

	divide = {
		value = modifier:diplomatic_capacity
		multiply = { value = 1 add = modifier:diplomatic_capacity_modifier }
		min = 1
	}

	min = 0
}
