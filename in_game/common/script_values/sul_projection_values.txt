# Power Projection — drag computations
# All values are SIGNED so the auto_modifier coefficient stays positive.
# Negative result = penalty (PP drops), zero = neutral, positive = bonus.
# Scope: country.

# Control-weighted population share, recentered around full control.
# (control_pop / total_pop) - 1
#   ratio 1.0 (full control everywhere) → 0
#   ratio 0.5 (half-controlled) → -0.5
#   ratio 0.0 (uncontrolled) → -1.0
# Multiplied in the auto_modifier by sul_projection_control_drag_constant.

sul_projection_control_drag = {
	value = total_control_scaled_population
	divide = total_population
	subtract = 1
}

# Cultural non-acceptance share, signed negative.
# (accepted_pop / total_pop) - 1
#   all pops in accepted cultures → 0
#   half pops accepted → -0.5
#   no pops accepted → -1.0

sul_projection_culture_drag = {
	value = total_accepted_culture_population
	divide = total_population
	subtract = 1
}

# Stability impact. Symmetric: negative stab penalises, positive stab rewards.
# Final PP contribution = stability × sul_projection_stability_constant.

sul_projection_stability_drag = {
	value = stability
}

# Religious friction — fraction of pops NOT in the country's own religion.
# Captures both heretics (same group) and heathens (different group), unlike
# heretic_population_fraction which only counts the same-group case.
# Mirrors the cultural drag shape:
#   100% in country religion → 0 (neutral)
#   50% in country religion → -0.5
#   0% in country religion → -1 (max drag)

sul_projection_religion_drag = {
	value = "religion_percentage_in_country(root.religion)"
	subtract = 1
}

# Administrative overhead — raw location count.
# More territory = more bureaucratic load regardless of composition.
# Coefficient of -0.05 in the auto_modifier means -1 PP per 20 locations.

sul_projection_size_drag = {
	value = num_locations
}

# Government rank → PP. County = 10, Duchy = 20, Kingdom = 30.

sul_projection_rank_value = {
	value = 0
	if = { limit = { country_rank = country_rank:rank_empire } add = 15 }
	else_if = { limit = { country_rank = country_rank:rank_kingdom } add = 10 }
	else_if = { limit = { country_rank = country_rank:rank_duchy } add = 5 }
}

# Per-subject-type weight applied to that subject's country_strength before
# summing into the overlord's drag total. Evaluated in the subject scope by
# sul_projection_recompute_subject_drag. Placeholder values — every entry
# defaults to 1.0; tune per type without touching the iteration code.

sul_projection_subject_type_weight = {
	value = 0
	if = { limit = { is_subject_type = vassal } add = 1.0 }
	else_if = { limit = { is_subject_type = march } add = 1.0 }
	else_if = { limit = { is_subject_type = appanage } add = 1.0 }
	else_if = { limit = { is_subject_type = dominion } add = 1.0 }
	else_if = { limit = { is_subject_type = fiefdom } add = 1.0 }
	else_if = { limit = { is_subject_type = tributary } add = 1.0 }
	else_if = { limit = { is_subject_type = colonial_nation } add = 1.0 }
	else_if = { limit = { is_subject_type = conquistador } add = 1.0 }
	else_if = { limit = { is_subject_type = trade_company } add = 1.0 }
	else_if = { limit = { is_subject_type = hanseatic_member } add = 1.0 }
	else_if = { limit = { is_subject_type = state_bank } add = 1.0 }
	else_if = { limit = { is_subject_type = samanta } add = 1.0 }
	else_if = { limit = { is_subject_type = tusi } add = 1.0 }
	else_if = { limit = { is_subject_type = uc_bey } add = 1.0 }
	else_if = { limit = { is_subject_type = secessionists } add = 1.0 }
}

# Subject relative-strength drag — read by the auto_modifier.
# The actual signed value is computed monthly by
# sul_projection_recompute_subject_drag and stored in
# var:sul_projection_subject_drag_value. This wrapper provides a safe 0
# fallback for countries that have not yet had the variable set.

sul_projection_subject_drag = {
	value = 0
	if = {
		limit = { has_variable = sul_projection_subject_drag_value }
		add = var:sul_projection_subject_drag_value
	}
}
