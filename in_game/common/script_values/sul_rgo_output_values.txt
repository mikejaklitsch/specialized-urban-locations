# RGO building level / output / max-level math.
#
# RGO cap = base + pop/50 + rgo_size + raw_material_output - building_density.
# All terms are additive. Modifier-based terms scale against the static base (5).

employment_size_RGO = 1

build_time_RGO = 365

sul_rgo_output_per_level = {
	value = raw_material_output

	divide = { value = sul_location_rgo_building_level min = 1 }
}

sul_location_rgo_building_level = {
	value = "location_building_level(var:sul_rgo_building_type)"
}

sul_rgo_levels_under_construction = {
	value = var:sul_rgo_constructing
}

sul_location_max_rgo_building_level = {
	value = sul_rgo_max_level_base
	min = 1
}

sul_location_max_rgo_building_level_rounded = {
	value = sul_rgo_max_level_base
	floor = yes
	min = 1
}

rgo_building_level = {
	value = 1

	add = { value = population divide = 50 }

	floor = yes
}

sul_rgo_max_level_base = {
	add = { desc = "BUILDING_LEVEL_BASE" value = 5 }

	add = {
		desc = "BUILDING_LEVEL_POPULATION"
		value = population
		divide = 50
	}

	if = {
		limit = { modifier:local_max_rgo_size_modifier != 0 }
		add = {
			desc = "SUL_RGO_SIZE"
			value = 5
			multiply = modifier:local_max_rgo_size_modifier
		}
	}

	if = {
		limit = { has_owner = yes owner = { modifier:global_max_rgo_size_modifier != 0 } }
		add = {
			desc = "SUL_RGO_SIZE_GLOBAL"
			value = 5
			multiply = owner.modifier:global_max_rgo_size_modifier
		}
	}

	if = {
		limit = {
			has_owner = yes
			owner = { modifier:global_max_rgo_size_modifier_in_rural != 0 }
			modifier:sul_is_rural_settlement_rank_tier = yes
		}
		add = {
			desc = "SUL_RGO_SIZE_RURAL"
			value = 5
			multiply = owner.modifier:global_max_rgo_size_modifier_in_rural
		}
	}

	if = {
		limit = {
			has_owner = yes
			owner = { modifier:global_max_rgo_size_modifier_in_non_rural != 0 }
			modifier:sul_is_rural_settlement_rank_tier = no
		}
		add = {
			desc = "SUL_RGO_SIZE_URBAN"
			value = 5
			multiply = owner.modifier:global_max_rgo_size_modifier_in_non_rural
		}
	}

	if = {
		limit = { modifier:local_raw_material_output != 0 }
		add = {
			desc = "SUL_RAW_MATERIAL_OUTPUT"
			value = 5
			multiply = modifier:local_raw_material_output
		}
	}

	if = {
		limit = { has_owner = yes owner = { modifier:global_raw_material_output != 0 } }
		add = {
			desc = "SUL_RAW_MATERIAL_OUTPUT_GLOBAL"
			value = 5
			multiply = owner.modifier:global_raw_material_output
		}
	}

	add = {
		desc = "SUL_BUILDING_DENSITY"
		value = total_building_levels
		multiply = -0.1
	}

	min = 0
}
