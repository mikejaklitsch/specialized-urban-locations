# RGO building level / output / max-level math.
#
# RGO cap = (base + population) × raw material output × extraction investment.
# Uses save_temporary_value_as pattern (from MnT) so every multiplier
# shows as a tooltipped additive line instead of a blank multiply.

employment_size_RGO = 1

build_time_RGO = 365

sul_rgo_output_per_level = {
	value = raw_material_output
	divide = { value = sul_location_rgo_building_level min = 1 }
}

sul_location_rgo_building_level = {
	value = "location_building_level(var:sul_rgo_building_type)"
}

sul_rgo_levels_under_construction = {
	value = var:sul_rgo_constructing
}

sul_location_max_rgo_building_level = {
	value = sul_rgo_max_level_base
	min = 1
}

sul_location_max_rgo_building_level_rounded = {
	value = sul_rgo_max_level_base
	floor = yes
	min = 1
}

rgo_building_level = {
	value = 1
	add = { value = population divide = 50 }
	floor = yes
}

sul_rgo_max_level_base = {
	save_temporary_value_as = { name = rgo_base value = 0 }

	# Base
	save_temporary_value_as = {
		name = rgo_base
		value = { value = scope:rgo_base add = 3 }
	}
	add = { desc = "BUILDING_LEVEL_BASE" value = 3 }

	# Population
	save_temporary_value_as = {
		name = rgo_base
		value = {
			value = scope:rgo_base
			add = { value = population multiply = 0.005 }
		}
	}
	add = { desc = "BUILDING_LEVEL_POPULATION" value = population multiply = 0.005 }

	# Snapshot base+pop before multiplicative scaling
	save_temporary_value_as = {
		name = rgo_scaled
		value = scope:rgo_base
	}

	# Raw material output bonus (multiplicative, displayed as additive)
	if = {
		limit = { modifier:local_raw_material_output != 0 }
		save_temporary_value_as = {
			name = rgo_scaled
			value = {
				value = scope:rgo_base
				multiply = { value = 1 add = modifier:local_raw_material_output }
			}
		}
		add = {
			desc = "SUL_RAW_MATERIAL_OUTPUT"
			value = scope:rgo_base
			multiply = modifier:local_raw_material_output
		}
	}
	if = {
		limit = { has_owner = yes owner = { modifier:global_raw_material_output != 0 } }
		save_temporary_value_as = {
			name = rgo_scaled
			value = {
				value = scope:rgo_scaled
				add = {
					value = scope:rgo_base
					multiply = owner.modifier:global_raw_material_output
				}
			}
		}
		add = {
			desc = "SUL_RAW_MATERIAL_OUTPUT_GLOBAL"
			value = scope:rgo_base
			multiply = owner.modifier:global_raw_material_output
		}
	}

	# Extraction investment (multiplicative, displayed as additive)
	# Each extraction building level adds modifier:sul_extraction_size = 1.
	# At 0 extraction: total zeroes out. At 1: unchanged. At N: ×N.
	add = {
		desc = "SUL_EXTRACTION_INVESTMENT"
		value = scope:rgo_scaled
		multiply = { value = modifier:sul_extraction_size subtract = 1 }
	}
}
