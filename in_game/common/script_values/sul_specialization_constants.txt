# Diversity pressure. Reads cached counts (sul_mining_count, etc.) updated
# only when specializations change.
# var:sul_total_weighted = farming×2 + others×1
# Result is positive if under-represented, negative if over-represented
# Returns 0 if variables not yet initialized (first action by this country)
# Mining diversity pressure: (expected - actual) / expected × 5

sul_mining_diversity = {
	if = {
		limit = {
			NOT = {
				has_variable = sul_total_weighted
			}
		}

		value = 0
	}
	else = {
		value = var:sul_total_weighted
		divide = 6
		subtract = var:sul_mining_count
		multiply = 6
		divide = { value = var:sul_total_weighted min = 1 }
		multiply = 5
	}
}

# Market pressure for AI weights. Formula: (demand/supply - 1) × scale.
# Positive = shortage, encourages production.
# Negative = surplus (supply > demand), discourages production
# Uses min = 0.1 on supply to avoid divide-by-zero
# Clamped to ±1 before scaling to prevent extreme swings
# Scale factor of 5 gives max ±5 AI weight impact
# Mining: Iron market pressure (called from market scope)

sul_iron_market_pressure = {
	value = "goods_demand_in_market(goods:iron)"

	divide = { value = "goods_supply_in_market(goods:iron)" min = 0.1 }

	subtract = 1
	max = 1
	min = -1
	multiply = 5
}

# Food price bonuses. Market scope. food_price scale:
# 0.01 = minimum, 0.02 = baseline, higher = shortage.
# Farming gets stronger food price pressure (main food producer)
# Gathering gets weaker food price pressure (supplementary food)
# Farming food price bonus: (food_price - 0.02) × 400 (strong)
# At 0.01 (minimum): -4 penalty
# At 0.02 (baseline): 0 bonus
# At 0.04 (shortage): +8 bonus
# At 0.045+: +10 bonus (capped)

sul_farming_food_price_bonus = {
	value = food_price
	subtract = 0.02
	multiply = 400
	min = -4
	max = 10
}

# Gathering food price bonus: (food_price - 0.02) × 200 (weak)
# At 0.01 (minimum): -2 penalty
# At 0.02 (baseline): 0 bonus
# At 0.04 (shortage): +4 bonus
# At 0.05+: +6 bonus (capped)

sul_gathering_food_price_bonus = {
	value = food_price
	subtract = 0.02
	multiply = 200
	min = -2
	max = 6
}
