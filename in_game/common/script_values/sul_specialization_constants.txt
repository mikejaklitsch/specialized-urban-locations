# Diversity pressure. Reads cached counts (sul_mining_count, etc.) updated
# only when specializations change.
# var:sul_total_weighted = farming×2 + others×1
# Result is positive if under-represented, negative if over-represented
# Returns 0 if variables not yet initialized (first action by this country)
# Mining diversity pressure: (expected - actual) / expected × 5

sul_mining_diversity = {
	if = {
		limit = {
			NOT = {
				has_variable = sul_total_weighted
			}
		}

		value = 0
	}
	else = {
		value = var:sul_total_weighted
		divide = 6
		subtract = var:sul_mining_count
		multiply = 6
		divide = { value = var:sul_total_weighted min = 1 }
		multiply = 5
	}
}

# ════════════════════════════════════════════════════════════════════
# AI SPECIALIZATION SCORING
# Two-part system: per-spec attractiveness (pure positive) vs
# cost of leaving current spec. AI respec action fires when
# best alternative attractiveness > cost of leaving.
# ════════════════════════════════════════════════════════════════════

# Raw material price premium: base price anchored, dampened downside.
# Location scope. Collapsed market prices (e.g. gold at 1 vs base 10)
# still reflect the good's inherent trade value — negative deviation is
# dampened to 1/3. Positive deviation passes through at full rate to
# encourage switching toward high-demand goods.
#
# Formula: base + delta * scale
#   delta >= 0 → scale = 1.0
#   delta <  0 → scale = 0.33

@sul_price_downside_dampening = 0.33

sul_raw_material_base_price = {
	value = "default_price(raw_material)"
}

sul_raw_material_market_price = {
	value = "market.location.variable_map(sul_market_prices|raw_material)"
}

sul_raw_material_price_delta = {
	value = sul_raw_material_market_price
	subtract = sul_raw_material_base_price
}

sul_raw_material_price_premium = {
	value = sul_raw_material_base_price
	# Above base: full positive signal
	if = {
		limit = { sul_raw_material_market_price >= sul_raw_material_base_price }
		add = sul_raw_material_price_delta
	}
	# Below base: dampened negative signal
	else = {
		add = {
			value = sul_raw_material_price_delta
			multiply = @sul_price_downside_dampening
		}
	}
}

# ── Per-spec attractiveness (location scope, pure positive) ──

sul_attractiveness_mining = {
	# Ineligible: hard block
	if = {
		limit = { NOT = { has_variable = sul_can_mine } }
		add = -10000
	}
	# Terrain
	if = {
		limit = { topography = wetlands }
		add = 3
	}
	else_if = {
		limit = { topography = mountains }
		add = 2
	}
	# Raw material pull (only if mining resource)
	if = {
		limit = { sul_has_mining_resources = yes }
		add = sul_raw_material_price_premium
	}
	# Finished goods demand (cached)
	market ?= {
		if = {
			limit = { is_key_in_global_variable_map = { name = sul_mining_finished_composite target = this } }
			add = "global_variable_map(sul_mining_finished_composite|this)"
		}
	}
	# Diversity
	add = owner.sul_mining_diversity
}

sul_attractiveness_farming = {
	# Ineligible: hard block
	if = {
		limit = { NOT = { has_variable = sul_can_farm } }
		add = -10000

		limit = { vegetation = farmland }
		add = 4
	}
	else_if = {
		limit = { topography = flatland vegetation = grasslands }
		add = 2
	}
	else_if = {
		limit = { vegetation = grasslands }
		add = 1
	}
	if = {
		limit = { has_river = yes }
		add = 1
	}
	# Government
	owner = {
		if = {
			limit = { OR = { government_type = government_type:tribe government_type = government_type:steppe_horde } }
			add = 2
		}
	}
	# Raw material pull (only if farming resource)
	if = {
		limit = { sul_has_farming_resources = yes }
		add = sul_raw_material_price_premium
	}
	# Finished goods demand (cached)

		if = {
			limit = { is_key_in_global_variable_map = { name = sul_farming_finished_composite target = this } }

# ── Current spec attractiveness (location scope) ──
# Returns the attractiveness of whichever spec this location currently has.
# 0 for unspecialized locations.

sul_current_attractiveness = {
	if = {
		limit = { sul_has_mining_specialization = yes }
		value = sul_attractiveness_mining
	}
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		value = sul_attractiveness_farming
	}
	else_if = {
		limit = { sul_has_woodland_specialization = yes }
		value = sul_attractiveness_woodland
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		value = sul_attractiveness_gathering
	}
	else_if = {
		limit = { sul_has_commercial_specialization = yes }
		value = sul_attractiveness_commercial
	}
}

# ── Cost of leaving current spec (location scope, always positive) ──
# Current spec attractiveness × 1.5 + mechanical friction.
# 0 for unspecialized locations.

sul_cost_of_leaving = {
	value = sul_current_attractiveness
	multiply = 1.5
	# Building levels (more buildings = more to lose)
	add = {
		value = total_building_levels
		multiply = 0.25
	}
	# Rank inertia
	if = {
		limit = { modifier:sul_is_town_rank_tier = yes }
		add = 5
	}
	else_if = {
		limit = { modifier:sul_is_city_rank_tier = yes }
		add = 10
	}
}

# ── Best alternative attractiveness (location scope) ──
# Max attractiveness across non-current specs. Uses min= as floor/max.

sul_best_alternative = {
	value = -1000
	if = {
		limit = { NOT = { sul_has_mining_specialization = yes } }
		min = sul_attractiveness_mining
	}
	if = {
		limit = { NOT = { sul_has_farming_specialization = yes } }
		min = sul_attractiveness_farming
	}
	if = {
		limit = { NOT = { sul_has_woodland_specialization = yes } }
		min = sul_attractiveness_woodland
	}
	if = {
		limit = { NOT = { sul_has_gathering_specialization = yes } }
		min = sul_attractiveness_gathering
	}
	if = {
		limit = { NOT = { sul_has_commercial_specialization = yes } }
		min = sul_attractiveness_commercial
	}
}

# Net respec score: best alternative minus cost of staying. Positive = should switch.
sul_respec_score = {
	value = sul_best_alternative
	subtract = sul_cost_of_leaving
}
