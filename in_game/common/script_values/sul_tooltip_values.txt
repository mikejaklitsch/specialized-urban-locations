# --- EFFECTIVE SHARES (tax base distribution, kept for backward compat) ---

sul_noble_effective_share = {
	value = 0
	if = {
		limit = { has_variable = sul_total_weighted }

		add = {
			value = { value = num_pop_type:nobles multiply = @sul_power_weight_nobles }

			add = {
				value = num_pop_type:peasants
				add = num_pop_type:laborers
				add = num_pop_type:soldiers
				multiply = { value = 1 subtract = var:sul_enfranchisement }
			}

			divide = var:sul_total_weighted
		}
	}
}

sul_clergy_effective_share = {
	value = 0
	if = {
		limit = { has_variable = sul_total_weighted }

		add = {
			value = { value = num_pop_type:clergy multiply = @sul_power_weight_clergy }
			divide = var:sul_total_weighted
		}
	}
}

sul_burgher_effective_share = {
	value = 0
	if = {
		limit = { has_variable = sul_total_weighted }

		add = {
			value = {
				value = num_pop_type:burghers
				multiply = @sul_power_weight_burghers
			}

			divide = var:sul_total_weighted
		}
	}
}

sul_commoner_effective_share = {
	value = 0
	if = {
		limit = { has_variable = sul_total_weighted }

		add = {
			value = num_pop_type:peasants
			add = num_pop_type:laborers
			add = num_pop_type:soldiers
			multiply = var:sul_enfranchisement
			divide = var:sul_total_weighted
		}
	}
}
