# Shows per-pop-type breakdown: base wages, capital returns (upper class),
# flat wealth (commoners), total WPP.

# --- WAGE COMPONENT PER POP (live: pool × power / pops) ---
# The share of the location's total wage output captured by this pop type.

sul_noble_wage_rate = {
	value = 0
	if = { limit = { num_pop_type:nobles > 0 }
		add = { value = var:sul_local_wages multiply = "local_relative_estate_power(estate_type:nobles_estate)" divide = num_pop_type:nobles }

# --- DEMAND PER POP (live-computed WPP) ---
# Recomputes WPP from current power/pop counts so the tooltip always
# shows what demand would be if the location updated right now.
# wage_pool × power / pops + capital_returns (upper) or flat_wealth (commoner)

@sul_return_rate = 0.02

sul_noble_demand_per_pop = {
	value = 0
	if = {
		limit = { num_pop_type:nobles > 0 }
		add = {
			value = var:sul_local_wages
			multiply = "local_relative_estate_power(estate_type:nobles_estate)"
			divide = num_pop_type:nobles
			add = { value = var:sul_noble_assets multiply = @sul_return_rate divide = { value = num_pop_type:nobles min = 0.001 } }
		}
	}
}
