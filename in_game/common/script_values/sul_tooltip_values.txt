# =============================================================================
# TOOLTIP SCRIPT VALUES (location scope)
# =============================================================================
# Display-only values for the location wealth tooltip.
# Shares are computed inline from sul_total_weighted + sul_enfranchisement + pop counts.
# =============================================================================

# Estate power weights (must match sul_gdp_update.txt)
@sul_power_weight_nobles = 100
@sul_power_weight_clergy = 25
@sul_power_weight_burghers = 20

# --- EFFECTIVE SHARES (reusable helpers) ---

sul_noble_effective_share = {
	value = 0
	if = {
		limit = { has_variable = sul_total_weighted }

		add = {
			value = { value = num_pop_type:nobles multiply = @sul_power_weight_nobles }

			add = {
				value = num_pop_type:peasants
				add = num_pop_type:laborers
				add = num_pop_type:soldiers
				multiply = { value = 1 subtract = var:sul_enfranchisement }
			}

			divide = var:sul_total_weighted
		}
	}
}

sul_clergy_effective_share = {
	value = 0
	if = {
		limit = { has_variable = sul_total_weighted }

		add = {
			value = { value = num_pop_type:clergy multiply = @sul_power_weight_clergy }
			divide = var:sul_total_weighted
		}
	}
}

sul_burgher_effective_share = {
	value = 0
	if = {
		limit = { has_variable = sul_total_weighted }

		add = {
			value = {
				value = num_pop_type:burghers
				multiply = @sul_power_weight_burghers
			}

}

sul_commoner_effective_share = {

}

# --- WEALTH PER POP (read from variable map) ---

sul_noble_per_capita = {
	value = 0
	if = {
		limit = {
			has_variable_map = sul_wealth_per_pop

			is_key_in_variable_map = {
				name = sul_wealth_per_pop
				target = pop_type:nobles
			}
		}

		add = "variable_map(sul_wealth_per_pop|pop_type:nobles)"

}

# --- ESTATE SHARE PERCENTAGES ---

sul_noble_share_pct = {
	value = sul_noble_effective_share
	multiply = 100
}

sul_clergy_share_pct = {
	value = sul_clergy_effective_share
	multiply = 100
}

sul_burgher_share_pct = {
	value = sul_burgher_effective_share
	multiply = 100
}

# Commoner sub-shares (proportional within commoner pool)

sul_soldier_share_pct = {
	value = sul_commoner_effective_share
	multiply = num_pop_type:soldiers

	divide = {
		value = num_pop_type:soldiers
		add = num_pop_type:laborers
		add = num_pop_type:peasants
		min = 0.001
	}

	multiply = 100
}

sul_laborer_share_pct = {
	value = sul_commoner_effective_share
	multiply = num_pop_type:laborers
