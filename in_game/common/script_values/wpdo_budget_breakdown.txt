# =============================================================================
# BUDGET BREAKDOWN SCRIPT VALUES (location scope)
# =============================================================================
# Per pop-type per tier: returns max(0, wpp - threshold) × num_pops.
# GUI multiplies by demand_add (from wpdo_da_* map) to get per-good quantity.
# Evaluated on tooltip hover only.
# =============================================================================

# Thresholds must match TIER_THRESHOLD in demand_calculator.py
@wpdo_tier_threshold_necessity = 0
@wpdo_tier_threshold_basic = 0.05
@wpdo_tier_threshold_common = 0.12
@wpdo_tier_threshold_upper = 0.25
@wpdo_tier_threshold_luxury = 0.5
@wpdo_tier_threshold_exotic = 1.5

# --- NOBLES ---

wpdo_nobles_necessity = {
	value = {
		value = "variable_map(wpdo_wealth_per_pop|pop_type:nobles)"
		subtract = @wpdo_tier_threshold_necessity
		min = 0
	}

	multiply = num_pop_type:nobles
}
