# Granary cap — scales with development and population
# ~3-4 in a small town, ~8-10 in a large city

wpdo_granary_max_level = {
	add = { desc = "BUILDING_LEVEL_BASE" value = 1 }
	add = {
		desc = "BUILDING_LEVEL_DEVELOPMENT"
		value = development
		multiply = 0.05
	}
	add = {
		desc = "BUILDING_LEVEL_POPULATION"
		value = population
		multiply = 0.01
	}
	if = {
		limit = { location_rank ?= location_rank:city }
		add = { desc = "BUILDING_LEVEL_IS_CITY" value = 2 }
	}
}

wpdo_rural_granary_max_level = {
	add = 1

	add = { value = development multiply = 0.05 }
}
