# =============================================================================
# TOOLTIP SCRIPT VALUES (location scope)
# =============================================================================
# Display-only values for the location wealth tooltip.
# Shares are computed inline from wpdo_total_weighted + wpdo_enfranchisement + pop counts.
# =============================================================================

# Estate power weights (must match wpdo_gdp_update.txt)
@wpdo_power_weight_nobles = 100
@wpdo_power_weight_clergy = 25
@wpdo_power_weight_burghers = 20

# --- EFFECTIVE SHARES (reusable helpers) ---

wpdo_noble_effective_share = {
	value = 0
	if = {
		limit = { has_variable = wpdo_total_weighted }

		add = {
			value = { value = num_pop_type:nobles multiply = @wpdo_power_weight_nobles }

			add = {
				value = num_pop_type:peasants
				add = num_pop_type:laborers
				add = num_pop_type:soldiers
				multiply = { value = 1 subtract = var:wpdo_enfranchisement }
			}

			divide = var:wpdo_total_weighted
		}
	}
}

wpdo_clergy_effective_share = {
	value = 0
	if = {
		limit = { has_variable = wpdo_total_weighted }

		add = {
			value = { value = num_pop_type:clergy multiply = @wpdo_power_weight_clergy }
			divide = var:wpdo_total_weighted
		}
	}
}

wpdo_burgher_effective_share = {
	value = 0
	if = {
		limit = { has_variable = wpdo_total_weighted }

		add = {
			value = {
				value = num_pop_type:burghers
				multiply = @wpdo_power_weight_burghers
			}

			divide = var:wpdo_total_weighted
		}
	}
}

wpdo_commoner_effective_share = {

			divide = var:wpdo_total_weighted
		}
	}
}

# --- WEALTH PER POP (read from variable map) ---

wpdo_noble_per_capita = {
	value = 0
	if = {
		limit = {
			has_variable_map = wpdo_wealth_per_pop

			is_key_in_variable_map = {
				name = wpdo_wealth_per_pop
				target = pop_type:nobles
			}
		}

		add = "variable_map(wpdo_wealth_per_pop|pop_type:nobles)"
	}
}

# --- ESTATE SHARE PERCENTAGES ---

wpdo_noble_share_pct = {
	value = wpdo_noble_effective_share
	multiply = 100
}

wpdo_clergy_share_pct = {
	value = wpdo_clergy_effective_share
	multiply = 100
}

wpdo_burgher_share_pct = {
	value = wpdo_burgher_effective_share
	multiply = 100
}

# Commoner sub-shares (proportional within commoner pool)

wpdo_soldier_share_pct = {
	value = wpdo_commoner_effective_share
	multiply = num_pop_type:soldiers

	divide = {
		value = num_pop_type:soldiers
		add = num_pop_type:laborers
		add = num_pop_type:peasants
		min = 0.001
	}

	multiply = 100
}

wpdo_laborer_share_pct = {
	value = wpdo_commoner_effective_share
	multiply = num_pop_type:laborers

	divide = {

		add = num_pop_type:peasants
		min = 0.001
	}

	multiply = 100
}
