# Decluttered War UI — Default Settings Initialization
# Called once per country on first game load with this mod
# Also called by sul_initialize_all at game start (before CMF loads)

dwui_init_settings = {
	set_variable = {
		name = dwui_initialized
		value = yes
	}

	# City label coloring ON by default
	set_variable = {
		name = dwui_cl
		value = yes
	}

	# Default color filters: capitals, province capitals, cities
	set_variable = {
		name = dwui_cap
		value = yes
	}
	set_variable = {
		name = dwui_pcap
		value = yes
	}
	set_variable = {
		name = dwui_city
		value = yes
	}

	# Default name visibility: capitals, province capitals
	set_variable = {
		name = dwui_nm_cap
		value = yes
	}
	set_variable = {
		name = dwui_nm_pcap
		value = yes
	}

	# Rank icon filters: show specialization icons for all ranks by default
	set_variable = {
		name = dwui_ri_city
		value = yes
	}
	set_variable = {
		name = dwui_ri_town
		value = yes
	}
	set_variable = {
		name = dwui_ri_rural
		value = yes
	}

	# Rank icon admin filters: show on capitals, province capitals, and all locations by default
	set_variable = {
		name = dwui_ri_cap
		value = yes
	}
	set_variable = {
		name = dwui_ri_pcap
		value = yes
	}
	set_variable = {
		name = dwui_ri_all
		value = yes
	}

	# Towns, rural, fort color filters OFF by default
	# Vanilla fort icon OFF by default (use building-specific)
}
