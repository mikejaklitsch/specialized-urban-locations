# Decluttered War UI — Unit Composition Calculation
# Counts sub-units by type (regular, mercenary, levy) per army.
# Called monthly and after battles.

dwui_calculate_unit_composition = {
	# Reset counters
	set_variable = {
		name = dwui_regular_str
		value = 0
	}
	set_variable = {
		name = dwui_merc_str
		value = 0
	}
	set_variable = {
		name = dwui_levy_str
		value = 0
	}

	# Count sub-units by type
	every_sub_unit = {
		if = {
			limit = { is_mercenary = yes }
			root = { change_variable = { name = dwui_merc_str add = 1 } }
		}
		else_if = {
			limit = { is_levy = yes }
			root = { change_variable = { name = dwui_levy_str add = 1 } }
		}
		else = {
			root = { change_variable = { name = dwui_regular_str add = 1 } }
		}
	}
}
