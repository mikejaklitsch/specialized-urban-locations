# Expanded Urban Locations - Scripted Effects

# ============================================
# BUILDING DESTRUCTION EFFECTS
# Each effect destroys all specialization-relevant buildings EXCEPT those matching the specialization
# root = location
# ============================================

# Mining Specialization - keeps mining raw + mining production (tools, weapons, cannons, firearms, steel, jewelry)
destroy_non_mining_buildings = {
	every_buildings_in_location = {
		limit = {
			is_destroyable_specialization_building = yes
			NOR = {
				# Mining raw buildings
				building_type = building_type:local_smelters
				building_type = building_type:mercury_patio
				building_type = building_type:stone_quarry
				building_type = building_type:bog_iron_smelter
				building_type = building_type:mason
				# Mining production - tools
				building_type = building_type:tools_guild
				building_type = building_type:tools_workshop
				building_type = building_type:iron_foundry
				building_type = building_type:iron_mill
				# Mining production - weapons
				building_type = building_type:weapon_guild
				building_type = building_type:weapon_workshop
				building_type = building_type:weapon_manufactory
				building_type = building_type:weapon_factory
				# Mining production - cannons
				building_type = building_type:cannon_maker
				building_type = building_type:cannon_workshop
				building_type = building_type:cannon_foundry
				building_type = building_type:cannons_factory
				# Mining production - firearms
				building_type = building_type:hand_cannon_guild
				building_type = building_type:gun_smith
				building_type = building_type:guns_workshop
				building_type = building_type:firearms_manufactory
				building_type = building_type:firearms_factory
				# Mining production - steel and jewelry
				building_type = building_type:steel_mill
				building_type = building_type:jewelry_guild
			}
		}
		location = { destroy_building = prev }
	}
}

# ============================================
# MODIFIER MANAGEMENT EFFECTS
# root = location
# ============================================

# Remove all specialization modifiers
clear_all_specialization_modifiers = {
	if = {
		limit = { has_location_modifier = mining_town_specialization }
		remove_location_modifier = mining_town_specialization
	}
	if = {
		limit = { has_location_modifier = mining_city_specialization }
		remove_location_modifier = mining_city_specialization
	}
	if = {
		limit = { has_location_modifier = farming_town_specialization }
		remove_location_modifier = farming_town_specialization
	}
	if = {
		limit = { has_location_modifier = farming_city_specialization }
		remove_location_modifier = farming_city_specialization
	}
	if = {
		limit = { has_location_modifier = gathering_town_specialization }
		remove_location_modifier = gathering_town_specialization
	}
	if = {
		limit = { has_location_modifier = gathering_city_specialization }
		remove_location_modifier = gathering_city_specialization
	}
	if = {
		limit = { has_location_modifier = forestry_town_specialization }
		remove_location_modifier = forestry_town_specialization
	}
	if = {
		limit = { has_location_modifier = forestry_city_specialization }
		remove_location_modifier = forestry_city_specialization
	}
	if = {
		limit = { has_location_modifier = commercial_town_specialization }
		remove_location_modifier = commercial_town_specialization
	}
	if = {
		limit = { has_location_modifier = commercial_city_specialization }
		remove_location_modifier = commercial_city_specialization
	}
}

# Apply correct modifier based on specialization variable and current rank
# root = location (must have specialization_designation variable set)
apply_specialization_modifier = {
	# Mining
	if = {
		limit = {
			var:specialization_designation = flag:mining
			location_rank = location_rank:town
		}
		add_location_modifier = {
			modifier = mining_town_specialization
		}
	}
	else_if = {
		limit = {
			var:specialization_designation = flag:mining
			location_rank = location_rank:city
		}
		add_location_modifier = {
			modifier = mining_city_specialization
		}
	}
	# Farming
	else_if = {
		limit = {
			var:specialization_designation = flag:farming
			location_rank = location_rank:town
		}
		add_location_modifier = {
			modifier = farming_town_specialization
		}
	}
	else_if = {
		limit = {
			var:specialization_designation = flag:farming
			location_rank = location_rank:city
		}
		add_location_modifier = {
			modifier = farming_city_specialization
		}
	}
	# Forestry
	else_if = {
		limit = {
			var:specialization_designation = flag:forestry
			location_rank = location_rank:town
		}
		add_location_modifier = {
			modifier = forestry_town_specialization
		}
	}
	else_if = {
		limit = {
			var:specialization_designation = flag:forestry
			location_rank = location_rank:city
		}
		add_location_modifier = {
			modifier = forestry_city_specialization
		}
	}
	# Gathering
	else_if = {
		limit = {
			var:specialization_designation = flag:gathering
			location_rank = location_rank:town
		}
		add_location_modifier = {
			modifier = gathering_town_specialization
		}
	}
	else_if = {
		limit = {
			var:specialization_designation = flag:gathering
			location_rank = location_rank:city
		}
		add_location_modifier = {
			modifier = gathering_city_specialization
		}
	}
	# Commercial
	else_if = {
		limit = {
			var:specialization_designation = flag:commercial
			location_rank = location_rank:town
		}
		add_location_modifier = {
			modifier = commercial_town_specialization
		}
	}
	else_if = {
		limit = {
			var:specialization_designation = flag:commercial
			location_rank = location_rank:city
		}
		add_location_modifier = {
			modifier = commercial_city_specialization
		}
	}
}
