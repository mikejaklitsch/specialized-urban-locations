# Mining Specialization - keeps mining raw + mining production (tools, weapons, cannons, firearms, steel, jewelry)
destroy_non_mining_buildings = {
	every_buildings_in_location = {
		limit = {
			is_destroyable_specialization_building = yes
			NOR = {
				# Mining raw buildings
				building_type = building_type:local_smelters
				building_type = building_type:mercury_patio
				building_type = building_type:stone_quarry
				building_type = building_type:bog_iron_smelter
				building_type = building_type:mason
				# Mining production - tools
				building_type = building_type:tools_guild
				building_type = building_type:tools_workshop
				building_type = building_type:iron_foundry
				building_type = building_type:iron_mill
				# Mining production - weapons
				building_type = building_type:weapon_guild
				building_type = building_type:weapon_workshop
				building_type = building_type:weapon_manufactory
				building_type = building_type:weapon_factory
				# Mining production - cannons
				building_type = building_type:cannon_maker
				building_type = building_type:cannon_workshop
				building_type = building_type:cannon_foundry
				building_type = building_type:cannons_factory
				# Mining production - firearms
				building_type = building_type:hand_cannon_guild
				building_type = building_type:gun_smith
				building_type = building_type:guns_workshop
				building_type = building_type:firearms_manufactory
				building_type = building_type:firearms_factory
				# Mining production - steel and jewelry
				building_type = building_type:steel_mill

		}
		location = { destroy_building = prev }
	}
}

# Grant tier 1 buildings for a specialization (guilds valid for towns/cities)
# Checks advance requirements - only grants buildings the owner has unlocked
# root = location (must have specialization_designation variable set)
# NOTE: Only runs for towns/cities, not rural settlements
grant_specialization_starter_buildings = {
	# Mining - grant tier 1 production guilds
	if = {
		limit = { has_mining_specialization = yes }
		# tools_guild requires metallurgy advance
		if = {
			limit = { owner = { has_advance = metallurgy } }
			change_building_level_in_location = {
				building = building_type:tools_guild
				value = 1
			}
		}
		# weapon_guild requires weapon_guild_advance
		if = {
			limit = { owner = { has_advance = weapon_guild_advance } }
			change_building_level_in_location = {
				building = building_type:weapon_guild
				value = 1
			}
		}
		# cannon_maker requires cannon_maker_advance
		if = {
			limit = { owner = { has_advance = cannon_maker_advance } }
			change_building_level_in_location = {
				building = building_type:cannon_maker
				value = 1
			}
		}
		# hand_cannon_guild requires hand_cannon_guild_advance (East Asia only)
		if = {
			limit = { owner = { has_advance = hand_cannon_guild_advance } }
			change_building_level_in_location = {
				building = building_type:hand_cannon_guild
				value = 1
			}
		}
		# jewelry_guild - no advance requirement
		change_building_level_in_location = {
			building = building_type:jewelry_guild
			value = 1
		}
	}
	# Farming - grant tier 1 production guilds
	else_if = {
		limit = { has_farming_specialization = yes }
		# No advance requirements for these
		change_building_level_in_location = {
			building = building_type:brewery
			value = 1
		}
		change_building_level_in_location = {
			building = building_type:winery
			value = 1
		}
		change_building_level_in_location = {
			building = building_type:distillers_guild
			value = 1
		}
		# saltpeter_guild requires livestock + clay in market
		if = {
			limit = {
				market ?= {
					is_produced_in_market = goods:livestock
					is_produced_in_market = goods:clay
				}
			}
			change_building_level_in_location = {
				building = building_type:saltpeter_guild
				value = 1
			}
		}
	}
	# Woodland - grant tier 1 production guilds
	else_if = {
		limit = { has_woodland_specialization = yes }
		# No advance requirements for these
		change_building_level_in_location = {
			building = building_type:furniture_guild
			value = 1
		}
		change_building_level_in_location = {
			building = building_type:dyes_guild
			value = 1
		}
		# lacquerware_guild requires lacquerware_guild_advance (East Asia only)
		if = {
			limit = { owner = { has_advance = lacquerware_guild_advance } }
			change_building_level_in_location = {
				building = building_type:lacquerware_guild
				value = 1
			}
		}
		change_building_level_in_location = {
			building = building_type:tannery
			value = 1
		}
	}
	# Gathering - grant tier 1 production guilds
	else_if = {
		limit = { has_gathering_specialization = yes }
		# glass_guild requires sand in market
		if = {
			limit = { market ?= { is_produced_in_market = goods:sand } }
			change_building_level_in_location = {
				building = building_type:glass_guild
				value = 1
			}
		}
		# No advance requirement
		change_building_level_in_location = {
			building = building_type:pottery_guild
			value = 1
		}
		# porcelain_guild requires porcelain_kiln_advance (East Asia only)
		if = {
			limit = { owner = { has_advance = porcelain_kiln_advance } }
			change_building_level_in_location = {
				building = building_type:porcelain_guild
				value = 1
			}
		}
		# apothecary requires apothecary_advance
		if = {
			limit = { owner = { has_advance = apothecary_advance } }
			change_building_level_in_location = {
				building = building_type:apothecary
				value = 1
			}
		}
	}
	# Commercial - grant tier 1 trade/production buildings
	else_if = {
		limit = { has_commercial_specialization = yes }
		# No advance requirement
		change_building_level_in_location = {
			building = building_type:marketplace
			value = 1
		}
		# scriptorium requires scriptorium_advance
		if = {
			limit = { owner = { has_advance = scriptorium_advance } }
			change_building_level_in_location = {
				building = building_type:scriptorium
				value = 1
			}
		}
		# No advance requirements
		change_building_level_in_location = {
			building = building_type:cloth_guild
			value = 1
		}
		change_building_level_in_location = {
			building = building_type:fine_cloth_guild
			value = 1
		}
		change_building_level_in_location = {
			building = building_type:naval_supplies_guild
			value = 1
		}
		# paper_guild requires paper_guild_advance
		if = {
			limit = { owner = { has_advance = paper_guild_advance } }
			change_building_level_in_location = {
				building = building_type:paper_guild
				value = 1
			}
		}
	}
}

# Destroy non-matching buildings based on specialization variable
# root = location (must have specialization_designation variable set)
# NOTE: Only runs for towns/cities, not rural settlements (rural keeps all buildings)
destroy_non_matching_buildings = {
	# Only destroy buildings for towns/cities, not rural
	if = {
		limit = { is_town_or_city = yes }
		if = {
			limit = { has_mining_specialization = yes }
			destroy_non_mining_buildings = yes
		}
		else_if = {
			limit = { has_farming_specialization = yes }
			destroy_non_farming_buildings = yes
		}
		else_if = {
			limit = { has_woodland_specialization = yes }
			destroy_non_woodland_buildings = yes
		}
		else_if = {
			limit = { has_gathering_specialization = yes }
			destroy_non_gathering_buildings = yes
		}
		else_if = {
			limit = { has_commercial_specialization = yes }
			destroy_non_commercial_buildings = yes
		}
	}
}

# Assign specialization based on local resources
# Priority: Commercial (special conditions) > Farming > Mining > Gathering > Woodland
# root = location (rural, town, or city)
# Uses constants from expanded_urban_locations_specialization_constants.txt
assign_specialization_by_resources = {
	# Commercial specialization (priority for market centers, harbors, republic capitals)
	if = {
		limit = {
			OR = {
				is_market_center = yes
				modifier:harbor_suitability > 0.7
				AND = {
					is_capital = yes
					location_rank = location_rank:city
					owner = { government_type = government_type:republic }
				}
			}
		}
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = specialization_designation value = eul_commercial_rural_value }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			set_variable = { name = specialization_designation value = eul_commercial_town_value }
		}
		else = {
			set_variable = { name = specialization_designation value = eul_commercial_city_value }
		}
	}
	# Farming specialization
	else_if = {
		limit = { has_farming_resources = yes }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = specialization_designation value = eul_farming_rural_value }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			set_variable = { name = specialization_designation value = eul_farming_town_value }
		}
		else = {
			set_variable = { name = specialization_designation value = eul_farming_city_value }
		}
	}
	# Mining specialization
	else_if = {
		limit = { has_mining_resources = yes }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = specialization_designation value = eul_mining_rural_value }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			set_variable = { name = specialization_designation value = eul_mining_town_value }
		}
		else = {
			set_variable = { name = specialization_designation value = eul_mining_city_value }
		}
	}
	# Gathering specialization
	else_if = {
		limit = { has_gathering_resources = yes }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = specialization_designation value = eul_gathering_rural_value }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			set_variable = { name = specialization_designation value = eul_gathering_town_value }
		}
		else = {
			set_variable = { name = specialization_designation value = eul_gathering_city_value }
		}
	}
	# Woodland specialization
	else_if = {
		limit = { has_woodland_resources = yes }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = specialization_designation value = eul_woodland_rural_value }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			set_variable = { name = specialization_designation value = eul_woodland_town_value }
		}
		else = {
			set_variable = { name = specialization_designation value = eul_woodland_city_value }
		}
	}
}

# Adds 1 level of appropriate village based on terrain
# root = location (rural settlement)
# Priority: coastal > forest/woods > farmland > market
# ============================================
grant_starter_village = {
	if = {
		limit = { is_coastal = yes }
		change_building_level_in_location = {
			building = building_type:fishing_village
			value = 1
		}
	}
	else_if = {
		limit = {
			OR = {
				vegetation = forest
				vegetation = woods
			}
		}
		change_building_level_in_location = {
			building = building_type:forest_village
			value = 1
		}
	}
	else_if = {
		limit = { vegetation = farmland }
		change_building_level_in_location = {
			building = building_type:farming_village
			value = 1
		}
	}
	else = {
		change_building_level_in_location = {
			building = building_type:market_village
			value = 1
		}
	}
}

eul_initialize_specialization_counts = {
	# Initialize all counts to 0
	set_variable = { name = eul_mining_count value = 0 }
	set_variable = { name = eul_farming_count value = 0 }
	set_variable = { name = eul_woodland_count value = 0 }
	set_variable = { name = eul_gathering_count value = 0 }
	set_variable = { name = eul_commercial_count value = 0 }
	set_variable = { name = eul_total_weighted value = 0 }

	# Count all existing specializations
	every_owned_location = {
		if = {
			limit = { has_mining_specialization = yes }
			owner = {
				change_variable = { name = eul_mining_count add = 1 }
				change_variable = { name = eul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { has_farming_specialization = yes }
			owner = {
				change_variable = { name = eul_farming_count add = 1 }
				change_variable = { name = eul_total_weighted add = 2 }
			}
		}
		else_if = {
			limit = { has_woodland_specialization = yes }
			owner = {
				change_variable = { name = eul_woodland_count add = 1 }
				change_variable = { name = eul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { has_gathering_specialization = yes }
			owner = {
				change_variable = { name = eul_gathering_count add = 1 }
				change_variable = { name = eul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { has_commercial_specialization = yes }
			owner = {
				change_variable = { name = eul_commercial_count add = 1 }
				change_variable = { name = eul_total_weighted add = 1 }
			}
		}
	}
}
