# Economy Bootstrap — demand-weighted building distribution at game start
# Replaces sul_restore_building_levels with a unified system that:
#   1. Places universal infrastructure (granary/storehouse, mason)
#   2. Places spec-appropriate village building
#   3. Grants terrain-specific RGO building
#   4. Splits remaining budget between rural cottages and guild buildings
#      using demand-derived weights (per-mille, sum to 1000 per tier)

# ═══════════════════════════════════════════════════════════════════════════
# BUILDING DISTRIBUTION WEIGHTS
# Generated from demand_calculator.py --weights
# Higher weight = more building levels allocated for that building
# ═══════════════════════════════════════════════════════════════════════════

# ═══════════════════════════════════════════════════════════════════════════
# MAIN ENTRY POINT
# root = location (must have sul_prior_building_levels and specialization set)
# Called after specialization assignment and building destruction
# ═══════════════════════════════════════════════════════════════════════════

sul_bootstrap_economy = {
	# ── Step 1: Grant terrain-specific RGO building ──
	sul_grant_specialization_rgo_building = yes

	# ── Step 2: Place universal infrastructure ──
	# Granary/storehouse — conservative (AI cannot destroy)
	if = {
		limit = { location_rank = location_rank:rural_settlement }

		change_building_level_in_location = {
			building = building_type:sul_storehouse
			value = 1
		}
	}
	else_if = {
		limit = { location_rank = location_rank:town }

		change_building_level_in_location = {
			building = building_type:granary
			value = 1
		}
	}
	else = {
		# Cities get 2 granaries
		change_building_level_in_location = {
			building = building_type:granary
			value = 2
		}
	}

	# Mason — towns and cities only
	if = {
		limit = { is_town_or_city = yes }

		change_building_level_in_location = {
			building = building_type:mason
			value = 1
		}
	}

	# ── Step 3: Place spec-appropriate village building ──
	if = {
		limit = { sul_has_farming_specialization = yes }

		change_building_level_in_location = {
			building = building_type:farming_village
			value = 1
		}
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }

		change_building_level_in_location = {
			building = building_type:fishing_village
			value = 1
		}
	}
	else_if = {
		limit = { sul_has_woodland_specialization = yes }

		change_building_level_in_location = {
			building = building_type:forest_village
			value = 1
		}
	}
	else_if = {
		limit = { sul_has_mining_specialization = yes }

		change_building_level_in_location = {
			building = building_type:sul_mining_village
			value = 1
		}
	}
	else_if = {
		limit = { sul_has_commercial_specialization = yes }

		change_building_level_in_location = {
			building = building_type:market_village
			value = 1
		}
	}

	# ── Step 4: Calculate target building level and budget ──
	# target = rank_base + development × 1.5 + prior_buildings × vanilla_scale
	if = {
		limit = { location_rank = location_rank:rural_settlement }

		set_variable = {
			name = sul_target

			value = {
				value = @sul_rank_base_rural
				add = { value = development multiply = 0.375 }
				add = { value = var:sul_prior_building_levels multiply = @sul_vanilla_scale }
			}
		}
	}
	else_if = {
		limit = { location_rank = location_rank:town }

		set_variable = {
			name = sul_target

			value = {
				value = @sul_rank_base_town
				add = { value = development multiply = 0.375 }
				add = { value = var:sul_prior_building_levels multiply = @sul_vanilla_scale }
			}
		}
	}
	else = {
		set_variable = {
			name = sul_target

			value = {
				value = @sul_rank_base_city
				add = { value = development multiply = 0.375 }
				add = { value = var:sul_prior_building_levels multiply = @sul_vanilla_scale }
			}
		}
	}

	# Population ceiling — can't overbuild beyond workforce
	set_variable = {
		name = sul_pop_ceiling

		value = { value = population divide = @sul_pop_per_level }
	}
	if = {
		limit = { var:sul_target > var:sul_pop_ceiling }
		set_variable = { name = sul_target value = var:sul_pop_ceiling }
	}

	# Budget = target minus what's already placed
	set_variable = {
		name = sul_budget

		value = {
			value = var:sul_target
			subtract = total_building_levels
			min = 0
			floor = yes
		}
	}

	# Nothing left to distribute
	if = {
		limit = { var:sul_budget <= 0 }
		sul_bootstrap_cleanup = yes
	}
	else = {
		# ── Step 5: Split budget between rural and guild tiers ──
		# Rural settlements: 100% rural
		# Towns: 25-50% guild (scales with development)
		# Cities: 55-75% guild (scales with development)

		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = sul_guild_budget value = 0 }
			set_variable = { name = sul_rural_budget value = var:sul_budget }
		}
		else_if = {
			limit = { location_rank = location_rank:town }

			set_variable = {
				name = sul_guild_permille
				value = { value = 250 add = { value = development multiply = 5 } max = 500 }
			}
			set_variable = {
				name = sul_guild_budget

				value = {
					value = var:sul_budget
					multiply = var:sul_guild_permille
					divide = 1000
					floor = yes
				}
			}
			set_variable = {
				name = sul_rural_budget
				value = { value = var:sul_budget subtract = var:sul_guild_budget }
			}
		}
		else = {
			# City
			set_variable = {
				name = sul_guild_permille
				value = { value = 550 add = { value = development multiply = 3 } max = 750 }
			}
			set_variable = {
				name = sul_guild_budget

				value = {
					value = var:sul_budget
					multiply = var:sul_guild_permille
					divide = 1000
					floor = yes
				}
			}
			set_variable = {
				name = sul_rural_budget
				value = { value = var:sul_budget subtract = var:sul_guild_budget }
			}
		}

		# ── Step 6: Distribute rural cottage buildings ──
		if = {
			limit = { var:sul_rural_budget > 0 }
			set_variable = { name = sul_tier_budget value = var:sul_rural_budget }

			if = {
				limit = { sul_has_mining_specialization = yes }
				sul_bootstrap_mining_rural = yes
			}
			else_if = {
				limit = { sul_has_farming_specialization = yes }
				sul_bootstrap_farming_rural = yes
			}
			else_if = {
				limit = { sul_has_gathering_specialization = yes }
				sul_bootstrap_gathering_rural = yes
			}
			else_if = {
				limit = { sul_has_woodland_specialization = yes }
				sul_bootstrap_woodland_rural = yes
			}
			else_if = {
				limit = { sul_has_commercial_specialization = yes }
				sul_bootstrap_commercial_rural = yes
			}
		}

		# ── Step 7: Distribute guild buildings ──
		if = {
			limit = { var:sul_guild_budget > 0 }
			set_variable = { name = sul_tier_budget value = var:sul_guild_budget }

			if = {
				limit = { sul_has_mining_specialization = yes }
				sul_bootstrap_mining_guild = yes
			}
			else_if = {
				limit = { sul_has_farming_specialization = yes }
				sul_bootstrap_farming_guild = yes
			}
			else_if = {
				limit = { sul_has_gathering_specialization = yes }
				sul_bootstrap_gathering_guild = yes
			}
			else_if = {
				limit = { sul_has_woodland_specialization = yes }
				sul_bootstrap_woodland_guild = yes
			}
			else_if = {
				limit = { sul_has_commercial_specialization = yes }
				sul_bootstrap_commercial_guild = yes
			}
		}

		sul_bootstrap_cleanup = yes
	}
}

# CLEANUP
# ═══════════════════════════════════════════════════════════════════════════

sul_bootstrap_cleanup = {
	remove_variable = sul_prior_building_levels
	remove_variable = sul_target
	remove_variable = sul_pop_ceiling
	remove_variable = sul_budget
	remove_variable = sul_guild_permille
	remove_variable = sul_guild_budget
	remove_variable = sul_rural_budget
	remove_variable = sul_tier_budget
	remove_variable = sul_tmp
	remove_variable = sul_bldg_weight
}

# ═══════════════════════════════════════════════════════════════════════════
# WEIGHTED ADD HELPER
# Allocates floor(tier_budget * weight / 1000) levels to a building
# and subtracts from sul_tier_budget. Remainder flows to the last building.
# root = location, expects sul_tier_budget and sul_bldg_weight set
# ═══════════════════════════════════════════════════════════════════════════

sul_bootstrap_weighted_add = {
	set_variable = {
		name = sul_tmp

		value = {
			value = var:sul_tier_budget
			multiply = var:sul_bldg_weight
			divide = 1000
			floor = yes
		}
	}
	if = {
		limit = { var:sul_tmp > 0 }

		change_building_level_in_location = {
			building = building_type:$BUILDING$
			value = var:sul_tmp
		}
	}
	change_variable = {
		name = sul_tier_budget
		subtract = var:sul_tmp
	}
}
