
remove_RGOs = {
	every_owned_location = { replace_rgos_with_buildings = yes }

	every_owned_location = { change_max_raw_material_workers = -100 }

	add_country_modifier = {
		modifier = force_promote_vacancies
		months = 2
		mode = add
		desc = force_promote_vacancies_desc
	}
}

# Converts vanilla RGO into corresponding building using the sul_rgo_map.
# Location scope. Map must be initialized before calling.

replace_rgos_with_buildings = {
	# Cache building_type on location for script value lookups
	set_variable = {
		name = sul_rgo_building_type
		value = "global_variable_map(sul_rgo_map|raw_material)"
	}

	change_building_level_in_location = {
		building = "global_variable_map(sul_rgo_map|raw_material)"
		value = rgo_building_level
	}

	# Seed the market stockpile: 6 of the good + 1.8 tools (3 years of demand) per RGO level
	save_temporary_scope_value_as = {
		name = sul_rgo_seed_goods_amount
		value = { value = rgo_building_level multiply = 6 }
	}
	save_temporary_scope_value_as = {
		name = sul_rgo_seed_tools_amount
		value = { value = rgo_building_level multiply = 1.8 }
	}

	market = {
		add_goods_supply = {
			goods = prev.raw_material
			amount = scope:sul_rgo_seed_goods_amount
		}
		add_goods_supply = {
			goods = goods:tools
			amount = scope:sul_rgo_seed_tools_amount
		}
	}
}

}

# Rebuilds 1 RGO level if all were destroyed (e.g. bankruptcy). Location scope.
# Called from on_destroyed callback on RGO buildings.

sul_rebuild_rgo_if_empty = {
	if = {
		limit = { exists = raw_material }

		"global_variable_map(sul_rgo_map|raw_material)" = {
			save_temporary_scope_as = sul_rgo_btype
		}

		if = {
			limit = {
				sul_location_rgo_building_level <= 0
				modifier:sul_rgo_size > 0
				# Verify destroyed building is still the correct RGO for this location
				prev = { building_type = scope:sul_rgo_btype }
			}

			change_building_level_in_location = {
				building = scope:sul_rgo_btype
				value = 1
			}
		}
	}
}

# Constructs one level of the location's RGO building. Location scope.
# Used by the scripted GUI button.

sul_construct_rgo_building = {
	construct_building = {
		building_type = "global_variable_map(sul_rgo_map|raw_material)"
	}
}

# Trim RGO building levels down to max if over cap. Location scope.
# Minimum 1 level always preserved for script safety.

sul_trim_rgo_levels = {
	if = {
		limit = {
			exists = raw_material
			sul_location_rgo_building_level > sul_location_max_rgo_building_level_rounded
		}

		change_building_level_in_location = {
			building = "global_variable_map(sul_rgo_map|raw_material)"

			value = {
				value = sul_location_max_rgo_building_level_rounded
				subtract = sul_location_rgo_building_level
			}
		}
	}
}

# Engine bug: add_gold_to_estate negates the value, so we negate first.

sul_pay_single_estate_wage = {
	if = {
		limit = {
			has_variable = sul_monthly_wages_$estate$
			var:sul_monthly_wages_$estate$ > 0
		}

		add_gold_to_estate = {
			estate_type = estate_type:$estate$_estate
			value = { value = var:sul_monthly_wages_$estate$ multiply = -1 }
		}
	}
}

# =============================================================================
# TRADE MAINTENANCE EFFICIENCY → COST CONVERSION
# =============================================================================
# Vanilla `merchant_maintenance_cost` is a flat additive cost modifier.
# We replaced every source with `sul_trade_maintenance_efficiency = -X` so
# the engine accumulates a single efficiency total. This effect reads that
# total each month and re-emits it as merchant_maintenance_cost via the
# multiplicative inverse formula:
#
#   final_cost = 1 / (1 + E) - 1
#
# E > 0  → cost reduction with diminishing returns
# E = 0  → no modifier (both carriers removed)
# E < 0  → cost increase (clamped to E > -0.99 to keep the divisor finite)
#
# Carriers: `sul_trade_maintenance_reduction` (merchant_maintenance_cost = -1)
# and `sul_trade_maintenance_penalty` (merchant_maintenance_cost = +1) — picked
# by sign because add_country_modifier `size` only takes positive values.
# Country scope. Called from monthly_country_pulse via sul_hardcoded.txt.

sul_trade_maintenance_apply = {
	remove_country_modifier = sul_trade_maintenance_reduction
	remove_country_modifier = sul_trade_maintenance_penalty

	# Snapshot and clamp the engine-summed efficiency.
	set_variable = {
		name = sul_trade_maintenance_efficiency_total

		value = { value = modifier:sul_trade_maintenance_efficiency min = -0.99 }
	}

	if = {
		limit = { var:sul_trade_maintenance_efficiency_total > 0 }

		# Cost reduction: |1/(1+E) - 1| = E/(1+E)
		add_country_modifier = {
			modifier = sul_trade_maintenance_reduction
			days = -1

			size = {
				value = var:sul_trade_maintenance_efficiency_total
				divide = { value = 1 add = var:sul_trade_maintenance_efficiency_total }
			}
		}
	}
	else_if = {
		limit = { var:sul_trade_maintenance_efficiency_total < 0 }

		# Cost increase: 1/(1+E) - 1 with E < 0 yields a positive number.
		add_country_modifier = {
			modifier = sul_trade_maintenance_penalty
			days = -1

			size = {
				value = 1
				divide = { value = 1 add = var:sul_trade_maintenance_efficiency_total }
				subtract = 1
			}
		}
	}
}
