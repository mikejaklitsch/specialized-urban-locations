# Each effect destroys all specialization-relevant buildings EXCEPT those matching the specialization
# root = location

# Mining Specialization - keeps mining buildings (uses is_mining_building trigger)
destroy_non_mining_buildings = {
	every_buildings_in_location = {
		limit = {
			is_destroyable_specialization_building = yes
			NOT = { is_mining_building = yes }
		}
		location = { destroy_building = prev }
	}
}

# Remove all specialization modifiers (rural, town, and city)
# Optimized: branch by rank first, then by specialization type
# Reduces from 15 checks to max 6 (1 rank + 5 specialization)
clear_all_specialization_modifiers = {
	if = {
		limit = { location_rank = location_rank:rural_settlement }
		# Rural modifiers
		if = {
			limit = { has_mining_specialization = yes }
			remove_location_modifier = mining_rural_specialization
		}
		else_if = {
			limit = { has_farming_specialization = yes }
			remove_location_modifier = farming_rural_specialization
		}
		else_if = {
			limit = { has_gathering_specialization = yes }
			remove_location_modifier = gathering_rural_specialization
		}
		else_if = {
			limit = { has_woodland_specialization = yes }
			remove_location_modifier = woodland_rural_specialization
		}
		else_if = {
			limit = { has_commercial_specialization = yes }
			remove_location_modifier = commercial_rural_specialization
		}
	}
	else_if = {
		limit = { location_rank = location_rank:town }
		# Town modifiers
		if = {
			limit = { has_mining_specialization = yes }
			remove_location_modifier = mining_town_specialization
		}
		else_if = {
			limit = { has_farming_specialization = yes }
			remove_location_modifier = farming_town_specialization
		}
		else_if = {
			limit = { has_gathering_specialization = yes }
			remove_location_modifier = gathering_town_specialization
		}
		else_if = {
			limit = { has_woodland_specialization = yes }
			remove_location_modifier = woodland_town_specialization
		}
		else_if = {
			limit = { has_commercial_specialization = yes }
			remove_location_modifier = commercial_town_specialization
		}
	}
	else_if = {
		limit = { location_rank = location_rank:city }
		# City modifiers
		if = {
			limit = { has_mining_specialization = yes }
			remove_location_modifier = mining_city_specialization
		}
		else_if = {
			limit = { has_farming_specialization = yes }
			remove_location_modifier = farming_city_specialization
		}
		else_if = {
			limit = { has_gathering_specialization = yes }
			remove_location_modifier = gathering_city_specialization
		}
		else_if = {
			limit = { has_woodland_specialization = yes }
			remove_location_modifier = woodland_city_specialization
		}
		else_if = {
			limit = { has_commercial_specialization = yes }
			remove_location_modifier = commercial_city_specialization
		}
	}
}

# Grant tier 1 buildings for a specialization (guilds valid for towns/cities)
# Checks advance requirements - only grants buildings the owner has unlocked
# root = location (must have specialization_designation variable set)
# NOTE: Only runs for towns/cities, not rural settlements
grant_specialization_starter_buildings = {
	# Mining - grant tier 1 production guilds
	if = {
		limit = { has_mining_specialization = yes }
		# tools_guild requires metallurgy advance
		if = {
			limit = { owner = { has_advance = metallurgy } }
			change_building_level_in_location = {
				building = building_type:tools_guild
				value = 1
			}
		}
		# weapon_guild requires weapon_guild_advance
		if = {
			limit = { owner = { has_advance = weapon_guild_advance } }
			change_building_level_in_location = {
				building = building_type:weapon_guild
				value = 1
			}
		}
		# cannon_maker requires cannon_maker_advance
		if = {
			limit = { owner = { has_advance = cannon_maker_advance } }
			change_building_level_in_location = {
				building = building_type:cannon_maker
				value = 1
			}
		}
		# hand_cannon_guild requires hand_cannon_guild_advance (East Asia only)
		if = {
			limit = { owner = { has_advance = hand_cannon_guild_advance } }
			change_building_level_in_location = {
				building = building_type:hand_cannon_guild

# Assign specialization based on local resources
# Priority: Commercial (special conditions) > Farming > Mining > Gathering > Woodland
# root = location (rural, town, or city)
# Uses constants from expanded_urban_locations_specialization_constants.txt
assign_specialization_by_resources = {
	# Commercial specialization (priority for market centers, harbors, republic capitals)
	if = {
		limit = {
			OR = {
				is_market_center = yes
				modifier:harbor_suitability > 0.7
				AND = {
					is_capital = yes
					location_rank = location_rank:city
					owner = { government_type = government_type:republic }
				}
			}
		}
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = specialization_designation value = sul_commercial_rural_value }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			set_variable = { name = specialization_designation value = sul_commercial_town_value }
		}
		else = {
			set_variable = { name = specialization_designation value = sul_commercial_city_value }
		}
	}
	# Farming specialization
	else_if = {
		limit = { has_farming_resources = yes }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = specialization_designation value = sul_farming_rural_value }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			set_variable = { name = specialization_designation value = sul_farming_town_value }
		}
		else = {
			set_variable = { name = specialization_designation value = sul_farming_city_value }
		}
	}
	# Mining specialization
	else_if = {
		limit = { has_mining_resources = yes }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = specialization_designation value = sul_mining_rural_value }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			set_variable = { name = specialization_designation value = sul_mining_town_value }
		}
		else = {
			set_variable = { name = specialization_designation value = sul_mining_city_value }
		}
	}
	# Gathering specialization
	else_if = {
		limit = { has_gathering_resources = yes }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = specialization_designation value = sul_gathering_rural_value }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			set_variable = { name = specialization_designation value = sul_gathering_town_value }
		}
		else = {
			set_variable = { name = specialization_designation value = sul_gathering_city_value }
		}
	}

	else_if = {

# STARTER VILLAGE GRANT
# Adds 1 level of appropriate village based on specialization AND terrain
# root = location (rural settlement with specialization already assigned)
# Must match building location_potential requirements
# ============================================
grant_starter_village = {
	# Gathering + coastal = fishing_village
	if = {
		limit = {
			has_gathering_specialization = yes
			is_coastal = yes
		}
		change_building_level_in_location = {
			building = building_type:fishing_village
			value = 1
		}
	}
	# Woodland + forest terrain = forest_village
	else_if = {
		limit = {
			has_woodland_specialization = yes
			OR = {
				raw_material = goods:wild_game
				vegetation = forest
				vegetation = woods
				vegetation = jungle
			}
		}
		change_building_level_in_location = {
			building = building_type:forest_village
			value = 1
		}
	}
	# Farming + farming raw materials = farming_village (matches base game location_potential)
	else_if = {
		limit = {
			has_farming_specialization = yes
			OR = {
				raw_material = goods:wheat
				raw_material = goods:maize
				raw_material = goods:rice
				raw_material = goods:millet
				raw_material = goods:legumes
				raw_material = goods:potato
				raw_material = goods:livestock
				raw_material = goods:olives
				raw_material = goods:fruit
				raw_material = goods:wool
			}
		}
		change_building_level_in_location = {
			building = building_type:farming_village
			value = 1
		}
	}
	# Fallback for commercial, mining, or non-matching terrain
	else = {
		change_building_level_in_location = {
			building = building_type:market_village
			value = 1
		}
	}
}

# ============================================
# SPECIALIZATION ELIGIBILITY CACHING
# Calculates and stores which specializations a location can have
# root = location
# Called on: game start, colonization, raw material change
# Note: This is purely terrain/resource based - tech requirements are checked
#       separately when actually designating a specialization
# ============================================
sul_calculate_specialization_eligibility = {
	# Mining eligibility: mining resources OR mountains OR wetlands OR lake adjacency
	if = {
		limit = {
			OR = {
				has_mining_resources = yes
				topography = mountains
				topography = wetlands
				is_adjacent_to_lake = yes
			}
		}
		set_variable = { name = sul_can_mine value = 1 }
	}
	else = {
		remove_variable = sul_can_mine
	}

	# Farming eligibility: farming resources OR farmland OR flatland with river/grasslands
	if = {
		limit = {
			OR = {
				has_farming_resources = yes
				vegetation = farmland
				# Flatland with river access
				AND = {
					topography = flatland
					has_river = yes
				}
				# Flatland with grasslands
				AND = {
					topography = flatland
					vegetation = grasslands
				}
			}
		}
		set_variable = { name = sul_can_farm value = 1 }
	}
	else = {
		remove_variable = sul_can_farm
	}

	# Gathering eligibility: gathering resources OR coastal OR river OR lake adjacency
	if = {
		limit = {
			OR = {
				has_gathering_resources = yes
				is_coastal = yes
				has_river = yes
				is_adjacent_to_lake = yes
			}
		}
		set_variable = { name = sul_can_gather value = 1 }
	}
	else = {
		remove_variable = sul_can_gather
	}

	# Woodland eligibility: woodland resources OR forest/woods/jungle
	if = {
		limit = {
			OR = {
				has_woodland_resources = yes
				vegetation = forest
				vegetation = woods
				vegetation = jungle
			}
		}
		set_variable = { name = sul_can_woodland value = 1 }
	}
	else = {
		remove_variable = sul_can_woodland
	}
}

# ============================================
# SPECIALIZATION VALIDATION
# Checks if current specialization is still valid after raw material change
# If invalid, switches to commercial
# root = location
# ============================================
sul_validate_specialization = {
	# Only validate if location has a specialization
	if = {
		limit = { has_any_specialization = yes }

		# Check if current specialization is still valid
		if = {
			limit = {
				has_mining_specialization = yes
				NOT = { has_variable = sul_can_mine }
			}
			# Mining no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				has_farming_specialization = yes
				NOT = { has_variable = sul_can_farm }
			}
			# Farming no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				has_gathering_specialization = yes
				NOT = { has_variable = sul_can_gather }
			}
			# Gathering no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				has_woodland_specialization = yes
				NOT = { has_variable = sul_can_woodland }
			}
			# Woodland no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
	}
}

# ============================================
# SWITCH TO COMMERCIAL
# Forces a location to commercial specialization
# root = location
# ============================================
sul_switch_to_commercial = {
	# Clear existing modifiers
	clear_all_specialization_modifiers = yes

	# Set commercial variable based on rank
	if = {
		limit = { location_rank = location_rank:rural_settlement }
		set_variable = { name = specialization_designation value = sul_commercial_rural_value }
	}
	else_if = {
		limit = { location_rank = location_rank:town }
		set_variable = { name = specialization_designation value = sul_commercial_town_value }
	}
	else = {
		set_variable = { name = specialization_designation value = sul_commercial_city_value }
	}

	# Apply new modifier
	apply_specialization_modifier = yes

	# Handle buildings for towns/cities
	if = {
		limit = { is_town_or_city = yes }
		destroy_non_matching_buildings = yes
		grant_specialization_starter_buildings = yes
	}
}

# ============================================
# RGO BUILDING GRANT FOR TOWNS/CITIES
# Adds one valid RGO building based on specialization + terrain
# root = location (town/city with specialization assigned)
# Must match building location_potential requirements
# ============================================
grant_specialization_rgo_building = {
	# Mining RGO buildings
	if = {
		limit = { has_mining_specialization = yes }
		# stone_quarry - mountains/plateau/hills
		if = {
			limit = {
				OR = {
					topography = mountains
					topography = plateau
					topography = hills
				}
			}
			change_building_level_in_location = {
				building = building_type:stone_quarry
				value = 1
			}
		}
		# bog_iron_smelter - lake/wetlands, not iron, requires iron_working
		else_if = {
			limit = {
				OR = {
					is_adjacent_to_lake = yes
					topography = wetlands
				}
				NOT = { raw_material = goods:iron }
				owner = { has_advance = iron_working }
			}
			change_building_level_in_location = {
				building = building_type:bog_iron_smelter
				value = 1
			}
		}
		# local_smelters - has metal raw material, requires local_smelters_advance
		else_if = {
			limit = {
				OR = {
					raw_material = goods:iron
					raw_material = goods:copper
					raw_material = goods:tin
					raw_material = goods:lead
				}
				owner = { has_advance = local_smelters_advance }
			}
			change_building_level_in_location = {
				building = building_type:local_smelters
				value = 1
			}
		}
	}
	# Farming RGO buildings
	else_if = {
		limit = { has_farming_specialization = yes }
		# windmill - grain raw materials, requires windmills_advance
		if = {
			limit = {
				OR = {
					raw_material = goods:wheat
					raw_material = goods:rice
					raw_material = goods:millet
				}
				owner = { has_advance = windmills_advance }
			}
			change_building_level_in_location = {
				building = building_type:windmill
				value = 1
			}
		}
		# sheep_farms - sparse/farmland/grasslands
		else_if = {
			limit = {
				OR = {
					vegetation = sparse
					vegetation = farmland
					vegetation = grasslands
				}
			}
			change_building_level_in_location = {
				building = building_type:sheep_farms
				value = 1
			}
		}
		# fruit_orchard - grasslands/farmland/woods/jungle
		else_if = {
			limit = {
				OR = {
					vegetation = grasslands
					vegetation = farmland
					vegetation = woods
					vegetation = jungle
				}
			}
			change_building_level_in_location = {
				building = building_type:fruit_orchard
				value = 1
			}
		}
		# fiber_crops_farm - not arctic, not fiber_crops
		else_if = {
			limit = {
				NOT = { raw_material = goods:fiber_crops }
				NOT = { climate = arctic }
			}
			change_building_level_in_location = {
				building = building_type:fiber_crops_farm
				value = 1
			}
		}
	}
	# Gathering RGO buildings
	else_if = {
		limit = { has_gathering_specialization = yes }
		# clay_pit - coastal/river/lake, not clay
		if = {
			limit = {
				OR = {
					is_coastal = yes
					has_river = yes
					is_adjacent_to_lake = yes
				}
				NOT = { raw_material = goods:clay }
			}
			change_building_level_in_location = {
				building = building_type:clay_pit
				value = 1
			}
		}
		# salt_collector - desert or coastal
		else_if = {
			limit = {
				OR = {
					vegetation = desert
					is_coastal = yes
				}
			}
			change_building_level_in_location = {
				building = building_type:salt_collector
				value = 1
			}
		}
		# sand_pit - not sand (fallback)
		else_if = {
			limit = {
				NOT = { raw_material = goods:sand }
			}
			change_building_level_in_location = {
				building = building_type:sand_pit
				value = 1
			}
		}
	}
	# Woodland RGO buildings
	else_if = {
		limit = { has_woodland_specialization = yes }
		# tar_kiln - woods/forest/jungle OR lumber
		if = {
			limit = {
				OR = {
					vegetation = woods
					vegetation = forest
					vegetation = jungle
					raw_material = goods:lumber
				}
			}
			change_building_level_in_location = {
				building = building_type:tar_kiln
				value = 1
			}
		}
		# lumber_mill - woods/forest/farmland/jungle, not lumber
		else_if = {
			limit = {
				OR = {
					vegetation = woods
					vegetation = forest
					vegetation = farmland
					vegetation = jungle
				}
				NOT = { raw_material = goods:lumber }
			}
			change_building_level_in_location = {
				building = building_type:lumber_mill
				value = 1
			}
		}
		# charcoal_maker - town/city only (our INJECT), fallback
		else = {
			change_building_level_in_location = {
				building = building_type:charcoal_maker
				value = 1
			}
		}
	}
	# Commercial - no RGO buildings, already has marketplace from guild grant
}

sul_initialize_specialization_counts = {
	# Initialize all counts to 0
	set_variable = { name = sul_mining_count value = 0 }
	set_variable = { name = sul_farming_count value = 0 }
	set_variable = { name = sul_woodland_count value = 0 }
	set_variable = { name = sul_gathering_count value = 0 }
	set_variable = { name = sul_commercial_count value = 0 }
	set_variable = { name = sul_total_weighted value = 0 }

	# Count all existing specializations
	every_owned_location = {
		if = {
			limit = { has_mining_specialization = yes }
			owner = {
				change_variable = { name = sul_mining_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { has_farming_specialization = yes }
			owner = {
				change_variable = { name = sul_farming_count add = 1 }
				change_variable = { name = sul_total_weighted add = 2 }
			}
		}
		else_if = {
			limit = { has_woodland_specialization = yes }
			owner = {
				change_variable = { name = sul_woodland_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { has_gathering_specialization = yes }
			owner = {
				change_variable = { name = sul_gathering_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { has_commercial_specialization = yes }
			owner = {
				change_variable = { name = sul_commercial_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
	}
}

# ============================================
# MARKET DEMAND CACHE UPDATE
# Calculates and stores supply/demand ratios and high-demand flags
# root = market
# Called on game start and monthly for all markets
# NOTE: Variables are stored on market center location because markets don't support variables
# Uses market.location to access the market center
# ============================================
sul_update_market_cache = {
	# Save market scope for price lookups, then switch to market center for variable storage
	save_temporary_scope_as = current_market
	location = {
		# Cache market pressure values (supply/demand ratios, ±5 scale)
		set_variable = { name = sul_iron_pressure value = "scope:current_market.sul_iron_market_pressure" }
		set_variable = { name = sul_wool_pressure value = "scope:current_market.sul_wool_market_pressure" }
		set_variable = { name = sul_wheat_pressure value = "scope:current_market.sul_wheat_market_pressure" }
		set_variable = { name = sul_lumber_pressure value = "scope:current_market.sul_lumber_market_pressure" }
		set_variable = { name = sul_salt_pressure value = "scope:current_market.sul_salt_market_pressure" }
		set_variable = { name = sul_clay_pressure value = "scope:current_market.sul_clay_market_pressure" }

		# Cache scaled price bonuses for specializations (read from script values)
		# Formula: (price / base_price - 1) × 8, capped at -8 to +16
		# Positive = shortage (high prices), negative = surplus (low prices)
		set_variable = { name = sul_mining_raw_price_bonus value = "scope:current_market.sul_mining_raw_price_bonus" }
		set_variable = { name = sul_mining_finished_price_bonus value = "scope:current_market.sul_mining_finished_price_bonus" }
		set_variable = { name = sul_farming_raw_price_bonus value = "scope:current_market.sul_farming_raw_price_bonus" }
		set_variable = { name = sul_farming_finished_price_bonus value = "scope:current_market.sul_farming_finished_price_bonus" }
		set_variable = { name = sul_woodland_raw_price_bonus value = "scope:current_market.sul_woodland_raw_price_bonus" }
		set_variable = { name = sul_woodland_finished_price_bonus value = "scope:current_market.sul_woodland_finished_price_bonus" }
		set_variable = { name = sul_gathering_raw_price_bonus value = "scope:current_market.sul_gathering_raw_price_bonus" }
		set_variable = { name = sul_gathering_finished_price_bonus value = "scope:current_market.sul_gathering_finished_price_bonus" }
	}
}

# ============================================
# TRADE SPREAD CALCULATION
# Iterates all goods and calculates total price deviation weighted by base price
# Higher spread = more trade opportunity (prices deviate from normal)
# root = market
# NOTE: Variables are stored on market center location because markets don't support variables
# Uses market.location to access the market center
# ============================================
sul_calculate_trade_spread = {
	# Reset accumulator on market center location
	location = {
		set_variable = { name = sul_trade_spread value = 0 }
	}
}

# Called from every_goods iterator to accumulate trade spread for each market
# root = goods, prev = calling scope
# Uses sul_goods_price_deviation script value: |price/base - 1| × base
sul_accumulate_trade_spread_for_markets = {
	save_temporary_scope_as = current_good
	every_market_in_world = {
		save_temporary_scope_as = current_market
		location = {
			# Get weighted deviation from goods scope, store on this location
			set_variable = { name = sul_temp_deviation value = "scope:current_good.sul_goods_price_deviation" }
			# Add to accumulator
			change_variable = { name = sul_trade_spread add = var:sul_temp_deviation }
		}
	}
}
