# Specialized Urban Locations - Scripted Effects
@sul_mining_type = 1
@sul_farming_type = 2
@sul_gathering_type = 3

@sul_rank_offset_town = 0
@sul_rank_offset_city = 10
@sul_rank_offset_rural = 20

# =============================================================================
# FULL REBUILD
# =============================================================================
# Called on game start and when mod version changes (via version check)
# Minimally invasive: preserves valid player specializations and buildings
# Only reassigns locations whose current spec violates updated rules

sul_full_rebuild = {
	sul_setup_goods_spec_map = yes

	# All ownable locations: recalculate eligibility, then branch
	every_location_in_the_world = {
		limit = { is_ownable = yes exists = raw_material }
		sul_calculate_specialization_eligibility = yes
		sul_rebuild_location = yes
	}

	# GDP per capita cache
	every_country = {
		limit = { sul_is_valid_location_owner = yes }
		sul_update_all_location_gdp = yes
	}

	# Market cache
	sul_refresh_market_cache = yes
}

# =============================================================================
# REBUILD SINGLE LOCATION
# =============================================================================
# Handles one location during rebuild. Three paths:
# 1. No specialization -> fresh assignment with buildings
# 2. Valid specialization -> refresh modifier only (handles value changes)
# 3. Invalid specialization -> reassign by resources + rebuild buildings
# root = location (eligibility already calculated)

sul_rebuild_location = {
	if = {
		limit = { sul_has_any_specialization = yes }

		if = {
			# Current spec still valid - just refresh the modifier
			limit = { sul_current_specialization_is_valid = yes }
			sul_refresh_specialization_modifier = yes
		}
		else = {
			# Current spec no longer valid - full reassignment
			sul_reassign_specialization = yes
		}
	}
	else = {
		# No specialization yet - first-time init
		sul_init_new_specialization = yes
	}
}

# =============================================================================
# REFRESH SPECIALIZATION MODIFIER
# =============================================================================
# Swaps the modifier for the current spec+rank without changing anything else
# root = location (must have sul_spec_type set)

sul_refresh_specialization_modifier = {
	sul_clear_all_specialization_modifiers = yes
	sul_update_designation = yes
	sul_apply_specialization_modifier = yes
}

# =============================================================================
# REASSIGN SPECIALIZATION
# =============================================================================
# Current spec is no longer valid under updated rules
# Clears old state, picks best match from resources, rebuilds buildings
# root = location

sul_reassign_specialization = {
	sul_clear_all_specialization_modifiers = yes
	sul_assign_specialization_by_resources = yes
	sul_apply_specialization_modifier = yes

	# Rebuild buildings for owned locations
	if = {
		limit = { exists = owner }
		sul_save_building_levels = yes
		sul_destroy_non_matching_buildings = yes
		sul_bootstrap_economy = yes
	}
}

# =============================================================================
# INIT NEW SPECIALIZATION
# =============================================================================
# First-time setup for a location that has never been specialized
# root = location

sul_init_new_specialization = {
	set_variable = {
		name = sul_local_gdp
		value = 0
	}

	sul_assign_specialization_by_resources = yes
	sul_apply_specialization_modifier = yes

	# Grant buildings for owned locations
	if = {
		limit = { exists = owner }
		sul_save_building_levels = yes
		sul_destroy_non_matching_buildings = yes
		sul_bootstrap_economy = yes
	}
}

# BUILDING DESTRUCTION EFFECTS
# Each effect destroys all specialization-relevant buildings EXCEPT those matching the specialization
# root = location
# NOTE: Universities are always preserved in cities regardless of specialization
#
# Uses add_to_list pattern to build list first, then destroy via every_in_list
# This avoids null reference errors from modifying collection during iteration
# ============================================
# Mining Specialization - keeps mining buildings

sul_destroy_non_mining_buildings = {
	every_buildings_in_location = {
		limit = {
			sul_is_destroyable_specialization_building = yes
			NOT = {
				sul_is_mining_building = yes
			}

			# Preserve universities in cities
			NAND = {
				building_type = building_type:university
				location = { location_rank = location_rank:city }
			}
		}

		location = { destroy_building = prev }
	}
}

# Destroy non-matching buildings based on specialization type
# root = location (must have sul_spec_type variable set)
# Runs for all location types (rural, town, city)

sul_destroy_non_matching_buildings = {
	if = {
		limit = { sul_has_mining_specialization = yes }
		sul_destroy_non_mining_buildings = yes
	}
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		sul_destroy_non_farming_buildings = yes
	}
	else_if = {
		limit = { sul_has_woodland_specialization = yes }
		sul_destroy_non_woodland_buildings = yes
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		sul_destroy_non_gathering_buildings = yes
	}
	else_if = {
		limit = { sul_has_commercial_specialization = yes }
		sul_destroy_non_commercial_buildings = yes
	}
}

# Assign specialization based on local resources
# Priority: Commercial (special conditions) > Farming > Mining > Gathering > Woodland
# root = location (rural, town, or city)
# Sets sul_spec_type to type only (1-5); rank derived from location_rank

sul_assign_specialization_by_resources = {
	# Commercial specialization (market centers always; cities require development > 30; towns pass through)
	if = {
		limit = {
			OR = {
				is_market_center = yes
				# Towns/rural qualify without dev check but owner must have more than one location
				AND = {
					NOT = {
						location_rank = location_rank:city
					}
					owner = { num_locations > 1 }
					OR = {
						modifier:harbor_suitability > 0.7
						AND = {
							is_capital = yes
							owner = { government_type = government_type:republic }
						}
					}
				}

				# Cities require development > 30
				AND = {
					location_rank = location_rank:city
					development > 30
					OR = {
						modifier:harbor_suitability > 0.7
						AND = {
							is_capital = yes
							owner = { government_type = government_type:republic }
						}
					}
				}
			}
		}

		set_variable = { name = sul_spec_type value = @sul_commercial_type }
	}

	# Farming specialization
	else_if = {
		limit = { sul_has_farming_resources = yes }
		set_variable = { name = sul_spec_type value = @sul_farming_type }
	}

	# Mining specialization
	else_if = {
		limit = { sul_has_mining_resources = yes }
		set_variable = { name = sul_spec_type value = @sul_mining_type }
	}

	# Gathering specialization
	else_if = {
		limit = { sul_has_gathering_resources = yes }
		set_variable = { name = sul_spec_type value = @sul_gathering_type }
	}

	# Woodland specialization
	else_if = {
		limit = { sul_has_woodland_resources = yes }
		set_variable = { name = sul_spec_type value = @sul_woodland_type }
	}

	# Update combined designation for GUI
	if = {
		limit = { has_variable = sul_spec_type }
		sul_update_designation = yes
	}
}

# Update sul_designation from sul_spec_type + current rank
# Encodes type+rank into one value for GUI (avoids rank lookups in GUI context)
# Town: 1-5, City: 11-15, Rural: 21-25
# root = location (must have sul_spec_type set)

sul_update_designation = {
	set_variable = {
		name = sul_designation
		value = var:sul_spec_type
	}
	if = {
		limit = { location_rank = location_rank:city }
		change_variable = { name = sul_designation add = @sul_rank_offset_city }
	}
	else_if = {
		limit = { location_rank = location_rank:rural_settlement }
		change_variable = { name = sul_designation add = @sul_rank_offset_rural }
	}
}

# Apply correct modifier based on specialization type and current rank
# root = location (must have sul_spec_type variable set)

sul_apply_specialization_modifier = {

	if = {
		limit = { var:sul_spec_type = @sul_mining_type }

			limit = { location_rank = location_rank:rural_settlement }
			add_location_modifier = { modifier = sul_mining_rural_specialization }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			add_location_modifier = { modifier = sul_mining_town_specialization }
		}
		else = { add_location_modifier = { modifier = sul_mining_city_specialization } }

	else_if = {
		limit = { var:sul_spec_type = @sul_farming_type }

		}
		else_if = {

			add_location_modifier = { modifier = sul_farming_town_specialization }
		}
		else = {
			add_location_modifier = { modifier = sul_farming_city_specialization }
		}
	}

	else_if = {

		if = {
			limit = { location_rank = location_rank:rural_settlement }
			add_location_modifier = { modifier = sul_woodland_rural_specialization }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			add_location_modifier = { modifier = sul_woodland_town_specialization }
		}
		else = {
			add_location_modifier = { modifier = sul_woodland_city_specialization }
		}
	}

	# Gathering
	else_if = {
		limit = { var:sul_spec_type = @sul_gathering_type }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			add_location_modifier = { modifier = sul_gathering_rural_specialization }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			add_location_modifier = { modifier = sul_gathering_town_specialization }
		}
		else = {
			add_location_modifier = { modifier = sul_gathering_city_specialization }
		}
	}

	# Commercial
	else_if = {
		limit = { var:sul_spec_type = @sul_commercial_type }

			limit = { location_rank = location_rank:rural_settlement }
			add_location_modifier = { modifier = sul_commercial_rural_specialization }

			limit = { location_rank = location_rank:town }
			add_location_modifier = { modifier = sul_commercial_town_specialization }
		}
		else = {

# ============================================
# DIVERSITY TRACKING INITIALIZATION
# Counts all existing specializations and populates cache variables
# root = country
# Called when variables don't exist (first run after mod update or for existing saves)
# ============================================
# ============================================
# SPECIALIZATION ELIGIBILITY CACHING
# Calculates and stores which specializations a location can have
# root = location
# Called on: game start, colonization, raw material change
# Note: This is purely terrain/resource based - tech requirements are checked
#       separately when actually designating a specialization
# ============================================

sul_calculate_specialization_eligibility = {
	# Mining eligibility: mining resources OR mountains OR wetlands OR lake adjacency
	if = {
		limit = {
			OR = {
				sul_has_mining_resources = yes
				topography = mountains
				topography = wetlands
				is_adjacent_to_lake = yes
			}
		}

		set_variable = { name = sul_can_mine value = 1 }
	}
	else = {
		remove_variable = sul_can_mine
	}

	# Farming eligibility: farming resources OR farmland OR flatland with river/grasslands
	if = {
		limit = {
			OR = {
				sul_has_farming_resources = yes
				vegetation = farmland
				# Flatland with river access
				AND = {
					topography = flatland
					has_river = yes
				}

				# Flatland with grasslands
				AND = {
					topography = flatland
					vegetation = grasslands
				}
			}
		}

		set_variable = { name = sul_can_farm value = 1 }
	}
	else = {
		remove_variable = sul_can_farm
	}

	# Gathering eligibility: gathering resources OR coastal OR river OR lake adjacency
	if = {
		limit = {
			OR = {
				sul_has_gathering_resources = yes
				is_coastal = yes
				has_river = yes
				is_adjacent_to_lake = yes
			}
		}

		set_variable = { name = sul_can_gather value = 1 }
	}
	else = {
		remove_variable = sul_can_gather
	}

	# Woodland eligibility: woodland resources OR forest/woods/jungle
	if = {
		limit = {
			OR = {
				sul_has_woodland_resources = yes
				vegetation = forest
				vegetation = woods
				vegetation = jungle
			}
		}

		set_variable = { name = sul_can_woodland value = 1 }
	}
	else = {
		remove_variable = sul_can_woodland
	}
}

# ============================================
# SPECIALIZATION VALIDATION
# Checks if current specialization is still valid after raw material change
# If invalid, switches to commercial
# root = location
# ============================================

sul_validate_specialization = {
	# Only validate if location has a specialization
	if = {
		limit = { sul_has_any_specialization = yes }
		# Check if current specialization is still valid
		if = {
			limit = {
				sul_has_mining_specialization = yes
				NOT = {
					has_variable = sul_can_mine
				}
			}

			# Mining no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				sul_has_farming_specialization = yes
				NOT = {
					has_variable = sul_can_farm
				}
			}

			# Farming no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				sul_has_gathering_specialization = yes
				NOT = {
					has_variable = sul_can_gather
				}
			}

			# Gathering no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				sul_has_woodland_specialization = yes
				NOT = {
					has_variable = sul_can_woodland
				}
			}

			# Woodland no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
	}
}

# ============================================
# SWITCH TO COMMERCIAL
# Forces a location to commercial specialization
# root = location
# ============================================

sul_switch_to_commercial = {
	# Clear existing modifiers
	sul_clear_all_specialization_modifiers = yes
	# Set commercial type
	set_variable = {
		name = sul_spec_type
		value = @sul_commercial_type
	}

	# Apply new modifier
	sul_apply_specialization_modifier = yes
	# Handle buildings for towns/cities
	if = {
		limit = {
			OR = {
				location_rank = location_rank:town
				location_rank = location_rank:city
			}
		}

		sul_destroy_non_matching_buildings = yes
	}
}

# ============================================
# RGO BUILDING GRANT FOR TOWNS/CITIES
# Adds one valid RGO building based on specialization + terrain
# root = location (town/city with specialization assigned)
# Must match building location_potential requirements
# ============================================

sul_grant_specialization_rgo_building = {
	# Mining RGO buildings
	if = {
		limit = { sul_has_mining_specialization = yes }
		# stone_quarry - mountains, plateau, or hills (not stone raw material)
		if = {
			limit = {
				OR = {
					topography = mountains
					topography = plateau
					topography = hills
				}
				NOT = {
					raw_material = goods:stone
				}
			}

			change_building_level_in_location = {
				building = building_type:stone_quarry
				value = 1
			}
		}

		# bog_iron_smelter - lake/wetlands, not iron, requires iron_working
		else_if = {
			limit = {
				OR = {
					is_adjacent_to_lake = yes
					topography = wetlands
				}
				NOT = {
					raw_material = goods:iron
				}
				owner = { has_advance = iron_working }
			}

			change_building_level_in_location = {
				building = building_type:bog_iron_smelter
				value = 1
			}
		}

		# local_smelters - has metal raw material, requires local_smelters_advance
		else_if = {
			limit = {
				OR = {
					raw_material = goods:iron
					raw_material = goods:copper
					raw_material = goods:tin
					raw_material = goods:lead
				}
				owner = { has_advance = local_smelters_advance }
			}

			change_building_level_in_location = {
				building = building_type:local_smelters
				value = 1
			}
		}
	}

	# Farming RGO buildings
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		# windmill - grain raw materials, requires windmills_advance
		if = {
			limit = {
				OR = {
					raw_material = goods:wheat
					raw_material = goods:rice
					raw_material = goods:millet
				}
				owner = { has_advance = windmills_advance }
			}

			change_building_level_in_location = {
				building = building_type:windmill
				value = 1
			}
		}

		# fruit_orchard - grasslands/farmland/woods/jungle
		else_if = {
			limit = {
				OR = {
					vegetation = grasslands
					vegetation = farmland
					vegetation = woods
					vegetation = jungle
				}
			}

			change_building_level_in_location = {
				building = building_type:fruit_orchard
				value = 1
			}
		}

		# fiber_crops_farm - not arctic, not fiber_crops
		else_if = {
			limit = {
				NOT = {
					raw_material = goods:fiber_crops
				}
				NOT = {
					climate = arctic
				}
			}

			change_building_level_in_location = {
				building = building_type:fiber_crops_farm
				value = 1
			}
		}
	}

	# Gathering RGO buildings
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		# sheep_farms - sparse/farmland/grasslands
		if = {
			limit = {
				OR = {
					vegetation = sparse
					vegetation = farmland
					vegetation = grasslands
				}
			}

			change_building_level_in_location = {
				building = building_type:sheep_farms
				value = 1
			}
		}

		# clay_pit - coastal/river/lake, not clay
		else_if = {
			limit = {
				OR = {
					is_coastal = yes
					has_river = yes
					is_adjacent_to_lake = yes
				}
				NOT = {
					raw_material = goods:clay
				}
			}

			change_building_level_in_location = {
				building = building_type:clay_pit
				value = 1
			}
		}

		# salt_collector - desert or coastal
		else_if = {
			limit = {
				OR = {
					vegetation = desert
					is_coastal = yes
				}
			}

			change_building_level_in_location = {
				building = building_type:salt_collector
				value = 1
			}
		}

		# sand_pit - not sand (fallback)
		else_if = {
			limit = {
				NOT = {
					raw_material = goods:sand
				}
			}

			change_building_level_in_location = {
				building = building_type:sand_pit
				value = 1
			}
		}
	}

	# Woodland RGO buildings
	else_if = {
		limit = { sul_has_woodland_specialization = yes }
		# tar_kiln - woods/forest/jungle OR lumber
		if = {
			limit = {
				OR = {
					vegetation = woods
					vegetation = forest
					vegetation = jungle
					raw_material = goods:lumber
				}
			}

			change_building_level_in_location = {
				building = building_type:tar_kiln
				value = 1
			}
		}

		# lumber_mill - woods/forest/farmland/jungle, not lumber
		else_if = {
			limit = {
				OR = {
					vegetation = woods
					vegetation = forest
					vegetation = farmland
					vegetation = jungle
				}
				NOT = {
					raw_material = goods:lumber
				}
			}

			change_building_level_in_location = {
				building = building_type:lumber_mill
				value = 1
			}
		}

		# charcoal_maker - town/city only (our INJECT), fallback
		else = {
			change_building_level_in_location = {
				building = building_type:charcoal_maker
				value = 1
			}
		}
	}

	# Commercial - no RGO buildings, already has marketplace from guild grant
}

sul_initialize_specialization_counts = {
	# Initialize all counts to 0
	set_variable = {
		name = sul_mining_count
		value = 0
	}
	set_variable = {
		name = sul_farming_count
		value = 0
	}
	set_variable = {
		name = sul_woodland_count
		value = 0
	}
	set_variable = {
		name = sul_gathering_count
		value = 0
	}
	set_variable = {
		name = sul_commercial_count
		value = 0
	}
	set_variable = {
		name = sul_total_weighted
		value = 0
	}

	# Count all existing specializations
	every_owned_location = {
		if = {
			limit = { sul_has_mining_specialization = yes }

			owner = {
				change_variable = { name = sul_mining_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { sul_has_farming_specialization = yes }

			owner = {
				change_variable = { name = sul_farming_count add = 1 }
				change_variable = { name = sul_total_weighted add = 2 }
			}
		}
		else_if = {
			limit = { sul_has_woodland_specialization = yes }

			owner = {
				change_variable = { name = sul_woodland_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { sul_has_gathering_specialization = yes }

			owner = {
				change_variable = { name = sul_gathering_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { sul_has_commercial_specialization = yes }

			owner = {
				change_variable = { name = sul_commercial_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
	}
}

sul_update_market_cache = {
	# Cache iron pressure for wetland bog iron bonus
	add_to_global_variable_map = {
		name = sul_iron_pressure
		key = this
		value = sul_iron_market_pressure
	}

	# Formula: average (demand/supply - 1) across all finished goods, ×5, capped ±5

		key = this
