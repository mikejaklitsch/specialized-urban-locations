# ============================================
# BUILDING LEVEL COUNTING
# Saves total building levels before specialization
# root = location
# Sets sul_prior_building_levels variable
# ============================================

sul_save_building_levels = {
	set_variable = { name = sul_prior_building_levels value = total_building_levels }
}

# ============================================
# BUILDING LEVEL RESTORATION
# After specialization, adds buildings until we reach prior count
# root = location (must have sul_prior_building_levels and specialization set)
#
# Uses math-based distribution: calculates levels per building upfront
# instead of cycling one-at-a-time. Deterministic and efficient.
# ============================================

sul_restore_building_levels = {
	# Grant terrain-specific RGO building first
	sul_grant_specialization_rgo_building = yes

	# Calculate how many more levels we need to add
	set_variable = {
		name = sul_levels_to_add
		value = {
			value = var:sul_prior_building_levels
			subtract = total_building_levels
			min = 0
		}
	}

	# Distribute buildings based on specialization
	if = {
		limit = { sul_has_mining_specialization = yes }
		sul_distribute_mining_buildings = yes
	}
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		sul_distribute_farming_buildings = yes
	}
	else_if = {
		limit = { sul_has_woodland_specialization = yes }
		sul_distribute_woodland_buildings = yes
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		sul_distribute_gathering_buildings = yes
	}
	else_if = {
		limit = { sul_has_commercial_specialization = yes }
		sul_distribute_commercial_buildings = yes
	}

	# Cleanup
	remove_variable = sul_prior_building_levels
	remove_variable = sul_levels_to_add
}

# ============================================
# MINING BUILDING DISTRIBUTION
# root = location with sul_levels_to_add set
# ============================================
sul_distribute_mining_buildings = {
	# Count valid building options (tech requirements only - max level caps naturally)
	# Note: hand_cannon_guild excluded - max level 1, not suitable for distribution
	set_variable = { name = sul_valid_count value = 0 }

	# jewelry_guild - always available
	change_variable = { name = sul_valid_count add = 1 }
	set_variable = { name = sul_b1_valid value = 1 }

	if = {
		limit = { owner = { has_advance = metallurgy } }
		change_variable = { name = sul_valid_count add = 1 }
		set_variable = { name = sul_b2_valid value = 1 }
	}
	if = {
		limit = { owner = { has_advance = weapon_guild_advance } }
		change_variable = { name = sul_valid_count add = 1 }
		set_variable = { name = sul_b3_valid value = 1 }
	}
	if = {
		limit = { owner = { has_advance = cannon_maker_advance } }
		change_variable = { name = sul_valid_count add = 1 }
		set_variable = { name = sul_b4_valid value = 1 }
	}

	# Calculate distribution (skip if no valid buildings)
	if = {
		limit = { var:sul_valid_count > 0 }
		set_variable = {
			name = sul_base
			value = { value = var:sul_levels_to_add divide = var:sul_valid_count floor = yes }
		}
		set_variable = {
			name = sul_remainder
			value = { value = var:sul_levels_to_add modulo = var:sul_valid_count }
		}

		# Add to each valid building
		if = {
			limit = { has_variable = sul_b1_valid }
			if = {
				limit = { var:sul_remainder > 0 }
				change_building_level_in_location = { building = building_type:jewelry_guild value = { value = var:sul_base add = 1 } }
				change_variable = { name = sul_remainder subtract = 1 }
			}
			else = {
				change_building_level_in_location = { building = building_type:jewelry_guild value = var:sul_base }
			}
		}
		if = {
			limit = { has_variable = sul_b2_valid }
			if = {
				limit = { var:sul_remainder > 0 }
				change_building_level_in_location = { building = building_type:tools_guild value = { value = var:sul_base add = 1 } }
				change_variable = { name = sul_remainder subtract = 1 }
			}
			else = {
				change_building_level_in_location = { building = building_type:tools_guild value = var:sul_base }
			}
		}
		if = {
			limit = { has_variable = sul_b3_valid }
			if = {
				limit = { var:sul_remainder > 0 }
				change_building_level_in_location = { building = building_type:weapon_guild value = { value = var:sul_base add = 1 } }
				change_variable = { name = sul_remainder subtract = 1 }
			}
			else = {
				change_building_level_in_location = { building = building_type:weapon_guild value = var:sul_base }
			}
		}
		if = {
			limit = { has_variable = sul_b4_valid }
			if = {
				limit = { var:sul_remainder > 0 }
				change_building_level_in_location = { building = building_type:cannon_maker value = { value = var:sul_base add = 1 } }
				change_variable = { name = sul_remainder subtract = 1 }
			}
			else = {
				change_building_level_in_location = { building = building_type:cannon_maker value = var:sul_base }
			}
		}
	}

	# Cleanup
	remove_variable = sul_valid_count
	remove_variable = sul_base
	remove_variable = sul_remainder
	remove_variable = sul_b1_valid
	remove_variable = sul_b2_valid
	remove_variable = sul_b3_valid
	remove_variable = sul_b4_valid
}

# BUILDING DESTRUCTION EFFECTS
# Each effect destroys all specialization-relevant buildings EXCEPT those matching the specialization
# root = location
# NOTE: Building destruction only applies to towns/cities, NOT rural settlements
# NOTE: Universities are always preserved in cities regardless of specialization
#
# Uses add_to_list pattern to build list first, then destroy via every_in_list
# This avoids null reference errors from modifying collection during iteration
# ============================================

# Mining Specialization - keeps mining buildings
sul_destroy_non_mining_buildings = {
	every_buildings_in_location = {
		limit = {
			sul_is_destroyable_specialization_building = yes
			NOT = { sul_is_mining_building = yes }
			# Preserve universities in cities
			NAND = {
				building_type = building_type:university
				location = { location_rank = location_rank:city }
			}
		}
		location = { destroy_building = prev }
	}
}

# MODIFIER MANAGEMENT EFFECTS
# root = location

# Remove all specialization modifiers (rural, town, and city)
# Checks specialization TYPE from variable and removes ALL rank variants that exist
sul_clear_all_specialization_modifiers = {
	if = {
		limit = { sul_has_mining_specialization = yes }
		if = { limit = { has_location_modifier = sul_mining_rural_specialization } remove_location_modifier = sul_mining_rural_specialization }
		if = { limit = { has_location_modifier = sul_mining_town_specialization } remove_location_modifier = sul_mining_town_specialization }
		if = { limit = { has_location_modifier = sul_mining_city_specialization } remove_location_modifier = sul_mining_city_specialization }
	}
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		if = { limit = { has_location_modifier = sul_farming_rural_specialization } remove_location_modifier = sul_farming_rural_specialization }
		if = { limit = { has_location_modifier = sul_farming_town_specialization } remove_location_modifier = sul_farming_town_specialization }

# ============================================
# SPECIALIZATION ELIGIBILITY CACHING
# Calculates and stores which specializations a location can have
# root = location
# Called on: game start, colonization, raw material change
# Note: This is purely terrain/resource based - tech requirements are checked
#       separately when actually designating a specialization
# ============================================
sul_calculate_specialization_eligibility = {
	# Mining eligibility: mining resources OR mountains OR wetlands OR lake adjacency
	if = {
		limit = {
			OR = {
				sul_has_mining_resources = yes
				topography = mountains
				topography = wetlands
				is_adjacent_to_lake = yes
			}
		}
		set_variable = { name = sul_can_mine value = 1 }
	}
	else = {
		remove_variable = sul_can_mine
	}

	# Farming eligibility: farming resources OR farmland OR flatland with river/grasslands
	if = {
		limit = {
			OR = {
				sul_has_farming_resources = yes
				vegetation = farmland
				# Flatland with river access
				AND = {
					topography = flatland
					has_river = yes
				}
				# Flatland with grasslands
				AND = {
					topography = flatland
					vegetation = grasslands
				}
			}
		}
		set_variable = { name = sul_can_farm value = 1 }
	}
	else = {
		remove_variable = sul_can_farm
	}

	# Gathering eligibility: gathering resources OR coastal OR river OR lake adjacency
	if = {
		limit = {
			OR = {
				sul_has_gathering_resources = yes
				is_coastal = yes
				has_river = yes
				is_adjacent_to_lake = yes
			}
		}
		set_variable = { name = sul_can_gather value = 1 }
	}
	else = {
		remove_variable = sul_can_gather
	}

	# Woodland eligibility: woodland resources OR forest/woods/jungle
	if = {
		limit = {
			OR = {
				sul_has_woodland_resources = yes
				vegetation = forest
				vegetation = woods
				vegetation = jungle
			}
		}
		set_variable = { name = sul_can_woodland value = 1 }
	}
	else = {
		remove_variable = sul_can_woodland
	}
}

# ============================================
# SPECIALIZATION VALIDATION
# Checks if current specialization is still valid after raw material change
# If invalid, switches to commercial
# root = location
# ============================================
sul_validate_specialization = {
	# Only validate if location has a specialization
	if = {
		limit = { sul_has_any_specialization = yes }

		# Check if current specialization is still valid
		if = {
			limit = {
				sul_has_mining_specialization = yes
				NOT = { has_variable = sul_can_mine }
			}
			# Mining no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				sul_has_farming_specialization = yes
				NOT = { has_variable = sul_can_farm }
			}
			# Farming no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				sul_has_gathering_specialization = yes
				NOT = { has_variable = sul_can_gather }
			}
			# Gathering no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				sul_has_woodland_specialization = yes
				NOT = { has_variable = sul_can_woodland }
			}
			# Woodland no longer valid - switch to commercial
			sul_switch_to_commercial = yes
		}
	}
}

# ============================================
# SWITCH TO COMMERCIAL
# Forces a location to commercial specialization
# root = location
# ============================================
sul_switch_to_commercial = {
	# Clear existing modifiers
	sul_clear_all_specialization_modifiers = yes

	# Set commercial variable based on rank
	if = {
		limit = { location_rank = location_rank:rural_settlement }
		set_variable = { name = sul_specialization_designation value = sul_commercial_rural_value }
	}
	else_if = {
		limit = { location_rank = location_rank:town }
		set_variable = { name = sul_specialization_designation value = sul_commercial_town_value }
	}
	else = {
		set_variable = { name = sul_specialization_designation value = sul_commercial_city_value }
	}

	# Apply new modifier
	sul_apply_specialization_modifier = yes

	# Handle buildings for towns/cities
	if = {
		limit = { OR = { location_rank = location_rank:town location_rank = location_rank:city } }
		sul_destroy_non_matching_buildings = yes
	}
}

# ============================================
# RGO BUILDING GRANT FOR TOWNS/CITIES
# Adds one valid RGO building based on specialization + terrain
# root = location (town/city with specialization assigned)
# Must match building location_potential requirements
# ============================================
sul_grant_specialization_rgo_building = {
	# Mining RGO buildings
	if = {
		limit = { sul_has_mining_specialization = yes }
		# stone_quarry - mountains only
		if = {
			limit = { topography = mountains }
			change_building_level_in_location = {
				building = building_type:stone_quarry
				value = 1
			}
		}
		# bog_iron_smelter - lake/wetlands, not iron, requires iron_working
		else_if = {
			limit = {
				OR = {
					is_adjacent_to_lake = yes
					topography = wetlands
				}
				NOT = { raw_material = goods:iron }
				owner = { has_advance = iron_working }
			}
			change_building_level_in_location = {
				building = building_type:bog_iron_smelter
				value = 1
			}
		}
		# local_smelters - has metal raw material, requires local_smelters_advance
		else_if = {
			limit = {
				OR = {
					raw_material = goods:iron
					raw_material = goods:copper
					raw_material = goods:tin
					raw_material = goods:lead
				}
				owner = { has_advance = local_smelters_advance }
			}
			change_building_level_in_location = {
				building = building_type:local_smelters
				value = 1
			}
		}
	}
	# Farming RGO buildings
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		# windmill - grain raw materials, requires windmills_advance
		if = {
			limit = {
				OR = {
					raw_material = goods:wheat
					raw_material = goods:rice
					raw_material = goods:millet
				}
				owner = { has_advance = windmills_advance }
			}
			change_building_level_in_location = {
				building = building_type:windmill
				value = 1
			}
		}
		# fruit_orchard - grasslands/farmland/woods/jungle
		else_if = {
			limit = {
				OR = {
					vegetation = grasslands
					vegetation = farmland
					vegetation = woods
					vegetation = jungle
				}
			}
			change_building_level_in_location = {
				building = building_type:fruit_orchard
				value = 1
			}
		}
		# fiber_crops_farm - not arctic, not fiber_crops
		else_if = {
			limit = {
				NOT = { raw_material = goods:fiber_crops }
				NOT = { climate = arctic }
			}
			change_building_level_in_location = {
				building = building_type:fiber_crops_farm
				value = 1
			}
		}
	}
	# Gathering RGO buildings
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		# sheep_farms - sparse/farmland/grasslands
		if = {
			limit = {
				OR = {
					vegetation = sparse
					vegetation = farmland
					vegetation = grasslands
				}
			}
			change_building_level_in_location = {
				building = building_type:sheep_farms
				value = 1
			}
		}
		# clay_pit - coastal/river/lake, not clay
		else_if = {
			limit = {
				OR = {
					is_coastal = yes
					has_river = yes
					is_adjacent_to_lake = yes
				}
				NOT = { raw_material = goods:clay }
			}
			change_building_level_in_location = {
				building = building_type:clay_pit
				value = 1
			}
		}
		# salt_collector - desert or coastal
		else_if = {
			limit = {
				OR = {
					vegetation = desert
					is_coastal = yes
				}
			}
			change_building_level_in_location = {
				building = building_type:salt_collector
				value = 1
			}
		}
		# sand_pit - not sand (fallback)
		else_if = {
			limit = {
				NOT = { raw_material = goods:sand }
			}
			change_building_level_in_location = {
				building = building_type:sand_pit
				value = 1
			}
		}
	}
	# Woodland RGO buildings
	else_if = {
		limit = { sul_has_woodland_specialization = yes }
		# tar_kiln - woods/forest/jungle OR lumber
		if = {
			limit = {
				OR = {
					vegetation = woods
					vegetation = forest
					vegetation = jungle
					raw_material = goods:lumber
				}
			}
			change_building_level_in_location = {
				building = building_type:tar_kiln
				value = 1
			}
		}
		# lumber_mill - woods/forest/farmland/jungle, not lumber
		else_if = {
			limit = {
				OR = {
					vegetation = woods
					vegetation = forest
					vegetation = farmland
					vegetation = jungle
				}
				NOT = { raw_material = goods:lumber }
			}
			change_building_level_in_location = {
				building = building_type:lumber_mill
				value = 1
			}
		}
		# charcoal_maker - town/city only (our INJECT), fallback
		else = {
			change_building_level_in_location = {
				building = building_type:charcoal_maker
				value = 1
			}
		}
	}
	# Commercial - no RGO buildings, already has marketplace from guild grant
}

# ============================================
# POPULATION CAPACITY PRESSURE UPDATE
# root = location
#
# Uses tiered modifiers (no dynamic size overhead)
# Variable updated every call, modifier only switches on tier change
# Tiers: 0, 10, 20, 30, 40, 50, 60, 70 (neutral), 80, 90, 100
# ============================================
sul_update_capacity_pressure = {
	# Always update variable (cheap)
	set_variable = {
		name = sul_pop_pct
		value = {
			value = location_population_percentage
			multiply = 100
			floor = yes
		}
	}

	# Calculate current tier (round down to nearest 10, clamp 0-100)
	set_variable = {
		name = sul_pop_tier_new
		value = {
			value = var:sul_pop_pct
			divide = 10
			floor = yes
			multiply = 10
			min = 0
			max = 100
		}
	}

	# Only switch modifier if tier changed
	if = {
		limit = {
			OR = {
				NOT = { has_variable = sul_pop_tier }
				NOT = { var:sul_pop_tier = var:sul_pop_tier_new }
			}
		}
		# Remove old modifier
		sul_remove_capacity_pressure = yes
		# Apply new modifier
		sul_apply_capacity_pressure_tier = yes
		# Update cached tier
		set_variable = { name = sul_pop_tier value = var:sul_pop_tier_new }
	}

	# Cleanup temp variable
	remove_variable = sul_pop_tier_new
}

# Remove any existing capacity pressure modifier

	if = { limit = { has_location_modifier = sul_capacity_pressure_0 } remove_location_modifier = sul_capacity_pressure_0 }
