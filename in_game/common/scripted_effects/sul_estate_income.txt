# =============================================================================
# ESTATE INCOME FROM CONTROL SHORTFALL
# =============================================================================
#
# Estates receive income from taxes the crown loses due to low control.
#
# ARCHITECTURE:
# - sul_update_all_location_gdp (monthly) calculates and caches:
#   - sul_local_gdp on each location (for pop demand scaling)
#   - sul_X_taxes_lost on country (per-estate, based on local tax base shares)
#
# - This file just reads the pre-calculated per-estate values.
#
# =============================================================================

# =============================================================================
# PER-ESTATE INCOME CALCULATION
# =============================================================================
# Called from estate scope. Reads pre-calculated sul_X_taxes_lost from monthly update.
# Distribution is based on LOCAL tax base shares, not national estate power.

sul_calculate_estate_income_lost = {
	# Guard: skip if taxes_lost not calculated yet (newly formed country, etc.)
	if = {
		limit = { owner = { NOT = { has_variable = sul_nobles_taxes_lost } } }
		# Variables not initialized - skip this tick, will be set next economy update
	}
	# Copy pre-calculated value to estate-specific var (for GUI and event use)
	else_if = { limit = { estate_type = estate_type:nobles_estate }
		owner = { set_variable = { name = sul_estate_nobles_income_lost value = { value = var:sul_nobles_taxes_lost min = 0 } } }
	}
	else_if = { limit = { estate_type = estate_type:clergy_estate }
		owner = { set_variable = { name = sul_estate_clergy_income_lost value = { value = var:sul_clergy_taxes_lost min = 0 } } }
	}
	else_if = { limit = { estate_type = estate_type:burghers_estate }
		owner = { set_variable = { name = sul_estate_burghers_income_lost value = { value = var:sul_burghers_taxes_lost min = 0 } } }
	}
	else_if = { limit = { estate_type = estate_type:peasants_estate }
		owner = { set_variable = { name = sul_estate_peasants_income_lost value = { value = var:sul_peasants_taxes_lost min = 0 } } }
	}
	else_if = { limit = { estate_type = estate_type:tribes_estate }
		owner = { set_variable = { name = sul_estate_tribes_income_lost value = { value = var:sul_tribes_taxes_lost min = 0 } } }
	}
}

# Grant monthly income to estate
sul_grant_estate_income_lost = {
	if = { limit = { estate_type = estate_type:nobles_estate }
		estate_add_gold = { value = owner.var:sul_estate_nobles_income_lost }
	}
	else_if = { limit = { estate_type = estate_type:clergy_estate }
		estate_add_gold = { value = owner.var:sul_estate_clergy_income_lost }
	}
	else_if = { limit = { estate_type = estate_type:burghers_estate }
		estate_add_gold = { value = owner.var:sul_estate_burghers_income_lost }
	}
	else_if = { limit = { estate_type = estate_type:peasants_estate }
		estate_add_gold = { value = owner.var:sul_estate_peasants_income_lost }
	}
	else_if = { limit = { estate_type = estate_type:tribes_estate }
		estate_add_gold = { value = owner.var:sul_estate_tribes_income_lost }
	}
}
