# =============================================================================
# ESTATE INCOME FROM CONTROL SHORTFALL
# =============================================================================
#
# Estates receive income from taxes the crown loses due to low control.
#
# ARCHITECTURE:
# - sul_update_all_location_gdp (monthly) calculates and caches:
#   - sul_local_gdp on each location (for pop demand scaling)
#   - sul_total_taxes_lost on country (for estate shortfall)
#
# - This file just reads the cached values and splits by estate power.
#
# =============================================================================

# =============================================================================
# PER-ESTATE INCOME CALCULATION
# =============================================================================
# Called from estate scope. Reads cached sul_total_taxes_lost from monthly update.
# Each estate gets: total_taxes_lost * estate_power

sul_calculate_estate_income_lost = {
	# Only calculate if the monthly GDP update has run
	if = {
		limit = { owner = { has_variable = sul_total_taxes_lost } }
		set_local_variable = {
			name = income_lost
			value = {
				value = owner.var:sul_total_taxes_lost
				multiply = power
				min = 0
			}
		}
	}
	else = {
		set_local_variable = { name = income_lost value = 0 }
	}

	# Store per-estate for GUI display
	if = { limit = { estate_type = estate_type:nobles_estate }
		owner = { set_variable = { name = sul_estate_nobles_income_lost value = local_var:income_lost } } }
	else_if = { limit = { estate_type = estate_type:burghers_estate }
		owner = { set_variable = { name = sul_estate_burghers_income_lost value = local_var:income_lost } } }
	else_if = { limit = { estate_type = estate_type:clergy_estate }
		owner = { set_variable = { name = sul_estate_clergy_income_lost value = local_var:income_lost } } }
	else_if = { limit = { estate_type = estate_type:peasants_estate }
		owner = { set_variable = { name = sul_estate_peasants_income_lost value = local_var:income_lost } } }
	else_if = { limit = { estate_type = estate_type:tribes_estate }
		owner = { set_variable = { name = sul_estate_tribes_income_lost value = local_var:income_lost } } }
	else_if = { limit = { estate_type = estate_type:dhimmi_estate }
		owner = { set_variable = { name = sul_estate_dhimmi_income_lost value = local_var:income_lost } } }
	else_if = { limit = { estate_type = estate_type:cossacks_estate }
		owner = { set_variable = { name = sul_estate_cossacks_income_lost value = local_var:income_lost } } }
}
