# =============================================================================
# LOCAL ESTATE POWER EFFECTS
# =============================================================================
# Updates estate power share variables for locations
# =============================================================================

# -----------------------------------------------------------------------------
# UPDATE SINGLE LOCATION (location scope)
# -----------------------------------------------------------------------------
# Stores estate power shares as decimals (0.0-1.0)
sul_update_location_estate_power = {
	set_variable = {
		name = sul_nobles_power_share
		value = sul_local_nobles_share
	}
	set_variable = {
		name = sul_clergy_power_share
		value = sul_local_clergy_share
	}
	set_variable = {
		name = sul_burghers_power_share
		value = sul_local_burghers_share
	}
	set_variable = {
		name = sul_peasants_power_share
		value = sul_local_peasants_share
	}
	set_variable = {
		name = sul_tribes_power_share
		value = sul_local_tribes_share
	}
}

# -----------------------------------------------------------------------------
# UPDATE ALL LOCATIONS FOR A COUNTRY (country scope)
# -----------------------------------------------------------------------------
sul_update_country_estate_power = {
	every_owned_location = {
		sul_update_location_estate_power = yes
	}
}

# -----------------------------------------------------------------------------
# UPDATE ALL COUNTRIES (no scope required)
# -----------------------------------------------------------------------------
sul_update_all_estate_power = {
	every_country = {
		limit = { sul_is_valid_location_owner = yes }
		sul_update_country_estate_power = yes
	}
}
