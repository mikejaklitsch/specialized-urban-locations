# Estate Wages - Monthly wage payouts to estates
#
# Wages accumulate in country variables sul_monthly_wages_<estate> (set
# elsewhere by GDP/economy code). This file just transfers the accumulated
# value to estate gold each month.
#
# Engine bug: add_gold_to_estate negates the value, so we negate first.
#
# Effects defined here:
#   sul_estate_wages_pay_single  - pay one estate (parameterized by $estate$)
#   sul_estate_wages_pay_all     - pay all 7 estates
#
# Wage rate constants are kept here for proximity to the only place that
# pays wages out. Per-pop accumulation lives in the GDP/economy update.

@sul_wage_nobles = 5.0
@sul_wage_clergy = 2.0
@sul_wage_burghers = 1.0
@sul_wage_soldiers = 0.5
@sul_wage_laborers = 0.25
@sul_wage_peasants = 0.05

sul_estate_wages_pay_single = {
	if = {
		limit = {
			has_variable = sul_monthly_wages_$estate$
			var:sul_monthly_wages_$estate$ > 0
		}

		add_gold_to_estate = {
			estate_type = estate_type:$estate$_estate
			value = { value = var:sul_monthly_wages_$estate$ multiply = -1 }
		}
	}
}

sul_estate_wages_pay_all = {
	sul_estate_wages_pay_single = { estate = nobles }
	sul_estate_wages_pay_single = { estate = clergy }
	sul_estate_wages_pay_single = { estate = burghers }
	sul_estate_wages_pay_single = { estate = peasants }
	sul_estate_wages_pay_single = { estate = dhimmi }
	sul_estate_wages_pay_single = { estate = cossacks }
	sul_estate_wages_pay_single = { estate = tribes }

	# Inter-estate transfer: upper estates fund commoner base wages
	sul_estate_wage_transfer = yes
}

# Inter-estate wage transfers — upper estates pay commoner base wages.
#
# Transfer amount = per-location sum of (commoner_wages × enfranchisement × commoner_power).
# Enfranchisement is the institutional lever (player policy), political power is the
# structural reality (organic, per-location). Both must be present for redistribution:
# high enfranchisement in a noble stronghold transfers little; low enfranchisement in
# a burgher trade city also transfers little.
#
# The weighted total is pre-accumulated in sul_transfer_weighted_total by the
# location iteration in sul_update_country_economy. Deducted proportionally from
# upper estates by their wage share, credited to peasant estate.

sul_estate_wage_transfer = {
	# --- DEBUG: transfer amounts ---
	if = {
		limit = { OR = { tag = ENG tag = FRA tag = VEN } }
		set_local_variable = { name = d_xfer value = var:sul_transfer_weighted_total }
		set_local_variable = { name = d_wN value = var:sul_monthly_wages_nobles }
		set_local_variable = { name = d_wC value = var:sul_monthly_wages_clergy }
		set_local_variable = { name = d_wB value = var:sul_monthly_wages_burghers }
		set_local_variable = { name = d_wP value = var:sul_monthly_wages_peasants }
		error_log = "::SUL_XFER::[GetCurrentYear]:[SCOPE.GetTag]:xfer=[SCOPE.GetLocalVariable('d_xfer').GetValue]:wN=[SCOPE.GetLocalVariable('d_wN').GetValue]:wC=[SCOPE.GetLocalVariable('d_wC').GetValue]:wB=[SCOPE.GetLocalVariable('d_wB').GetValue]:wP=[SCOPE.GetLocalVariable('d_wP').GetValue]"
	}

	if = {
		limit = {
			has_variable = sul_transfer_weighted_total
			var:sul_transfer_weighted_total > 0.01
		}

		set_local_variable = {
			name = sul_upper_total
			value = {
				value = var:sul_monthly_wages_nobles
				add = var:sul_monthly_wages_clergy
				add = var:sul_monthly_wages_burghers
				min = 0.001
			}
		}

		# Deduct from each upper estate proportional to their wage share
		# add_gold_to_estate negates the value, so positive = drain
		if = {
			limit = { var:sul_monthly_wages_nobles > 0 }
			add_gold_to_estate = {
				estate_type = estate_type:nobles_estate
				value = {
					value = var:sul_transfer_weighted_total
					multiply = { value = var:sul_monthly_wages_nobles divide = local_var:sul_upper_total }
				}
			}
		}
		if = {
			limit = { var:sul_monthly_wages_clergy > 0 }
			add_gold_to_estate = {
				estate_type = estate_type:clergy_estate
				value = {
					value = var:sul_transfer_weighted_total
					multiply = { value = var:sul_monthly_wages_clergy divide = local_var:sul_upper_total }
				}
			}
		}
		if = {
			limit = { var:sul_monthly_wages_burghers > 0 }
			add_gold_to_estate = {
				estate_type = estate_type:burghers_estate
				value = {
					value = var:sul_transfer_weighted_total
					multiply = { value = var:sul_monthly_wages_burghers divide = local_var:sul_upper_total }
				}
			}
		}

		# Credit to peasant estate (negate because add_gold_to_estate negates)
		add_gold_to_estate = {
			estate_type = estate_type:peasants_estate
			value = { value = var:sul_transfer_weighted_total multiply = -1 }
		}
	}
}
