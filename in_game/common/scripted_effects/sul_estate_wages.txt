# Estate Wages - Monthly wage payouts to estates
#
# Wages accumulate in country variables sul_monthly_wages_<estate> (set
# elsewhere by GDP/economy code). This file just transfers the accumulated
# value to estate gold each month.
#
# Engine bug: add_gold_to_estate negates the value, so we negate first.
#
# Effects defined here:
#   sul_estate_wages_pay_single  - pay one estate (parameterized by $estate$)
#   sul_estate_wages_pay_all     - pay all 7 estates
#
# Wage rate constants are kept here for proximity to the only place that
# pays wages out. Per-pop accumulation lives in the GDP/economy update.

@sul_wage_nobles = 5.0
@sul_wage_clergy = 2.0
@sul_wage_burghers = 1.0
@sul_wage_soldiers = 0.5
@sul_wage_laborers = 0.25
@sul_wage_peasants = 0.05

sul_estate_wages_pay_single = {
	if = {
		limit = {
			has_variable = sul_monthly_wages_$estate$
			var:sul_monthly_wages_$estate$ > 0
		}

		add_gold_to_estate = {
			estate_type = estate_type:$estate$_estate
			value = { value = var:sul_monthly_wages_$estate$ multiply = -1 }
		}
	}
}

sul_estate_wages_pay_all = {
	sul_estate_wages_pay_single = { estate = nobles }
	sul_estate_wages_pay_single = { estate = clergy }
	sul_estate_wages_pay_single = { estate = burghers }
	sul_estate_wages_pay_single = { estate = peasants }
	sul_estate_wages_pay_single = { estate = dhimmi }
	sul_estate_wages_pay_single = { estate = cossacks }
	sul_estate_wages_pay_single = { estate = tribes }
}
