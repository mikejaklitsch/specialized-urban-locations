# GDP yearly computation — country scope.
#
# Gross output per location = sum of (goods_output × market_price) for each
# good tracked in the sul_local_goods variable map. Building hooks maintain
# this map as buildings are constructed and destroyed, so only goods actually
# produced at a location get checked. Asset share = local GDP / national GDP
# drives upper-class capital returns allocation.

# Location scope: iterate the local goods map and sum output × price.
# The variable map keys are goods scopes; every_key_in_variable_map iterates
# them, and we price each against the location's market.

sul_gdp_compute_location = {
	save_scope_as = sul_gdp_loc
	set_variable = { name = sul_local_gdp_output value = 0 }

	if = {
		limit = { has_variable_map = sul_local_goods }

		every_key_in_variable_map = {
			variable = sul_local_goods
			save_scope_as = sul_gdp_good

			scope:sul_gdp_loc = {
				change_variable = {
					name = sul_local_gdp_output
					add = {
						value = "goods_output(scope:sul_gdp_good)"
						multiply = "scope:sul_gdp_good.price_in_market(scope:sul_gdp_loc.market)"
					}
				}
			}
		}
	}

	if = {
		limit = { var:sul_local_gdp_output < 0.01 }
		set_variable = { name = sul_local_gdp_output value = 0.01 }
	}
}

sul_gdp_yearly_update = {
	set_variable = { name = sul_national_gdp value = 0 }

	every_owned_location = {
		sul_gdp_compute_location = yes

		owner ?= {
			change_variable = { name = sul_national_gdp add = prev.var:sul_local_gdp_output }
		}
	}

	if = {
		limit = { var:sul_national_gdp < 1 }
		set_variable = { name = sul_national_gdp value = 1 }
	}

	every_owned_location = {
		set_variable = {
			name = sul_asset_share
			value = {
				value = var:sul_local_gdp_output
				divide = owner.var:sul_national_gdp
			}
		}
	}

	# --- DEBUG: GDP summary ---
	if = {
		limit = { OR = { tag = ENG tag = FRA tag = VEN } }
		set_local_variable = { name = d_natgdp value = var:sul_national_gdp }
		error_log = "::SUL_GDP::[GetCurrentYear]:[SCOPE.GetTag]:natGDP=[SCOPE.GetLocalVariable('d_natgdp').GetValue]"
		capital = {
			set_local_variable = { name = d_locgdp value = var:sul_local_gdp_output }
			set_local_variable = { name = d_asset value = var:sul_asset_share }
			error_log = "::SUL_GDP_CAP::[GetCurrentYear]:locGDP=[SCOPE.GetLocalVariable('d_locgdp').GetValue]:asset=[SCOPE.GetLocalVariable('d_asset').GetValue]"
		}
	}
}

# Populates sul_local_goods from pre-existing buildings, then computes
# the first GDP values so asset shares are available before the first
# yearly pulse fires.

sul_gdp_init = {
	every_owned_location = {
		every_buildings_in_location = {
			sul_gdp_init_building = yes
		}
	}
	sul_gdp_yearly_update = yes

	# --- DEBUG: init confirmation ---
	if = {
		limit = { OR = { tag = ENG tag = FRA tag = VEN } }
		set_local_variable = { name = d_natgdp value = var:sul_national_gdp }
		error_log = "::SUL_GDP_INIT::[GetCurrentYear]:[SCOPE.GetTag]:natGDP=[SCOPE.GetLocalVariable('d_natgdp').GetValue]"
	}
}
