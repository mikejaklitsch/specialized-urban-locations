# -----------------------------------------------------------------------------
# SINGLE LOCATION UPDATE (location scope)
# -----------------------------------------------------------------------------
# Updates estate power, GDP, and gdp_pc for one location
sul_update_location_economy = {
	sul_update_location_estate_power = yes
	sul_update_location_gdp = yes
	sul_cache_gdp_per_capita = yes
}

# -----------------------------------------------------------------------------
# SINGLE LOCATION GDP (location scope)
# -----------------------------------------------------------------------------
# Sets sul_local_gdp based on control level and market access
sul_update_location_gdp = {
	if = {
		limit = { market_access <= 0 }
		set_variable = { name = sul_local_gdp value = 0 }
	}
	else_if = {
		limit = { local_control > 0 }
		set_variable = {
			name = sul_local_gdp
			value = {
				value = location_tax_base
				divide = local_control
			}
		}
	}
	else = {
		# Zero control - estimate from country averages (if available)
		if = {
			limit = { exists = owner owner = { has_variable = sul_avg_rgo_value_per_output } }
			set_variable = {
				name = sul_local_gdp
				value = {
					value = owner.var:sul_avg_rgo_value_per_output
					multiply = raw_material_output
					multiply = market_access
				}
			}
			change_variable = {
				name = sul_local_gdp
				add = {
					value = owner.var:sul_avg_building_value_per_level
					multiply = total_building_levels
					multiply = market_access
				}
			}
		}
		else = {
			# No averages yet - use 0 as fallback
			set_variable = { name = sul_local_gdp value = 0 }
		}
	}
}

# -----------------------------------------------------------------------------
# COUNTRY GDP UPDATE (country scope)
# -----------------------------------------------------------------------------
# Updates all locations and accumulates taxes lost
sul_update_all_location_gdp = {
	# =========================================================================
	# INITIALIZE TRACKING VARIABLES
	# =========================================================================

	# Per-estate taxes lost
	set_variable = { name = sul_nobles_taxes_lost value = 0 }
	set_variable = { name = sul_clergy_taxes_lost value = 0 }
	set_variable = { name = sul_burghers_taxes_lost value = 0 }
	set_variable = { name = sul_peasants_taxes_lost value = 0 }
	set_variable = { name = sul_tribes_taxes_lost value = 0 }

	# Initialize averages if not set (first run fallback for zero-control estimation)
	if = {
		limit = { NOT = { has_variable = sul_avg_rgo_value_per_output } }
		set_variable = { name = sul_avg_rgo_value_per_output value = 0 }
		set_variable = { name = sul_avg_building_value_per_level value = 0 }
	}

	# For calculating NEW averages (to be stored for next month)
	set_local_variable = { name = rgo_value_sum value = 0 }
	set_local_variable = { name = rgo_output_sum value = 0 }
	set_local_variable = { name = building_value_sum value = 0 }
	set_local_variable = { name = building_level_sum value = 0 }

	# =========================================================================
	# SINGLE PASS: ALL LOCATIONS
	# =========================================================================
	# Each location: estate power -> GDP -> gdp_pc cache
	# CL uses direct calculation, ZCL uses stored averages from previous month

	every_owned_location = {
		# Update estate power, GDP, and gdp_pc in one call
		sul_update_location_economy = yes

		# Accumulate taxes lost and averaging data (only for market-connected locations)
		if = {
			limit = { market_access > 0 }
			save_scope_as = cur_loc

			# For controlled locations: accumulate for averaging
			if = {
				limit = { local_control > 0 }

				# Calculate RGO value for averaging
				set_local_variable = { name = loc_rgo_value value = 0 }
				raw_material ?= {
					set_local_variable = {
						name = loc_rgo_value
						value = {
							value = "price_in_market(scope:cur_loc.market)"
							multiply = scope:cur_loc.raw_material_output
							multiply = scope:cur_loc.market_access
						}
					}
				}

				# Calculate building value (GDP minus RGO)
				set_local_variable = {
					name = loc_building_value
					value = {
						value = var:sul_local_gdp
						subtract = local_var:loc_rgo_value
						min = 0
					}
				}

				# Accumulate for next month's averages
				owner ?= {
					change_local_variable = { name = rgo_value_sum add = local_var:loc_rgo_value }
					change_local_variable = { name = rgo_output_sum add = prev.raw_material_output }
					change_local_variable = { name = building_value_sum add = local_var:loc_building_value }
					change_local_variable = { name = building_level_sum add = prev.total_building_levels }
				}

				# Calculate taxes lost (GDP - actual tax base)
				set_local_variable = {
					name = loc_taxes_lost
					value = {
						value = var:sul_local_gdp
						subtract = location_tax_base
					}
				}
			}
			else = {
				# Zero-control: full GDP is taxes lost (crown gets nothing)
				set_local_variable = { name = loc_taxes_lost value = var:sul_local_gdp }
			}

			# Accumulate taxes lost by estate (both CL and ZCL)
			owner ?= {
				change_variable = {
					name = sul_nobles_taxes_lost
					add = { value = local_var:loc_taxes_lost multiply = prev.var:sul_nobles_power_share }
				}
				change_variable = {
					name = sul_clergy_taxes_lost
					add = { value = local_var:loc_taxes_lost multiply = prev.var:sul_clergy_power_share }
				}
				change_variable = {
					name = sul_burghers_taxes_lost
					add = { value = local_var:loc_taxes_lost multiply = prev.var:sul_burghers_power_share }
				}
				change_variable = {
					name = sul_peasants_taxes_lost
					add = { value = local_var:loc_taxes_lost multiply = prev.var:sul_peasants_power_share }
				}
				change_variable = {
					name = sul_tribes_taxes_lost
					add = { value = local_var:loc_taxes_lost multiply = prev.var:sul_tribes_power_share }
				}
			}
		}
	}

	# =========================================================================
	# UPDATE AVERAGES FOR NEXT MONTH
	# =========================================================================

	set_variable = {
		name = sul_avg_rgo_value_per_output
		value = {
			if = {
				limit = { local_var:rgo_output_sum > 0 }
				value = local_var:rgo_value_sum
				divide = local_var:rgo_output_sum
			}
			else = { value = 0 }
		}
	}

	set_variable = {
		name = sul_avg_building_value_per_level
		value = {
			if = {
				limit = { local_var:building_level_sum > 0 }
				value = local_var:building_value_sum
				divide = local_var:building_level_sum
			}
			else = { value = 0 }
		}
	}

}

# =============================================================================
# CACHE POP DEMAND MULTIPLIER (location scope)
# =============================================================================
# Pre-calculates demand multiplier for each pop type so pop demand reads one
# variable instead of computing the formula each time.
#
# FORMULA: demand = floor + (ceiling - floor) × (X / (X + k))
#          Equivalent to: ceiling - (ceiling - floor) × k / (X + k)
#          Where X = gdp × power_share
#
# All pop types have floor=1.0 (baseline demand). Ceiling determines max scaling.
#
#   Pop Type    | Floor | Ceiling | k   | (c-f)*k
#   ------------|-------|---------|-----|--------
#   Nobles      | 1.0   | 5.0     | 200 | 800
#   Clergy      | 1.0   | 4.0     | 150 | 450
#   Burghers    | 1.0   | 3.5     | 150 | 375
#   Commoners   | 1.0   | 2.5     | 100 | 150
#   Tribesmen   | 1.0   | 2.0     | 75  | 75
#   Slaves      | 1.0   | 1.0     | -   | 0 (fixed at 1.0)
#
# =============================================================================

sul_cache_gdp_per_capita = {
	# Nobles: floor=1.0, ceiling=5.0, k=200
	set_variable = {
		name = sul_noble_gdp_pc
		value = {
			value = 5.0
			subtract = {
				value = 800
				divide = { value = var:sul_local_gdp multiply = var:sul_nobles_power_share add = 200 }
			}
		}
	}

	# Clergy: floor=1.0, ceiling=4.0, k=150
