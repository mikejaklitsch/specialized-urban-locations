# =============================================================================
# GDP UPDATE SYSTEM
# =============================================================================
# -----------------------------------------------------------------------------
# SINGLE LOCATION UPDATE (location scope)
# -----------------------------------------------------------------------------
# Updates estate power, GDP, and gdp_pc for one location

sul_update_location_economy = {
	sul_update_location_estate_power = yes
	sul_update_location_gdp = yes
	sul_cache_gdp_per_capita = yes
}

# -----------------------------------------------------------------------------
# SINGLE LOCATION GDP (location scope)
# -----------------------------------------------------------------------------
# Sets sul_local_gdp based on control level and market access

sul_update_location_gdp = {
	if = {
		limit = { market_access <= 0 }
		set_variable = { name = sul_local_gdp value = 0 }
	}
	else_if = {
		limit = { local_control > 0 }
		set_variable = {
			name = sul_local_gdp
			value = { value = location_tax_base divide = local_control }
		}
	}
	else = {
		# Zero control - estimate from country averages (if available)
		if = {
			limit = {
				exists = owner
				owner = { has_variable = sul_avg_rgo_value_per_output }
			}
			set_variable = {
				name = sul_local_gdp
				value = {
					value = owner.var:sul_avg_rgo_value_per_output
					multiply = raw_material_output
					multiply = market_access
				}
			}
			change_variable = {
				name = sul_local_gdp
				add = {
					value = owner.var:sul_avg_building_value_per_level
					multiply = total_building_levels
					multiply = market_access
				}
			}
		}
		else = {
			# No averages yet - use 0 as fallback
			set_variable = { name = sul_local_gdp value = 0 }
		}
	}
}

# -----------------------------------------------------------------------------
# COUNTRY GDP UPDATE (country scope)
# -----------------------------------------------------------------------------
# Updates all locations and caches GDP/gdp_pc values

sul_update_all_location_gdp = {
	# Initialize averages if not set (first run fallback for zero-control estimation)
	if = {
		limit = {
			NOT = {
				has_variable = sul_avg_rgo_value_per_output
			}
		}
		set_variable = { name = sul_avg_rgo_value_per_output value = 0 }
		set_variable = { name = sul_avg_building_value_per_level value = 0 }
	}

	# For calculating NEW averages (to be stored for next month)
	set_local_variable = {
		name = rgo_value_sum
		value = 0
	}
	set_local_variable = {
		name = rgo_output_sum
		value = 0
	}
	set_local_variable = {
		name = building_value_sum
		value = 0
	}
	set_local_variable = {
		name = building_level_sum
		value = 0
	}

	# =========================================================================
	# SINGLE PASS: ALL LOCATIONS
	# =========================================================================
	# Each location: estate power -> GDP -> gdp_pc cache
	# CL uses direct calculation, ZCL uses stored averages from previous month
	every_owned_location = {
		# Update estate power, GDP, and gdp_pc in one call
		sul_update_location_economy = yes
		# Accumulate averaging data (only for controlled, market-connected locations)
		if = {
			limit = { market_access > 0 local_control > 0 }
			save_scope_as = cur_loc
			# Calculate RGO value for averaging
			set_local_variable = { name = loc_rgo_value value = 0 }
			raw_material ?= {
				set_local_variable = {
					name = loc_rgo_value
					value = {
						value = "price_in_market(scope:cur_loc.market)"
						multiply = scope:cur_loc.raw_material_output
						multiply = scope:cur_loc.market_access
					}
				}
			}
			# Calculate building value (GDP minus RGO)
			set_local_variable = {
				name = loc_building_value
				value = {
					value = var:sul_local_gdp
					subtract = local_var:loc_rgo_value
					min = 0
				}
			}
			# Accumulate for next month's averages
			owner ?= {
				change_local_variable = { name = rgo_value_sum add = local_var:loc_rgo_value }
				change_local_variable = {
					name = rgo_output_sum
					add = prev.raw_material_output
				}
				change_local_variable = {
					name = building_value_sum
					add = local_var:loc_building_value
				}
				change_local_variable = {
					name = building_level_sum
					add = prev.total_building_levels
				}
			}
		}
	}

	# =========================================================================
	# UPDATE AVERAGES FOR NEXT MONTH
	# =========================================================================
	set_variable = {
		name = sul_avg_rgo_value_per_output
		value = {
			if = {
				limit = { local_var:rgo_output_sum > 0 }
				value = local_var:rgo_value_sum
				divide = local_var:rgo_output_sum
			}
			else = { value = 0 }
		}
	}
	set_variable = {
		name = sul_avg_building_value_per_level
		value = {
			if = {
				limit = { local_var:building_level_sum > 0 }
				value = local_var:building_value_sum
				divide = local_var:building_level_sum
			}
			else = { value = 0 }
		}
	}
}

# =============================================================================
# CACHE POP DEMAND MULTIPLIER (location scope)
# =============================================================================
# Pre-calculates demand multiplier for each pop type so pop demand reads one
# variable instead of computing the formula each time.
#
# FORMULA: demand = floor + (ceiling - floor) × (X / (X + k))
#          Equivalent to: ceiling - (ceiling - floor) × k / (X + k)
#          Where X = gdp × power_share
#
# All pop types have floor=0.8 (below vanilla baseline). Ceiling determines max scaling.
#
#   Pop Type    | Floor | Ceiling | k   | (c-f)*k
#   ------------|-------|---------|-----|--------
#   Nobles      | 0.8   | 5.0     | 200 | 840
#   Clergy      | 0.8   | 4.0     | 150 | 480
#   Burghers    | 0.8   | 3.5     | 150 | 405
#   Commoners   | 0.8   | 2.5     | 100 | 170
#   Tribesmen   | 0.8   | 2.0     | 75  | 90
#   Slaves      | 0.8   | 0.8     | -   | 0 (fixed at 0.8)
#
# ENFRANCHISEMENT: Adjusts peasant/noble shares based on wealth transfer.
#   - Peasant effective share = peasant_share × enfranchisement
#   - Noble effective share = noble_share + peasant_share × (1 - enfranchisement)
#   - Enfranchisement clamped to 10%-100%
#
# =============================================================================

sul_cache_gdp_per_capita = {
	# -------------------------------------------------------------------------
	# ENFRANCHISEMENT ADJUSTMENT
	# -------------------------------------------------------------------------
	# Get enfranchisement (clamped 0.1 to 1.0)
	set_variable = {
		name = sul_enfranchisement
		value = { value = modifier:local_peasant_enfranchisment min = 0.1 max = 1.0 }
	}

	# Calculate adjusted shares for wealth transfer (stored for GUI display)
	set_variable = {
		name = sul_adjusted_peasant_share
		value = {
			value = var:sul_peasants_power_share
			multiply = var:sul_enfranchisement
		}
	}
	set_variable = {
		name = sul_adjusted_noble_share
		value = {
			value = var:sul_nobles_power_share
			add = {
				value = var:sul_peasants_power_share
				multiply = { value = 1 subtract = var:sul_enfranchisement }
			}
		}
	}

	# -------------------------------------------------------------------------
	# GDP PER CAPITA CACHING
	# -------------------------------------------------------------------------
	# Nobles: floor=0.8, ceiling=5.0, k=200 (uses adjusted share)
	set_variable = {
		name = sul_noble_gdp_pc
		value = {
			value = 5.0
			subtract = {
				value = 840
				divide = {
					value = var:sul_local_gdp
					multiply = var:sul_adjusted_noble_share
					add = 200
				}
			}
		}
	}

	# Clergy: floor=0.8, ceiling=4.0, k=150
	set_variable = {
		name = sul_clergy_gdp_pc
		value = {
			value = 4.0
			subtract = {
				value = 480
				divide = {
					value = var:sul_local_gdp
					multiply = var:sul_clergy_power_share
					add = 150
				}
			}
		}
	}

	# Burghers: floor=0.8, ceiling=3.5, k=150
	set_variable = {
		name = sul_burgher_gdp_pc
		value = {
			value = 3.5
			subtract = {
				value = 405
				divide = {
					value = var:sul_local_gdp
					multiply = var:sul_burghers_power_share
					add = 150
				}
			}
		}
	}

	# Soldiers: floor=0.8, ceiling=2.5, k=100 (uses adjusted peasant share)
	set_variable = {
		name = sul_soldier_gdp_pc
		value = {
			value = 2.5
			subtract = {
				value = 170
				divide = {
					value = var:sul_local_gdp
					multiply = var:sul_adjusted_peasant_share
					add = 100
				}
			}
		}
	}

	# Laborers: floor=0.8, ceiling=2.5, k=100 (uses adjusted peasant share)
	set_variable = {
		name = sul_laborer_gdp_pc
		value = {
			value = 2.5
			subtract = {
				value = 170
				divide = {
					value = var:sul_local_gdp
					multiply = var:sul_adjusted_peasant_share
					add = 100
				}
			}
		}
	}

	# Peasants: floor=0.8, ceiling=2.5, k=100 (uses adjusted peasant share)
	set_variable = {
		name = sul_peasant_gdp_pc
		value = {
			value = 2.5
			subtract = {
				value = 170
				divide = {
					value = var:sul_local_gdp
					multiply = var:sul_adjusted_peasant_share
					add = 100
				}
			}
		}
	}

	# Tribesmen: floor=0.8, ceiling=2.0, k=75
	set_variable = {
		name = sul_tribesman_gdp_pc
		value = {
			value = 2.0
			subtract = {
				value = 90
				divide = {
					value = var:sul_local_gdp
					multiply = var:sul_tribes_power_share
					add = 75
				}
			}
		}
	}

	# Slaves: fixed at 0.8 (no GDP scaling)
	set_variable = {
		name = sul_slave_gdp_pc
		value = 0.8
	}
}
