# =============================================================================
# GDP-BASED ECONOMY: WPP, LOCAL ASSETS, CONVERGENCE
# =============================================================================
#
# WPP Formula (per location, per estate pop type):
#
#   gdp_wages        = @wage_share × sul_local_gdp × (estate_power / total_power) / pops
#   local_returns    = @asset_return_rate × sul_local_assets × (estate_power / total_power) / pops
#   national_returns = @estate_return_rate × estate_gold / national_estate_pops
#   provisions_cost  = (food_consumption / 30) × provisions_market_price
#
#   WPP = gdp_wages + local_returns + national_returns - provisions_cost
#
# Local Assets:
#   max_assets = @asset_max_multiplier × sul_local_gdp × (1 + prosperity/100)
#   Recovery:  converge toward max at @asset_converge_rate per month, clamped
#   Occupied:  drain @asset_drain_rate of remaining per month, no recovery
#
# Provisions price is cached per market in a global variable map, cleared monthly.

# --- Tuning macros ---
@base_wage_share = 0.50
@dev_midpoint = 0.50
@emp_midpoint = 0.50
@asset_max_multiplier = 5
@asset_converge_rate = 0.05
@asset_return_rate = 0.05
@asset_drain_rate = 0.025
@estate_return_rate = 0.025

# Demand saturation curve: WPP / (intercept + slope × WPP)
@sul_demand_intercept = 3.0
@sul_demand_slope = 1.0

# Fraction of WPP that becomes estate savings (remainder goes to consumption)
@sul_enrichment_weight = 0.53

# Provisions food value per unit (denominator for food_consumption → demand)
@provisions_food_value = 30

# Per-pop food consumption (must match pop_types/sul_pop_food_consumption.txt)
@food_nobles = 24.0
@food_clergy = 6.0
@food_burghers = 6.0
@food_soldiers = 4.5
@food_laborers = 1.5

# Commoner WPP weights (soldiers and laborers get larger share of estate pool)
@commoner_weight_soldiers = 12
@commoner_weight_laborers = 4
@commoner_weight_peasants = 1

# Provisions quantity per 1k pops (food_consumption / food_value, precalculated)
@prov_qty_nobles = @[24/30]
@prov_qty_clergy = @[6/30]
@prov_qty_burghers = @[6/30]
@prov_qty_soldiers = @[4.5/30]
@prov_qty_laborers = @[1.5/30]

# Looting
@sul_loot_extraction_rate = 0.25
@sul_loot_destroyed_base = 0.50

# -----------------------------------------------------------------------------
# PROVISIONS CACHE REBUILD (scopeless, called on init + monthly)
# -----------------------------------------------------------------------------
# Precomputes provisions price and per-pop costs per market.

sul_provisions_cache_rebuild = {
	clear_global_variable_map = sul_provisions_price_cache
	clear_global_variable_map = sul_prov_cost_nobles
	clear_global_variable_map = sul_prov_cost_clergy
	clear_global_variable_map = sul_prov_cost_burghers
	clear_global_variable_map = sul_prov_cost_soldiers
	clear_global_variable_map = sul_prov_cost_laborers

	every_market_in_world = {
		save_scope_as = sul_prov_market
		set_local_variable = { name = p value = "goods:provisions.price_in_market(scope:sul_prov_market)" }
		set_local_variable = { name = c_nobles value = { value = local_var:p multiply = @prov_qty_nobles } }
		set_local_variable = { name = c_clergy value = { value = local_var:p multiply = @prov_qty_clergy } }
		set_local_variable = { name = c_burghers value = { value = local_var:p multiply = @prov_qty_burghers } }
		set_local_variable = { name = c_soldiers value = { value = local_var:p multiply = @prov_qty_soldiers } }
		set_local_variable = { name = c_laborers value = { value = local_var:p multiply = @prov_qty_laborers } }

		add_to_global_variable_map = {
			name = sul_provisions_price_cache
			key = scope:sul_prov_market
			value = local_var:p
		}
		add_to_global_variable_map = {
			name = sul_prov_cost_nobles
			key = scope:sul_prov_market
			value = local_var:c_nobles
		}
		add_to_global_variable_map = {
			name = sul_prov_cost_clergy
			key = scope:sul_prov_market
			value = local_var:c_clergy
		}
		add_to_global_variable_map = {
			name = sul_prov_cost_burghers
			key = scope:sul_prov_market
			value = local_var:c_burghers
		}
		add_to_global_variable_map = {
			name = sul_prov_cost_soldiers
			key = scope:sul_prov_market
			value = local_var:c_soldiers
		}
		add_to_global_variable_map = {
			name = sul_prov_cost_laborers
			key = scope:sul_prov_market
			value = local_var:c_laborers
		}
	}
}

# -----------------------------------------------------------------------------

sul_cache_provisions_price = {
	set_local_variable = { name = sul_provisions_price value = "global_variable_map(sul_provisions_price_cache|market)" }
}

# -----------------------------------------------------------------------------
# SINGLE LOCATION WPP UPDATE (location scope)
# -----------------------------------------------------------------------------

sul_update_location_wpp = {
	sul_cache_provisions_price = yes

	# Dynamic wage share: base × urbanization efficiency × employment efficiency × market access
	set_local_variable = { name = sul_urbanization value = { value = development divide = 100 } }
	set_local_variable = {
		name = sul_dev_factor
		value = {
			value = 2
			multiply = local_var:sul_urbanization
			divide = { value = @dev_midpoint add = local_var:sul_urbanization }
		}
	}
	set_local_variable = {
		name = sul_emp_factor
		value = {
			value = 2
			multiply = employment_percentage
			divide = { value = @emp_midpoint add = employment_percentage }
		}
	}

	set_variable = {
		name = sul_wage_share
		value = {
			value = @base_wage_share
			multiply = local_var:sul_dev_factor
			multiply = local_var:sul_emp_factor
			multiply = market_access
		}
	}

	# GDP wages pool for this location
	set_local_variable = { name = sul_gdp_wage_pool value = { value = var:sul_local_gdp multiply = var:sul_wage_share } }

	# Local asset returns pool
	set_local_variable = { name = sul_asset_pool value = 0 }

	if = {
		limit = { has_variable = sul_local_assets }
		set_local_variable = { name = sul_asset_pool value = { value = var:sul_local_assets multiply = @asset_return_rate } }
	}

	# --- Upper estates: nobles, clergy, burghers ---
	# WPP = (gdp_wages + asset_returns) × power_share / pops + national_returns - provisions

	set_local_variable = { name = sul_noble_wpp value = 0 }

	if = {
		limit = { num_pop_type:nobles > 0 }
		set_local_variable = {
			name = sul_noble_wpp
			value = {
				value = local_var:sul_gdp_wage_pool
				add = local_var:sul_asset_pool
				multiply = "local_relative_estate_power(estate_type:nobles_estate)"
				divide = { value = num_pop_type:nobles min = 0.001 }

				add = {
					value = owner.var:sul_estate_gold_nobles
					multiply = @estate_return_rate
					divide = { value = owner.var:sul_country_pop_nobles min = 1 }
				}

				subtract = "global_variable_map(sul_prov_cost_nobles|market)"

				min = 0
			}
		}
	}

	set_local_variable = { name = sul_clergy_wpp value = 0 }

	if = {
		limit = { num_pop_type:clergy > 0 }
		set_local_variable = {
			name = sul_clergy_wpp
			value = {
				value = local_var:sul_gdp_wage_pool
				add = local_var:sul_asset_pool
				multiply = "local_relative_estate_power(estate_type:clergy_estate)"
				divide = { value = num_pop_type:clergy min = 0.001 }

				add = {
					value = owner.var:sul_estate_gold_clergy
					multiply = @estate_return_rate
					divide = { value = owner.var:sul_country_pop_clergy min = 1 }
				}

				subtract = "global_variable_map(sul_prov_cost_clergy|market)"

				min = 0
			}
		}
	}

	set_local_variable = { name = sul_burgher_wpp value = 0 }

	if = {
		limit = { num_pop_type:burghers > 0 }
		set_local_variable = {
			name = sul_burgher_wpp
			value = {
				value = local_var:sul_gdp_wage_pool
				add = local_var:sul_asset_pool
				multiply = "local_relative_estate_power(estate_type:burghers_estate)"
				divide = { value = num_pop_type:burghers min = 0.001 }

				add = {
					value = owner.var:sul_estate_gold_burghers
					multiply = @estate_return_rate
					divide = { value = owner.var:sul_country_pop_burghers min = 1 }
				}

				subtract = "global_variable_map(sul_prov_cost_burghers|market)"

				min = 0
			}
		}
	}

	# --- Commoners: soldiers, laborers, peasants (share peasants_estate) ---
	# Weighted distribution: soldiers 8x, laborers 4x, peasants 1x

	set_local_variable = { name = sul_commoner_base value = 0 }
	set_local_variable = { name = sul_soldier_wpp value = 0 }
	set_local_variable = { name = sul_laborer_wpp value = 0 }
	set_local_variable = { name = sul_peasant_wpp value = 0 }

	if = {
		limit = {
			OR = {
				num_pop_type:soldiers > 0
				num_pop_type:laborers > 0
				num_pop_type:peasants > 0
			}
		}

		set_local_variable = {
			name = sul_commoner_weighted_total
			value = {
				value = { value = num_pop_type:soldiers multiply = @commoner_weight_soldiers }
				add = { value = num_pop_type:laborers multiply = @commoner_weight_laborers }
				add = { value = num_pop_type:peasants multiply = @commoner_weight_peasants }
				min = 0.001
			}
		}

		# Base per weighted pop: (GDP wages + asset returns) × power / weighted_total + national returns / weighted_national
		set_local_variable = {
			name = sul_commoner_base
			value = {
				value = local_var:sul_gdp_wage_pool
				add = local_var:sul_asset_pool
				multiply = "local_relative_estate_power(estate_type:peasants_estate)"
				divide = local_var:sul_commoner_weighted_total

				min = 0
			}
		}

		# National returns (unweighted, same for all commoner types)
		set_local_variable = {
			name = sul_commoner_national_return
			value = {
				value = owner.var:sul_estate_gold_peasants
				multiply = @estate_return_rate

				divide = {
					value = owner.var:sul_country_pop_soldiers
					add = owner.var:sul_country_pop_laborers
					add = owner.var:sul_country_pop_peasants
					min = 1
				}
			}
		}

		if = {
			limit = { num_pop_type:soldiers > 0 }
			set_local_variable = {
				name = sul_soldier_wpp
				value = {
					value = local_var:sul_commoner_base
					multiply = @commoner_weight_soldiers
					add = local_var:sul_commoner_national_return

					subtract = "global_variable_map(sul_prov_cost_soldiers|market)"

					min = 0
				}
			}
		}

		if = {
			limit = { num_pop_type:laborers > 0 }
			set_local_variable = {
				name = sul_laborer_wpp
				value = {
					value = local_var:sul_commoner_base
					multiply = @commoner_weight_laborers
					add = local_var:sul_commoner_national_return

					subtract = "global_variable_map(sul_prov_cost_laborers|market)"

					min = 0
				}
			}
		}

		# Peasants: 1x weight, zero food consumption
		if = {
			limit = { num_pop_type:peasants > 0 }
			set_local_variable = {
				name = sul_peasant_wpp
				value = {
					value = local_var:sul_commoner_base
					multiply = @commoner_weight_peasants
					add = local_var:sul_commoner_national_return
					min = 0
				}
			}
		}
	}

	# --- Tribesmen (own estate, zero provisions cost) ---
	set_local_variable = { name = sul_tribesmen_wpp value = 0 }

	if = {
		limit = { num_pop_type:tribesmen > 0 }
		set_local_variable = {
			name = sul_tribesmen_wpp
			value = {
				value = local_var:sul_gdp_wage_pool
				add = local_var:sul_asset_pool
				multiply = "local_relative_estate_power(estate_type:tribes_estate)"
				divide = { value = num_pop_type:tribesmen min = 0.001 }

				add = {
					value = owner.var:sul_estate_gold_tribes
					multiply = @estate_return_rate
					divide = { value = owner.var:sul_country_pop_tribesmen min = 1 }
				}

				min = 0
			}
		}
	}

	# --- Wage bills (per-estate, for national accumulation + estate payment) ---
	set_variable = {
		name = sul_wage_bill_nobles
		value = { value = local_var:sul_gdp_wage_pool multiply = "local_relative_estate_power(estate_type:nobles_estate)" }
	}
	set_variable = {
		name = sul_wage_bill_clergy
		value = { value = local_var:sul_gdp_wage_pool multiply = "local_relative_estate_power(estate_type:clergy_estate)" }
	}
	set_variable = {
		name = sul_wage_bill_burghers
		value = { value = local_var:sul_gdp_wage_pool multiply = "local_relative_estate_power(estate_type:burghers_estate)" }
	}
	set_variable = {
		name = sul_wage_bill_commoners
		value = { value = local_var:sul_gdp_wage_pool multiply = "local_relative_estate_power(estate_type:peasants_estate)" }
	}

	# Total local wages (tooltip)
	set_variable = {
		name = sul_local_wages
		value = {
			value = var:sul_wage_bill_nobles
			add = var:sul_wage_bill_clergy
			add = var:sul_wage_bill_burghers
			add = var:sul_wage_bill_commoners
		}
	}

	# --- Local demand budget = sum of WPP × pop counts (tooltip metric) ---
	set_variable = {
		name = sul_local_demand_budget
		value = {
			value = { value = local_var:sul_noble_wpp multiply = num_pop_type:nobles }
			add = { value = local_var:sul_clergy_wpp multiply = num_pop_type:clergy }
			add = { value = local_var:sul_burgher_wpp multiply = num_pop_type:burghers }
			add = { value = local_var:sul_soldier_wpp multiply = num_pop_type:soldiers }
			add = { value = local_var:sul_laborer_wpp multiply = num_pop_type:laborers }
			add = { value = local_var:sul_peasant_wpp multiply = num_pop_type:peasants }
			add = { value = local_var:sul_tribesmen_wpp multiply = num_pop_type:tribesmen }
		}
	}

	# --- Demand saturation: WPP / (intercept + slope × WPP) → 0..1 curve ---
	set_local_variable = {
		name = sul_demand_base_nobles
		value = {
			value = local_var:sul_noble_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_noble_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_clergy
		value = {
			value = local_var:sul_clergy_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_clergy_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_burghers
		value = {
			value = local_var:sul_burgher_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_burgher_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_soldiers
		value = {
			value = local_var:sul_soldier_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_soldier_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_laborers
		value = {
			value = local_var:sul_laborer_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_laborer_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_peasants
		value = {
			value = local_var:sul_peasant_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_peasant_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_tribesmen
		value = {
			value = local_var:sul_tribesmen_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_tribesmen_wpp multiply = @sul_demand_slope } }
		}
	}

	# --- Export WPP + demand base to variable maps (read by pop_demands) ---
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:nobles
		value = local_var:sul_noble_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:nobles
		value = local_var:sul_demand_base_nobles
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:clergy
		value = local_var:sul_clergy_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:clergy
		value = local_var:sul_demand_base_clergy
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:burghers
		value = local_var:sul_burgher_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:burghers
		value = local_var:sul_demand_base_burghers
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:soldiers
		value = local_var:sul_soldier_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:soldiers
		value = local_var:sul_demand_base_soldiers
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:laborers
		value = local_var:sul_laborer_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:laborers
		value = local_var:sul_demand_base_laborers
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:peasants
		value = local_var:sul_peasant_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:peasants
		value = local_var:sul_demand_base_peasants
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:tribesmen
		value = local_var:sul_tribesmen_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:tribesmen
		value = local_var:sul_demand_base_tribesmen
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:slaves
		value = 0
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:slaves
		value = 0
	}

	# --- Enrichment: demand_base × WPP × weight × pops = estate savings after demand ---
	set_variable = {
		name = sul_enrichment_nobles
		value = {
			value = local_var:sul_demand_base_nobles
			multiply = local_var:sul_noble_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:nobles

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_nobles } }
				multiply = owner.var:sul_enrich_rate_nobles
			}
		}
	}
	set_variable = {
		name = sul_enrichment_clergy
		value = {
			value = local_var:sul_demand_base_clergy
			multiply = local_var:sul_clergy_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:clergy

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_clergy } }
				multiply = owner.var:sul_enrich_rate_clergy
			}
		}
	}
	set_variable = {
		name = sul_enrichment_burghers
		value = {
			value = local_var:sul_demand_base_burghers
			multiply = local_var:sul_burgher_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:burghers

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_burghers } }
				multiply = owner.var:sul_enrich_rate_burghers
			}
		}
	}
	set_variable = {
		name = sul_enrichment_commoners
		value = {
			value = local_var:sul_demand_base_soldiers
			multiply = local_var:sul_soldier_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:soldiers

			add = {
				value = local_var:sul_demand_base_laborers
				multiply = local_var:sul_laborer_wpp
				multiply = @sul_enrichment_weight
				multiply = num_pop_type:laborers
			}
			add = {
				value = local_var:sul_demand_base_peasants
				multiply = local_var:sul_peasant_wpp
				multiply = @sul_enrichment_weight
				multiply = num_pop_type:peasants
			}

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_commoners } }
				multiply = owner.var:sul_enrich_rate_commoners
			}
		}
	}

	set_variable = {
		name = sul_enrichment_tribesmen
		value = {
			value = local_var:sul_demand_base_tribesmen
			multiply = local_var:sul_tribesmen_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:tribesmen

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_tribesmen } }
				multiply = owner.var:sul_enrich_rate_tribesmen
			}
		}
	}
}

# -----------------------------------------------------------------------------
# BUDGET PRESSURE + ENRICHMENT RATES (country scope)
# -----------------------------------------------------------------------------
# Cached before the location loop so WPP can read via owner.var:sul_enrich_rate_*.
# Budget pressure: (gold - 1000) × 0.00025 + 1, min 1 — richer estates spend more.
# Enrichment rate: (1 + estate_enrichment) × (1 + per-estate modifier).

sul_compute_budget_pressure = {
	"estate(estate_type:nobles_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_nobles
				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	"estate(estate_type:clergy_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_clergy
				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	"estate(estate_type:burghers_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_burghers
				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	"estate(estate_type:peasants_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_peasants
				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	# Per-estate enrichment rate multipliers
	set_local_variable = { name = sul_base_enrich value = { value = 1 add = modifier:estate_enrichment } }

	set_variable = {
		name = sul_enrich_rate_nobles
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_nobles_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_clergy
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_clergy_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_burghers
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_burghers_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_commoners
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_commoners_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_tribesmen
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_tribesmen_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_dhimmi
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_dhimmi_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_cossacks
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_cossacks_enrichment_rate } }
	}
}

# -----------------------------------------------------------------------------
# INIT ECONOMY FULL PASS (country scope)
# -----------------------------------------------------------------------------
# Game-start full pass: budget pressure → per-location WPP →
# accumulate wage bills + enrichment → pay enrichment → cache estate gold.

sul_init_economy_full_pass = {
	set_variable = {
		name = sul_country_pop_nobles
		value = { value = "pop_type_population_in_country(pop_type:nobles)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_clergy
		value = { value = "pop_type_population_in_country(pop_type:clergy)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_burghers
		value = { value = "pop_type_population_in_country(pop_type:burghers)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_soldiers
		value = { value = "pop_type_population_in_country(pop_type:soldiers)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_laborers
		value = { value = "pop_type_population_in_country(pop_type:laborers)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_peasants
		value = { value = "pop_type_population_in_country(pop_type:peasants)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_tribesmen
		value = { value = "pop_type_population_in_country(pop_type:tribesmen)" min = 1 }
	}

	sul_compute_budget_pressure = yes

	# Cache estate gold for national returns
	set_variable = { name = sul_estate_gold_nobles value = "estate(estate_type:nobles_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_clergy value = "estate(estate_type:clergy_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_burghers value = "estate(estate_type:burghers_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_peasants value = 0 }

	"estate(estate_type:peasants_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_peasants value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_tribes value = 0 }

	"estate(estate_type:tribes_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_tribes value = prev.estate_gold } }
	}

	# Accumulators: wage bills, enrichment savings, enfranchisement-weighted transfer
	set_local_variable = { name = sul_noble_wages value = 0 }
	set_local_variable = { name = sul_clergy_wages value = 0 }
	set_local_variable = { name = sul_burgher_wages value = 0 }
	set_local_variable = { name = sul_commoner_wages value = 0 }
	set_local_variable = { name = sul_enrich_nobles value = 0 }
	set_local_variable = { name = sul_enrich_clergy value = 0 }
	set_local_variable = { name = sul_enrich_burghers value = 0 }
	set_local_variable = { name = sul_enrich_commoners value = 0 }
	set_local_variable = { name = sul_enrich_tribesmen value = 0 }
	set_local_variable = { name = sul_transfer_weighted value = 0 }

	every_owned_location = {
		if = {
			limit = { market_access > 0 }
			sul_update_location_wpp = yes

			if = {
				limit = { var:sul_local_wages > 0 }
				owner ?= {
					change_local_variable = { name = sul_noble_wages add = prev.var:sul_wage_bill_nobles }
					change_local_variable = { name = sul_clergy_wages add = prev.var:sul_wage_bill_clergy }
					change_local_variable = { name = sul_burgher_wages add = prev.var:sul_wage_bill_burghers }
					change_local_variable = { name = sul_commoner_wages add = prev.var:sul_wage_bill_commoners }

					# Enrichment: net savings (wages - demand) per estate
					change_local_variable = { name = sul_enrich_nobles add = prev.var:sul_enrichment_nobles }
					change_local_variable = { name = sul_enrich_clergy add = prev.var:sul_enrichment_clergy }
					change_local_variable = { name = sul_enrich_burghers add = prev.var:sul_enrichment_burghers }
					change_local_variable = { name = sul_enrich_commoners add = prev.var:sul_enrichment_commoners }
					change_local_variable = { name = sul_enrich_tribesmen add = prev.var:sul_enrichment_tribesmen }

					# Enfranchisement × commoner political power
					change_local_variable = {
						name = sul_transfer_weighted
						add = {
							value = prev.var:sul_wage_bill_commoners

							multiply = {
								value = prev.modifier:local_peasant_enfranchisment
								add = modifier:global_peasant_enfranchisment
								min = 0.1
								max = 1.0
							}

							multiply = "prev.local_relative_estate_power(estate_type:peasants_estate)"
						}
					}
				}
			}
		}
	}

	set_variable = { name = sul_transfer_weighted_total value = local_var:sul_transfer_weighted }

	# Pay enrichment to estates (negative = withdraw from estate gold)
	if = {
		limit = { local_var:sul_enrich_nobles > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:nobles_estate
			value = { value = local_var:sul_enrich_nobles multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_clergy > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:clergy_estate
			value = { value = local_var:sul_enrich_clergy multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_burghers > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:burghers_estate
			value = { value = local_var:sul_enrich_burghers multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_commoners > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:peasants_estate
			value = { value = local_var:sul_enrich_commoners multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_tribesmen > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:tribes_estate
			value = { value = local_var:sul_enrich_tribesmen multiply = -1 }
		}
	}

	# Cache estate gold after enrichment payments
	set_variable = { name = sul_estate_gold_nobles value = "estate(estate_type:nobles_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_clergy value = "estate(estate_type:clergy_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_burghers value = "estate(estate_type:burghers_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_peasants value = 0 }

	"estate(estate_type:peasants_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_peasants value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_tribes value = 0 }

	"estate(estate_type:tribes_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_tribes value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_crown value = gold }

	set_variable = { name = sul_monthly_wages_nobles value = local_var:sul_noble_wages }
	set_variable = { name = sul_monthly_wages_clergy value = local_var:sul_clergy_wages }
	set_variable = { name = sul_monthly_wages_burghers value = local_var:sul_burgher_wages }
	set_variable = { name = sul_monthly_wages_peasants value = local_var:sul_commoner_wages }

	set_variable = { name = sul_monthly_wages_dhimmi value = 0 }
	set_variable = { name = sul_monthly_wages_cossacks value = 0 }
	set_variable = { name = sul_monthly_wages_tribes value = 0 }

	sul_estate_wages_pay_all = yes
}

# -----------------------------------------------------------------------------
# WEALTH SEED (location scope)
# -----------------------------------------------------------------------------
# Sets sul_local_assets to max on game start so day-1 values are correct.

sul_wealth_seed_location = {
	set_local_variable = {
		name = sul_max_assets
		value = {
			value = var:sul_local_gdp
			multiply = @asset_max_multiplier
			multiply = { value = 1 add = { value = prosperity divide = 100 } }
			min = 0
		}
	}

	set_variable = { name = sul_local_assets value = local_var:sul_max_assets }
}

# -----------------------------------------------------------------------------
# ASSET CONVERGENCE (location scope)
# -----------------------------------------------------------------------------
# Assets converge toward max at @asset_converge_rate per month.
# Occupied locations drain at @asset_drain_rate of remaining, no recovery.

sul_converge_location_assets = {
	# Seed on first encounter (new conquest, game start stragglers)
	if = {
		limit = { NOT = { has_variable = sul_local_assets } }
		sul_wealth_seed_location = yes
	}

	set_local_variable = { name = sul_prev_assets value = var:sul_local_assets }

	set_local_variable = {
		name = sul_max_assets
		value = {
			value = var:sul_local_gdp
			multiply = @asset_max_multiplier
			multiply = { value = 1 add = { value = prosperity divide = 100 } }
			min = 0
		}
	}

	if = {
		limit = { is_occupied = yes }
		# Occupation: drain percentage of remaining, no recovery
		change_variable = { name = sul_local_assets multiply = { value = 1 subtract = @asset_drain_rate } }
	}
	else = {
		# Recovery: converge toward max, clamped to prevent oscillation
		set_local_variable = { name = sul_delta value = { value = local_var:sul_max_assets subtract = var:sul_local_assets } }

		# Clamp delta: magnitude <= @asset_converge_rate × max
		set_local_variable = {
			name = sul_max_step
			value = { value = local_var:sul_max_assets multiply = @asset_converge_rate }
		}
		set_local_variable = { name = sul_neg_step value = { value = local_var:sul_max_step multiply = -1 } }

		if = {
			limit = { local_var:sul_delta > local_var:sul_max_step }
			set_local_variable = { name = sul_delta value = local_var:sul_max_step }
		}
		else_if = {
			limit = { local_var:sul_delta < local_var:sul_neg_step }
			set_local_variable = { name = sul_delta value = local_var:sul_neg_step }
		}

		change_variable = { name = sul_local_assets add = local_var:sul_delta }
	}

	# Floor at zero
	if = { limit = { var:sul_local_assets < 0 } set_variable = { name = sul_local_assets value = 0 } }

	# Store delta for tooltip display
	set_variable = {
		name = sul_asset_delta
		value = { value = var:sul_local_assets subtract = local_var:sul_prev_assets }
	}
}

# -----------------------------------------------------------------------------
# WEALTH LOOT (location scope, called on occupation)
# -----------------------------------------------------------------------------
# Drains @sul_loot_extraction_rate of local assets. Controller keeps a fraction.

sul_wealth_loot_location = {
	set_local_variable = {
		name = sul_loot_total
		value = { value = var:sul_local_assets multiply = @sul_loot_extraction_rate }
	}

	change_variable = { name = sul_local_assets subtract = local_var:sul_loot_total }

	# Crown income = loot × (1 - destroyed_base + looting_efficiency)
	controller ?= {
		set_local_variable = {
			name = sul_loot_kept
			value = {
				value = 1
				subtract = @sul_loot_destroyed_base
				add = modifier:sul_looting_efficiency
				min = 0
				max = 1
			}
		}
		add_gold = { value = local_var:sul_loot_total multiply = local_var:sul_loot_kept }
	}

	set_variable = { name = sul_wealth_loot_amount value = local_var:sul_loot_total }
}

		value = { value = "pop_type_population_in_country(pop_type:nobles)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_clergy

	set_variable = {
		name = sul_country_pop_burghers
		value = { value = "pop_type_population_in_country(pop_type:burghers)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_soldiers
		value = { value = "pop_type_population_in_country(pop_type:soldiers)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_laborers
		value = { value = "pop_type_population_in_country(pop_type:laborers)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_peasants
		value = { value = "pop_type_population_in_country(pop_type:peasants)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_tribesmen
		value = { value = "pop_type_population_in_country(pop_type:tribesmen)" min = 1 }
	}

	set_local_variable = { name = sul_noble_wages value = 0 }
	set_local_variable = { name = sul_clergy_wages value = 0 }
	set_local_variable = { name = sul_burgher_wages value = 0 }
	set_local_variable = { name = sul_commoner_wages value = 0 }

	sul_compute_budget_pressure = yes

	# Cache estate gold for national returns in WPP
	set_variable = { name = sul_estate_gold_nobles value = "estate(estate_type:nobles_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_clergy value = "estate(estate_type:clergy_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_burghers value = "estate(estate_type:burghers_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_peasants value = 0 }

	"estate(estate_type:peasants_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_peasants value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_tribes value = 0 }

	"estate(estate_type:tribes_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_tribes value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_crown value = gold }

	every_owned_location = {
		sul_converge_location_assets = yes
		owner ?= {
			# Wages: accumulate cached wage bills
			if = {
				limit = { prev = { has_variable = sul_wage_bill_nobles var:sul_local_wages > 0 } }

				change_local_variable = { name = sul_noble_wages add = prev.var:sul_wage_bill_nobles }

			}
		}
	}

	# Wages: finalize monthly totals
	sul_compute_budget_pressure = yes

	set_variable = { name = sul_monthly_wages_nobles value = local_var:sul_noble_wages }
	set_variable = { name = sul_monthly_wages_clergy value = local_var:sul_clergy_wages }
	set_variable = { name = sul_monthly_wages_burghers value = local_var:sul_burgher_wages }
	set_variable = { name = sul_monthly_wages_peasants value = local_var:sul_commoner_wages }

	set_variable = { name = sul_monthly_wages_cossacks value = 0 }
	set_variable = { name = sul_monthly_wages_tribes value = 0 }
