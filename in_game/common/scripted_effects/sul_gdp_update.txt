# WPP pipeline: capital returns for upper estates, wage-based for commoners.
#
# Upper class (nobles, clergy, burghers):
#   WPP = base_wage + capital_return × asset_share × power_share
#   capital_return = per_pop_capital × return_rate (set monthly at country level)
#   asset_share = location's fraction of national GDP (set yearly)
#   power_share = local_relative_estate_power (engine's 0-1 political influence)
#
# Commoners (soldiers, laborers, peasants):
#   WPP = base_wage × (1 + wage_modifiers + enfranchisement) + flat_wealth
#   flat_wealth = country-level supplement from commoner estate savings
#
# Stored per location:
#   sul_wage_bill_*       per-pop-type wage bills (6 types)
#   sul_local_wages       total wage bill (sum of above)
#   sul_local_gdp         WPP-based economic activity (sum of WPP × pop counts)
#   sul_asset_share       location's fraction of national GDP (yearly)
#   sul_wealth_per_pop    variable map read by pop_demands
#   sul_demand_base       variable map read by demand tier formulas

# Demand saturation curve: WPP / (intercept + slope × WPP)
@sul_demand_intercept = 3.0
@sul_demand_slope = 1.0

# Fraction of WPP that becomes estate savings (remainder goes to consumption)
@sul_enrichment_weight = 0.53

# Capital return rate: estates earn this fraction of local wealth per month
@sul_return_rate = 0.02

# Minority power_per_pop constants (inverted to estimate pop count from power)
@sul_dhimmi_power_per_pop = 0.02
@sul_cossacks_power_per_pop = 0.5

# Dhimmi return distribution weights by pop type
@sul_weight_nobles = 25
@sul_weight_clergy = 10
@sul_weight_burghers = 2
@sul_weight_commoners = 0.1
@sul_weight_tribesmen = 0.01

# -----------------------------------------------------------------------------
# MINORITY RETURNS (location scope, called by WPP refresh + wealth seed)
# -----------------------------------------------------------------------------

sul_compute_minority_returns = {
	set_variable = { name = sul_dhimmi_return_per_pop value = 0 }
	set_variable = { name = sul_cossacks_return_per_pop value = 0 }
	set_variable = { name = sul_enrichment_dhimmi value = 0 }
	set_variable = { name = sul_enrichment_cossacks value = 0 }

	if = {
		limit = { has_variable = sul_asset_dhimmi }

		set_local_variable = { name = sul_dhimmi_power value = "local_estate_power(estate_type:dhimmi_estate)" }

		set_variable = {
			name = sul_dhimmi_return_per_pop
			value = {
				value = var:sul_asset_dhimmi
				multiply = @sul_return_rate
				multiply = @sul_dhimmi_power_per_pop
				divide = { value = local_var:sul_dhimmi_power min = 0.001 }
			}
		}
		set_variable = {
			name = sul_enrichment_dhimmi
			value = {
				value = var:sul_dhimmi_return_per_pop

				divide = {
					value = @sul_demand_intercept
					add = { value = var:sul_dhimmi_return_per_pop multiply = @sul_demand_slope }
				}

				multiply = var:sul_dhimmi_return_per_pop
				multiply = @sul_enrichment_weight

				multiply = { value = local_var:sul_dhimmi_power divide = { value = @sul_dhimmi_power_per_pop min = 0.001 } }

				if = {
					limit = { owner = { has_variable = sul_enrich_rate_dhimmi } }
					multiply = owner.var:sul_enrich_rate_dhimmi
				}
			}
		}
	}

	if = {
		limit = { has_variable = sul_asset_cossacks }

		set_local_variable = { name = sul_cossacks_power value = "local_estate_power(estate_type:cossacks_estate)" }

		set_variable = {
			name = sul_cossacks_return_per_pop
			value = {
				value = var:sul_asset_cossacks
				multiply = @sul_return_rate
				multiply = @sul_cossacks_power_per_pop
				divide = { value = local_var:sul_cossacks_power min = 0.001 }
			}
		}
		set_variable = {
			name = sul_enrichment_cossacks
			value = {
				value = var:sul_cossacks_return_per_pop

				divide = {
					value = @sul_demand_intercept
					add = { value = var:sul_cossacks_return_per_pop multiply = @sul_demand_slope }
				}

				multiply = var:sul_cossacks_return_per_pop
				multiply = @sul_enrichment_weight

				multiply = { value = local_var:sul_cossacks_power divide = { value = @sul_cossacks_power_per_pop min = 0.001 } }

				if = {
					limit = { owner = { has_variable = sul_enrich_rate_cossacks } }
					multiply = owner.var:sul_enrich_rate_cossacks
				}
			}
		}
	}
}

# -----------------------------------------------------------------------------
# SINGLE LOCATION UPDATE (location scope)
# -----------------------------------------------------------------------------

sul_update_location_wpp = {
	# 1. Per-pop-type wage bills (pop × rate × market_access)
	set_variable = {
		name = sul_wage_bill_nobles
		value = {
			value = num_pop_type:nobles
			multiply = var:sul_wages_nobles
			multiply = market_access
		}
	}
	set_variable = {
		name = sul_wage_bill_clergy
		value = {
			value = num_pop_type:clergy
			multiply = var:sul_wages_clergy
			multiply = market_access
		}
	}
	set_variable = {
		name = sul_wage_bill_burghers
		value = {
			value = num_pop_type:burghers
			multiply = var:sul_wages_burghers
			multiply = market_access
		}
	}
	set_variable = {
		name = sul_wage_bill_soldiers
		value = {
			value = num_pop_type:soldiers
			multiply = var:sul_wages_soldiers
			multiply = market_access
		}
	}
	set_variable = {
		name = sul_wage_bill_laborers
		value = {
			value = num_pop_type:laborers
			multiply = var:sul_wages_laborers
			multiply = market_access
		}
	}
	set_variable = {
		name = sul_wage_bill_peasants
		value = {
			value = num_pop_type:peasants
			multiply = var:sul_wages_peasants
			multiply = market_access
		}
	}

	# 2. Total local wages
	set_variable = {
		name = sul_local_wages
		value = {
			value = var:sul_wage_bill_nobles
			add = var:sul_wage_bill_clergy
			add = var:sul_wage_bill_burghers
			add = var:sul_wage_bill_soldiers
			add = var:sul_wage_bill_laborers
			add = var:sul_wage_bill_peasants
		}
	}

	# 3. Per-pop WPP: wage pool × estate power share / pop count + capital returns.
	# Guarded on pop type existing — local_relative_estate_power() returns none
	# for absent estates, which poisons the inline value.
	set_local_variable = { name = sul_wage_pool value = var:sul_local_wages }

	# Dhimmi returns distributed to all pops weighted by political importance
	set_local_variable = { name = sul_dhimmi_pool value = 0 }

	if = {
		limit = { has_variable = sul_asset_dhimmi }
		set_local_variable = { name = sul_dhimmi_pool value = { value = var:sul_asset_dhimmi multiply = @sul_return_rate } }
	}

	set_local_variable = {
		name = sul_total_weight
		value = {
			value = { value = num_pop_type:nobles multiply = @sul_weight_nobles }
			add = { value = num_pop_type:clergy multiply = @sul_weight_clergy }
			add = { value = num_pop_type:burghers multiply = @sul_weight_burghers }
			add = { value = num_pop_type:soldiers multiply = @sul_weight_commoners }
			add = { value = num_pop_type:laborers multiply = @sul_weight_commoners }
			add = { value = num_pop_type:peasants multiply = @sul_weight_commoners }
			add = { value = num_pop_type:tribesmen multiply = @sul_weight_tribesmen }
			min = 0.001
		}
	}

	set_local_variable = { name = sul_noble_wpp value = 0 }

	if = {
		limit = { num_pop_type:nobles > 0 }
		set_local_variable = {
			name = sul_noble_wpp
			value = {
				value = local_var:sul_wage_pool
				multiply = "local_relative_estate_power(estate_type:nobles_estate)"
				divide = { value = num_pop_type:nobles min = 0.001 }

				if = {
					limit = { has_variable = sul_asset_nobles }
					add = {
						value = var:sul_asset_nobles
						multiply = @sul_return_rate
						divide = { value = num_pop_type:nobles min = 0.001 }
					}
				}

				add = {
					value = local_var:sul_dhimmi_pool
					multiply = @sul_weight_nobles
					divide = local_var:sul_total_weight
				}

				min = 0
			}
		}
	}

	set_local_variable = { name = sul_clergy_wpp value = 0 }

	if = {
		limit = { num_pop_type:clergy > 0 }
		set_local_variable = {
			name = sul_clergy_wpp
			value = {
				value = local_var:sul_wage_pool
				multiply = "local_relative_estate_power(estate_type:clergy_estate)"
				divide = { value = num_pop_type:clergy min = 0.001 }

				if = {
					limit = { has_variable = sul_asset_nobles }
					add = {
						value = var:sul_asset_clergy
						multiply = @sul_return_rate
						divide = { value = num_pop_type:clergy min = 0.001 }
					}
				}

				add = {
					value = local_var:sul_dhimmi_pool
					multiply = @sul_weight_clergy
					divide = local_var:sul_total_weight
				}

				min = 0
			}
		}
	}

	set_local_variable = { name = sul_burgher_wpp value = 0 }

	if = {
		limit = { num_pop_type:burghers > 0 }
		set_local_variable = {
			name = sul_burgher_wpp
			value = {
				value = local_var:sul_wage_pool
				multiply = "local_relative_estate_power(estate_type:burghers_estate)"
				divide = { value = num_pop_type:burghers min = 0.001 }

				if = {
					limit = { has_variable = sul_asset_nobles }
					add = {
						value = var:sul_asset_burghers
						multiply = @sul_return_rate
						divide = { value = num_pop_type:burghers min = 0.001 }
					}
				}

				add = {
					value = local_var:sul_dhimmi_pool
					multiply = @sul_weight_burghers
					divide = local_var:sul_total_weight
				}

				min = 0
			}
		}
	}

	# Commoner WPP: peasant estate's share distributed by wage bill.
	# Each type's share = their wage bill / total commoner bills.
	# Capital returns from peasant assets shared across all commoner types.
	set_local_variable = { name = sul_commoner_pool value = 0 }
	set_local_variable = { name = sul_commoner_bill_total value = 0.001 }
	set_local_variable = { name = sul_commoner_return value = 0 }
	set_local_variable = { name = sul_soldier_wpp value = 0 }
	set_local_variable = { name = sul_laborer_wpp value = 0 }
	set_local_variable = { name = sul_peasant_wpp value = 0 }

	if = {
		limit = {
			OR = {
				num_pop_type:soldiers > 0
				num_pop_type:laborers > 0
				num_pop_type:peasants > 0
			}
		}

		set_local_variable = {
			name = sul_commoner_pool
			value = { value = local_var:sul_wage_pool multiply = "local_relative_estate_power(estate_type:peasants_estate)" }
		}
		set_local_variable = {
			name = sul_commoner_bill_total
			value = {
				value = var:sul_wage_bill_soldiers
				add = var:sul_wage_bill_laborers
				add = var:sul_wage_bill_peasants
				min = 0.001
			}
		}

		if = {
			limit = { has_variable = sul_asset_nobles }
			set_local_variable = {
				name = sul_commoner_return
				value = {
					value = var:sul_asset_peasants
					multiply = @sul_return_rate

					divide = {
						value = num_pop_type:soldiers
						add = num_pop_type:laborers
						add = num_pop_type:peasants
						min = 0.001
					}
				}
			}
		}

		change_local_variable = {
			name = sul_commoner_return
			add = {
				value = local_var:sul_dhimmi_pool
				multiply = @sul_weight_commoners
				divide = local_var:sul_total_weight
			}
		}

		set_local_variable = {
			name = sul_soldier_wpp
			value = {
				value = local_var:sul_commoner_pool

				multiply = { value = var:sul_wage_bill_soldiers divide = local_var:sul_commoner_bill_total }

				divide = { value = num_pop_type:soldiers min = 0.001 }

				add = local_var:sul_commoner_return

				min = 0
			}
		}
		set_local_variable = {
			name = sul_laborer_wpp
			value = {
				value = local_var:sul_commoner_pool

				multiply = { value = var:sul_wage_bill_laborers divide = local_var:sul_commoner_bill_total }

				divide = { value = num_pop_type:laborers min = 0.001 }

				add = local_var:sul_commoner_return

				min = 0
			}
		}
		set_local_variable = {
			name = sul_peasant_wpp
			value = {
				value = local_var:sul_commoner_pool

				multiply = { value = var:sul_wage_bill_peasants divide = local_var:sul_commoner_bill_total }

				divide = { value = num_pop_type:peasants min = 0.001 }

				add = local_var:sul_commoner_return

				min = 0
			}
		}
	}

	# Tribesmen: own estate, no wage bill, draws from pool via power + capital returns
	# Also receives cossack returns (cossacks distribute only to tribes)
	set_local_variable = { name = sul_tribesmen_wpp value = 0 }

	if = {
		limit = { num_pop_type:tribesmen > 0 }
		set_local_variable = {
			name = sul_tribesmen_wpp
			value = {
				value = local_var:sul_wage_pool
				multiply = "local_relative_estate_power(estate_type:tribes_estate)"
				divide = { value = num_pop_type:tribesmen min = 0.001 }

				if = {
					limit = { has_variable = sul_asset_nobles }
					add = {
						value = var:sul_asset_tribes
						multiply = @sul_return_rate
						divide = { value = num_pop_type:tribesmen min = 0.001 }
					}
				}

				add = {
					value = local_var:sul_dhimmi_pool
					multiply = @sul_weight_tribesmen
					divide = local_var:sul_total_weight
				}

				if = {
					limit = { has_variable = sul_asset_nobles has_variable = sul_asset_cossacks }
					add = {
						value = var:sul_asset_cossacks
						multiply = @sul_return_rate
						divide = { value = num_pop_type:tribesmen min = 0.001 }
					}
				}

				min = 0
			}
		}
	}

	# 4. Local GDP = sum of WPP × pop counts (tooltip metric)
	set_variable = {
		name = sul_local_gdp
		value = {
			value = { value = local_var:sul_noble_wpp multiply = num_pop_type:nobles }
			add = { value = local_var:sul_clergy_wpp multiply = num_pop_type:clergy }
			add = { value = local_var:sul_burgher_wpp multiply = num_pop_type:burghers }
			add = { value = local_var:sul_soldier_wpp multiply = num_pop_type:soldiers }
			add = { value = local_var:sul_laborer_wpp multiply = num_pop_type:laborers }
			add = { value = local_var:sul_peasant_wpp multiply = num_pop_type:peasants }
			add = { value = local_var:sul_tribesmen_wpp multiply = num_pop_type:tribesmen }
		}
	}

	# 5. Demand saturation: WPP / (intercept + slope × WPP) → 0..1 curve
	set_local_variable = {
		name = sul_demand_base_nobles
		value = {
			value = local_var:sul_noble_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_noble_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_clergy
		value = {
			value = local_var:sul_clergy_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_clergy_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_burghers
		value = {
			value = local_var:sul_burgher_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_burgher_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_soldiers
		value = {
			value = local_var:sul_soldier_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_soldier_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_laborers
		value = {
			value = local_var:sul_laborer_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_laborer_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_peasants
		value = {
			value = local_var:sul_peasant_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_peasant_wpp multiply = @sul_demand_slope } }
		}
	}
	set_local_variable = {
		name = sul_demand_base_tribesmen
		value = {
			value = local_var:sul_tribesmen_wpp
			divide = { value = @sul_demand_intercept add = { value = local_var:sul_tribesmen_wpp multiply = @sul_demand_slope } }
		}
	}

	# 6. Export WPP + demand base to variable maps (read by pop_demands)
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:nobles
		value = local_var:sul_noble_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:nobles
		value = local_var:sul_demand_base_nobles
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:clergy
		value = local_var:sul_clergy_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:clergy
		value = local_var:sul_demand_base_clergy
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:burghers
		value = local_var:sul_burgher_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:burghers
		value = local_var:sul_demand_base_burghers
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:soldiers
		value = local_var:sul_soldier_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:soldiers
		value = local_var:sul_demand_base_soldiers
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:laborers
		value = local_var:sul_laborer_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:laborers
		value = local_var:sul_demand_base_laborers
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:peasants
		value = local_var:sul_peasant_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:peasants
		value = local_var:sul_demand_base_peasants
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:tribesmen
		value = local_var:sul_tribesmen_wpp
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:tribesmen
		value = local_var:sul_demand_base_tribesmen
	}
	cmf_change_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:slaves
		value = 0
	}
	cmf_change_variable_map = {
		name = sul_demand_base
		key = pop_type:slaves
		value = 0
	}

	# 7. Enrichment: demand_base × WPP × weight × pops = estate savings after demand
	set_variable = {
		name = sul_enrichment_nobles
		value = {
			value = local_var:sul_demand_base_nobles
			multiply = local_var:sul_noble_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:nobles

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_nobles } }
				multiply = owner.var:sul_enrich_rate_nobles
			}
		}
	}
	set_variable = {
		name = sul_enrichment_clergy
		value = {
			value = local_var:sul_demand_base_clergy
			multiply = local_var:sul_clergy_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:clergy

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_clergy } }
				multiply = owner.var:sul_enrich_rate_clergy
			}
		}
	}
	set_variable = {
		name = sul_enrichment_burghers
		value = {
			value = local_var:sul_demand_base_burghers
			multiply = local_var:sul_burgher_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:burghers

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_burghers } }
				multiply = owner.var:sul_enrich_rate_burghers
			}
		}
	}
	set_variable = {
		name = sul_enrichment_commoners
		value = {
			value = local_var:sul_demand_base_soldiers
			multiply = local_var:sul_soldier_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:soldiers

			add = {
				value = local_var:sul_demand_base_laborers
				multiply = local_var:sul_laborer_wpp
				multiply = @sul_enrichment_weight
				multiply = num_pop_type:laborers
			}
			add = {
				value = local_var:sul_demand_base_peasants
				multiply = local_var:sul_peasant_wpp
				multiply = @sul_enrichment_weight
				multiply = num_pop_type:peasants
			}

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_commoners } }
				multiply = owner.var:sul_enrich_rate_commoners
			}
		}
	}

	set_variable = {
		name = sul_enrichment_tribesmen
		value = {
			value = local_var:sul_demand_base_tribesmen
			multiply = local_var:sul_tribesmen_wpp
			multiply = @sul_enrichment_weight
			multiply = num_pop_type:tribesmen

			if = {
				limit = { owner = { has_variable = sul_enrich_rate_tribesmen } }
				multiply = owner.var:sul_enrich_rate_tribesmen
			}
		}
	}

	# 8. Minority estate returns and enrichment (dhimmi/cossacks)
	sul_compute_minority_returns = yes
}

# -----------------------------------------------------------------------------
# PER-LOCATION WAGE RATE (location scope)
# -----------------------------------------------------------------------------
# wage = base × (1 + global_mod + local_mod + pop_cap_factor × sensitivity)
# Pop capacity pulls wages up/down based on distance from 50% capacity.
# Sensitivity: nobles 0, clergy ±10%, burghers ±25%, commoners ±50%.
# Commoners also get enfranchisement: (enfr - 0.5) scaled to ±50%.

@sul_pop_cap_midpoint = 0.5
@sul_enfr_midpoint = 0.5

sul_calculate_location_wages = {
	set_local_variable = {
		name = sul_pop_cap_factor
		value = {
			value = @sul_pop_cap_midpoint
			subtract = location_population_percentage
			multiply = 2
		}
	}

	set_local_variable = {
		name = sul_enfr_factor
		value = {
			value = modifier:local_peasant_enfranchisment
			add = owner.modifier:global_peasant_enfranchisment
			min = 0.1
			max = 1.0
			subtract = @sul_enfr_midpoint
			multiply = 2
			multiply = 0.5
		}
	}

	set_variable = {
		name = sul_wages_nobles
		value = {
			value = owner.modifier:sul_nobles_base_wages

			multiply = {
				value = 1
				add = owner.modifier:sul_global_nobles_wages_modifier
				add = modifier:sul_local_nobles_wages_modifier
			}

			min = 0
		}
	}
	set_variable = {
		name = sul_wages_clergy
		value = {
			value = owner.modifier:sul_clergy_base_wages

			multiply = {
				value = 1
				add = owner.modifier:sul_global_clergy_wages_modifier
				add = modifier:sul_local_clergy_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.1 }
			}

			min = 0
		}
	}
	set_variable = {
		name = sul_wages_burghers
		value = {
			value = owner.modifier:sul_burghers_base_wages

			multiply = {
				value = 1
				add = owner.modifier:sul_global_burghers_wages_modifier
				add = modifier:sul_local_burghers_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.25 }
			}

			min = 0
		}
	}
	set_variable = {
		name = sul_wages_soldiers
		value = {
			value = owner.modifier:sul_soldiers_base_wages

			multiply = {
				value = 1
				add = owner.modifier:sul_global_soldiers_wages_modifier
				add = modifier:sul_local_soldiers_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.5 }
				add = local_var:sul_enfr_factor
			}

			min = 0
		}
	}
	set_variable = {
		name = sul_wages_laborers
		value = {
			value = owner.modifier:sul_laborers_base_wages

			multiply = {
				value = 1
				add = owner.modifier:sul_global_laborers_wages_modifier
				add = modifier:sul_local_laborers_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.5 }
				add = local_var:sul_enfr_factor
			}

			min = 0
		}
	}
	set_variable = {
		name = sul_wages_peasants
		value = {
			value = owner.modifier:sul_peasants_base_wages

			multiply = {
				value = 1
				add = owner.modifier:sul_global_peasants_wages_modifier
				add = modifier:sul_local_peasants_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.5 }
				add = local_var:sul_enfr_factor
			}

			min = 0
		}
	}
}

# -----------------------------------------------------------------------------
# BUDGET PRESSURE + ENRICHMENT RATES (country scope)
# -----------------------------------------------------------------------------
# Cached before the location loop so WPP can read via owner.var:sul_enrich_rate_*.
# Budget pressure: (gold - 1000) × 0.00025 + 1, min 1 — richer estates spend more.
# Enrichment rate: (1 + estate_enrichment) × (1 + per-estate modifier).

sul_compute_budget_pressure = {
	"estate(estate_type:nobles_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_nobles
				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	"estate(estate_type:clergy_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_clergy
				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	"estate(estate_type:burghers_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_burghers
				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	"estate(estate_type:peasants_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_peasants
				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	# Per-estate enrichment rate multipliers
	set_local_variable = { name = sul_base_enrich value = { value = 1 add = modifier:estate_enrichment } }

	set_variable = {
		name = sul_enrich_rate_nobles
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_nobles_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_clergy
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_clergy_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_burghers
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_burghers_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_commoners
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_commoners_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_tribesmen
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_tribesmen_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_dhimmi
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_dhimmi_enrichment_rate } }
	}
	set_variable = {
		name = sul_enrich_rate_cossacks
		value = { value = local_var:sul_base_enrich multiply = { value = 1 add = modifier:sul_cossacks_enrichment_rate } }
	}
}

# -----------------------------------------------------------------------------
# INIT ECONOMY FULL PASS (country scope)
# -----------------------------------------------------------------------------
# Game-start full pass: pop counts → budget pressure → per-location WPP →
# accumulate national power + wage bills → pay enrichment → cache estate gold.

sul_init_economy_full_pass = {
	set_variable = {
		name = sul_country_pop_nobles
		value = { value = "pop_type_population_in_country(pop_type:nobles)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_clergy
		value = { value = "pop_type_population_in_country(pop_type:clergy)" min = 1 }
	}
	set_variable = {
		name = sul_country_pop_burghers
		value = { value = "pop_type_population_in_country(pop_type:burghers)" min = 1 }
	}

	sul_compute_budget_pressure = yes

	# Accumulators: wage bills, enrichment savings, enfranchisement-weighted transfer
	set_local_variable = { name = sul_noble_wages value = 0 }
	set_local_variable = { name = sul_clergy_wages value = 0 }
	set_local_variable = { name = sul_burgher_wages value = 0 }
	set_local_variable = { name = sul_commoner_wages value = 0 }
	set_local_variable = { name = sul_enrich_nobles value = 0 }
	set_local_variable = { name = sul_enrich_clergy value = 0 }
	set_local_variable = { name = sul_enrich_burghers value = 0 }
	set_local_variable = { name = sul_enrich_commoners value = 0 }
	set_local_variable = { name = sul_enrich_tribesmen value = 0 }
	set_local_variable = { name = sul_enrich_dhimmi value = 0 }
	set_local_variable = { name = sul_enrich_cossacks value = 0 }
	set_local_variable = { name = sul_transfer_weighted value = 0 }

	# Zero national power counters for re-accumulation
	set_variable = { name = sul_national_power_nobles value = 0 }
	set_variable = { name = sul_national_power_clergy value = 0 }
	set_variable = { name = sul_national_power_burghers value = 0 }
	set_variable = { name = sul_national_power_peasants value = 0 }
	set_variable = { name = sul_national_power_dhimmi value = 0 }
	set_variable = { name = sul_national_power_cossacks value = 0 }
	set_variable = { name = sul_national_power_tribes value = 0 }

	every_owned_location = {
		owner ?= {
			change_variable = { name = sul_national_power_nobles add = "prev.local_estate_power(estate_type:nobles_estate)" }
			change_variable = { name = sul_national_power_clergy add = "prev.local_estate_power(estate_type:clergy_estate)" }
			change_variable = { name = sul_national_power_burghers add = "prev.local_estate_power(estate_type:burghers_estate)" }
			change_variable = { name = sul_national_power_peasants add = "prev.local_estate_power(estate_type:peasants_estate)" }

			if = {
				limit = { country_has_estate = estate_type:dhimmi_estate }
				change_variable = { name = sul_national_power_dhimmi add = "prev.local_estate_power(estate_type:dhimmi_estate)" }
			}

			if = {
				limit = { country_has_estate = estate_type:cossacks_estate }
				change_variable = { name = sul_national_power_cossacks add = "prev.local_estate_power(estate_type:cossacks_estate)" }
			}

			if = {
				limit = { country_has_estate = estate_type:tribes_estate }
				change_variable = { name = sul_national_power_tribes add = "prev.local_estate_power(estate_type:tribes_estate)" }
			}
		}
		if = {
			limit = { market_access > 0 }
			sul_update_location_wpp = yes

			if = {
				limit = { var:sul_local_wages > 0 }
				owner ?= {
					change_local_variable = { name = sul_noble_wages add = prev.var:sul_wage_bill_nobles }
					change_local_variable = { name = sul_clergy_wages add = prev.var:sul_wage_bill_clergy }
					change_local_variable = { name = sul_burgher_wages add = prev.var:sul_wage_bill_burghers }
					change_local_variable = {
						name = sul_commoner_wages
						add = {
							value = prev.var:sul_wage_bill_soldiers
							add = prev.var:sul_wage_bill_laborers
							add = prev.var:sul_wage_bill_peasants
						}
					}

					# Enrichment: net savings (wages - demand) per estate
					change_local_variable = { name = sul_enrich_nobles add = prev.var:sul_enrichment_nobles }
					change_local_variable = { name = sul_enrich_clergy add = prev.var:sul_enrichment_clergy }
					change_local_variable = { name = sul_enrich_burghers add = prev.var:sul_enrichment_burghers }
					change_local_variable = { name = sul_enrich_commoners add = prev.var:sul_enrichment_commoners }
					change_local_variable = { name = sul_enrich_tribesmen add = prev.var:sul_enrichment_tribesmen }
					change_local_variable = { name = sul_enrich_dhimmi add = prev.var:sul_enrichment_dhimmi }
					change_local_variable = { name = sul_enrich_cossacks add = prev.var:sul_enrichment_cossacks }

					# Enfranchisement × commoner political power
					change_local_variable = {
						name = sul_transfer_weighted
						add = {
							value = prev.var:sul_wage_bill_soldiers
							add = prev.var:sul_wage_bill_laborers
							add = prev.var:sul_wage_bill_peasants

							multiply = {
								value = prev.modifier:local_peasant_enfranchisment
								add = modifier:global_peasant_enfranchisment
								min = 0.1
								max = 1.0
							}

							multiply = "prev.local_relative_estate_power(estate_type:peasants_estate)"
						}
					}
				}
			}
		}
	}

	set_variable = { name = sul_transfer_weighted_total value = local_var:sul_transfer_weighted }

	# Wealth: derive crown from estate totals
	set_variable = { name = sul_national_power_crown value = var:sul_national_power_nobles }

	change_variable = { name = sul_national_power_crown add = var:sul_national_power_clergy }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_burghers }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_peasants }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_dhimmi }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_cossacks }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_tribes }

	# Pay enrichment to estates (negative = withdraw from estate gold)
	if = {
		limit = { local_var:sul_enrich_nobles > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:nobles_estate
			value = { value = local_var:sul_enrich_nobles multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_clergy > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:clergy_estate
			value = { value = local_var:sul_enrich_clergy multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_burghers > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:burghers_estate
			value = { value = local_var:sul_enrich_burghers multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_commoners > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:peasants_estate
			value = { value = local_var:sul_enrich_commoners multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_tribesmen > 0.01 }
		add_gold_to_estate = {
			estate_type = estate_type:tribes_estate
			value = { value = local_var:sul_enrich_tribesmen multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_dhimmi > 0.01 country_has_estate = estate_type:dhimmi_estate }
		add_gold_to_estate = {
			estate_type = estate_type:dhimmi_estate
			value = { value = local_var:sul_enrich_dhimmi multiply = -1 }
		}
	}

	if = {
		limit = { local_var:sul_enrich_cossacks > 0.01 country_has_estate = estate_type:cossacks_estate }
		add_gold_to_estate = {
			estate_type = estate_type:cossacks_estate
			value = { value = local_var:sul_enrich_cossacks multiply = -1 }
		}
	}

	# Cache estate gold after enrichment payments
	set_variable = { name = sul_estate_gold_nobles value = "estate(estate_type:nobles_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_clergy value = "estate(estate_type:clergy_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_burghers value = "estate(estate_type:burghers_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_peasants value = 0 }

	"estate(estate_type:peasants_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_peasants value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_dhimmi value = 0 }

	"estate(estate_type:dhimmi_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_dhimmi value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_cossacks value = 0 }

	"estate(estate_type:cossacks_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_cossacks value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_tribes value = 0 }

	"estate(estate_type:tribes_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_tribes value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_crown value = gold }

	set_variable = { name = sul_monthly_wages_nobles value = local_var:sul_noble_wages }
	set_variable = { name = sul_monthly_wages_clergy value = local_var:sul_clergy_wages }
	set_variable = { name = sul_monthly_wages_burghers value = local_var:sul_burgher_wages }
	set_variable = { name = sul_monthly_wages_peasants value = local_var:sul_commoner_wages }

	set_variable = { name = sul_monthly_wages_dhimmi value = 0 }
	set_variable = { name = sul_monthly_wages_cossacks value = 0 }
	set_variable = { name = sul_monthly_wages_tribes value = 0 }

	sul_estate_wages_pay_all = yes
}

		set_variable = { name = sul_potential_cossacks value = local_var:sul_val }
	}

	if = {

				multiply = owner.var:sul_estate_gold_tribes

		set_variable = { name = sul_potential_tribes value = local_var:sul_val }
	}

		name = sul_val
		value = {
			value = local_var:sul_crown_local_power
			divide = { value = owner.var:sul_national_power_crown min = 0.001 }

		}
	}

	set_variable = { name = sul_asset_crown value = local_var:sul_val }
	set_variable = { name = sul_potential_crown value = local_var:sul_val }

	sul_compute_minority_returns = yes

	set_variable = { name = sul_delta_nobles value = 0 }
	set_variable = { name = sul_delta_clergy value = 0 }
	set_variable = { name = sul_delta_burghers value = 0 }
	set_variable = { name = sul_delta_peasants value = 0 }
	set_variable = { name = sul_delta_crown value = 0 }
}

# -----------------------------------------------------------------------------
# WEALTH CONVERGENCE (location scope)
# -----------------------------------------------------------------------------
# assets += (target - assets) × 2.5% per month (asymptotic approach).
# Uses prev-month national power to avoid same-tick feedback.
# Occupied locations drain assets at 2%/month; potential freezes.

sul_converge_location_assets = {
	if = {
		limit = { NOT = { has_variable = sul_converge_count } }
		set_variable = { name = sul_converge_count value = 0 }
	}

	change_variable = { name = sul_converge_count add = 1 }

	# Seed on first encounter (new conquest, game start stragglers)
	if = {
		limit = { NOT = { has_variable = sul_asset_nobles } }
		sul_wealth_seed_location = yes
	}

	set_variable = { name = sul_wealth_loot_amount value = 0 }

	# Snapshot pre-convergence values for delta calculation
	set_local_variable = { name = sul_prev_nobles value = var:sul_asset_nobles }
	set_local_variable = { name = sul_prev_clergy value = var:sul_asset_clergy }
	set_local_variable = { name = sul_prev_burghers value = var:sul_asset_burghers }
	set_local_variable = { name = sul_prev_peasants value = var:sul_asset_peasants }
	set_local_variable = { name = sul_prev_crown value = var:sul_asset_crown }
	set_local_variable = { name = sul_prev_dhimmi value = 0 }
	set_local_variable = { name = sul_prev_cossacks value = 0 }
	set_local_variable = { name = sul_prev_tribes value = 0 }

	if = {
		limit = { has_variable = sul_asset_dhimmi }
		set_local_variable = { name = sul_prev_dhimmi value = var:sul_asset_dhimmi }
	}

	if = {
		limit = { has_variable = sul_asset_cossacks }
		set_local_variable = { name = sul_prev_cossacks value = var:sul_asset_cossacks }
	}

	if = {
		limit = { has_variable = sul_asset_tribes }
		set_local_variable = { name = sul_prev_tribes value = var:sul_asset_tribes }
	}

	# Occupation: drain assets, skip convergence (potential freezes)
	if = {
		limit = { is_occupied = yes }
		change_variable = { name = sul_asset_nobles multiply = { value = 1 subtract = @sul_wealth_occupation_drain } }
		change_variable = { name = sul_asset_clergy multiply = { value = 1 subtract = @sul_wealth_occupation_drain } }
		change_variable = { name = sul_asset_burghers multiply = { value = 1 subtract = @sul_wealth_occupation_drain } }
		change_variable = { name = sul_asset_peasants multiply = { value = 1 subtract = @sul_wealth_occupation_drain } }
		change_variable = { name = sul_asset_crown multiply = { value = 1 subtract = @sul_wealth_occupation_drain } }

		if = {
			limit = { has_variable = sul_asset_dhimmi }
			change_variable = { name = sul_asset_dhimmi multiply = { value = 1 subtract = @sul_wealth_occupation_drain } }
		}

		if = {
			limit = { has_variable = sul_asset_cossacks }
			change_variable = { name = sul_asset_cossacks multiply = { value = 1 subtract = @sul_wealth_occupation_drain } }
		}

		if = {
			limit = { has_variable = sul_asset_tribes }
			change_variable = { name = sul_asset_tribes multiply = { value = 1 subtract = @sul_wealth_occupation_drain } }
		}
	}

	# Convergence: target = (local_power / national_power) × estate_gold
	else = {
		set_local_variable = { name = sul_crown_power value = 0 }

		# Nobles
		set_local_variable = { name = sul_estate_power value = "local_estate_power(estate_type:nobles_estate)" }

		change_local_variable = { name = sul_crown_power add = local_var:sul_estate_power }

		set_local_variable = {
			name = sul_wealth_target
			value = {
				value = local_var:sul_estate_power
				divide = { value = owner.var:sul_national_power_prev_nobles min = 0.001 }
				multiply = owner.var:sul_estate_gold_nobles
			}
		}
		set_local_variable = {
			name = sul_val
			value = {
				value = local_var:sul_prev_nobles

				add = {
					value = local_var:sul_wealth_target
					subtract = local_var:sul_prev_nobles
					multiply = @sul_wealth_converge_rate
				}

				min = 0
			}
		}

		set_variable = { name = sul_asset_nobles value = local_var:sul_val }
		set_variable = { name = sul_potential_nobles value = local_var:sul_wealth_target }

		# Clergy
		set_local_variable = { name = sul_estate_power value = "local_estate_power(estate_type:clergy_estate)" }

		change_local_variable = { name = sul_crown_power add = local_var:sul_estate_power }

		set_local_variable = {
			name = sul_wealth_target
			value = {
				value = local_var:sul_estate_power
				divide = { value = owner.var:sul_national_power_prev_clergy min = 0.001 }
				multiply = owner.var:sul_estate_gold_clergy
			}
		}
		set_local_variable = {
			name = sul_val
			value = {
				value = local_var:sul_prev_clergy

				add = {
					value = local_var:sul_wealth_target
					subtract = local_var:sul_prev_clergy
					multiply = @sul_wealth_converge_rate
				}

				min = 0
			}
		}

		set_variable = { name = sul_asset_clergy value = local_var:sul_val }
		set_variable = { name = sul_potential_clergy value = local_var:sul_wealth_target }

		# Burghers
		set_local_variable = { name = sul_estate_power value = "local_estate_power(estate_type:burghers_estate)" }

		change_local_variable = { name = sul_crown_power add = local_var:sul_estate_power }

		set_local_variable = {
			name = sul_wealth_target
			value = {
				value = local_var:sul_estate_power
				divide = { value = owner.var:sul_national_power_prev_burghers min = 0.001 }
				multiply = owner.var:sul_estate_gold_burghers
			}
		}
		set_local_variable = {
			name = sul_val
			value = {
				value = local_var:sul_prev_burghers

				add = {
					value = local_var:sul_wealth_target
					subtract = local_var:sul_prev_burghers
					multiply = @sul_wealth_converge_rate
				}

				min = 0
			}
		}

		set_variable = { name = sul_asset_burghers value = local_var:sul_val }
		set_variable = { name = sul_potential_burghers value = local_var:sul_wealth_target }

		# Peasants
		set_local_variable = { name = sul_estate_power value = "local_estate_power(estate_type:peasants_estate)" }

		change_local_variable = { name = sul_crown_power add = local_var:sul_estate_power }

		set_local_variable = {
			name = sul_wealth_target
			value = {
				value = local_var:sul_estate_power
				divide = { value = owner.var:sul_national_power_prev_peasants min = 0.001 }
				multiply = owner.var:sul_estate_gold_peasants
			}
		}
		set_local_variable = {
			name = sul_val
			value = {
				value = local_var:sul_prev_peasants

				add = {
					value = local_var:sul_wealth_target
					subtract = local_var:sul_prev_peasants
					multiply = @sul_wealth_converge_rate
				}

				min = 0
			}
		}

		set_variable = { name = sul_asset_peasants value = local_var:sul_val }
		set_variable = { name = sul_potential_peasants value = local_var:sul_wealth_target }

		# Dhimmi (conditional)
		if = {
			limit = { has_variable = sul_asset_nobles has_variable = sul_asset_dhimmi }

			set_local_variable = { name = sul_estate_power value = "local_estate_power(estate_type:dhimmi_estate)" }

			change_local_variable = { name = sul_crown_power add = local_var:sul_estate_power }

			set_local_variable = {
				name = sul_wealth_target
				value = {
					value = local_var:sul_estate_power
					divide = { value = owner.var:sul_national_power_prev_dhimmi min = 0.001 }
					multiply = owner.var:sul_estate_gold_dhimmi
				}
			}
			set_local_variable = {
				name = sul_val
				value = {
					value = local_var:sul_prev_dhimmi

					add = {
						value = local_var:sul_wealth_target
						subtract = local_var:sul_prev_dhimmi
						multiply = @sul_wealth_converge_rate
					}

					min = 0
				}
			}

			set_variable = { name = sul_asset_dhimmi value = local_var:sul_val }
			set_variable = { name = sul_potential_dhimmi value = local_var:sul_wealth_target }
		}

		# Cossacks (conditional)
		if = {
			limit = { has_variable = sul_asset_nobles has_variable = sul_asset_cossacks }

			set_local_variable = { name = sul_estate_power value = "local_estate_power(estate_type:cossacks_estate)" }

			change_local_variable = { name = sul_crown_power add = local_var:sul_estate_power }

			set_local_variable = {
				name = sul_wealth_target
				value = {
					value = local_var:sul_estate_power
					divide = { value = owner.var:sul_national_power_prev_cossacks min = 0.001 }
					multiply = owner.var:sul_estate_gold_cossacks
				}

					value = local_var:sul_prev_cossacks

					multiply = @sul_wealth_converge_rate
				}

				min = 0

		}

		set_variable = { name = sul_potential_crown value = local_var:sul_wealth_target }
	}

	# Total clamp: if sum(assets) > sum(potential), scale all estates proportionally
	set_local_variable = {
		name = sul_total_assets
		value = {
			value = var:sul_asset_nobles
			add = var:sul_asset_clergy
			add = var:sul_asset_burghers

		change_variable = { name = sul_asset_clergy multiply = local_var:sul_clamp_ratio }
		change_variable = { name = sul_asset_burghers multiply = local_var:sul_clamp_ratio }
		change_variable = { name = sul_asset_peasants multiply = local_var:sul_clamp_ratio }
		change_variable = { name = sul_asset_crown multiply = local_var:sul_clamp_ratio }
	}

	set_local_variable = {
		name = sul_delta
		value = { value = var:sul_asset_nobles subtract = local_var:sul_prev_nobles }

	set_local_variable = {
		name = sul_delta
		value = { value = var:sul_asset_clergy subtract = local_var:sul_prev_clergy }
	}

	set_variable = { name = sul_delta_clergy value = local_var:sul_delta }

	set_local_variable = {
		name = sul_delta
		value = { value = var:sul_asset_burghers subtract = local_var:sul_prev_burghers }
	}

	set_variable = { name = sul_delta_burghers value = local_var:sul_delta }

	set_local_variable = {
		name = sul_delta
		value = { value = var:sul_asset_peasants subtract = local_var:sul_prev_peasants }
	}

	set_variable = { name = sul_delta_peasants value = local_var:sul_delta }

	set_local_variable = { name = sul_delta value = { value = var:sul_asset_crown subtract = local_var:sul_prev_crown } }

	set_variable = { name = sul_delta_crown value = local_var:sul_delta }

	if = {
		limit = { has_variable = sul_asset_dhimmi }

		set_local_variable = {
			name = sul_delta
			value = { value = var:sul_asset_dhimmi subtract = local_var:sul_prev_dhimmi }

		set_variable = { name = sul_delta_dhimmi value = local_var:sul_delta }
	}

	if = {
		limit = { has_variable = sul_asset_cossacks }

		set_local_variable = {
			name = sul_delta
			value = { value = var:sul_asset_cossacks subtract = local_var:sul_prev_cossacks }

	if = {
		limit = { has_variable = sul_asset_tribes }

		}

	}

# -----------------------------------------------------------------------------
# WEALTH LOOT (location scope, called on occupation)
# -----------------------------------------------------------------------------
# Extracts 25% of each estate's assets. Controller keeps loot × (1 - destroyed + efficiency).

@sul_loot_extraction_rate = 0.25
@sul_loot_destroyed_base = 0.50

sul_wealth_loot_location = {
	set_local_variable = { name = sul_loot_total value = 0 }

	# Nobles
	set_local_variable = { name = sul_loot value = { value = var:sul_asset_nobles multiply = @sul_loot_extraction_rate } }
	set_local_variable = { name = sul_val value = { value = var:sul_asset_nobles subtract = local_var:sul_loot } }

	set_variable = { name = sul_asset_nobles value = local_var:sul_val }

	change_local_variable = { name = sul_loot_total add = local_var:sul_loot }

	# Clergy
	set_local_variable = { name = sul_loot value = { value = var:sul_asset_clergy multiply = @sul_loot_extraction_rate } }
	set_local_variable = { name = sul_val value = { value = var:sul_asset_clergy subtract = local_var:sul_loot } }

	set_variable = { name = sul_asset_clergy value = local_var:sul_val }

	change_local_variable = { name = sul_loot_total add = local_var:sul_loot }

	# Burghers
	set_local_variable = {
		name = sul_loot
		value = { value = var:sul_asset_burghers multiply = @sul_loot_extraction_rate }
	}
	set_local_variable = { name = sul_val value = { value = var:sul_asset_burghers subtract = local_var:sul_loot } }

	set_variable = { name = sul_asset_burghers value = local_var:sul_val }

	change_local_variable = { name = sul_loot_total add = local_var:sul_loot }

	# Peasants
	set_local_variable = {
		name = sul_loot
		value = { value = var:sul_asset_peasants multiply = @sul_loot_extraction_rate }
	}
	set_local_variable = { name = sul_val value = { value = var:sul_asset_peasants subtract = local_var:sul_loot } }

	set_variable = { name = sul_asset_peasants value = local_var:sul_val }

	change_local_variable = { name = sul_loot_total add = local_var:sul_loot }

	# Dhimmi (conditional)
	if = {
		limit = { has_variable = sul_asset_dhimmi }

		set_local_variable = { name = sul_loot value = { value = var:sul_asset_dhimmi multiply = @sul_loot_extraction_rate } }
		set_local_variable = { name = sul_val value = { value = var:sul_asset_dhimmi subtract = local_var:sul_loot } }

		set_variable = { name = sul_asset_dhimmi value = local_var:sul_val }

		change_local_variable = { name = sul_loot_total add = local_var:sul_loot }
	}

	# Cossacks (conditional)
	if = {
		limit = { has_variable = sul_asset_cossacks }

		set_local_variable = {
			name = sul_loot
			value = { value = var:sul_asset_cossacks multiply = @sul_loot_extraction_rate }
		}
		set_local_variable = { name = sul_val value = { value = var:sul_asset_cossacks subtract = local_var:sul_loot } }

		set_variable = { name = sul_asset_cossacks value = local_var:sul_val }

		change_local_variable = { name = sul_loot_total add = local_var:sul_loot }
	}

	# Tribes (conditional)
	if = {
		limit = { has_variable = sul_asset_tribes }

		set_local_variable = { name = sul_loot value = { value = var:sul_asset_tribes multiply = @sul_loot_extraction_rate } }
		set_local_variable = { name = sul_val value = { value = var:sul_asset_tribes subtract = local_var:sul_loot } }

		set_variable = { name = sul_asset_tribes value = local_var:sul_val }

		change_local_variable = { name = sul_loot_total add = local_var:sul_loot }
	}

	# Crown
	set_local_variable = { name = sul_loot value = { value = var:sul_asset_crown multiply = @sul_loot_extraction_rate } }
	set_local_variable = { name = sul_val value = { value = var:sul_asset_crown subtract = local_var:sul_loot } }

	set_variable = { name = sul_asset_crown value = local_var:sul_val }

	change_local_variable = { name = sul_loot_total add = local_var:sul_loot }

	# Crown income = loot × (1 - destroyed_base + looting_efficiency)
	controller ?= {
		set_local_variable = {
			name = sul_loot_kept
			value = {
				value = 1
				subtract = @sul_loot_destroyed_base
				add = modifier:sul_looting_efficiency
				min = 0
				max = 1
			}
		}
		add_gold = { value = local_var:sul_loot_total multiply = local_var:sul_loot_kept }
	}

	change_variable = { name = sul_wealth_loot_amount add = local_var:sul_loot_total }
}

	set_local_variable = { name = sul_clergy_wages value = 0 }
	set_local_variable = { name = sul_burgher_wages value = 0 }
	set_local_variable = { name = sul_commoner_wages value = 0 }

	set_variable = { name = sul_national_power_prev_nobles value = var:sul_national_power_nobles }
	set_variable = { name = sul_national_power_prev_clergy value = var:sul_national_power_clergy }
	set_variable = { name = sul_national_power_prev_burghers value = var:sul_national_power_burghers }
	set_variable = { name = sul_national_power_prev_peasants value = var:sul_national_power_peasants }
	set_variable = { name = sul_national_power_prev_dhimmi value = var:sul_national_power_dhimmi }
	set_variable = { name = sul_national_power_prev_cossacks value = var:sul_national_power_cossacks }
	set_variable = { name = sul_national_power_prev_tribes value = var:sul_national_power_tribes }
	set_variable = { name = sul_national_power_prev_crown value = var:sul_national_power_crown }
	set_variable = { name = sul_national_power_nobles value = 0 }
	set_variable = { name = sul_national_power_clergy value = 0 }
	set_variable = { name = sul_national_power_burghers value = 0 }
	set_variable = { name = sul_national_power_peasants value = 0 }
	set_variable = { name = sul_national_power_dhimmi value = 0 }
	set_variable = { name = sul_national_power_cossacks value = 0 }
	set_variable = { name = sul_national_power_tribes value = 0 }

	sul_compute_budget_pressure = yes

	# Cache estate gold for convergence targets
	set_variable = { name = sul_estate_gold_nobles value = "estate(estate_type:nobles_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_clergy value = "estate(estate_type:clergy_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_burghers value = "estate(estate_type:burghers_estate).estate_gold" }
	set_variable = { name = sul_estate_gold_peasants value = 0 }

	"estate(estate_type:peasants_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_peasants value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_dhimmi value = 0 }

	"estate(estate_type:dhimmi_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_dhimmi value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_cossacks value = 0 }

	"estate(estate_type:cossacks_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_cossacks value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_tribes value = 0 }

	"estate(estate_type:tribes_estate)" ?= {
		owner = { set_variable = { name = sul_estate_gold_tribes value = prev.estate_gold } }
	}

	set_variable = { name = sul_estate_gold_crown value = gold }

	every_owned_location = {
		sul_converge_location_assets = yes
		owner ?= {
			# Wealth: accumulate raw estate power
			change_variable = { name = sul_national_power_nobles add = "prev.local_estate_power(estate_type:nobles_estate)" }
			change_variable = { name = sul_national_power_clergy add = "prev.local_estate_power(estate_type:clergy_estate)" }
			change_variable = { name = sul_national_power_burghers add = "prev.local_estate_power(estate_type:burghers_estate)" }
			change_variable = { name = sul_national_power_peasants add = "prev.local_estate_power(estate_type:peasants_estate)" }

			if = {
				limit = { country_has_estate = estate_type:dhimmi_estate }
				change_variable = { name = sul_national_power_dhimmi add = "prev.local_estate_power(estate_type:dhimmi_estate)" }
			}

			if = {
				limit = { country_has_estate = estate_type:cossacks_estate }
				change_variable = { name = sul_national_power_cossacks add = "prev.local_estate_power(estate_type:cossacks_estate)" }
			}

			if = {
				limit = { country_has_estate = estate_type:tribes_estate }
				change_variable = { name = sul_national_power_tribes add = "prev.local_estate_power(estate_type:tribes_estate)" }
			}

			# Wages: accumulate cached wage bills
			if = {
				limit = { prev = { has_variable = sul_wage_bill_nobles var:sul_local_wages > 0 } }

				change_local_variable = { name = sul_noble_wages add = prev.var:sul_wage_bill_nobles }
				change_local_variable = { name = sul_clergy_wages add = prev.var:sul_wage_bill_clergy }
				change_local_variable = { name = sul_burgher_wages add = prev.var:sul_wage_bill_burghers }
				change_local_variable = {
					name = sul_commoner_wages
					add = {
						value = prev.var:sul_wage_bill_soldiers
						add = prev.var:sul_wage_bill_laborers
						add = prev.var:sul_wage_bill_peasants
					}
				}
			}
		}
	}

	# Wealth: derive crown from estate totals
	set_variable = { name = sul_national_power_crown value = var:sul_national_power_nobles }

	change_variable = { name = sul_national_power_crown add = var:sul_national_power_clergy }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_burghers }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_peasants }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_dhimmi }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_cossacks }
	change_variable = { name = sul_national_power_crown add = var:sul_national_power_tribes }

	# Wages: finalize monthly totals
	sul_compute_budget_pressure = yes

	set_variable = { name = sul_monthly_wages_nobles value = local_var:sul_noble_wages }
	set_variable = { name = sul_monthly_wages_clergy value = local_var:sul_clergy_wages }
	set_variable = { name = sul_monthly_wages_burghers value = local_var:sul_burgher_wages }
	set_variable = { name = sul_monthly_wages_peasants value = local_var:sul_commoner_wages }
	set_variable = { name = sul_monthly_wages_dhimmi value = 0 }
	set_variable = { name = sul_monthly_wages_cossacks value = 0 }
	set_variable = { name = sul_monthly_wages_tribes value = 0 }
}
