# WPP pipeline: tax base distributed by estate power + direct wage rates.
#
# Tax base income is split among pops via estate power weights and
# enfranchisement (unchanged from original system). Wages go directly
# to pops based on their group wage rate (set yearly by sul_yearly_wage_recalc).
#
# WPP = tax_base_share + wage_rate
#
# Stored per location:
#   sul_wage_bill_*       per-pop-type wage bills (6 types)
#   sul_local_wages       total wage bill (sum of above)
#   sul_total_weighted    tax base distribution denominator
#   sul_local_gdp         tooltip + GUI visibility
#   sul_enfranchisement   noble/commoner tax base split
#   sul_wealth_per_pop    variable map read by pop_demands

# Estate power weights
@sul_power_weight_nobles = 100
@sul_power_weight_clergy = 25
@sul_power_weight_burghers = 20

# Enfranchisement bounds
@sul_enfranchisement_min = 0.1
@sul_enfranchisement_max = 1.0

# Demand model denominator: A + B × WPP (must match sul_pop_demand_wpp.txt)
@sul_denom_A = 1.0
@sul_denom_B = 1.0

# -----------------------------------------------------------------------------
# SINGLE LOCATION UPDATE (location scope)
# -----------------------------------------------------------------------------

sul_update_location_wpp = {
	# 1. Per-pop-type wage bills (pop × rate × market_access)
	set_variable = {
		name = sul_wage_bill_nobles
		value = { value = num_pop_type:nobles multiply = var:sul_wages_nobles multiply = market_access }
	}
	set_variable = {
		name = sul_wage_bill_clergy
		value = { value = num_pop_type:clergy multiply = var:sul_wages_clergy multiply = market_access }
	}
	set_variable = {
		name = sul_wage_bill_burghers
		value = { value = num_pop_type:burghers multiply = var:sul_wages_burghers multiply = market_access }
	}
	set_variable = {
		name = sul_wage_bill_soldiers
		value = { value = num_pop_type:soldiers multiply = var:sul_wages_soldiers multiply = market_access }
	}
	set_variable = {
		name = sul_wage_bill_laborers
		value = { value = num_pop_type:laborers multiply = var:sul_wages_laborers multiply = market_access }
	}
	set_variable = {
		name = sul_wage_bill_peasants
		value = { value = num_pop_type:peasants multiply = var:sul_wages_peasants multiply = market_access }
	}

	# 2. Total local wages
	set_variable = {
		name = sul_local_wages

		value = {
			value = var:sul_wage_bill_nobles
			add = var:sul_wage_bill_clergy
			add = var:sul_wage_bill_burghers
			add = var:sul_wage_bill_soldiers
			add = var:sul_wage_bill_laborers
			add = var:sul_wage_bill_peasants
		}
	}

	# 3. Total weighted estate power (for tax base distribution)
	set_variable = {
		name = sul_total_weighted

		value = {
			value = { value = num_pop_type:nobles multiply = @sul_power_weight_nobles }
			add = { value = num_pop_type:clergy multiply = @sul_power_weight_clergy }
			add = { value = num_pop_type:burghers multiply = @sul_power_weight_burghers }
			add = num_pop_type:peasants
			add = num_pop_type:laborers
			add = num_pop_type:soldiers
			add = num_pop_type:slaves
			add = { value = num_pop_type:tribesmen multiply = 0.01 }
			min = 0.001
		}
	}

	# 4. GDP (tax base + wages; zero-control fallback)
	save_scope_as = cur_loc

	if = {
		limit = { local_control > 0 }

		set_variable = {
			name = sul_local_gdp

			value = {
				value = scope:cur_loc.location_tax_base
				divide = local_control
				add = var:sul_local_wages
			}
		}

		# Tax base income (GDP minus wages) for estate distribution
		set_local_variable = {
			name = sul_tax_base_income
			value = { value = var:sul_local_gdp subtract = var:sul_local_wages }
		}
	}
	else = {
		set_variable = { name = sul_local_gdp value = 0 }
		set_local_variable = { name = sul_tax_base_income value = 0 }
	}

	# 5. Enfranchisement (controls noble/commoner tax base split)
	set_variable = {
		name = sul_enfranchisement

		value = {
			value = modifier:local_peasant_enfranchisment
			add = owner.modifier:global_peasant_enfranchisment
			min = @sul_enfranchisement_min
			max = @sul_enfranchisement_max
		}
	}

	# 6. Tax base wealth unit = tax_base_income / total_weighted
	set_local_variable = {
		name = sul_tax_wealth_base
		value = { value = local_var:sul_tax_base_income divide = var:sul_total_weighted }
	}

	# 7. WPP = tax base share (power-weighted) + direct wage rate
	set_local_variable = {
		name = sul_noble_wpp

		value = {
			value = local_var:sul_tax_wealth_base
			multiply = @sul_power_weight_nobles

			add = {
				value = local_var:sul_tax_wealth_base
				multiply = { value = 1 subtract = var:sul_enfranchisement }
				multiply = {
					value = num_pop_type:peasants
					add = num_pop_type:laborers
					add = num_pop_type:soldiers
					add = num_pop_type:slaves
				}

				divide = { value = num_pop_type:nobles min = 0.001 }
			}

			add = var:sul_wages_nobles
		}
	}
	set_local_variable = {
		name = sul_clergy_wpp

		value = {
			value = local_var:sul_tax_wealth_base
			multiply = @sul_power_weight_clergy
			add = var:sul_wages_clergy
		}
	}
	set_local_variable = {
		name = sul_burgher_wpp

		value = {
			value = local_var:sul_tax_wealth_base
			multiply = @sul_power_weight_burghers
			add = var:sul_wages_burghers
		}
	}

	# Commoner tax base share, then per-type WPP with individual wage rates
	set_local_variable = {
		name = sul_commoner_tax_share

		value = {
			value = local_var:sul_tax_wealth_base
			multiply = var:sul_enfranchisement
		}
	}
	set_local_variable = {
		name = sul_soldier_wpp
		value = { value = local_var:sul_commoner_tax_share add = var:sul_wages_soldiers }
	}
	set_local_variable = {
		name = sul_laborer_wpp
		value = { value = local_var:sul_commoner_tax_share add = var:sul_wages_laborers }
	}
	set_local_variable = {
		name = sul_peasant_wpp
		value = { value = local_var:sul_commoner_tax_share add = var:sul_wages_peasants }
	}

	# 8. Write WPP + demand base to variable maps
	add_to_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:nobles
		value = local_var:sul_noble_wpp
	}
	add_to_variable_map = {
		name = sul_demand_base
		key = pop_type:nobles
		value = { value = local_var:sul_noble_wpp divide = { value = @sul_denom_A add = { value = local_var:sul_noble_wpp multiply = @sul_denom_B } } }
	}
	add_to_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:clergy
		value = local_var:sul_clergy_wpp
	}
	add_to_variable_map = {
		name = sul_demand_base
		key = pop_type:clergy
		value = { value = local_var:sul_clergy_wpp divide = { value = @sul_denom_A add = { value = local_var:sul_clergy_wpp multiply = @sul_denom_B } } }
	}
	add_to_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:burghers
		value = local_var:sul_burgher_wpp
	}
	add_to_variable_map = {
		name = sul_demand_base
		key = pop_type:burghers
		value = { value = local_var:sul_burgher_wpp divide = { value = @sul_denom_A add = { value = local_var:sul_burgher_wpp multiply = @sul_denom_B } } }
	}
	add_to_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:soldiers
		value = local_var:sul_soldier_wpp
	}
	add_to_variable_map = {
		name = sul_demand_base
		key = pop_type:soldiers
		value = { value = local_var:sul_soldier_wpp divide = { value = @sul_denom_A add = { value = local_var:sul_soldier_wpp multiply = @sul_denom_B } } }
	}
	add_to_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:laborers
		value = local_var:sul_laborer_wpp
	}
	add_to_variable_map = {
		name = sul_demand_base
		key = pop_type:laborers
		value = { value = local_var:sul_laborer_wpp divide = { value = @sul_denom_A add = { value = local_var:sul_laborer_wpp multiply = @sul_denom_B } } }
	}
	add_to_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:peasants
		value = local_var:sul_peasant_wpp
	}
	add_to_variable_map = {
		name = sul_demand_base
		key = pop_type:peasants
		value = { value = local_var:sul_peasant_wpp divide = { value = @sul_denom_A add = { value = local_var:sul_peasant_wpp multiply = @sul_denom_B } } }
	}
}

# -----------------------------------------------------------------------------
# PER-LOCATION WAGE RATE CALCULATION (location scope)
# -----------------------------------------------------------------------------
# Called yearly (sul_yearly_wage_recalc) and at game start.
# wage = base × (1 + global_mod + local_mod + pop_cap + enfranchisement)
#
# Pop capacity: (0.5 - pop%) × 2 × scale
#   Scale: commoners ×0.5 (±50%), burghers ×0.25 (±25%), clergy ×0.1 (±10%), nobles 0
#
# Enfranchisement: (enfr - 0.5) × 2 × 0.5 = ±50% for commoners only

@sul_pop_cap_midpoint = 0.5
@sul_enfr_midpoint = 0.5

sul_calculate_location_wages = {
	set_local_variable = {
		name = sul_pop_cap_factor
		value = {
			value = @sul_pop_cap_midpoint
			subtract = location_population_percentage
			multiply = 2
		}
	}

	set_local_variable = {
		name = sul_enfr_factor
		value = {
			value = modifier:local_peasant_enfranchisment
			add = owner.modifier:global_peasant_enfranchisment
			min = 0.1
			max = 1.0
			subtract = @sul_enfr_midpoint
			multiply = 2
			multiply = 0.5
		}
	}

	set_variable = {
		name = sul_wages_nobles
		value = {
			value = owner.modifier:sul_nobles_base_wages
			multiply = { value = 1 add = owner.modifier:sul_global_nobles_wages_modifier add = modifier:sul_local_nobles_wages_modifier }
			min = 0
		}
	}
	set_variable = {
		name = sul_wages_clergy
		value = {
			value = owner.modifier:sul_clergy_base_wages
			multiply = {
				value = 1
				add = owner.modifier:sul_global_clergy_wages_modifier
				add = modifier:sul_local_clergy_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.1 }
			}
			min = 0
		}
	}
	set_variable = {
		name = sul_wages_burghers
		value = {
			value = owner.modifier:sul_burghers_base_wages
			multiply = {
				value = 1
				add = owner.modifier:sul_global_burghers_wages_modifier
				add = modifier:sul_local_burghers_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.25 }
			}
			min = 0
		}
	}
	set_variable = {
		name = sul_wages_soldiers
		value = {
			value = owner.modifier:sul_soldiers_base_wages
			multiply = {
				value = 1
				add = owner.modifier:sul_global_soldiers_wages_modifier
				add = modifier:sul_local_soldiers_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.5 }
				add = local_var:sul_enfr_factor
			}
			min = 0
		}
	}
	set_variable = {
		name = sul_wages_laborers
		value = {
			value = owner.modifier:sul_laborers_base_wages
			multiply = {
				value = 1
				add = owner.modifier:sul_global_laborers_wages_modifier
				add = modifier:sul_local_laborers_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.5 }
				add = local_var:sul_enfr_factor
			}
			min = 0
		}
	}
	set_variable = {
		name = sul_wages_peasants
		value = {
			value = owner.modifier:sul_peasants_base_wages
			multiply = {
				value = 1
				add = owner.modifier:sul_global_peasants_wages_modifier
				add = modifier:sul_local_peasants_wages_modifier
				add = { value = local_var:sul_pop_cap_factor multiply = 0.5 }
				add = local_var:sul_enfr_factor
			}
			min = 0
		}
	}
}

# Init non-participating pop types to zero WPP

sul_init_zero_wpp = {
	add_to_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:slaves
		value = 0
	}
	add_to_variable_map = {
		name = sul_wealth_per_pop
		key = pop_type:tribesmen
		value = 0
	}
	add_to_variable_map = {
		name = sul_demand_base
		key = pop_type:slaves
		value = 0
	}
	add_to_variable_map = {
		name = sul_demand_base
		key = pop_type:tribesmen
		value = 0
	}
}

# -----------------------------------------------------------------------------
# BUDGET PRESSURE (country scope)
# -----------------------------------------------------------------------------
# Per-estate budget pressure from gold reserves.
# Every 1000 gold after the first 1000 adds +25% demand.
# Formula: max(1, 1 + (estate_gold - 1000) × 0.00025)

sul_compute_budget_pressure = {
	"estate(estate_type:nobles_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_nobles

				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	"estate(estate_type:clergy_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_clergy

				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	"estate(estate_type:burghers_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_burghers

				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}

	"estate(estate_type:peasants_estate)" ?= {
		owner = {
			set_variable = {
				name = sul_budget_pressure_peasants

				value = {
					value = prev.estate_gold
					subtract = 1000
					multiply = 0.00025
					add = 1
					min = 1
				}
			}
		}
	}
}

# -----------------------------------------------------------------------------
# COUNTRY ECONOMY UPDATE (country scope)
# -----------------------------------------------------------------------------
# Full pass: updates all locations + accumulates estate wages.
# Called monthly for player, on init for everyone.

sul_update_country_economy = {
	set_local_variable = { name = sul_noble_wages value = 0 }
	set_local_variable = { name = sul_clergy_wages value = 0 }
	set_local_variable = { name = sul_burgher_wages value = 0 }
	set_local_variable = { name = sul_commoner_wages value = 0 }

	every_owned_location = {
		if = {
			limit = { market_access > 0 }
			sul_update_location_wpp = yes

			if = {
				limit = { var:sul_local_wages > 0 }

				owner ?= {
					change_local_variable = { name = sul_noble_wages add = prev.var:sul_wage_bill_nobles }
					change_local_variable = { name = sul_clergy_wages add = prev.var:sul_wage_bill_clergy }
					change_local_variable = { name = sul_burgher_wages add = prev.var:sul_wage_bill_burghers }
					change_local_variable = {
						name = sul_commoner_wages

						add = {
							value = prev.var:sul_wage_bill_soldiers
							add = prev.var:sul_wage_bill_laborers
							add = prev.var:sul_wage_bill_peasants
						}
					}
				}
			}
		}
	}

	sul_compute_budget_pressure = yes

	set_variable = {
		name = sul_monthly_wages_nobles
		value = local_var:sul_noble_wages
	}
	set_variable = {
		name = sul_monthly_wages_clergy
		value = local_var:sul_clergy_wages
	}
	set_variable = {
		name = sul_monthly_wages_burghers
		value = local_var:sul_burgher_wages
	}
	set_variable = {
		name = sul_monthly_wages_peasants
		value = local_var:sul_commoner_wages
	}

	set_variable = { name = sul_monthly_wages_dhimmi value = 0 }
	set_variable = { name = sul_monthly_wages_cossacks value = 0 }
	set_variable = { name = sul_monthly_wages_tribes value = 0 }

	sul_estate_wages_pay_all = yes
}

# -----------------------------------------------------------------------------
# WAGE ACCUMULATION FROM CACHED VALUES (country scope)
# -----------------------------------------------------------------------------
# Reads stored per-pop-type wage bills without recomputing WPP.
# Player: monthly. AI: yearly (January).

sul_accumulate_wages = {
	set_local_variable = { name = sul_noble_wages value = 0 }
	set_local_variable = { name = sul_clergy_wages value = 0 }
	set_local_variable = { name = sul_burgher_wages value = 0 }
	set_local_variable = { name = sul_commoner_wages value = 0 }

	every_owned_location = {
		limit = { has_variable = sul_wage_bill_nobles var:sul_local_wages > 0 }

		owner ?= {
			change_local_variable = { name = sul_noble_wages add = prev.var:sul_wage_bill_nobles }
			change_local_variable = { name = sul_clergy_wages add = prev.var:sul_wage_bill_clergy }
			change_local_variable = { name = sul_burgher_wages add = prev.var:sul_wage_bill_burghers }
			change_local_variable = {
				name = sul_commoner_wages

				add = {
					value = prev.var:sul_wage_bill_soldiers
					add = prev.var:sul_wage_bill_laborers
					add = prev.var:sul_wage_bill_peasants
				}
			}
		}
	}

	sul_compute_budget_pressure = yes

	set_variable = {
		name = sul_monthly_wages_nobles
		value = local_var:sul_noble_wages
	}
	set_variable = {
		name = sul_monthly_wages_clergy
		value = local_var:sul_clergy_wages
	}
	set_variable = {
		name = sul_monthly_wages_burghers
		value = local_var:sul_burgher_wages
	}
	set_variable = {
		name = sul_monthly_wages_peasants
		value = local_var:sul_commoner_wages
	}

	set_variable = { name = sul_monthly_wages_dhimmi value = 0 }
	set_variable = { name = sul_monthly_wages_cossacks value = 0 }
	set_variable = { name = sul_monthly_wages_tribes value = 0 }
}
