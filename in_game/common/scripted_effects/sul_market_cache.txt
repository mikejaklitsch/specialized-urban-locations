# Market Cache - Shared price cache
# Populated each month, read by:
#   - Passthrough pulse (sul_market_prices variable_map per market)
#   - Minting price cache (reads sul_market_prices)
#   - Commercial spec scoring (sul_trade_value_per_capita)
# Composites (finished goods demand/supply) computed inline by attractiveness
# script_values — no longer cached monthly.

# Per-market cache write. root = market.

sul_market_cache_update = {
	add_to_global_variable_map = {
		name = sul_trade_value_per_capita
		key = this
		value = sul_trade_value_per_capita
	}
}

# Full world pass. Used for game-start cache warmup.
# Monthly refresh now handled by sul_passthrough_apply (merged loop).

sul_market_cache_refresh_all = {
	if = {
		limit = {
			NOT = {
				has_global_variable = sul_market_demand_cached
			}
		}

		every_market_in_world = {
			save_temporary_scope_as = current_market
			sul_market_cache_update = yes

			location = { clear_variable_map = sul_market_prices }

			every_goods = {
				save_temporary_scope_as = current_good

				scope:current_market.location = {
					add_to_variable_map = {
						name = sul_market_prices
						key = scope:current_good
						value = "scope:current_good.price_in_market(scope:current_market)"
					}
				}
			}
		}

		set_global_variable = { name = sul_market_demand_cached value = 1 }
	}
}
