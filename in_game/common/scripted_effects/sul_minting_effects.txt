# The Gold Standard - Scripted Effects
# Scope: country

# ---- Parameterized Helpers ----

# Initialize a variable if not already set.

sul_minting_init_variable = {
	if = {
		limit = {
			NOT = {
				has_variable = $name$
			}
		}

		set_variable = { name = $name$ value = $value$ }
	}
}

# Check a minting good modifier and add to the goods list if active.

sul_minting_check_minting_good = {
	if = {
		limit = { modifier:$good$_used_for_minting = yes }

		add_to_variable_list = { name = sul_minting_goods target = goods:$good$ }
	}
}

# Adjust a currency level (debasement or rebasement) and reset the other.

sul_minting_adjust_currency = {
	change_variable = { name = sul_minting_$type$_level add = $delta$ }

	clamp_variable = { name = sul_minting_$type$_level min = 0 max = 2.0 }

	set_variable = { name = sul_minting_$other$_level value = 0 }
}

# ---- Per-Country Initialization ----
# Sets default values for all TGS variables.

sul_minting_initialize_country = {
	sul_minting_init_variable = { name = sul_minting_debasement_level value = 0 }
	sul_minting_init_variable = { name = sul_minting_rebasement_level value = 0 }
	sul_minting_init_variable = { name = sul_minting_market_cost value = 0 }
	sul_minting_init_variable = { name = sul_minting_efficiency value = 0 }

	sul_minting_init_variable = {
		name = sul_minting_real_minting_efficiency
		value = 0
	}

	sul_minting_rebuild_minting_goods = yes
	sul_minting_cache_market_center = yes
}

# Resolves and caches the market center location reference on the country
# (during init and when market changes). Avoids repeated
# random_location_in_market searches at runtime.

sul_minting_cache_market_center = {
	capital ?= {
		market ?= { location = { save_scope_as = sul_minting_mc_ref } }
	}

	set_variable = {
		name = sul_minting_market_center
		value = scope:sul_minting_mc_ref
	}
}

# ---- Price Cache Update ----
# Sums the market price of one unit of each good in sul_minting_goods via
# price_in_market(), then recomputes E and real E from the result.
# Self-contained: no dependency on the shared sul_market_prices cache.
# Scope: country (requires capital with market)

sul_minting_update_price_cache = {
	var:sul_minting_market_center = { market ?= { save_temporary_scope_as = sul_minting_mkt } }

	set_local_variable = { name = sul_minting_tmp_mc value = 0 }

	every_in_list = {
		variable = sul_minting_goods

		change_local_variable = {
			name = sul_minting_tmp_mc
			add = "price_in_market(scope:sul_minting_mkt)"
		}
	}

	set_variable = {
		name = sul_minting_market_cost
		value = local_var:sul_minting_tmp_mc
	}

	set_variable = {
		name = sul_minting_efficiency
		value = sul_minting_efficiency_from_cached_cost
	}
	set_variable = {
		name = sul_minting_real_minting_efficiency
		value = sul_minting_real_efficiency_from_cached_cost
	}
}
