# Passthrough Pulse - Global warehouse stockpile pulse
# Implements the yosiu_market_stockpiles workshop mod's approach compressed
# via $GOOD$ argument substitution. Called each weather tick from scopeless.
#
# Branches:
#   - HIGH branch (stockpile > upper_value = 5% capacity):
#     Apply negative _stockpile temp demand to push price down.
#     Conditionally cancel the warehouse building's output via _oversupply.
#   - LOW branch (stockpile < lower_value = 4.9% capacity, good in use):
#     Apply negative _stockpile temp demand scaled inverse + direct
#     add_goods_supply injection to keep the market from starving.
#
# Iterates every_market_in_world once, skipping unowned markets.
#
# Effects defined here:
#   sul_passthrough_apply        - global pulse entry point
#                                  (called by on_action sul_passthrough_pulse)
#   sul_passthrough_per_good     - per-good HIGH/LOW branch handler

sul_passthrough_apply = {
	# Market cache guard: rebuild prices when month changes.
	if = {
		limit = {
			has_global_variable = sul_market_demand_cached
			has_global_variable = sul_cache_month
			NOT = { global_var:sul_cache_month = current_month }
		}
		remove_global_variable = sul_market_demand_cached
	}
	set_global_variable = { name = sul_cache_month value = current_month }
	set_local_variable = { name = sul_pt_rebuild_cache value = 0 }
	if = {
		limit = { NOT = { has_global_variable = sul_market_demand_cached } }
		set_local_variable = { name = sul_pt_rebuild_cache value = 1 }
	}

	every_market_in_world = {
		limit = { location = { exists = owner } }
		save_scope_as = current_market

		# ---- Market cache rebuild (when flagged) ----
		if = {
			limit = { local_var:sul_pt_rebuild_cache = 1 }
			sul_market_cache_update = yes

			location = { clear_variable_map = sul_market_prices }

			every_goods = {
				save_temporary_scope_as = current_good

				scope:current_market.location = {
					add_to_variable_map = {
						name = sul_market_prices
						key = scope:current_good
						value = "scope:current_good.price_in_market(scope:current_market)"
					}
				}
			}
		}

		# ---- Remove only previously-applied demands ----
		location = {
			if = {
				limit = { has_variable_list = sul_pt_active }
				every_in_list = {
					variable = sul_pt_active
					scope:current_market = {
						remove_temporary_demand = "global_variable_map(sul_pt_stockpile_demands|prev)"
						remove_temporary_demand = "global_variable_map(sul_pt_oversupply_demands|prev)"
					}
				}
				clear_variable_list = sul_pt_active
			}
		}

		# ---- Evaluate all goods, apply demands, track active ----
		set_local_variable = {
			name = sul_pt_cap
			value = location.sul_passthrough_local_capacity
		}

		every_goods = {
			limit = { is_key_in_global_variable_map = { name = sul_pt_stockpile_demands target = this } }
			save_scope_as = current_good

			scope:current_market = {
				set_local_variable = {
					name = sul_pt_fill
					value = {
						value = "stockpile_in_market(scope:current_good)"
						divide = local_var:sul_pt_cap
					}
				}

				if = {
					limit = { local_var:sul_pt_fill > 0.05 }

					add_temporary_demand = {
						type = "global_variable_map(sul_pt_stockpile_demands|scope:current_good)"
						scale = sul_passthrough_supply_effect
						months = 1
					}

					location = { add_to_variable_list = { name = sul_pt_active target = scope:current_good } }

					if = {
						limit = {
							location = { has_building = "global_variable_map(sul_pt_warehouses|scope:current_good)" }
							goods_supply_in_market = { goods = scope:current_good value < 1 }
						}

						add_temporary_demand = {
							type = "global_variable_map(sul_pt_oversupply_demands|scope:current_good)"
							scale = sul_passthrough_warehouse_correction
							months = 1
						}
					}
				}

				else_if = {
					limit = {
						local_var:sul_pt_fill < 0.049
						OR = {
							scope:current_good = { is_demanded_in_market_by_pops = scope:current_market }
							scope:current_good = { is_demanded_in_market_by_buildings = scope:current_market }
							scope:current_good = { is_demanded_in_market_by_constructions = scope:current_market }
							scope:current_good = goods:goods_gold
							scope:current_good = goods:silver
							scope:current_good = goods:copper
							scope:current_good = goods:lumber
							scope:current_good = goods:masonry
						}
					}

					add_temporary_demand = {
						type = "global_variable_map(sul_pt_stockpile_demands|scope:current_good)"
						scale = sul_passthrough_low_supply_effect
						months = 1
					}

					add_goods_supply = {
						goods = scope:current_good
						amount = sul_passthrough_low_stockpile_effect
					}

					location = { add_to_variable_list = { name = sul_pt_active target = scope:current_good } }
				}
			}
		}
	}

	# Stamp cache after loop completes
	if = {
		limit = { local_var:sul_pt_rebuild_cache = 1 }
		set_global_variable = { name = sul_market_demand_cached value = 1 }
	}
}
