# Per-good passthrough logic, compressed via $GOOD$ substitution.
# One body serves all 75 goods, called 75 times from sul_passthrough_apply.
# This is the one place where we can compress the workshop mod's per-good
# switch unrolling — everything else (templates, buildings) is data and
# must be declared per-good.
#
# Called from market scope with scope:current_market saved.

sul_passthrough_per_good = {
	goods:$GOOD$ = { save_scope_as = current_good }

	scope:current_market = {
		# HIGH branch: fill ratio is above 5%.
		# Apply negative _stockpile demand to push price down.
		# If the warehouse exists and supply is low, also apply the paired
		# _oversupply demand to cancel the warehouse's +0.03 output at the
		# flow layer (workshop mod's "warehouse trick").
		if = {
			limit = { sul_passthrough_fill_ratio > 0.05 }

			add_temporary_demand = {
				type = demand:sul_$GOOD$_stockpile
				scale = sul_passthrough_supply_effect
				months = 1
			}

			if = {
				limit = {
					location = { has_building = building_type:sul_$GOOD$_warehouse }
					goods_supply_in_market = { goods = scope:current_good value < 1 }
				}

				add_temporary_demand = {
					type = demand:sul_$GOOD$_oversupply
					scale = sul_passthrough_warehouse_correction
					months = 1
				}
			}
		}

		# LOW branch: fill ratio is below 4.9% AND this good is actually
		# relevant in this market (has meaningful demand or is a critical
		# resource). Apply negative _stockpile demand with inverse scale
		# to push price up, and inject real supply via add_goods_supply to
		# prevent starvation/shortage cascades.
		else_if = {
			limit = {
				sul_passthrough_fill_ratio < 0.049
				OR = {
					scope:current_good = { is_demanded_in_market_by_pops = scope:current_market }
					scope:current_good = {
						is_demanded_in_market_by_buildings = scope:current_market
					}
					scope:current_good = {
						is_demanded_in_market_by_constructions = scope:current_market
					}

					scope:current_good = goods:goods_gold
					scope:current_good = goods:silver
					scope:current_good = goods:copper
					scope:current_good = goods:lumber
					scope:current_good = goods:masonry
				}
			}

			add_temporary_demand = {
				type = demand:sul_$GOOD$_stockpile
				scale = sul_passthrough_low_supply_effect
				months = 1
			}

			add_goods_supply = {
				goods = scope:current_good
				amount = sul_passthrough_low_stockpile_effect
			}
		}
	}
