# Power Projection — scripted effects.
# Per-country monthly recompute for drag sources that need iteration,
# plus conquest-debt accumulation and decay.

# Sums every subject's country_strength weighted by its type constant, then
# divides by overlord country_strength and negates. Result: -1 per "multiple"
# of the overlord's strength in subjects. Auto_modifier coefficient of 25
# gives -25 PP per multiple. Linear, no cap.

sul_projection_recompute_subject_drag = {
	set_local_variable = { name = sul_projection_subject_load value = 0 }

	every_subject = {
		change_local_variable = {
			name = sul_projection_subject_load

			add = {
				value = country_strength
				multiply = sul_projection_subject_type_weight
			}
		}
	}

	if = {
		limit = { local_var:sul_projection_subject_load > 0 }

		set_local_variable = {
			name = sul_projection_overlord_strength

			value = { value = country_strength min = 1 }
		}

		set_variable = {
			name = sul_projection_subject_drag_value

			value = {
				value = local_var:sul_projection_subject_load
				divide = local_var:sul_projection_overlord_strength
				multiply = -1
			}
		}
	}
	else = {
		set_variable = { name = sul_projection_subject_drag_value value = 0 }
	}
}
