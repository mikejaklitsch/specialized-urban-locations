# Per-location cache write. Location scope.
# Stores sul_cached_respec_score and sul_cached_desired_spec.

sul_cache_respec_location = {
	set_variable = { name = sul_cached_respec_score value = sul_respec_score }

	# Tournament: find best non-current spec
	set_local_variable = { name = best value = -1000 }
	set_local_variable = { name = winner value = 0 }

	if = {
		limit = { NOT = { sul_has_mining_specialization = yes } }
		set_local_variable = { name = score value = sul_attractiveness_mining }
		if = {
			limit = { local_var:score > local_var:best }
			set_local_variable = { name = best value = local_var:score }
			set_local_variable = { name = winner value = 1 }
		}
	}
	if = {
		limit = { NOT = { sul_has_farming_specialization = yes } }
		set_local_variable = { name = score value = sul_attractiveness_farming }
		if = {
			limit = { local_var:score > local_var:best }
			set_local_variable = { name = best value = local_var:score }
			set_local_variable = { name = winner value = 2 }
		}
	}
	if = {
		limit = { NOT = { sul_has_woodland_specialization = yes } }
		set_local_variable = { name = score value = sul_attractiveness_woodland }
		if = {
			limit = { local_var:score > local_var:best }
			set_local_variable = { name = best value = local_var:score }
			set_local_variable = { name = winner value = 4 }
		}
	}
	if = {
		limit = { NOT = { sul_has_gathering_specialization = yes } }
		set_local_variable = { name = score value = sul_attractiveness_gathering }
		if = {
			limit = { local_var:score > local_var:best }
			set_local_variable = { name = best value = local_var:score }
			set_local_variable = { name = winner value = 3 }
		}
	}
	if = {
		limit = { NOT = { sul_has_commercial_specialization = yes } }
		set_local_variable = { name = score value = sul_attractiveness_commercial }
		if = {
			limit = { local_var:score > local_var:best }
			set_local_variable = { name = best value = local_var:score }
			set_local_variable = { name = winner value = 5 }
		}
	}

	if = {
		limit = { var:sul_cached_respec_score > 0 }
		set_variable = { name = sul_cached_desired_spec value = local_var:winner }
	}
	else = {
		set_variable = { name = sul_cached_desired_spec value = 0 }
	}

	set_variable = { name = sul_last_respec_cache value = current_year }
}
