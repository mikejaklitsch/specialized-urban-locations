sul_respec_to_mining = {
	if = {
		limit = { modifier:sul_is_rural_settlement_rank_tier = yes }
		construct_building = { building_type = building_type:sul_respec_mining_rural }
	}
	else_if = {
		limit = { modifier:sul_is_town_rank_tier = yes }
		construct_building = { building_type = building_type:sul_respec_mining_town }
	}
	else = {
		construct_building = { building_type = building_type:sul_respec_mining_city }
	}
}
