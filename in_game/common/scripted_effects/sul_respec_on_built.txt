# Shared on_built logic for specialization carrier buildings.
# Root = building. Parameters: $SPEC$ (mining/farming/gathering/woodland/commercial), $COUNT_VAR$, $WEIGHT$

sul_respec_on_built_effect = {
	save_temporary_scope_as = sul_carrier
	location = {
		# Count bookkeeping
		if = {
			limit = { exists = owner }
			owner = {
				if = {
					limit = { NOT = { has_variable = sul_mining_count } }
					sul_specialization_count_initialize = yes
				}
				prev = { save_temporary_scope_as = sul_loc }
				sul_specialization_count_decrement = yes
			}
		}

		# Destroy market if leaving commercial
		if = {
			limit = {
				is_market_center = yes
				NOT = { sul_has_$SPEC$_specialization = yes }
			}
			sul_migration_on_market_destroyed = yes
			market = { destroy_market = {} }
		}

		# Set rank (preserves tier)
		sul_rank_set_$SPEC$ = yes

		# Create market if becoming commercial
		if = {
			limit = {
				sul_has_commercial_specialization = yes
				NOT = { is_market_center = yes }
			}
			create_market = {
				builder = owner
				location = prev
				price = price:create_market
				price_modifier = 0
				instant = yes
			}
			sul_migration_on_market_created = yes
		}

		# Destroy non-matching buildings + apply respec penalty
		sul_specialization_destroy_and_penalize = yes

		# Count increment (after rank is set)
		if = {
			limit = { exists = owner }
			owner = {
				prev = { save_temporary_scope_as = sul_loc }
				change_variable = { name = $COUNT_VAR$ add = 1 }
				change_variable = { name = sul_total_weighted add = $WEIGHT$ }
			}
		}

		# Update GUI bridge
		sul_specialization_update_designation = yes
	}
}

# root = location. Queues the correct resource-based spec carrier.
sul_construct_resource_spec_carrier = {
	if = {
		limit = { sul_has_farming_resources = yes }
		if = {
			limit = { modifier:sul_is_rural_settlement_rank_tier = yes }
			construct_building = { building_type = building_type:sul_respec_farming_rural }
		}
		else_if = {
			limit = { modifier:sul_is_town_rank_tier = yes }
			construct_building = { building_type = building_type:sul_respec_farming_town }
		}
		else = {
			construct_building = { building_type = building_type:sul_respec_farming_city }
		}
	}
	else_if = {
		limit = { sul_has_mining_resources = yes }
		if = {
			limit = { modifier:sul_is_rural_settlement_rank_tier = yes }
			construct_building = { building_type = building_type:sul_respec_mining_rural }
		}
		else_if = {
			limit = { modifier:sul_is_town_rank_tier = yes }
			construct_building = { building_type = building_type:sul_respec_mining_town }
		}
		else = {
			construct_building = { building_type = building_type:sul_respec_mining_city }
		}
	}
	else_if = {
		limit = { sul_has_gathering_resources = yes }
		if = {
			limit = { modifier:sul_is_rural_settlement_rank_tier = yes }
			construct_building = { building_type = building_type:sul_respec_gathering_rural }
		}
		else_if = {
			limit = { modifier:sul_is_town_rank_tier = yes }
			construct_building = { building_type = building_type:sul_respec_gathering_town }
		}
		else = {
			construct_building = { building_type = building_type:sul_respec_gathering_city }
		}
	}
	else_if = {
		limit = { sul_has_woodland_resources = yes }
		if = {
			limit = { modifier:sul_is_rural_settlement_rank_tier = yes }
			construct_building = { building_type = building_type:sul_respec_woodland_rural }
		}
		else_if = {
			limit = { modifier:sul_is_town_rank_tier = yes }
			construct_building = { building_type = building_type:sul_respec_woodland_town }
		}
		else = {
			construct_building = { building_type = building_type:sul_respec_woodland_city }
		}
	}
}
