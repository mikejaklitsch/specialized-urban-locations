# Specialized Urban Locations - Building Distribution
#
# Redistributes building levels destroyed by
# sul_specialization_destroy_non_matching_buildings into surviving-spec
# buildings, partitioned by pop_type so pops keep employment in their
# original class (no cross-class promotion needed).
#
# root = location. Call order inside init_new / reassign:
#   1. sul_specialization_save_building_levels   (per-pop-type snapshot)
#   2. sul_specialization_destroy_non_matching_buildings
#   3. sul_distribute_specialization_buildings   (this file)
#
# Pop-type partitioned slot maps per spec. Each spec's sub-effect distributes
# each pop_type's budget across that spec's buildings of matching pop_type.
# Guild buildings (burghers/clergy) are town/city only. Rural/RGO slots are
# anywhere. Buildings are selected round-robin; per-building counter prevents
# exceeding cap. Triggers mirror each building's location_potential (minus
# the spec gate, which is already satisfied by the if-chain dispatch).

sul_distribute_specialization_buildings = {
	# Snapshot post-destroy levels per pop_type.
	sul_dist_snapshot_current = yes

	# Per-pop-type budget = prior - current (levels destroyed, by class).
	set_variable = {
		name = sul_laborer_budget

		value = {
			value = var:sul_prior_laborer_levels
			subtract = var:sul_cur_laborer_levels
			min = 0
		}
	}
	set_variable = {
		name = sul_burgher_budget

		value = {
			value = var:sul_prior_burgher_levels
			subtract = var:sul_cur_burgher_levels
			min = 0
		}
	}
	set_variable = {
		name = sul_peasant_budget

		value = {
			value = var:sul_prior_peasant_levels
			subtract = var:sul_cur_peasant_levels
			min = 0
		}
	}
	set_variable = {
		name = sul_clergy_budget

		value = {
			value = var:sul_prior_clergy_levels
			subtract = var:sul_cur_clergy_levels
			min = 0
		}
	}

	# Dispatch: only the surviving spec's distributor runs.
	if = {
		limit = { sul_has_mining_specialization = yes }
		sul_distribute_mining_buildings = yes
	}
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		sul_distribute_farming_buildings = yes
	}
	else_if = {
		limit = { sul_has_woodland_specialization = yes }
		sul_distribute_woodland_buildings = yes
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		sul_distribute_gathering_buildings = yes
	}
	else_if = {
		limit = { sul_has_commercial_specialization = yes }
		sul_distribute_commercial_buildings = yes
	}

	sul_dist_cleanup = yes
}

# Snapshot current levels by pop_type (called by distributor before deltas).

sul_dist_snapshot_current = {
	set_variable = { name = sul_cur_laborer_levels value = 0 }
	set_variable = { name = sul_cur_burgher_levels value = 0 }
	set_variable = { name = sul_cur_peasant_levels value = 0 }
	set_variable = { name = sul_cur_clergy_levels value = 0 }

	every_buildings_in_location = {
		if = {
			limit = { building_pop_type = pop_type:laborers }

			location ?= {
				change_variable = { name = sul_cur_laborer_levels add = prev.building_level }
			}
		}
		else_if = {
			limit = { building_pop_type = pop_type:burghers }

			location ?= {
				change_variable = { name = sul_cur_burgher_levels add = prev.building_level }
			}
		}
		else_if = {
			limit = { building_pop_type = pop_type:peasants }

			location ?= {
				change_variable = { name = sul_cur_peasant_levels add = prev.building_level }
			}
		}
		else_if = {
			limit = { building_pop_type = pop_type:clergy }

			location ?= {
				change_variable = { name = sul_cur_clergy_levels add = prev.building_level }
			}
		}
	}
}

sul_dist_cleanup = {
	remove_variable = sul_prior_laborer_levels
	remove_variable = sul_prior_burgher_levels
	remove_variable = sul_prior_peasant_levels
	remove_variable = sul_prior_clergy_levels
	remove_variable = sul_cur_laborer_levels
	remove_variable = sul_cur_burgher_levels
	remove_variable = sul_cur_peasant_levels
	remove_variable = sul_cur_clergy_levels
	remove_variable = sul_laborer_budget
	remove_variable = sul_burgher_budget
	remove_variable = sul_peasant_budget
	remove_variable = sul_clergy_budget
	remove_variable = sul_dist_cap
	remove_variable = sul_guild_cap
	remove_variable = sul_c0
	remove_variable = sul_c1
	remove_variable = sul_c2
	remove_variable = sul_c3
	remove_variable = sul_c4
	remove_variable = sul_g0
	remove_variable = sul_g1
	remove_variable = sul_g2
	remove_variable = sul_g3
}

# ─────────────────────────────────────────────────────────────────────────────
# Mining
#   Laborers:  stone_quarry, bog_iron_smelter, local_smelters, sul_iron_mine,
#              sul_marble_quarry
#   Burghers:  tools_guild, weapon_guild, jewelry_guild  (town/city only)
# ─────────────────────────────────────────────────────────────────────────────

sul_distribute_mining_buildings = {
	# Laborer distribution
	if = {
		limit = { var:sul_laborer_budget > 0 }

		set_variable = {
			name = sul_dist_cap
			value = { value = rural_building_cap floor = yes }
		}
		set_variable = { name = sul_c0 value = 0 }
		set_variable = { name = sul_c1 value = 0 }
		set_variable = { name = sul_c2 value = 0 }
		set_variable = { name = sul_c3 value = 0 }
		set_variable = { name = sul_c4 value = 0 }

		while = {
			limit = { var:sul_laborer_budget > 0 }
			count = 50

			if = {
				limit = {
					var:sul_laborer_budget > 0
					var:sul_c0 < var:sul_dist_cap
					OR = {
						topography = mountains
						topography = plateau
						topography = hills
					}
					NOT = { raw_material = goods:stone }
				}

				change_building_level_in_location = { building = building_type:stone_quarry value = 1 }
				change_variable = { name = sul_laborer_budget subtract = 1 }
				change_variable = { name = sul_c0 add = 1 }
			}

			if = {
				limit = {
					var:sul_laborer_budget > 0
					var:sul_c1 < var:sul_dist_cap
					OR = {
						is_adjacent_to_lake = yes
						topography = wetlands
					}
					NOT = { raw_material = goods:iron }
				}

				change_building_level_in_location = { building = building_type:bog_iron_smelter value = 1 }
				change_variable = { name = sul_laborer_budget subtract = 1 }
				change_variable = { name = sul_c1 add = 1 }
			}

			if = {
				limit = {
					var:sul_laborer_budget > 0
					var:sul_c2 < var:sul_dist_cap
					OR = {
						raw_material = goods:iron
						raw_material = goods:copper
						raw_material = goods:tin
						raw_material = goods:lead
					}
				}

				change_building_level_in_location = { building = building_type:local_smelters value = 1 }
				change_variable = { name = sul_laborer_budget subtract = 1 }
				change_variable = { name = sul_c2 add = 1 }
			}

			if = {
				limit = {
					var:sul_laborer_budget > 0
					var:sul_c3 < var:sul_dist_cap
					OR = {
						topography = mountains
						topography = hills
					}
					NOT = { raw_material = goods:iron }
				}

				change_building_level_in_location = { building = building_type:sul_iron_mine value = 1 }
				change_variable = { name = sul_laborer_budget subtract = 1 }
				change_variable = { name = sul_c3 add = 1 }
			}

			if = {
				limit = {
					var:sul_laborer_budget > 0
					var:sul_c4 < var:sul_dist_cap
					topography = plateau
					NOT = { raw_material = goods:marble }
				}

				change_building_level_in_location = { building = building_type:sul_marble_quarry value = 1 }
				change_variable = { name = sul_laborer_budget subtract = 1 }
				change_variable = { name = sul_c4 add = 1 }
			}
		}
	}

	# Burgher distribution (guilds — town/city only)
	if = {
		limit = { var:sul_burgher_budget > 0 is_town_or_city = yes }

		set_variable = {
			name = sul_guild_cap
			value = { value = guild_max_level floor = yes }
		}
		set_variable = { name = sul_g0 value = 0 }
		set_variable = { name = sul_g1 value = 0 }
		set_variable = { name = sul_g2 value = 0 }

		while = {
			limit = { var:sul_burgher_budget > 0 }
			count = 30

			if = {
				limit = { var:sul_burgher_budget > 0 var:sul_g0 < var:sul_guild_cap }

				change_building_level_in_location = { building = building_type:tools_guild value = 1 }
				change_variable = { name = sul_burgher_budget subtract = 1 }
				change_variable = { name = sul_g0 add = 1 }
			}

			if = {
				limit = { var:sul_burgher_budget > 0 var:sul_g1 < var:sul_guild_cap }

				change_building_level_in_location = { building = building_type:weapon_guild value = 1 }
				change_variable = { name = sul_burgher_budget subtract = 1 }
				change_variable = { name = sul_g1 add = 1 }
			}

			if = {
				limit = { var:sul_burgher_budget > 0 var:sul_g2 < var:sul_guild_cap }

				change_building_level_in_location = { building = building_type:jewelry_guild value = 1 }
				change_variable = { name = sul_burgher_budget subtract = 1 }
				change_variable = { name = sul_g2 add = 1 }
			}
		}
	}
}

# Farming
#   Peasants:  fruit_orchard, fiber_crops_farm
#   Burghers:  brewery, winery, distillers_guild, tannery  (town/city only)
# ─────────────────────────────────────────────────────────────────────────────

sul_distribute_farming_buildings = {
	# Peasant distribution
	if = {
		limit = { var:sul_peasant_budget > 0 }

		set_variable = {
			name = sul_dist_cap
			value = { value = rural_building_cap floor = yes }
		}

		while = {
			limit = { var:sul_peasant_budget > 0 }
			count = 50

			if = {
				limit = {
					var:sul_peasant_budget > 0
					var:sul_c0 < var:sul_dist_cap
					OR = {
						vegetation = farmland
						AND = {
							OR = {
								vegetation = grasslands
								vegetation = woods
								vegetation = jungle
							}
							OR = {
								has_river = yes
								is_adjacent_to_lake = yes
							}
						}
					}
					OR = {
						climate = mediterranean
						climate = subtropical
						climate = tropical
					}
				}

				change_building_level_in_location = { building = building_type:fruit_orchard value = 1 }
				change_variable = { name = sul_peasant_budget subtract = 1 }
				change_variable = { name = sul_c0 add = 1 }
			}

			if = {
				limit = {
					var:sul_peasant_budget > 0
					var:sul_c1 < var:sul_dist_cap
					NOT = { raw_material = goods:fiber_crops }
					NOT = { climate = arctic }
				}

				change_building_level_in_location = { building = building_type:fiber_crops_farm value = 1 }
				change_variable = { name = sul_peasant_budget subtract = 1 }
				change_variable = { name = sul_c1 add = 1 }
			}
		}
	}

	# Burgher distribution (guilds — town/city only)
	if = {
		limit = { var:sul_burgher_budget > 0 is_town_or_city = yes }

		set_variable = {
			name = sul_guild_cap
			value = { value = guild_max_level floor = yes }
		}
		set_variable = { name = sul_g0 value = 0 }
		set_variable = { name = sul_g1 value = 0 }
		set_variable = { name = sul_g2 value = 0 }
		set_variable = { name = sul_g3 value = 0 }

		while = {
			limit = { var:sul_burgher_budget > 0 }
			count = 30

			if = {
				limit = { var:sul_burgher_budget > 0 var:sul_g0 < var:sul_guild_cap }

				change_building_level_in_location = { building = building_type:brewery value = 1 }
				change_variable = { name = sul_burgher_budget subtract = 1 }
				change_variable = { name = sul_g0 add = 1 }
			}

			if = {
				limit = { var:sul_burgher_budget > 0 var:sul_g1 < var:sul_guild_cap }

				change_building_level_in_location = { building = building_type:winery value = 1 }
				change_variable = { name = sul_burgher_budget subtract = 1 }
				change_variable = { name = sul_g1 add = 1 }

					var:sul_g2 < var:sul_guild_cap

					market ?= {
						is_produced_in_market = goods:livestock
						is_produced_in_market = goods:clay
					}
				}

				change_building_level_in_location = { building = building_type:saltpeter_guild value = 1 }
				change_variable = { name = sul_burgher_budget subtract = 1 }
				change_variable = { name = sul_g2 add = 1 }
			}

			if = {
				limit = { var:sul_burgher_budget > 0 var:sul_g3 < var:sul_guild_cap }

				change_building_level_in_location = { building = building_type:apothecary value = 1 }
				change_variable = { name = sul_burgher_budget subtract = 1 }
				change_variable = { name = sul_g3 add = 1 }
			}
		}
	}
}

sul_distribute_commercial_buildings = {
	if = {
		limit = { var:sul_burgher_budget > 0 is_town_or_city = yes }

		set_variable = {
			name = sul_guild_cap
			value = { value = guild_max_level floor = yes }
		}
		set_variable = { name = sul_g0 value = 0 }
		set_variable = { name = sul_g1 value = 0 }

		while = {
			limit = { var:sul_burgher_budget > 0 }
			count = 30

			if = {
				limit = { var:sul_burgher_budget > 0 var:sul_g0 < var:sul_guild_cap }

				change_building_level_in_location = { building = building_type:marketplace value = 1 }
				change_variable = { name = sul_burgher_budget subtract = 1 }
				change_variable = { name = sul_g0 add = 1 }

			if = {
				limit = { var:sul_burgher_budget > 0 var:sul_g1 < var:sul_guild_cap }

				change_building_level_in_location = { building = building_type:cloth_guild value = 1 }
				change_variable = { name = sul_burgher_budget subtract = 1 }
				change_variable = { name = sul_g1 add = 1 }
			}
		}
	}
}
