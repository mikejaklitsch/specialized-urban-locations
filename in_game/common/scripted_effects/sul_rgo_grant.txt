# Specialized Urban Locations - RGO Building Grants
# Round-robin: while budget remains, each qualifying RGO gets 1 level per round.
# Per-building counters prevent exceeding rural_building_cap.
# root = location with sul_rgo_levels set

# ============================================
# ROUTER
# ============================================

sul_grant_specialization_rgo_building = {
	if = {
		limit = { sul_has_mining_specialization = yes }
		sul_grant_mining_rgo = yes
	}
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		sul_grant_farming_rgo = yes
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		sul_grant_gathering_rgo = yes
	}
	else_if = {
		limit = { sul_has_woodland_specialization = yes }
		sul_grant_woodland_rgo = yes
	}

	# Commercial — no RGO buildings, marketplace comes from guild grant
}

# ============================================
# CLEANUP helper — removes cap/counter variables
# ============================================

sul_rgo_grant_cleanup = {
	remove_variable = sul_rgo_cap
	remove_variable = sul_c0
	remove_variable = sul_c1
	remove_variable = sul_c2
	remove_variable = sul_c3
	remove_variable = sul_c4
}

# ============================================
# MINING RGO
# Terrain-based independent checks (multiple can fire per iteration)
# c0=local_smelters c1=sul_iron_mine c2=sul_marble_quarry
# c3=stone_quarry c4=bog_iron_smelter
# ============================================

sul_grant_mining_rgo = {
	set_variable = {
		name = sul_rgo_cap
		value = { value = rural_building_cap floor = yes }
	}
	set_variable = {
		name = sul_c0
		value = 0
	}
	set_variable = {
		name = sul_c1
		value = 0
	}
	set_variable = {
		name = sul_c2
		value = 0
	}
	set_variable = {
		name = sul_c3
		value = 0
	}
	set_variable = {
		name = sul_c4
		value = 0
	}

	while = {
		limit = { var:sul_rgo_levels > 0 }
		count = 20
		if = {
			limit = {
				var:sul_c0 < var:sul_rgo_cap
				OR = {
					raw_material = goods:iron
					raw_material = goods:copper
					raw_material = goods:tin
					raw_material = goods:lead
				}
				owner = { has_advance = local_smelters_advance }
			}

			change_building_level_in_location = {
				building = building_type:local_smelters
				value = 1
			}

			change_variable = { name = sul_rgo_levels subtract = 1 }
			change_variable = { name = sul_c0 add = 1 }
		}
		if = {
			limit = {
				var:sul_rgo_levels > 0
				var:sul_c1 < var:sul_rgo_cap
				OR = {
					topography = mountains
					topography = hills
				}
				NOT = {
					raw_material = goods:iron
				}
			}

			change_building_level_in_location = {
				building = building_type:sul_iron_mine
				value = 1
			}

			change_variable = { name = sul_rgo_levels subtract = 1 }
			change_variable = { name = sul_c1 add = 1 }
		}
		if = {
			limit = {
				var:sul_rgo_levels > 0
				var:sul_c2 < var:sul_rgo_cap
				topography = plateau
				NOT = {
					raw_material = goods:marble
				}
			}

			change_building_level_in_location = {
				building = building_type:sul_marble_quarry
				value = 1
			}

			change_variable = { name = sul_rgo_levels subtract = 1 }
			change_variable = { name = sul_c2 add = 1 }
		}
		if = {
			limit = {
				var:sul_rgo_levels > 0
				var:sul_c3 < var:sul_rgo_cap
				OR = {
					topography = mountains
					topography = plateau
					topography = hills
				}
				NOT = {
					raw_material = goods:stone
				}
			}

			change_building_level_in_location = {
				building = building_type:stone_quarry
				value = 1
			}

			change_variable = { name = sul_rgo_levels subtract = 1 }
			change_variable = { name = sul_c3 add = 1 }
		}
		if = {
			limit = {
				var:sul_rgo_levels > 0
				var:sul_c4 < var:sul_rgo_cap
				OR = {
					is_adjacent_to_lake = yes
					topography = wetlands
				}
				NOT = {
					raw_material = goods:iron
				}
				owner = { has_advance = iron_working }
			}

			change_building_level_in_location = {
				building = building_type:bog_iron_smelter
				value = 1
			}

			change_variable = { name = sul_rgo_levels subtract = 1 }
			change_variable = { name = sul_c4 add = 1 }
		}
	}

	sul_rgo_grant_cleanup = yes
}
