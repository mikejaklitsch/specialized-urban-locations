# Specialized Urban Locations - Building Distribution
# Round-robin: distributes destroyed building levels across matching buildings.
# Each qualifying building gets 1 level per round until budget is spent.
# Per-building counters prevent exceeding caps.
#
# root = location (call AFTER sul_save_building_levels + sul_destroy_non_matching_buildings)
#
# To add a building: add a new if-block in the relevant specialization effect.
# Conditions should mirror the building's location_potential (minus the spec gate).
# RGO buildings use sul_dist_cap (rural_building_cap).
# Guild buildings use sul_guild_cap (guild_max_level), gated by is_town_or_city.

# ============================================
# ROUTER
# ============================================

sul_distribute_specialization_buildings = {
	# Budget = how many levels were destroyed
	set_variable = {
		name = sul_dist_budget

		value = {
			value = var:sul_prior_building_levels
			subtract = total_building_levels
			min = 0
		}
	}

	if = {
		limit = { var:sul_dist_budget > 0 }

		if = {
			limit = { sul_has_mining_specialization = yes }
			sul_distribute_mining_buildings = yes
		}
		else_if = {
			limit = { sul_has_farming_specialization = yes }
			sul_distribute_farming_buildings = yes
		}
		else_if = {
			limit = { sul_has_woodland_specialization = yes }
			sul_distribute_woodland_buildings = yes
		}
		else_if = {
			limit = { sul_has_gathering_specialization = yes }
			sul_distribute_gathering_buildings = yes
		}
		else_if = {
			limit = { sul_has_commercial_specialization = yes }
			sul_distribute_commercial_buildings = yes
		}
	}

	sul_dist_cleanup = yes
}

# ============================================
# CLEANUP — removes temporary variables
# ============================================

sul_dist_cleanup = {
	remove_variable = sul_dist_budget
	remove_variable = sul_dist_cap
	remove_variable = sul_guild_cap
	remove_variable = sul_c0
	remove_variable = sul_c1
	remove_variable = sul_c2
	remove_variable = sul_c3
	remove_variable = sul_g0
	remove_variable = sul_g1
	remove_variable = sul_g2
	remove_variable = sul_g3
}

# ============================================
# MINING
# RGO: c0=stone_quarry c1=bog_iron_smelter c2=local_smelters c3=sul_iron_mine c4=sul_marble_quarry
# Guild: g0=tools_guild g1=weapon_guild g2=jewelry_guild
# ============================================

sul_distribute_mining_buildings = {
	set_variable = {
		name = sul_dist_cap
		value = { value = rural_building_cap floor = yes }
	}
	set_variable = { name = sul_c0 value = 0 }
	set_variable = { name = sul_c1 value = 0 }
	set_variable = { name = sul_c2 value = 0 }
	set_variable = { name = sul_c3 value = 0 }
	set_variable = { name = sul_c4 value = 0 }

	while = {
		limit = { var:sul_dist_budget > 0 }
		count = 50

		# Stone quarry — mountains, plateau, or hills (not raw stone)
		if = {
			limit = {
				var:sul_dist_budget > 0
				var:sul_c0 < var:sul_dist_cap
				OR = {
					topography = mountains
					topography = plateau
					topography = hills
				}
				NOT = {
					raw_material = goods:stone
				}
			}

			change_building_level_in_location = {
				building = building_type:stone_quarry
				value = 1
			}

			change_variable = { name = sul_dist_budget subtract = 1 }
			change_variable = { name = sul_c0 add = 1 }
		}

		# Bog iron smelter — wetlands or lake (not raw iron)
		if = {
			limit = {
				var:sul_dist_budget > 0
				var:sul_c1 < var:sul_dist_cap
				OR = {
					is_adjacent_to_lake = yes
					topography = wetlands
				}
				NOT = {
					raw_material = goods:iron
				}
			}

			change_building_level_in_location = {
				building = building_type:bog_iron_smelter
				value = 1
			}

			change_variable = { name = sul_dist_budget subtract = 1 }
			change_variable = { name = sul_c1 add = 1 }
		}

		# Local smelters — ore-producing locations
		if = {
			limit = {
				var:sul_dist_budget > 0
				var:sul_c2 < var:sul_dist_cap
				OR = {
					raw_material = goods:iron
					raw_material = goods:copper
					raw_material = goods:tin
					raw_material = goods:lead
				}
			}

			change_building_level_in_location = {
				building = building_type:local_smelters
				value = 1
			}

			change_variable = { name = sul_dist_budget subtract = 1 }
			change_variable = { name = sul_c2 add = 1 }
		}

		# Iron mine — mountains or hills (not raw iron)
		if = {
			limit = {
				var:sul_dist_budget > 0
				var:sul_c3 < var:sul_dist_cap
				OR = {
					topography = mountains
					topography = hills
				}
				NOT = {
					raw_material = goods:iron
				}
			}

			change_building_level_in_location = {
				building = building_type:sul_iron_mine
				value = 1
			}

			change_variable = { name = sul_dist_budget subtract = 1 }
			change_variable = { name = sul_c3 add = 1 }
		}

		# Marble quarry — plateau (not raw marble)
		if = {
			limit = {
				var:sul_dist_budget > 0
				var:sul_c4 < var:sul_dist_cap
				topography = plateau
				NOT = {
					raw_material = goods:marble
				}
			}

			change_building_level_in_location = {
				building = building_type:sul_marble_quarry
				value = 1
			}

			change_variable = { name = sul_dist_budget subtract = 1 }
			change_variable = { name = sul_c4 add = 1 }
		}
	}

	# Guild buildings — towns and cities only
	if = {
		limit = { is_town_or_city = yes var:sul_dist_budget > 0 }

		set_variable = {
			name = sul_guild_cap
			value = { value = guild_max_level floor = yes }
		}
		set_variable = { name = sul_g0 value = 0 }
		set_variable = { name = sul_g1 value = 0 }
		set_variable = { name = sul_g2 value = 0 }

		while = {
			limit = { var:sul_dist_budget > 0 }
			count = 30

			# Tools guild
			if = {
				limit = { var:sul_dist_budget > 0 var:sul_g0 < var:sul_guild_cap }

				change_building_level_in_location = {
					building = building_type:tools_guild
					value = 1
				}

				change_variable = { name = sul_dist_budget subtract = 1 }
				change_variable = { name = sul_g0 add = 1 }
			}

			# Weapon guild
			if = {
				limit = { var:sul_dist_budget > 0 var:sul_g1 < var:sul_guild_cap }

				change_building_level_in_location = {
					building = building_type:weapon_guild
					value = 1
				}

				change_variable = { name = sul_dist_budget subtract = 1 }
				change_variable = { name = sul_g1 add = 1 }
			}

			# Jewelry guild
			if = {
				limit = { var:sul_dist_budget > 0 var:sul_g2 < var:sul_guild_cap }

				change_building_level_in_location = {
					building = building_type:jewelry_guild
					value = 1
				}

				change_variable = { name = sul_dist_budget subtract = 1 }
				change_variable = { name = sul_g2 add = 1 }
			}
		}
	}
}
