# Economy Seeding — demand-weighted building distribution at game start
# Unified initialization system that:
#   1. Places universal infrastructure (granary/storehouse, mason)
#   2. Places spec-appropriate village building
#   3. Grants terrain-specific RGO building
#   4. Splits remaining budget between rural cottages and guild buildings
#      using demand-derived weights (per-mille, sum to 1000 per tier)

# ═══════════════════════════════════════════════════════════════════════════
# BUILDING DISTRIBUTION WEIGHTS
# Generated from demand_calculator.py --weights
# Higher weight = more building levels allocated for that building
# ═══════════════════════════════════════════════════════════════════════════

# MINING
# BUDGET PARAMETERS
# target = rank_base + development × 0.375 + prior_buildings × vanilla_scale
# rgo_share = rank-dependent (rural 70%, town 50%, city 30%)
# budget = target - already_placed
@sul_rank_base_rural = 1
@sul_rank_base_town = 2
@sul_rank_base_city = 3
@sul_vanilla_scale = 1
@sul_rgo_share_rural = 700          # 70% of target goes to RGO buildings for rural (per-mille)
@sul_rgo_share_town = 500           # 50% of target goes to RGO buildings for towns (per-mille)
@sul_rgo_share_city = 300           # 30% of target goes to RGO buildings for cities (per-mille)
@sul_village_share = 150            # 15% of target goes to village buildings (per-mille)

@sul_w_sul_rural_blacksmith = 522
@sul_w_sul_rural_weaponmaker = 359
@sul_w_sul_rural_jeweler = 119

@sul_w_weapon_guild = 320
@sul_w_cannon_maker = 107

@sul_w_sul_rural_winemaker = 213

@sul_w_naval_supplies_guild = 127

# ═══════════════════════════════════════════════════════════════════════════
# MAIN ENTRY POINT
# root = location (must have sul_prior_building_levels and specialization set)
# Called after specialization assignment and building destruction
# ═══════════════════════════════════════════════════════════════════════════

sul_seed_economy = {
	# ── Steps 1-2: Calculate target building level and RGO share ──
	# target = rank_base + development × 0.375 + prior_buildings × vanilla_scale
	# RGO gets a rank-dependent share (higher for rural, lower for cities)
	if = {
		limit = { location_rank = location_rank:rural_settlement }

		set_variable = {
			name = sul_target

			value = {
				value = @sul_rank_base_rural
				add = { value = development multiply = 0.375 }
				add = { value = var:sul_prior_building_levels multiply = @sul_vanilla_scale }
				min = 1
				max = { value = population multiply = 0.25 }
				floor = yes
			}
		}
		set_variable = {
			name = sul_rgo_levels

			value = {
				value = var:sul_target
				multiply = @sul_rgo_share_rural
				divide = 1000
				min = 1
				floor = yes
			}
		}
	}
	else_if = {
		limit = { location_rank = location_rank:town }

		set_variable = {
			name = sul_target

			value = {
				value = @sul_rank_base_town
				add = { value = development multiply = 0.375 }
				add = { value = var:sul_prior_building_levels multiply = @sul_vanilla_scale }
				min = 1
				max = { value = population multiply = 0.25 }
				floor = yes
			}
		}
		set_variable = {
			name = sul_rgo_levels

			value = {
				value = var:sul_target
				multiply = @sul_rgo_share_town
				divide = 1000
				min = 1
				floor = yes
			}
		}
	}
	else = {
		set_variable = {
			name = sul_target

			value = {
				value = @sul_rank_base_city
				add = { value = development multiply = 0.375 }
				add = { value = var:sul_prior_building_levels multiply = @sul_vanilla_scale }
				min = 1
				max = { value = population multiply = 0.25 }
				floor = yes
			}
		}
		set_variable = {
			name = sul_rgo_levels

			value = {
				value = var:sul_target
				multiply = @sul_rgo_share_city
				divide = 1000
				min = 1
				floor = yes
			}
		}
	}

	sul_grant_specialization_rgo_building = yes

	# ── Step 3: Place universal infrastructure ──
	# Granaries only in capitals and provincial capitals
	# Rural granary: 1 per 5k pop (min 1), provincial capitals only
	# Granary (town/city): 1 per 40k pop, capitals/provincial capitals only
	if = {
		limit = {
			location_rank = location_rank:rural_settlement
			is_province_capital = yes
		}

		set_variable = {
			name = sul_granary_levels

			value = {
				value = population
				divide = 5
				floor = yes
				min = 1
				max = rural_building_cap
			}
		}

		change_building_level_in_location = {
			building = building_type:sul_rural_granary
			value = { value = var:sul_granary_levels floor = yes }
		}

		remove_variable = sul_granary_levels
	}
	else_if = {
		limit = {
			is_town_or_city = yes
			population >= 20
			OR = {
				is_capital = yes
				is_province_capital = yes
			}
		}

		set_variable = {
			name = sul_granary_levels
			value = { value = population divide = 40 floor = yes min = 1 }
		}

		change_building_level_in_location = {
			building = building_type:granary
			value = { value = var:sul_granary_levels floor = yes }
		}

		remove_variable = sul_granary_levels
	}

	# Mason — towns and cities only
	if = {
		limit = { is_town_or_city = yes }

		change_building_level_in_location = {
			building = building_type:mason
			value = 1
		}
	}

	# ── Step 4: Place spec-appropriate village building ──
	# Village gets a proportional share of target, scaling with location size
	set_variable = {
		name = sul_village_levels

		value = {
			value = var:sul_target
			multiply = @sul_village_share
			divide = 1000
			min = 1
			max = rural_building_cap
			floor = yes
		}
	}

	if = {
		limit = { sul_has_farming_specialization = yes }

		change_building_level_in_location = {
			building = building_type:farming_village
			value = { value = var:sul_village_levels floor = yes }
		}
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }

		change_building_level_in_location = {
			building = building_type:fishing_village
			value = { value = var:sul_village_levels floor = yes }
		}
	}
	else_if = {
		limit = { sul_has_woodland_specialization = yes }

		change_building_level_in_location = {
			building = building_type:forest_village
			value = { value = var:sul_village_levels floor = yes }
		}
	}
	else_if = {
		limit = { sul_has_mining_specialization = yes }

		change_building_level_in_location = {
			building = building_type:sul_mining_village
			value = { value = var:sul_village_levels floor = yes }
		}
	}
	else_if = {
		limit = { sul_has_commercial_specialization = yes }

		change_building_level_in_location = {
			building = building_type:market_village
			value = { value = var:sul_village_levels floor = yes }
		}
	}

	remove_variable = sul_village_levels

	# ── Step 5: Calculate remaining budget ──
	# Budget = target minus what's already placed (RGO + infra + village)
	set_variable = {
		name = sul_budget

		value = {
			value = var:sul_target
			subtract = total_building_levels
			min = 0
			floor = yes
		}
	}

	# Nothing left to distribute
	if = {
		limit = { var:sul_budget <= 0 }
		sul_seed_cleanup = yes
	}
	else = {
		# ── Step 6: Split remaining budget between rural and guild tiers ──
		# Rural settlements: 100% rural
		# Towns: 25-50% guild (scales with development)
		# Cities: 55-75% guild (scales with development)

		if = {
			limit = { location_rank = location_rank:rural_settlement }
			set_variable = { name = sul_guild_budget value = 0 }
			set_variable = { name = sul_rural_budget value = var:sul_budget }
		}
		else_if = {
			limit = { location_rank = location_rank:town }

			set_variable = {
				name = sul_guild_permille
				value = { value = 250 add = { value = development multiply = 5 } max = 500 }
			}
			set_variable = {
				name = sul_guild_budget

				value = {
					value = var:sul_budget
					multiply = var:sul_guild_permille
					divide = 1000
					floor = yes
				}
			}
			set_variable = {
				name = sul_rural_budget
				value = { value = var:sul_budget subtract = var:sul_guild_budget }
			}
		}
		else = {
			# City
			set_variable = {
				name = sul_guild_permille
				value = { value = 550 add = { value = development multiply = 3 } max = 750 }
			}
			set_variable = {
				name = sul_guild_budget

				value = {
					value = var:sul_budget
					multiply = var:sul_guild_permille
					divide = 1000
					floor = yes
				}
			}
			set_variable = {
				name = sul_rural_budget
				value = { value = var:sul_budget subtract = var:sul_guild_budget }
			}
		}

		# ── Steps 7-8: Distribute rural and guild buildings by specialization ──
		if = {
			limit = { sul_has_mining_specialization = yes }
			if = {
				limit = { var:sul_rural_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_rural_budget }
				sul_seed_mining_rural = yes
			}
			if = {
				limit = { var:sul_guild_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_guild_budget }
				sul_seed_mining_guild = yes
			}
		}
		else_if = {
			limit = { sul_has_farming_specialization = yes }
			if = {
				limit = { var:sul_rural_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_rural_budget }
				sul_seed_farming_rural = yes
			}
			if = {
				limit = { var:sul_guild_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_guild_budget }
				sul_seed_farming_guild = yes
			}
		}
		else_if = {
			limit = { sul_has_gathering_specialization = yes }
			if = {
				limit = { var:sul_rural_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_rural_budget }
				sul_seed_gathering_rural = yes
			}
			if = {
				limit = { var:sul_guild_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_guild_budget }
				sul_seed_gathering_guild = yes
			}
		}
		else_if = {
			limit = { sul_has_woodland_specialization = yes }
			if = {
				limit = { var:sul_rural_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_rural_budget }
				sul_seed_woodland_rural = yes
			}
			if = {
				limit = { var:sul_guild_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_guild_budget }
				sul_seed_woodland_guild = yes
			}
		}
		else_if = {
			limit = { sul_has_commercial_specialization = yes }
			if = {
				limit = { var:sul_rural_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_rural_budget }
				sul_seed_commercial_rural = yes
			}
			if = {
				limit = { var:sul_guild_budget > 0 }
				set_variable = { name = sul_tier_budget value = var:sul_guild_budget }
				sul_seed_commercial_guild = yes
			}
		}

		sul_seed_cleanup = yes
	}
}

# ═══════════════════════════════════════════════════════════════════════════
# CLEANUP

sul_seed_cleanup = {
	remove_variable = sul_prior_building_levels
	remove_variable = sul_target
	remove_variable = sul_rgo_levels
	remove_variable = sul_budget
	remove_variable = sul_guild_permille
	remove_variable = sul_guild_budget
	remove_variable = sul_rural_budget
	remove_variable = sul_tier_budget
	remove_variable = sul_tmp
	remove_variable = sul_bldg_weight
}

# ═══════════════════════════════════════════════════════════════════════════
# WEIGHTED ADD HELPER
# Allocates floor(tier_budget * weight / 1000) levels to a building
# and subtracts from sul_tier_budget. Remainder flows to the last building.
# root = location, expects sul_tier_budget and sul_bldg_weight set
# ═══════════════════════════════════════════════════════════════════════════

sul_seed_weighted_add = {
	set_variable = {
		name = sul_tmp

		value = {
			value = var:sul_tier_budget
			multiply = var:sul_bldg_weight
			divide = 1000
			max = guild_max_level
			floor = yes
		}
	}
	if = {
		limit = { var:sul_tmp > 0 }

		change_building_level_in_location = {
			building = building_type:$BUILDING$
			value = { value = var:sul_tmp floor = yes }
		}
	}
	change_variable = {
		name = sul_tier_budget
		subtract = var:sul_tmp
	}
}

# ═══════════════════════════════════════════════════════════════════════════
# RURAL TIER DISTRIBUTIONS
# Each processes buildings highest-weight first, last gets remainder
# root = location with sul_tier_budget set
# ═══════════════════════════════════════════════════════════════════════════

sul_seed_mining_rural = {
	# weaponmaker: needs iron/lumber in market + weapon_guild_advance
	if = {
		limit = {
			owner = { has_advance = weapon_guild_advance }
			OR = {
				is_produced_in_location_market = goods:iron
				is_produced_in_location_market = goods:lumber
			}
		}

		set_variable = { name = sul_bldg_weight value = @sul_w_sul_rural_weaponmaker }
		sul_seed_weighted_add = { BUILDING = sul_rural_weaponmaker }
	}

	# blacksmith: needs iron or stone in market
	if = {
		limit = {
			OR = {
				is_produced_in_location_market = goods:iron
				is_produced_in_location_market = goods:stone
			}
		}

		set_variable = { name = sul_bldg_weight value = @sul_w_sul_rural_blacksmith }
		sul_seed_weighted_add = { BUILDING = sul_rural_blacksmith }
	}

	# Remainder to jeweler: needs stone/gold/silver/gems in market
	if = {
		limit = {
			var:sul_tier_budget > 0
			OR = {
				is_produced_in_location_market = goods:stone
				is_produced_in_location_market = goods:goods_gold
				is_produced_in_location_market = goods:silver
				is_produced_in_location_market = goods:gems
			}
		}

		set_variable = {
			name = sul_tmp
			value = { value = var:sul_tier_budget max = guild_max_level floor = yes }
		}

		change_building_level_in_location = {
			building = building_type:sul_rural_jeweler
			value = { value = var:sul_tmp floor = yes }
		}
	}
}

# GUILD TIER DISTRIBUTIONS
# Conditional buildings (require advances) checked at runtime.
# If advance missing, those levels stay in budget and flow to last building.
# root = location with sul_tier_budget set
# ═══════════════════════════════════════════════════════════════════════════

sul_seed_mining_guild = {
	# Conditional: tools_guild(305), weapon_guild(305), cannon_maker(305)
	# Unconditional: jewelry_guild(85) — always last (gets remainder)
	if = {
		limit = { owner = { has_advance = metallurgy } }
		set_variable = { name = sul_bldg_weight value = @sul_w_tools_guild }
		sul_seed_weighted_add = { BUILDING = tools_guild }
	}
	if = {
		limit = { owner = { has_advance = weapon_guild_advance } }
		set_variable = { name = sul_bldg_weight value = @sul_w_weapon_guild }
		sul_seed_weighted_add = { BUILDING = weapon_guild }
	}
	if = {
