# Specialization System - Core Effects
# Init, rebuild, eligibility, validation, and cache management
@sul_mining_type = 1
@sul_farming_type = 2
@sul_gathering_type = 3
@sul_woodland_type = 4
@sul_commercial_type = 5
@sul_rank_offset_town = 0
@sul_rank_offset_city = 10
@sul_rank_offset_rural = 20
@sul_farming_weight = 2

# Called on game start and on version change. Minimally invasive: preserves
# valid player specializations and buildings, only reassigns locations whose
# current spec violates the updated rules.

sul_specialization_full_rebuild = {
	sul_setup_goods_spec_map = yes

	# Cleanup: destroy fruit orchards in locations that no longer meet climate gates
	every_location_in_the_world = {
		limit = {
			has_building = building_type:fruit_orchard
			NOT = {
				OR = {
					climate = mediterranean
					climate = subtropical
					climate = tropical
				}
			}
		}

		every_buildings_in_location = {
			limit = { building_type = building_type:fruit_orchard }
			location = { destroy_building = prev }
		}
	}

	# All ownable locations: recalculate eligibility, then branch
	every_location_in_the_world = {
		limit = { is_ownable = yes exists = raw_material }
		sul_specialization_calculate_eligibility = yes
		sul_specialization_rebuild_location = yes
		sul_specialization_set_init_production_variables = yes
	}

	# Market cache
	sul_market_cache_refresh_all = yes
}

# root = location (eligibility already calculated). Three paths:
#   no spec        -> fresh assignment with buildings
#   valid spec     -> refresh modifier only (handles tuning changes)
#   invalid spec   -> reassign by resources + rebuild buildings

sul_specialization_rebuild_location = {
	if = {
		limit = { sul_has_any_specialization = yes }
		if = {
			# Current spec still valid - just refresh the modifier
			limit = { sul_current_specialization_is_valid = yes }
			sul_specialization_refresh_modifier = yes
		}
		else = {
			# Current spec no longer valid - full reassignment
			sul_specialization_reassign = yes
		}
	}
	else = {
		# No specialization yet - first-time init
		sul_specialization_init_new = yes
	}
}

# root = location with sul_spec_type. Swaps the modifier for the current
# spec+rank without changing the spec itself.

sul_specialization_refresh_modifier = {
	sul_specialization_clear_all_modifiers = yes
	sul_specialization_update_designation = yes
	sul_specialization_apply_modifier = yes
}

# root = location. Used when the current spec is no longer valid: clears old
# state, picks best match from resources, rebuilds buildings.

sul_specialization_reassign = {
	sul_specialization_clear_all_modifiers = yes
	sul_specialization_assign_by_resources = yes
	sul_specialization_apply_modifier = yes

	# Replace incompatible buildings with matching ones for owned locations
	if = {
		limit = { exists = owner }
		sul_specialization_save_building_levels = yes
		sul_specialization_destroy_non_matching_buildings = yes
		sul_distribute_specialization_buildings = yes
	}
}

# root = location. First-time setup for a location that has never been specialized.

sul_specialization_init_new = {
	sul_specialization_assign_by_resources = yes
	sul_specialization_apply_modifier = yes

	# Replace incompatible buildings with matching ones for owned locations
	if = {
		limit = { exists = owner }
		sul_specialization_save_building_levels = yes
		sul_specialization_destroy_non_matching_buildings = yes
		sul_distribute_specialization_buildings = yes
	}
}

# root = location. Snapshots total_building_levels into sul_prior_building_levels
# so post-rebuild logic can compare before/after counts.

sul_specialization_save_building_levels = {
	set_variable = {
		name = sul_prior_building_levels
		value = total_building_levels
	}
}

# root = location. Caches which specializations the location is eligible for.
# Called on game start, colonization, and raw material change.

sul_specialization_calculate_eligibility = {
	# Mining: mining resources OR mountains OR wetlands OR lake adjacency
	if = {
		limit = {
			OR = {
				sul_has_mining_resources = yes
				topography = mountains
				topography = wetlands
				is_adjacent_to_lake = yes
			}
		}

		set_variable = { name = sul_can_mine value = 1 }
	}
	else = { remove_variable = sul_can_mine }

	# Farming: farming resources OR farmland OR flatland with river/grasslands
	if = {
		limit = {
			OR = {
				sul_has_farming_resources = yes
				vegetation = farmland
				AND = {
					topography = flatland
					has_river = yes
				}
				AND = {
					topography = flatland
					vegetation = grasslands
				}
			}
		}

		set_variable = { name = sul_can_farm value = 1 }
	}
	else = { remove_variable = sul_can_farm }

	# Gathering: gathering resources OR coastal OR river OR lake adjacency
	if = {
		limit = {
			OR = {
				sul_has_gathering_resources = yes
				is_coastal = yes
				has_river = yes
				is_adjacent_to_lake = yes
			}
		}

		set_variable = { name = sul_can_gather value = 1 }
	}
	else = { remove_variable = sul_can_gather }

	# Woodland: woodland resources OR forest/woods/jungle
	if = {
		limit = {
			OR = {
				sul_has_woodland_resources = yes
				vegetation = forest
				vegetation = woods
				vegetation = jungle
			}
		}

		set_variable = { name = sul_can_woodland value = 1 }
	}
	else = { remove_variable = sul_can_woodland }
}

# root = location. Re-checks current spec after a raw material change;
# falls back to commercial if invalidated.

sul_specialization_validate = {
	if = {
		limit = { sul_has_any_specialization = yes }
		if = {
			limit = {
				sul_has_mining_specialization = yes
				NOT = {
					has_variable = sul_can_mine
				}
			}

			sul_specialization_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				sul_has_farming_specialization = yes
				NOT = {
					has_variable = sul_can_farm
				}
			}

			sul_specialization_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				sul_has_gathering_specialization = yes
				NOT = {
					has_variable = sul_can_gather
				}
			}

			sul_specialization_switch_to_commercial = yes
		}
		else_if = {
			limit = {
				sul_has_woodland_specialization = yes
				NOT = {
					has_variable = sul_can_woodland
				}
			}

			sul_specialization_switch_to_commercial = yes
		}
	}
}

sul_specialization_switch_to_commercial = {
	sul_specialization_clear_all_modifiers = yes

	set_variable = { name = sul_spec_type value = @sul_commercial_type }

	sul_specialization_apply_modifier = yes
	if = {
		limit = {
			OR = {
				location_rank = location_rank:town
				location_rank = location_rank:city
			}
		}

		sul_specialization_destroy_non_matching_buildings = yes
	}
}

# root = location with sul_spec_type. Sets per-good vars (1-day expiry) that
# mirror the sul_allows_* static-modifier flags. Required because modifier:
# reads aren't available during initialize_from_bookmark, so building
# location_potential checks need a variable fallback during init.

		limit = { sul_has_mining_specialization = yes }

		set_variable = { name = sul_allows_jewelry_production value = 1 days = 1 }
		set_variable = { name = sul_allows_steel_production value = 1 days = 1 }
	}
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		set_variable = { name = sul_allows_beer_production value = 1 days = 1 }
		set_variable = { name = sul_allows_wine_production value = 1 days = 1 }
		set_variable = { name = sul_allows_liquor_production value = 1 days = 1 }
		set_variable = { name = sul_allows_leather_production value = 1 days = 1 }
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		set_variable = { name = sul_allows_glass_production value = 1 days = 1 }
		set_variable = { name = sul_allows_pottery_production value = 1 days = 1 }
		set_variable = { name = sul_allows_porcelain_production value = 1 days = 1 }
		set_variable = { name = sul_allows_saltpeter_production value = 1 days = 1 }
		set_variable = { name = sul_allows_apothecary_production value = 1 days = 1 }

			name = sul_allows_naval_supplies_production
			value = 1
			days = 1
		}
		set_variable = { name = sul_allows_commerce_buildings value = 1 days = 1 }
	}
}

# Specialized Urban Locations - Building Destruction Effects
# Each effect destroys all specialization-relevant buildings EXCEPT those matching the specialization
# root = location
# NOTE: Universities are always preserved in cities regardless of specialization

sul_specialization_destroy_non_mining_buildings = {
	every_buildings_in_location = {
		limit = {
			sul_is_destroyable_specialization_building = yes
			NOT = {
				sul_is_mining_building = yes
			}
			NAND = {
				building_type = building_type:university
				location = { location_rank = location_rank:city }
			}
		}

		location = { destroy_building = prev }
	}
}

# Dispatcher to the per-spec destroy effect. root = location with sul_spec_type.

sul_specialization_destroy_non_matching_buildings = {
	if = {
		limit = { sul_has_mining_specialization = yes }
		sul_specialization_destroy_non_mining_buildings = yes
	}
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		sul_specialization_destroy_non_farming_buildings = yes
	}
	else_if = {
		limit = { sul_has_woodland_specialization = yes }
		sul_specialization_destroy_non_woodland_buildings = yes
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		sul_specialization_destroy_non_gathering_buildings = yes
	}
	else_if = {
		limit = { sul_has_commercial_specialization = yes }
		sul_specialization_destroy_non_commercial_buildings = yes
	}
}

# root = location (any rank). Priority order:
# Commercial (special conditions) > Farming > Mining > Gathering > Woodland.
# Writes sul_spec_type only (1-5); rank is read live from location_rank.

sul_specialization_assign_by_resources = {
	# Commercial specialization (market centers always; cities require development > 30; towns pass through)
	if = {
		limit = {
			OR = {
				is_market_center = yes
				AND = {
					NOT = {
						location_rank = location_rank:city
					}
					owner ?= { num_locations > 1 }
					OR = {
						modifier:harbor_suitability > 0.7
						AND = {
							is_capital = yes
							owner ?= { government_type = government_type:republic }
						}
					}
				}
				AND = {
					location_rank = location_rank:city
					development > 30
					OR = {
						modifier:harbor_suitability > 0.7
						AND = {
							is_capital = yes
							owner ?= { government_type = government_type:republic }
						}
					}
				}
			}
		}

		set_variable = { name = sul_spec_type value = @sul_commercial_type }
	}
	else_if = {
		limit = { sul_has_farming_resources = yes }
		set_variable = { name = sul_spec_type value = @sul_farming_type }
	}
	else_if = {
		limit = { sul_has_mining_resources = yes }
		set_variable = { name = sul_spec_type value = @sul_mining_type }
	}
	else_if = {
		limit = { sul_has_gathering_resources = yes }
		set_variable = { name = sul_spec_type value = @sul_gathering_type }
	}
	else_if = {
		limit = { sul_has_woodland_resources = yes }
		set_variable = { name = sul_spec_type value = @sul_woodland_type }
	}

	# Update combined designation for GUI
	if = {
		limit = { has_variable = sul_spec_type }
		sul_specialization_update_designation = yes
	}
}

# Encodes type+rank into a single value so the GUI doesn't need to query
# location_rank from a non-location scope. Town 1-5, City 11-15, Rural 21-25.

sul_specialization_update_designation = {
	set_variable = { name = sul_designation value = var:sul_spec_type }
	if = {
		limit = { location_rank = location_rank:city }
		change_variable = { name = sul_designation add = @sul_rank_offset_city }
	}
	else_if = {
		limit = { location_rank = location_rank:rural_settlement }
		change_variable = { name = sul_designation add = @sul_rank_offset_rural }
	}
}

# root = location with sul_spec_type. Applies the modifier matching spec+rank.

sul_specialization_apply_modifier = {
	if = {
		limit = { var:sul_spec_type = @sul_mining_type }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			add_location_modifier = { modifier = sul_mining_rural_specialization }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			add_location_modifier = { modifier = sul_mining_town_specialization }
		}
		else = { add_location_modifier = { modifier = sul_mining_city_specialization } }
	}
	else_if = {
		limit = { var:sul_spec_type = @sul_farming_type }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			add_location_modifier = { modifier = sul_farming_rural_specialization }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			add_location_modifier = { modifier = sul_farming_town_specialization }
		}
		else = {
			add_location_modifier = { modifier = sul_farming_city_specialization }
		}
	}
	else_if = {
		limit = { var:sul_spec_type = @sul_woodland_type }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			add_location_modifier = { modifier = sul_woodland_rural_specialization }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			add_location_modifier = { modifier = sul_woodland_town_specialization }
		}
		else = {
			add_location_modifier = { modifier = sul_woodland_city_specialization }
		}
	}
	else_if = {
		limit = { var:sul_spec_type = @sul_gathering_type }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			add_location_modifier = { modifier = sul_gathering_rural_specialization }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			add_location_modifier = { modifier = sul_gathering_town_specialization }
		}
		else = {
			add_location_modifier = { modifier = sul_gathering_city_specialization }
		}
	}
	else_if = {
		limit = { var:sul_spec_type = @sul_commercial_type }
		if = {
			limit = { location_rank = location_rank:rural_settlement }
			add_location_modifier = { modifier = sul_commercial_rural_specialization }
		}
		else_if = {
			limit = { location_rank = location_rank:town }
			add_location_modifier = { modifier = sul_commercial_town_specialization }
		}
		else = {
			add_location_modifier = { modifier = sul_commercial_city_specialization }
		}
	}
}
