# Specialization System - Core Effects
# Init, rebuild, eligibility, validation, and cache management
# Rank-based: specialization is encoded in location_rank (mining_rural, farming_town, etc.)

# Called on game start and on version change. Minimally invasive: preserves
# valid player specializations and buildings, only reassigns locations whose
# current spec violates the updated rules.

sul_specialization_full_rebuild = {
	sul_setup_goods_spec_map = yes

	# Cleanup: destroy fruit orchards in locations that no longer meet climate gates
	every_location_in_the_world = {
		limit = {
			has_building = building_type:fruit_orchard
			NOT = {
				OR = {
					climate = mediterranean
					climate = subtropical
					climate = tropical
				}
			}
		}

		every_buildings_in_location = {
			limit = { building_type = building_type:fruit_orchard }
			location = { destroy_building = prev }
		}
	}

	# All ownable locations: recalculate eligibility cache
	every_location_in_the_world = {
		limit = { is_ownable = yes exists = raw_material }
		sul_specialization_calculate_eligibility = yes
	}

	# Market cache
	sul_market_cache_refresh_all = yes
}

# root = location (eligibility already calculated). Two paths:
#   has custom rank  -> validate, reassign if invalid
#   no custom rank   -> first-time init

sul_specialization_rebuild_location = {
	if = {
		limit = { sul_has_any_specialization = yes }
		if = {
			limit = { NOT = { sul_current_specialization_is_valid = yes } }
			sul_specialization_reassign = yes
		}
	}
	else = {
		sul_specialization_init_new = yes
	}
}

# Bridge: set sul_spec_type from rank boolean modifiers for GUI visibility checks.
# Call after any rank change.
sul_specialization_update_designation = {
	if = { limit = { modifier:sul_mining_specialization = yes }
		set_variable = { name = sul_spec_type value = @sul_mining_type }
	}
	else_if = { limit = { modifier:sul_farming_specialization = yes }
		set_variable = { name = sul_spec_type value = @sul_farming_type }
	}
	else_if = { limit = { modifier:sul_gathering_specialization = yes }
		set_variable = { name = sul_spec_type value = @sul_gathering_type }
	}
	else_if = { limit = { modifier:sul_woodland_specialization = yes }
		set_variable = { name = sul_spec_type value = @sul_woodland_type }
	}
	else_if = { limit = { modifier:sul_commercial_specialization = yes }
		set_variable = { name = sul_spec_type value = @sul_commercial_type }
	}
}

# root = location. Used when the current spec is no longer valid: picks best
# match from resources via change_location_rank, rebuilds buildings.

sul_specialization_reassign = {
	sul_rank_assign_by_resources = yes
	sul_specialization_tech_cleanup = yes
}

# root = location. First-time setup for a location that has never been specialized.

sul_specialization_init_new = {
	sul_rank_assign_by_resources = yes
	if = {
		limit = { exists = owner }
		sul_specialization_tech_cleanup = yes
	}
}

# root = location. Caches which specializations the location is eligible for.
# Called on game start, colonization, and raw material change.

sul_specialization_calculate_eligibility = {
	# Mining: mining resources OR mountains OR wetlands OR lake adjacency
	if = {
		limit = {
			OR = {
				sul_has_mining_resources = yes
				topography = mountains
				topography = wetlands
				is_adjacent_to_lake = yes
			}
		}

		set_variable = { name = sul_can_mine value = 1 }
	}
	else = { remove_variable = sul_can_mine }

	# Farming: farming resources OR farmland OR flatland with river/grasslands
	if = {
		limit = {
			OR = {
				sul_has_farming_resources = yes
				vegetation = farmland
				AND = {
					topography = flatland
					has_river = yes
				}
				AND = {
					topography = flatland
					vegetation = grasslands
				}
			}
		}

		set_variable = { name = sul_can_farm value = 1 }
	}
	else = { remove_variable = sul_can_farm }

	# Gathering: gathering resources OR coastal OR river OR lake adjacency
	if = {
		limit = {
			OR = {
				sul_has_gathering_resources = yes
				is_coastal = yes
				has_river = yes
				is_adjacent_to_lake = yes
			}
		}

		set_variable = { name = sul_can_gather value = 1 }
	}
	else = { remove_variable = sul_can_gather }

	# Woodland: woodland resources OR forest/woods/jungle
	if = {
		limit = {
			OR = {
				sul_has_woodland_resources = yes
				vegetation = forest
				vegetation = woods
				vegetation = jungle
			}
		}

		set_variable = { name = sul_can_woodland value = 1 }
	}
	else = { remove_variable = sul_can_woodland }
}

# root = location. Re-checks current spec after a raw material change;
# reassigns if current spec is no longer valid.

sul_specialization_validate = {
	if = {
		limit = { NOT = { sul_current_specialization_is_valid = yes } }
		sul_specialization_reassign_or_commercial = yes
	}
}

sul_specialization_reassign_or_commercial = {
	if = {
		limit = { is_market_center = yes }
		sul_specialization_switch_to_commercial = yes
	}
	else = {
		sul_specialization_calculate_eligibility = yes
		sul_specialization_reassign = yes
	}
}

sul_specialization_switch_to_commercial = {
	sul_rank_set_commercial = yes
}

# root = location. Sets per-good vars (1-day expiry) that mirror the
# Specialized Urban Locations - Building Destruction Effects
# Each effect destroys all specialization-relevant buildings EXCEPT those matching the specialization
# root = location
# NOTE: Universities are always preserved in cities regardless of specialization

sul_specialization_destroy_non_mining_buildings = {
	every_buildings_in_location = {
		limit = {
			sul_is_destroyable_specialization_building = yes
			NOT = {
				sul_is_mining_building = yes
			}
			NAND = {
				building_type = building_type:university
				location = { modifier:sul_is_city_rank_tier = yes }
			}
		}

		location = { destroy_building = prev }
	}
}

# Count non-matching buildings, destroy them, apply respec penalty scaled by count.
# root = location with rank already set to the NEW spec.

sul_specialization_destroy_and_penalize = {
	set_variable = { name = sul_respec_destroyed value = total_building_levels }

	sul_specialization_destroy_non_matching_buildings = yes

	change_variable = { name = sul_respec_destroyed subtract = total_building_levels }

	if = {
		limit = { var:sul_respec_destroyed > 0 }
		add_location_modifier = {
			modifier = sul_recently_respecialized
			years = 5
			size = var:sul_respec_destroyed
		}
	}
	else = {
		add_location_modifier = { modifier = sul_recently_respecialized years = 5 }
	}

	remove_variable = sul_respec_destroyed
}

# Dispatcher to the per-spec destroy effect. root = location.

sul_specialization_destroy_non_matching_buildings = {
	if = {
		limit = { sul_has_mining_specialization = yes }
		sul_specialization_destroy_non_mining_buildings = yes
	}
	else_if = {
		limit = { sul_has_farming_specialization = yes }
		sul_specialization_destroy_non_farming_buildings = yes
	}
	else_if = {
		limit = { sul_has_woodland_specialization = yes }
		sul_specialization_destroy_non_woodland_buildings = yes
	}
	else_if = {
		limit = { sul_has_gathering_specialization = yes }
		sul_specialization_destroy_non_gathering_buildings = yes
	}
	else_if = {
		limit = { sul_has_commercial_specialization = yes }
		sul_specialization_destroy_non_commercial_buildings = yes
	}
}

# root = location (any rank). Priority order:
# Commercial (market centers) > Farming > Mining > Gathering > Woodland.
# Sets specialization via change_location_rank, preserving current tier.

sul_rank_assign_by_resources = {
	if = {
		limit = { is_market_center = yes }
		sul_rank_set_commercial = yes
	}
	else_if = {
		limit = { sul_has_farming_resources = yes }
		sul_rank_set_farming = yes
	}
	else_if = {
		limit = { sul_has_mining_resources = yes }
		sul_rank_set_mining = yes
	}
	else_if = {
		limit = { sul_has_gathering_resources = yes }
		sul_rank_set_gathering = yes
	}
	else_if = {
		limit = { sul_has_woodland_resources = yes }
		sul_rank_set_woodland = yes
	}
	else = {
		sul_rank_set_commercial = yes
	}
}

# root = location. Destroys pre-placed buildings the current owner can't build
# due to missing advances. Called from sul_specialization_full_rebuild before
# the spec-mismatch destroy pass, so tech-invalid buildings are cleared first.
# The setup generator places a universal pool everywhere; this effect prunes
# the ones the country hasn't unlocked.
sul_specialization_tech_cleanup = {
	every_buildings_in_location = {
		limit = {
			OR = {
				AND = {
					building_type = building_type:sul_iron_mine
					location = { owner ?= { NOT = { has_advance = iron_working } } }
				}
				AND = {
					building_type = building_type:weapon_guild
					location = { owner ?= { NOT = { has_advance = weapon_guild_advance } } }
				}
				AND = {
					building_type = building_type:tools_guild
					location = { owner ?= { NOT = { has_advance = metallurgy } } }
				}
				AND = {
					building_type = building_type:porcelain_guild
					location = { owner ?= { NOT = { has_advance = porcelain_kiln_advance } } }
				}
				AND = {
					building_type = building_type:apothecary
					location = { owner ?= { NOT = { has_advance = apothecary_advance } } }
				}
				AND = {
					building_type = building_type:lacquerware_guild
					location = { owner ?= { NOT = { has_advance = lacquerware_guild_advance } } }
				}
				AND = {
					building_type = building_type:paper_guild
					location = { owner ?= { NOT = { has_advance = paper_guild_advance } } }
				}
				AND = {
					building_type = building_type:scriptorium
					location = { owner ?= { NOT = { has_advance = scriptorium_advance } } }
				}
			}
		}

		location = { destroy_building = prev }
	}
}
