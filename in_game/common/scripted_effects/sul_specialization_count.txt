# Specialization Count - Per-country diversity tracking
# Maintains per-country counts of how many locations hold each specialization,
# plus a weighted total used by diversity-pressure calculations.
#
# Country variables maintained here (do NOT rename - persisted in saves):
#   sul_mining_count, sul_farming_count, sul_woodland_count,
#   sul_gathering_count, sul_commercial_count, sul_total_weighted
#
# Effects defined here:
#   sul_specialization_count_initialize - full recount from owned locations
#   sul_specialization_count_increment  - +1 for the spec on scope:sul_loc
#   sul_specialization_count_decrement  - -1 for the spec on scope:sul_loc
#
# Constants are file-local in PDX script - redeclared here to keep this
# file self-contained.

@sul_mining_type = 1
@sul_farming_type = 2
@sul_gathering_type = 3
@sul_woodland_type = 4
@sul_commercial_type = 5
@sul_farming_weight = 2

# =============================================================================
# INITIALIZE COUNTS
# Counts all existing specializations and populates cache variables. Called
# when variables don't exist (first run after mod update or existing saves).
# root = country
# =============================================================================

sul_specialization_count_initialize = {
	set_variable = { name = sul_mining_count value = 0 }
	set_variable = { name = sul_farming_count value = 0 }
	set_variable = { name = sul_woodland_count value = 0 }
	set_variable = { name = sul_gathering_count value = 0 }
	set_variable = { name = sul_commercial_count value = 0 }
	set_variable = { name = sul_total_weighted value = 0 }

	every_owned_location = {
		limit = { has_variable = sul_spec_type }

		if = {
			limit = { var:sul_spec_type = @sul_mining_type }

			owner = {
				change_variable = { name = sul_mining_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { var:sul_spec_type = @sul_farming_type }

			owner = {
				change_variable = { name = sul_farming_count add = 1 }
				change_variable = { name = sul_total_weighted add = @sul_farming_weight }
			}
		}
		else_if = {
			limit = { var:sul_spec_type = @sul_woodland_type }

			owner = {
				change_variable = { name = sul_woodland_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { var:sul_spec_type = @sul_gathering_type }

			owner = {
				change_variable = { name = sul_gathering_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
		else_if = {
			limit = { var:sul_spec_type = @sul_commercial_type }

			owner = {
				change_variable = { name = sul_commercial_count add = 1 }
				change_variable = { name = sul_total_weighted add = 1 }
			}
		}
	}
}

# =============================================================================
# DECREMENT SPEC COUNT
# Country scope. Decrements count for location saved as scope:sul_loc.
# Call from country scope after: save_temporary_scope_as = sul_loc (from location)
# =============================================================================

sul_specialization_count_decrement = {
	if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_mining_type } }
		change_variable = { name = sul_mining_count subtract = 1 }
		change_variable = { name = sul_total_weighted subtract = 1 }
	}
	else_if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_farming_type } }
		change_variable = { name = sul_farming_count subtract = 1 }
		change_variable = { name = sul_total_weighted subtract = @sul_farming_weight }
	}
	else_if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_woodland_type } }
		change_variable = { name = sul_woodland_count subtract = 1 }
		change_variable = { name = sul_total_weighted subtract = 1 }
	}
	else_if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_gathering_type } }
		change_variable = { name = sul_gathering_count subtract = 1 }
		change_variable = { name = sul_total_weighted subtract = 1 }
	}
	else_if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_commercial_type } }
		change_variable = { name = sul_commercial_count subtract = 1 }
		change_variable = { name = sul_total_weighted subtract = 1 }
	}
}

# =============================================================================
# INCREMENT SPEC COUNT
# Country scope. Increments count for location saved as scope:sul_loc.
# Call from country scope after: save_temporary_scope_as = sul_loc (from location)
# =============================================================================

sul_specialization_count_increment = {
	if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_mining_type } }
		change_variable = { name = sul_mining_count add = 1 }
		change_variable = { name = sul_total_weighted add = 1 }
	}
	else_if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_farming_type } }
		change_variable = { name = sul_farming_count add = 1 }
		change_variable = { name = sul_total_weighted add = @sul_farming_weight }
	}
	else_if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_woodland_type } }
		change_variable = { name = sul_woodland_count add = 1 }
		change_variable = { name = sul_total_weighted add = 1 }
	}
	else_if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_gathering_type } }
		change_variable = { name = sul_gathering_count add = 1 }
		change_variable = { name = sul_total_weighted add = 1 }
	}
	else_if = {
		limit = { scope:sul_loc = { var:sul_spec_type = @sul_commercial_type } }
		change_variable = { name = sul_commercial_count add = 1 }
		change_variable = { name = sul_total_weighted add = 1 }
	}
}
