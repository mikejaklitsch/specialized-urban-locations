# Combat Overcrowding - Core Effects
#
# Penalizes the larger side's frontage when the smaller side can't fill
# the battle's max frontage. Scales with age:
#   Age 1 (Traditions):  125% of smaller army
#   Age 2 (Renaissance): 130%
#   Age 3 (Discovery):   135%
#   Age 4 (Reformation): 140%
#   Age 5 (Absolutism):  145%
#   Age 6 (Revolutions): 150%
#
# Counts sub_unit frontage on flanks (left + center + right).
# Both sides get a penalty clamping their frontage to their effective value.
#
# Optimization: caches counts on the character (supreme commander) and
# skips the expensive clear/apply when counts are unchanged.

# Helper: add a sub_unit's frontage to a counter variable.
# Called in sub_unit scope. $SIDE$ is the local_variable name to increment.

sul_war_add_subunit_frontage = {
	if = {
		limit = { sub_unit_type = unit_type:a_schiltron }
		change_local_variable = { name = $SIDE$ add = 0.75 }
	}
	else = { change_local_variable = { name = $SIDE$ add = 1 } }
}

sul_war_battle_frontage_check = {
	# root = character (supreme commander in battle)
	save_temporary_scope_as = sul_war_commander

	location = {
		if = {
			limit = { has_combat = yes }
			combat ?= { sul_war_calculate_frontage_penalty = yes }
		}
	}
}

sul_war_calculate_frontage_penalty = {
	# root = combat scope

	# Save both combat sides
	combat_attacker ?= { save_temporary_scope_as = sul_war_atk }

	combat_defender ?= { save_temporary_scope_as = sul_war_def }

	# Count frontage on attacker flanks (left + center + right)
	set_local_variable = { name = sul_war_atk_count value = 0 }

	scope:sul_war_atk = {
		every_center = { sul_war_add_subunit_frontage = { SIDE = sul_war_atk_count } }

		every_left_flank = {
			sul_war_add_subunit_frontage = { SIDE = sul_war_atk_count }
		}

		every_right_flank = {
			sul_war_add_subunit_frontage = { SIDE = sul_war_atk_count }
		}
	}

	# Count frontage on defender flanks (left + center + right)
	set_local_variable = { name = sul_war_def_count value = 0 }

	scope:sul_war_def = {
		every_center = { sul_war_add_subunit_frontage = { SIDE = sul_war_def_count } }

		every_left_flank = {
			sul_war_add_subunit_frontage = { SIDE = sul_war_def_count }
		}

		every_right_flank = {
			sul_war_add_subunit_frontage = { SIDE = sul_war_def_count }
		}
	}

	# Lift counts into scope values so they're accessible across scope switches
	save_temporary_scope_value_as = {
		name = sul_war_cur_atk
		value = local_var:sul_war_atk_count
	}
	save_temporary_scope_value_as = {
		name = sul_war_cur_def
		value = local_var:sul_war_def_count
	}

	# Skip if counts haven't changed since last application.
	# sul_war_in_battle flag ensures stale variables from prior battles don't match.
	if = {
		limit = {
			scope:sul_war_commander = {
				has_variable = sul_war_in_battle
				has_variable = sul_war_prev_atk
				has_variable = sul_war_prev_def
				var:sul_war_prev_atk = scope:sul_war_cur_atk
				var:sul_war_prev_def = scope:sul_war_cur_def
			}
		}
	}
	else = {
		# Cache counts and set active-battle flag on the commander
		scope:sul_war_commander = {
			set_variable = { name = sul_war_in_battle value = yes }
			set_variable = { name = sul_war_prev_atk value = scope:sul_war_cur_atk }
			set_variable = { name = sul_war_prev_def value = scope:sul_war_cur_def }
		}

		# Determine age multiplier
		set_local_variable = { name = sul_war_age_mult value = 1.25 }
		if = {
			limit = { current_later_age_2_renaissance_trigger = yes }
			set_local_variable = { name = sul_war_age_mult value = 1.30 }
		}
		if = {
			limit = { current_later_age_3_discovery_trigger = yes }
			set_local_variable = { name = sul_war_age_mult value = 1.35 }
		}
		if = {
			limit = { current_later_age_4_reformation_trigger = yes }
			set_local_variable = { name = sul_war_age_mult value = 1.40 }
		}
		if = {
			limit = { current_later_age_5_absolutism_trigger = yes }
			set_local_variable = { name = sul_war_age_mult value = 1.45 }
		}
		if = {
			limit = { current_later_age_6_revolutions_trigger = yes }
			set_local_variable = { name = sul_war_age_mult value = 1.50 }
		}

		# Determine smaller/larger
		if = {
			limit = { local_var:sul_war_atk_count >= local_var:sul_war_def_count }

			set_local_variable = {
				name = sul_war_larger
				value = local_var:sul_war_atk_count
			}
			set_local_variable = {
				name = sul_war_smaller
				value = local_var:sul_war_def_count
			}
		}
		else = {
			set_local_variable = {
				name = sul_war_larger
				value = local_var:sul_war_def_count
			}
			set_local_variable = {
				name = sul_war_smaller
				value = local_var:sul_war_atk_count
			}
		}

		# Penalty: 1 - (smaller * age_mult / larger)
		set_local_variable = {
			name = sul_war_penalty

			value = {
				add = 1

				subtract = {
					add = local_var:sul_war_smaller
					multiply = local_var:sul_war_age_mult
					divide = { value = local_var:sul_war_larger min = 0.01 }
				}

				min = 0
			}
		}

		# Clear existing penalty from both sides
		scope:sul_war_atk = {
			every_participating_units = {
				limit = { has_unit_modifier = sul_war_frontage_penalty }
				remove_unit_modifier = sul_war_frontage_penalty
			}
		}

		scope:sul_war_def = {
			every_participating_units = {
				limit = { has_unit_modifier = sul_war_frontage_penalty }
				remove_unit_modifier = sul_war_frontage_penalty
			}
		}

		# Apply penalty to the larger side only
		if = {
			limit = { local_var:sul_war_atk_count >= local_var:sul_war_def_count }

			scope:sul_war_atk = {
				every_participating_units = {
					add_unit_modifier = {
						modifier = sul_war_frontage_penalty
						days = 5
						size = local_var:sul_war_penalty
					}
				}
			}
		}
		else = {
			scope:sul_war_def = {
				every_participating_units = {
					add_unit_modifier = {
						modifier = sul_war_frontage_penalty
						days = 5
						size = local_var:sul_war_penalty
					}
				}
			}
		}
	}
}
