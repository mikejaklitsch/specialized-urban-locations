# Decluttered War UI — Scripted GUIs

# ---- CMF Action Bar: Toggle settings panel ----

dwui_settings_config = {
	scope = country

	is_shown = {
		is_ai = no
	}

	effect = {
		if = {
			limit = { has_variable = dwui_show_panel }
			remove_variable = dwui_show_panel
		}
		else = { set_variable = { name = dwui_show_panel value = yes } }
	}
}
