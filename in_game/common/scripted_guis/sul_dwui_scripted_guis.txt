# SUL — Ported Decluttered War UI Scripted GUIs
# Internal identifiers prefixed sul_dwui_ to coexist with standalone DWUI mod.
# Country-scope state variables keep the dwui_ prefix so both mods share state.

# ==== Panel toggle ====

sul_dwui_settings_config = {
	scope = country

	is_shown = { is_ai = no }

	is_valid = { is_ai = no }

	effect = {
		if = { limit = { has_variable = sul_dwui_show_panel } remove_variable = sul_dwui_show_panel }
		else = { set_variable = { name = sul_dwui_show_panel value = yes } }
	}
}

# ==== Fort icon toggle (vanilla generic vs location-specific fort building) ====
