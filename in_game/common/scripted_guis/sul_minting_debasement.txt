# ============================================
# CURRENCY DEBASEMENT SCRIPTED GUIS
# Range: 0 to 2.00. Setting debasement resets rebasement (mutually exclusive).
# ============================================

sul_minting_debasement_inc_small = {
	scope = country

	is_valid = { has_variable = sul_minting_debasement_level var:sul_minting_debasement_level < 2.0 }

	effect = {
		sul_minting_adjust_currency = {
			type = debasement
			other = rebasement
			delta = 0.01
		}
	}
}
