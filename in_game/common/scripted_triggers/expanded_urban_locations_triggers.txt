# ============================================
# SPECIALIZATION DESIGNATION TRIGGERS
# ============================================
# These triggers check the specialization_designation variable.
# We use NUMERIC values (not flags) because the GUI needs to read
# this variable via GetVariable().GetValue, which only works with numbers.
# Using triggers keeps the script code readable while allowing GUI access.
#
# Number mapping:
#   Towns:  1=mining, 2=farming, 3=gathering, 4=forestry, 5=commercial
#   Cities: 11=mining, 12=farming, 13=gathering, 14=forestry, 15=commercial
# ============================================

# Individual town checks (using constants from expanded_urban_locations_specialization_constants.txt)
is_mining_town = {
	has_variable = specialization_designation
	var:specialization_designation = eul_mining_town_value
}

	var:specialization_designation = eul_farming_town_value
}

is_gathering_town = {
