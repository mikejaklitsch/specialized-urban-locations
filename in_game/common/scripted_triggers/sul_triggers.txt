# VALID LAND LOCATION CHECK
# Used to filter out sea and wastelands for eligibility calculation
# Note: is_land = yes already excludes lakes and ocean
# ============================================

	is_land = yes

		# Wastelands

# ============================================
# BUILDING CLASSIFICATION TRIGGERS
# root = building
# Used by destruction effects to determine if a building belongs to a specialization
# Optimized: uses building_produced_goods for production buildings (72% fewer checks)
# ============================================

# Mining buildings: raw ore enhancement + metal/tool production
sul_is_mining_building = {
	OR = {
		# Enhancement buildings (no produced goods - must be explicit)
		building_type = building_type:local_smelters
		building_type = building_type:mercury_patio
		# Production buildings (check by output goods)
		building_produced_goods = goods:iron
		building_produced_goods = goods:stone
		building_produced_goods = goods:tools
		building_produced_goods = goods:weaponry
		building_produced_goods = goods:cannons
		building_produced_goods = goods:firearms

# ============================================
# BUILDING DESTRUCTION TRIGGER
# root = building
# Returns yes if this building type can be destroyed when assigning a specialization
# Optimized: uses specialization triggers instead of 336 explicit building checks
# ============================================

sul_is_destroyable_specialization_building = {
	# Never destroy village buildings (peasant-worked rural buildings)
	NOT = { building_category = village_category }
	OR = {
		sul_is_mining_building = yes
		sul_is_farming_building = yes
		sul_is_woodland_building = yes
		sul_is_gathering_building = yes
		sul_is_commercial_building = yes
	}
}

# These triggers check the specialization_designation variable.
# We use NUMERIC values (not flags) because the GUI needs to read
# this variable via GetVariable().GetValue, which only works with numbers.
# Using triggers keeps the script code readable while allowing GUI access.
#
# Number mapping:
#   Rural:  21=mining, 22=farming, 23=gathering, 24=woodland, 25=commercial
#   Towns:  1=mining, 2=farming, 3=gathering, 4=woodland, 5=commercial
#   Cities: 11=mining, 12=farming, 13=gathering, 14=woodland, 15=commercial
# ============================================
