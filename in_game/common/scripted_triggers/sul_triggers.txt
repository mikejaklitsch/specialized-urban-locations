# Check if current specialization is still allowed under current eligibility rules
# root = location (must have sul_spec_type and eligibility variables calculated)

sul_current_specialization_is_valid = {
	OR = {
		AND = {
			sul_has_mining_specialization = yes
			has_variable = sul_can_mine
		}

		AND = {
			sul_has_farming_specialization = yes
			has_variable = sul_can_farm
		}

		AND = {
			sul_has_gathering_specialization = yes
			has_variable = sul_can_gather
		}

		AND = {
			sul_has_woodland_specialization = yes
			has_variable = sul_can_woodland
		}

		# Commercial is always valid (fallback specialization)
		sul_has_commercial_specialization = yes
	}
}

# Check if location can respecialize (no cooldown active)

sul_can_respecialize = {
	NOT = {
		modifier:sul_can_respecialize = no
	}
}

# Check if country is a valid location owner (for country iterations)
# Filters out invalid/special country types and countries with no locations

sul_is_valid_location_owner = {
	exists = yes
	OR = {
		country_type = location
		country_type = army
		country_type = navy
	}

	num_locations > 0
}
