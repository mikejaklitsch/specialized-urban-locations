# ============================================
# BUILDING DESTRUCTION TRIGGER
# root = building
# Returns yes if this building type can be destroyed when assigning a specialization
# Optimized: uses specialization triggers instead of 336 explicit building checks
# ============================================

sul_is_destroyable_specialization_building = {
	# Never destroy village buildings (peasant-worked rural buildings)
	NOT = { building_category = village_category }
	OR = {
		sul_is_mining_building = yes
		sul_is_farming_building = yes
		sul_is_woodland_building = yes
		sul_is_gathering_building = yes
		sul_is_commercial_building = yes
	}
}

# ============================================
# SPECIALIZATION DESIGNATION TRIGGERS
# ============================================
# These triggers check the sul_specialization_designation variable.
# We use NUMERIC values (not flags) because the GUI needs to read
# this variable via GetVariable().GetValue, which only works with numbers.
# Using triggers keeps the script code readable while allowing GUI access.
#
# Number mapping:
#   Rural:  21=mining, 22=farming, 23=gathering, 24=woodland, 25=commercial
#   Towns:  1=mining, 2=farming, 3=gathering, 4=woodland, 5=commercial
#   Cities: 11=mining, 12=farming, 13=gathering, 14=woodland, 15=commercial
# ============================================
