# AI Builds Navies — England flavor event override
# Replaces dock construction with sul_dock

namespace = flavor_eng

flavor_eng.70 = {
	type = country_event
	fire_only_once = yes
	title = flavor_eng.70.title
	desc = flavor_eng.70.desc

	dynamic_historical_event = {
		tag = ENG
		tag = GBR
		from = 1350.1.1
		to = 1500.1.1
		monthly_chance = 1
	}

	trigger = {
		province_definition:kent_province = {
			any_location_in_province_definition = {
				count > 1
				owner ?= root
				has_building_with_at_least_one_level = cinque_port
			}
		}
		OR = {
			location:southampton = {
				owner ?= root
				NOR = {
					has_building_with_at_least_one_level = dock
					has_building_with_at_least_one_level = sul_dock
				}
			}

			location:portsmouth = {
				owner ?= root
				NOR = {
					has_building_with_at_least_one_level = dock
					has_building_with_at_least_one_level = sul_dock
				}
			}

			location:rochester = {
				owner ?= root
				NOR = {
					has_building_with_at_least_one_level = dock
					has_building_with_at_least_one_level = sul_dock
				}
			}
		}
	}

	illustration_tags = { 10 = regular 10 = interior }

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:nobles_estate
		}

		province_definition:kent_province = {
			random_location_in_province_definition = {
				limit = { owner ?= root has_building_with_at_least_one_level = cinque_port }
				destroy_building = "building(building_type:cinque_port|owner)"
				save_scope_as = target_location
			}
		}
	}

	option = {
		name = flavor_eng.70.a

		trigger = {
			location:southampton = {
				owner ?= root
				NOR = {
					has_building = building_type:dock
					has_building = building_type:sul_dock
				}
			}
		}

		historical_option = yes

		location:southampton = {
			construct_building = { building_type = building_type:sul_dock }
		}
	}

	option = {
		name = flavor_eng.70.b

		trigger = {
			location:portsmouth = {
				owner ?= root
				NOR = {
					has_building = building_type:dock
					has_building = building_type:sul_dock
				}
			}
		}

		location:portsmouth = {
			construct_building = { building_type = building_type:sul_dock }
		}
	}

	option = {
		name = flavor_eng.70.c

		trigger = {
			location:rochester = {
				owner ?= root
				NOR = {
					has_building = building_type:dock
					has_building = building_type:sul_dock
				}
			}
		}

		location:rochester = {
			construct_building = { building_type = building_type:sul_dock }
		}
	}

	option = {
		name = flavor_eng.70.d
		custom_tooltip = flavor_eng.70.tt1
		if = {
			limit = {
				NOT = {
					has_variable = lord_warden_show
				}
			}

			custom_tooltip = flavor_eng.70.tt2
		}
		set_variable = { name = decline_of_cinque_ports value = 1 }
	}

	historical_info = flavor_eng.70.historical_info
}
