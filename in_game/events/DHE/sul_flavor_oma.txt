# AI Builds Navies — Oman flavor event override
# Replaces dock checks with sul_dock

namespace = flavor_oma

flavor_oma.8 = {
	hide_portraits = yes
	type = country_event
	title = flavor_oma.8.title
	desc = flavor_oma.8.desc
	fire_only_once = yes

	dynamic_historical_event = {
		tag = OMA
		from = 1500.1.1
		to = 1700.1.1
		monthly_chance = 5
	}

	trigger = {
		OR = {
			total_effective_building_levels:dock > 0
			total_effective_building_levels:sul_dock > 0
		}
		any_cabinet_character = { count > 0 }
		societal_value:land_vs_naval >= 0.3
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_incense.dds"

	immediate = {
		ruler_or_regent ?= { save_scope_as = target_character }

		ordered_owned_location = {
			limit = {
				OR = {
					has_building = building_type:dock
					has_building = building_type:sul_dock
				}
			}

			order_by = population
			max = 1
			check_range_bounds = no
			save_scope_as = target_location
		}

		ordered_cabinet_character = {
			order_by = dip
			max = 1
			check_range_bounds = no
			save_scope_as = target_character2
		}
	}

	option = {
		name = flavor_oma.8.a

		change_gold_effect = {
			scale = -2
		}

		scope:target_location = { change_development = development_mild_bonus }

		change_societal_value = {
			type = land_vs_naval
			value = societal_value_move_to_right
		}

		add_navy_tradition = navy_tradition_severe_bonus

		unlock_cabinet_action_effect = {
			type = oma_bolster_omani_navy
		}
	}

	option = {
		name = flavor_oma.8.b

		change_societal_value = {
			type = land_vs_naval
			value = societal_value_move_to_left
		}

		add_army_tradition = army_tradition_severe_bonus
	}
}
