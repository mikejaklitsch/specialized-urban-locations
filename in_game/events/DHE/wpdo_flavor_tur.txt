# AI Builds Navies — Turkey flavor event override
# Replaces naval_supplies_guild with wpdo_naval_supplies_guild

REPLACE:flavor_tur.15 = {
	type = country_event
	title = flavor_tur.15.title
	desc = flavor_tur.15.desc
	fire_only_once = yes

	dynamic_historical_event = {
		tag = TUR
		from = 1437.1.1
		to = 1537.1.1
		monthly_chance = 5
	}

	historical_info = flavor_tur.15.historical_info

	trigger = { tag = TUR owns = location:constantinople }

	option = {
		name = flavor_tur.15.a
		historical_option = yes

		location:constantinople = {
			construct_building = { building_type = building_type:imperial_halic_shipyards }
		}
		if = {
			limit = {
				location:constantinople = {
					OR = {
						has_building_with_at_least_one_level = naval_supplies_guild
						has_building_with_at_least_one_level = wpdo_naval_supplies_guild
					}
				}
			}

			location:constantinople = {
				change_building_level_in_location = {
					building = building_type:wpdo_naval_supplies_guild
					value = 1
				}
			}
		}
		else = {
			location:constantinople = {
				construct_building = {
					building_type = building_type:wpdo_naval_supplies_guild
					cost_multiplier = 0.5
					cost_multiplier_reason = "military_cheap_sponsorship"
				}
			}
		}
		change_societal_value = { type = land_vs_naval value = 5 }
	}

	option = {
		name = flavor_tur.15.b
		change_societal_value = { type = land_vs_naval value = 2 }

		capital.market = {
			add_temporary_demand = { type = demand:galley_construction months = 48 }
		}
	}

	option = {
		name = flavor_tur.15.c

		change_societal_value = {
			type = land_vs_naval
			value = societal_value_move_to_left
		}
	}
}
