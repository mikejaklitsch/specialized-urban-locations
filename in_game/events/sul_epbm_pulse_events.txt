namespace = sul_epbm_pulse

# EPBM player monthly: recalculate + charge
sul_epbm_pulse.0 = {
	type = country_event
	hidden = yes
	title = "sul_epbm_pulse.0"

	trigger = {
		has_global_variable = sul_epbm_version
		is_ai = no
		any_estate = { estate_type != estate_type:crown_estate }
	}

	immediate = {
		sul_epbm_collect_maintenance = yes
	}
}

# EPBM AI monthly: charge from saved rates
sul_epbm_pulse.1 = {
	type = country_event
	hidden = yes
	title = "sul_epbm_pulse.1"

	trigger = {
		has_global_variable = sul_epbm_version
		is_ai = yes
		is_real_country = yes
		any_estate = { estate_type != estate_type:crown_estate }
	}

	immediate = {

	}
}

# EPBM AI yearly: expensive maintenance recalculation
sul_epbm_pulse.2 = {
	type = country_event
	hidden = yes
	title = "sul_epbm_pulse.2"

	trigger = {
		has_global_variable = sul_epbm_version
		global_var:sul_epbm_version >= sul_epbm_version_value
		is_ai = yes
		is_real_country = yes
		any_estate = { estate_type != estate_type:crown_estate }
	}

	immediate = {
		sul_epbm_calculate_maintenance = yes
	}
}

# EPBM decade rebuild: structural integrity check every 10 years
sul_epbm_pulse.3 = {
	type = country_event
	hidden = yes
	title = "sul_epbm_pulse.3"

	trigger = {
		has_global_variable = sul_epbm_version
		is_real_country = yes
		any_estate = { estate_type != estate_type:crown_estate }
	}

	immediate = {
		if = {
			limit = {
				has_variable = sul_epbm_rebuild_counter
				var:sul_epbm_rebuild_counter >= 9
			}
			sul_epbm_structural_rebuild = yes
			set_variable = { name = sul_epbm_rebuild_counter value = 0 }
		}
		else_if = {
			limit = { has_variable = sul_epbm_rebuild_counter }
			change_variable = { name = sul_epbm_rebuild_counter add = 1 }
		}
	}
}
