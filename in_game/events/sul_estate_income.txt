namespace = sul_estate_income

# Hidden event to calculate and grant control shortfall payments to estates
# Calculates total taxes lost from low/zero control locations, then splits by estate power
sul_estate_income.1 = {
	type = country_event
	title = sul_estate_income.1.title
	hidden = yes

	trigger = {
		always = yes
	}

	immediate = {
		every_estate = {
			limit = {
				estate_type != estate_type:crown_estate
				power > 0
			}
			sul_calculate_estate_income_lost = yes
			sul_grant_estate_income_lost = yes
		}
	}
}
