namespace = sul_estate_income

# Hidden event to calculate and grant control shortfall payments to estates
# Calculates total taxes lost from low/zero control locations, then splits by estate power
sul_estate_income.1 = {
	type = country_event
	title = sul_estate_income.1.title
	hidden = yes

	trigger = {
		is_ai = no
	}

	immediate = {
		# Calculate total taxes lost at country level first
		sul_calculate_total_taxes_lost_from_low_control = yes

		every_estate = {
			limit = {
				estate_type != estate_type:crown_estate
				power > 0
			}

			sul_calculate_estate_income_lost = yes

			estate_add_gold = { value = local_var:income_lost }
		}
	}
}

# AI version - runs yearly with 12x multiplier
sul_estate_income.2 = {
	type = country_event
	title = sul_estate_income.2.title
	hidden = yes

	trigger = {
		is_ai = yes
	}

	immediate = {
		# Calculate total taxes lost at country level first
		sul_calculate_total_taxes_lost_from_low_control = yes

		every_estate = {
			limit = {
				estate_type != estate_type:crown_estate
				power > 0
			}

			sul_calculate_estate_income_lost = yes

			estate_add_gold = {
				value = {
					add = local_var:income_lost
					multiply = 12
				}
			}
		}
	}
}
