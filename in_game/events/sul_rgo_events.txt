namespace = sul_rgo_pulse

# Delayed RGO rebuild after destruction (e.g. bankruptcy).
# 1-day delay lets the engine finish cleanup before we check levels.
sul_rgo_pulse.2 = {
	type = location_event
	hidden = yes
	title = "sul_rgo_pulse.2"

	immediate = {
		if = {
			limit = {
				exists = raw_material
				sul_location_rgo_building_level <= 0
				modifier:sul_rgo_size > 0
			}

			"global_variable_map(sul_rgo_map|raw_material)" = {
				save_temporary_scope_as = sul_rgo_btype
			}

			if = {
				limit = { can_build_building = scope:sul_rgo_btype }

				change_building_level_in_location = {
					building = scope:sul_rgo_btype
					value = 1
				}
			}
		}
	}
}

# Yearly RGO trim
sul_rgo_pulse.0 = {
	type = country_event
	hidden = yes
	title = "sul_rgo_pulse.0"

	trigger = { is_real_country = yes }

	immediate = {
		every_owned_location = {
			limit = {
				exists = raw_material
				sul_location_rgo_building_level > sul_location_max_rgo_building_level_rounded
			}
			sul_trim_rgo_levels = yes
		}
	}
}

# Cleanup dead AI army units
sul_rgo_pulse.1 = {
	type = country_event
	hidden = yes
	title = "sul_rgo_pulse.1"

	immediate = {
		if = {
			limit = { is_ai = yes }
			every_army = { limit = { strength_percentage <= 0 } destroy_unit = yes }
		}
	}
}
