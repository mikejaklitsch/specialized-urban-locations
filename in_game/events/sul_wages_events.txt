namespace = sul_wages

# Player wages: accumulate + pay monthly
sul_wages.0 = {
	type = country_event
	hidden = yes
	title = "sul_wages.0"

	trigger = {
		is_ai = no
		is_real_country = yes
		has_variable = sul_spending_nobles
	}

	immediate = {
		sul_accumulate_wages = yes
		sul_estate_wages_pay_all = yes
	}
}

# AI wages: pay from yearly-cached rates
sul_wages.1 = {
	type = country_event
	hidden = yes
	title = "sul_wages.1"

	trigger = {
		is_ai = yes
		is_real_country = yes
		has_variable = sul_monthly_wages_nobles
	}

	immediate = {
		sul_estate_wages_pay_all = yes
	}
}

# Yearly wage recalc + AI accumulation (merged — AI needs fresh rates)
sul_wages.2 = {
	type = country_event
	hidden = yes
	title = "sul_wages.2"

	trigger = { is_real_country = yes }

	immediate = {
		every_owned_location = { sul_calculate_location_wages = yes }

		if = {
			limit = {
				is_ai = yes
				has_variable = sul_spending_nobles
			}
			sul_accumulate_wages = yes
		}
	}
}
