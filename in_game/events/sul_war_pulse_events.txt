namespace = sul_war_pulse

# War monthly pulse
sul_war_pulse.0 = {
	type = country_event
	hidden = yes
	title = "sul_war_pulse.0"

	trigger = {
		OR = {
			at_war = yes
			has_variable = sul_war_momentum_army_size
		}
	}

	immediate = {
		sul_war_monthly_update = yes
	}
}

# Force rebellions during war
sul_war_pulse.1 = {
	type = country_event
	hidden = yes
	title = "sul_war_pulse.1"

	trigger = { at_war = yes }

	immediate = {
		every_rebel = {
			limit = { rebel_progress >= 1.0 }
			start_civil_war = {}
		}
	}
}
