# ============================================
# EXPANDED URBAN LOCATIONS - MAP MODE OVERRIDE
# Adds specialization overlay colors to the cities map mode
# ============================================

# Variable definitions (copied from base game)
@zoom_step_near = 2

@data_gradient_alpha_inside = 1
@data_gradient_alpha_outside = 1
@data_gradient_width = 0.875
@data_gradient_color_mult = 0.9
@data_edge_width = 0
@data_edge_sharpness = 0.01
@data_edge_alpha = 0
@data_edge_color_mult = 0
@data_before_lighting_blend = 0.5
@data_after_lighting_blend = 0.5

REPLACE:cities = {
	# Primary color: specialization
	map_color = {
		if = {
			limit = {
				sul_has_mining_specialization = yes
			}
			value = rgb { 128 0 0 }
		}
		else_if = {
			limit = {
				sul_has_gathering_specialization = yes
			}
			value = rgb { 0 0 128 }
		}
		else_if = {
			limit = {
				sul_has_woodland_specialization = yes
			}
			value = rgb { 85 34 0 }
		}
		else_if = {
			limit = {
				sul_has_commercial_specialization = yes
			}
			value = rgb { 128 0 128 }
		}
		else_if = {
			limit = {
				sul_has_farming_specialization = yes
			}
			value = rgb { 34 85 0 }
		}
	}

	# Slashed line overlay: location rank colors
	# Guard with exists check to prevent errors on sea/wasteland tiles
	secondary_map_color = {
		if = {
			limit = {
				exists = location_rank
				location_rank = location_rank:city
			}
			value = rgb { 255 230 128 }
		}
		else_if = {
			limit = {
				exists = location_rank
				location_rank = location_rank:town
			}
			value = rgb { 40 40 40 }
		}
		# Rural: no stripes
	}

	tooltip_key = {
		if = {
			limit = { sul_has_mining_specialization = yes }
			value = "sul_mining_specialization"
		}
		else_if = {
			limit = { sul_has_gathering_specialization = yes }
			value = "sul_gathering_specialization"
		}
		else_if = {
			limit = { sul_has_woodland_specialization = yes }
			value = "sul_woodland_specialization"
		}
		else_if = {
			limit = { sul_has_commercial_specialization = yes }
			value = "sul_commercial_specialization"
		}
		else_if = {
			limit = { sul_has_farming_specialization = yes }
			value = "sul_farming_specialization"
		}
	}

	small_map_names = location
	medium_map_names = province
	large_map_names = country

	small_tooltip_context = location
	medium_tooltip_context = location
	large_tooltip_context = location

	fill_in_impassable = no
	flatmap_behaviour = Always
	use_fow = no

	category = economy
	index = 1

	map_markers = {
		market_marker = no
		fort_marker = no
		toll_marker = no
	}

	gradient_parameters = {
		zoom_step = @zoom_step_near

		gradient_alpha_inside = @data_gradient_alpha_inside
		gradient_alpha_outside = @data_gradient_alpha_outside
		gradient_width = @data_gradient_width
		gradient_color_mult = @data_gradient_color_mult
		edge_width = @data_edge_width
		edge_sharpness = @data_edge_sharpness
		edge_alpha = @data_edge_alpha
		edge_color_mult = @data_edge_color_mult
		before_lighting_blend = @data_before_lighting_blend
		after_lighting_blend = @data_after_lighting_blend
	}

	# Specialization legend keys (primary colors)
	legend_key = {
		desc = "EUL_LEGEND_MINING"
		color = rgb { 128 0 0 }
	}
	legend_key = {
		desc = "EUL_LEGEND_GATHERING"
		color = rgb { 0 0 128 }
	}
	legend_key = {
		desc = "EUL_LEGEND_WOODLAND"
		color = rgb { 85 34 0 }
	}
	legend_key = {
		desc = "EUL_LEGEND_COMMERCIAL"
		color = rgb { 128 0 128 }
	}
	legend_key = {
		desc = "EUL_LEGEND_FARMING"
		color = rgb { 34 85 0 }
	}

	# Location rank legend keys (secondary/stripes)
	legend_key = {
		desc = "EUL_LEGEND_CITY"
		color = rgb { 255 230 128 }
	}
	legend_key = {
		desc = "EUL_LEGEND_TOWN"
		color = rgb { 40 40 40 }
	}

	# Added Month for periodic specialization updates
	color_refresh_counters = { Month }
	color_and_names_refresh_counters = { LocationOwnerChanged CountryStatus }
}
