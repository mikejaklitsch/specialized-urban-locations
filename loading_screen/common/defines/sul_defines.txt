# Specialized Urban Locations - Building Upkeep Fix
# Vanilla sets BUILDING_UPKEEP_FACTOR = 0.2 (20%), which means countries only pay 20% of building upkeep
# Crown power modifiers adjust this, but starting from 20% makes the system ineffective
# This fix sets it to 100% so buildings cost their full upkeep, and crown power can meaningfully reduce it

NEconomy = {
	BUILDING_UPKEEP_FACTOR = 1.0
	# 0.5% production efficiency bonus per building level (vanilla is 1%)
	PRODUCTION_EFFIENCY_BONUS_PER_LEVEL = 0.005
	# Double inflation from gold/silver RGOs (vanilla is 0.01)
	INFLATION_RGO_IMPACT_FACTOR = 0.02
}
