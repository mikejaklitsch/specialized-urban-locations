# Specialized Urban Locations - Map Marker Icon Concepts
# Maps numeric keys to icon textures via GetConceptTexture()
#
# Key formula: sul_icon_{designation + capital_offset}
#   designation = sul_designation variable (type + rank encoding)
#     Town:  1=mining, 2=farming, 3=gathering, 4=woodland, 5=commercial
#     City:  11=mining, 12=farming, 13=gathering, 14=woodland, 15=commercial
#     Rural: 21=mining, 22=farming, 23=gathering, 24=woodland, 25=commercial
#   capital_offset: 0=regular, 100=province capital, 200=country capital
#
# Color mapping: red=mining, green=farming, blue=gathering, brown=woodland, purple=commercial
