# Game-start pop rush. Massively accelerates pop promotion for 45 days so
# starting buildings get filled before the economy ticks for the first time.

sul_init_pop_rush = {
	game_data = { category = country }

	global_pop_promotion_speed_modifier = 10000
	block_tribal_promotion = yes
}

# Decaying respec cooldown. Size = buildings destroyed during respec.
# Per-building: -2.5% build cost, +2.5% construction speed, +10% dev growth.
# Dev growth modifier amplifies spring decay, making inflated dev drop faster.
# sul_can_respecialize = no gates the respec UI.

sul_recently_respecialized = {
	game_data = { category = location decaying = yes }

	sul_can_respecialize = no
	local_construction_cost = -0.025
	local_construction_speed = 0.025
	local_monthly_development_modifier = 0.10
}
