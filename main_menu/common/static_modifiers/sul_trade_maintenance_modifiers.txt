# Trade maintenance carriers + vanilla source overrides.
#
# Two carriers because add_country_modifier size only takes positive values:
#   sul_trade_maintenance_reduction is applied with positive size when
#   efficiency is net-positive, sul_trade_maintenance_penalty when net-negative.
#   sul_trade_maintenance_apply picks one or the other each month based on the
#   sign of root.modifier:sul_trade_maintenance_efficiency.
#
# The vanilla TRY_INJECTs below cancel each vanilla `merchant_maintenance_cost`
# source and re-emit it as `sul_trade_maintenance_efficiency = -X`. The dynamic
# apply effect then converts the engine-summed efficiency back into a
# multiplicatively-compounded cost via the 1/(1+E) - 1 formula.

sul_trade_maintenance_reduction = {
	game_data = { category = country }

	merchant_maintenance_cost = -1.0
}

sul_trade_maintenance_penalty = {
	game_data = { category = country }

	merchant_maintenance_cost = 1.0
}
