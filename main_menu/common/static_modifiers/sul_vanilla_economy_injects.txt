# Vanilla static modifier injects for the SUL economy/food/RGO overhaul.
#
# These cancel vanilla's abstract local_monthly_food_modifier (and related
# RGO/development boosts) and replace them with concrete per-good output
# modifiers, so food and raw materials come from actual production rather
# than magic numbers.
#
# Each TRY_INJECT here is paired with a vanilla modifier of the same name -
# the engine merges the two blocks at load time. Removing one of these
# injects will silently re-enable the vanilla bypass for that source.

# ===================
# BUILDING LEVEL DRAG
# ===================
# Per total building level at a location - creates diminishing returns
# for building density, offset by per-building efficiency bonus (define).

TRY_INJECT:building_levels = {
	local_population_growth = -0.0001
	local_migration_attraction = 0.0001
	local_production_efficiency = -0.005
}

# ===================
# BASELINE FOOD CAPACITY
# ===================

# can survive without needing granary infrastructure.

	local_max_rgo_size = -2				# nullify vanilla +2 (RGO removal)
}

}

# ===================
# DEVELOPMENT & DEVASTATION - Food Modifier Removal
# ===================
# Vanilla development gives +0.01 local_monthly_food_modifier per dev point.
# Vanilla devastation gives +0.50 local_monthly_food_modifier.
# Both are abstract food boosts that bypass the provisions system.

TRY_INJECT:development = {
	local_monthly_food_modifier = -0.01          # nullify vanilla +0.01

}

	local_monthly_food_modifier = -0.5           # nullify vanilla +0.50
}

# Zero out vanilla RGO size scaling from blocks not touched by sul_food_modifiers.txt.
# These are load-bearing: without them, vanilla RGO workers would still spawn from
# population and rural development scaling.

INJECT:total_population = {
	local_max_rgo_size = -0.025				# nullify vanilla +0.025 - DO NOT CHANGE without verifying vanilla value
	local_pop_food_consumption = 0.0025		# each pop increases local food consumption
	sul_rgo_size = 0.04						# +1 per 25k pop (25 in code)
}

INJECT:rural_development_modifier = {
	local_max_rgo_size = -1			# nullify vanilla +1 - DO NOT CHANGE without verifying vanilla value
}
