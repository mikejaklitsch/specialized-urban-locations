# can survive without needing granary infrastructure.

# ===================
# DEVELOPMENT & DEVASTATION — Food Modifier Removal
# ===================
# Vanilla development gives +0.01 local_monthly_food_modifier per dev point.
# Vanilla devastation gives +0.50 local_monthly_food_modifier.
# Both are abstract food boosts that bypass the provisions system.

TRY_INJECT:development = {
	local_monthly_food_modifier = -0.01          # nullify vanilla +0.01
	local_max_rgo_size = -0.1                    # nullify vanilla +0.1 (RGO removal)
}

TRY_INJECT:devastation = {
	local_monthly_food_modifier = -0.5           # nullify vanilla +0.50
}
