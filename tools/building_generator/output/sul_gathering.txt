	is_foreign = no
	max_levels = rural_building_cap
	pop_type = laborers
	employment_size = rural_peasant_produce_employment
	rural_settlement = yes
	town = yes
	city = yes

	location_potential = {
		OR = {
			is_coastal = yes

			is_adjacent_to_lake = yes
		}
		NOT = {
			raw_material = goods:clay
		}

		sul_has_gathering_specialization = yes

		}

	modifier = {
		# Raw Extraction
		local_raw_material_output = 0.02
		local_max_rgo_size_modifier = 0.02
	}
}
